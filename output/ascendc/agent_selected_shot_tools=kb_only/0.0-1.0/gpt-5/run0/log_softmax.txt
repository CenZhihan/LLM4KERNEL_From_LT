host_operator_src="""
#include "log_softmax_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace optiling {

static ge::graphStatus TilingFunc(gert::TilingContext *context)
{
    LogSoftmaxCustomTilingData tiling;

    const gert::RuntimeAttrs *attrs = context->GetAttrs();
    // dim attribute
    const int64_t *dimPtr = attrs->GetAttrPointer<int64_t>(0);
    int64_t dimAttr = (dimPtr == nullptr) ? 1 : *dimPtr;

    // Get input origin shape
    const gert::Shape *inShape = context->GetInputShape(0);
    const gert::OriginShape originShape = inShape->GetOriginShape();
    const std::vector<int64_t> dims = originShape.GetDims();
    int64_t rank = static_cast<int64_t>(dims.size());
    if (rank == 0) {
        return ge::GRAPH_FAILED;
    }
    int64_t axis = dimAttr < 0 ? (dimAttr + rank) : dimAttr;
    if (axis < 0 || axis >= rank) {
        return ge::GRAPH_FAILED;
    }

    // Compute outter (M) and inner (K, last-axis length for reduction)
    uint64_t outter = 1;
    for (int64_t i = 0; i < rank; ++i) {
        if (i != axis) {
            outter *= static_cast<uint64_t>(dims[i]);
        }
    }
    uint64_t inner = static_cast<uint64_t>(dims[axis]);

    // Data type size (float32)
    const uint32_t typeSize = 4;
    // element number per 32-byte block
    const uint32_t elementNumPerBlk = 32 / typeSize;

    // UB capacity
    auto platformInfo = context->GetPlatformInfo();
    auto ascendcPlatform = platform_ascendc::PlatformAscendC(platformInfo);
    uint64_t ubSize = 0;
    ascendcPlatform.GetCoreMemSize(platform_ascendc::CoreMemType::UB, ubSize);

    // Choose tileM (rows per tile). Start from 1 for safety.
    uint32_t tileM = 1;

    // Compute temporary buffer size using tiling API for current tile shape
    {
        std::vector<int64_t> tileShapeDims = {static_cast<int64_t>(tileM), static_cast<int64_t>(inner)};
        ge::Shape tileShape(tileShapeDims);
        uint32_t maxTmp = AscendC::GetLogSoftMaxMaxTmpSize(tileShape, typeSize, false);

        // Estimate UB footprint: src + dst + sum + max + tmp
        // src: tileM * inner * typeSize
        // dst: tileM * inner * typeSize
        // sum: tileM * elementNumPerBlk * typeSize
        // max: tileM * elementNumPerBlk * typeSize
        uint64_t needBytes = static_cast<uint64_t>(tileM) * static_cast<uint64_t>(inner) * typeSize; // src
        needBytes += static_cast<uint64_t>(tileM) * static_cast<uint64_t>(inner) * typeSize;         // dst
        needBytes += static_cast<uint64_t>(tileM) * elementNumPerBlk * typeSize;                     // sum
        needBytes += static_cast<uint64_t>(tileM) * elementNumPerBlk * typeSize;                     // max
        needBytes += maxTmp;                                                                         // tmp

        // If UB is smaller than needBytes, fallback remains tileM=1.
        // Otherwise use computed maxTmp.
        uint32_t tmpBufferSize = (needBytes <= ubSize) ? maxTmp : maxTmp;

        // Fill tiling struct using LogSoftMaxTilingFunc
        AscendC::LogSoftMaxTilingFunc(tileShape, typeSize, tmpBufferSize, tiling.logSoftmaxTilingData);
        tiling.set_outter(static_cast<uint32_t>(outter));
        tiling.set_inner(static_cast<uint32_t>(inner));
        tiling.set_tileM(tileM);
        tiling.set_tmpBufferSize(tmpBufferSize);
    }

    context->SetBlockDim(1);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());
    context->SetTilingKey(1);
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}

} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext *context)
{
    const gert::Shape *x_shape = context->GetInputShape(0);
    gert::Shape *y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const ge::DataType x_dtype = context->GetInputDataType(0);
    context->SetOutputDataType(0, x_dtype);
    return GRAPH_SUCCESS;
}
} // namespace ge

namespace ops {
class LogSoftmaxCustom : public OpDef {
public:
    explicit LogSoftmaxCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)

            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("y")
            .ParamType(REQUIRED)

            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Attr("dim").AttrType(OPTIONAL).Int(1);
        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);

OP_ADD(LogSoftmaxCustom);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "log_softmax_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelLogSoftmax {
public:
    __aicore__ inline KernelLogSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t outter, uint32_t inner,
                                uint32_t tileM, uint32_t tmpBufferSize,
                                const optiling::LogSoftMaxTiling &tilingIn)
    {
        this->M = outter;
        this->K = inner;
        this->tileM = tileM;
        this->tmpSize = tmpBufferSize;
        this->elementNumPerBlk = 32 / sizeof(float);
        this->tiling = tilingIn;

        xGm.SetGlobalBuffer((__gm__ float *)x, static_cast<uint64_t>(M) * K);
        yGm.SetGlobalBuffer((__gm__ float *)y, static_cast<uint64_t>(M) * K);

        // UB buffers for one tile
        pipe.InitBuffer(inQueueSrc, BUFFER_NUM, tileM * K * sizeof(float));
        pipe.InitBuffer(outQueueDst, BUFFER_NUM, tileM * K * sizeof(float));
        pipe.InitBuffer(sumQueue, BUFFER_NUM, tileM * elementNumPerBlk * sizeof(float));
        pipe.InitBuffer(maxQueue, BUFFER_NUM, tileM * elementNumPerBlk * sizeof(float));
        pipe.InitBuffer(tmpQueue, BUFFER_NUM, tmpSize);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t m0 = 0; m0 < M; m0 += tileM) {
            uint32_t curM = (m0 + tileM <= M) ? tileM : (M - m0);

            // CopyIn
            AscendC::LocalTensor<float> srcLocal = inQueueSrc.AllocTensor<float>();
            AscendC::DataCopy(srcLocal, xGm[m0 * K], curM * K);
            inQueueSrc.EnQue(srcLocal);

            // Compute
            AscendC::LocalTensor<float> srcLocalDeq = inQueueSrc.DeQue<float>();
            AscendC::LocalTensor<float> dstLocal = outQueueDst.AllocTensor<float>();
            AscendC::LocalTensor<float> sumLocal = sumQueue.AllocTensor<float>();
            AscendC::LocalTensor<float> maxLocal = maxQueue.AllocTensor<float>();
            AscendC::LocalTensor<uint8_t> tmpLocal = tmpQueue.AllocTensor<uint8_t>();

            AscendC::SoftMaxShapeInfo shapeInfo = {curM, K, curM, K};
            AscendC::LogSoftMax<float, false>(dstLocal, sumLocal, maxLocal, srcLocalDeq,
                                              tmpLocal, tiling, shapeInfo);

            outQueueDst.EnQue<float>(dstLocal);
            sumQueue.FreeTensor(sumLocal);
            maxQueue.FreeTensor(maxLocal);
            inQueueSrc.FreeTensor(srcLocalDeq);
            tmpQueue.FreeTensor(tmpLocal);

            // CopyOut
            AscendC::LocalTensor<float> dstLocalDeq = outQueueDst.DeQue<float>();
            AscendC::DataCopy(yGm[m0 * K], dstLocalDeq, curM * K);
            outQueueDst.FreeTensor(dstLocalDeq);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueSrc;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueDst;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> sumQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> maxQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> tmpQueue;
    AscendC::GlobalTensor<float> xGm, yGm;

    uint32_t M = 0;
    uint32_t K = 0;
    uint32_t tileM = 1;
    uint32_t tmpSize = 0;
    uint32_t elementNumPerBlk = 0;
    optiling::LogSoftMaxTiling tiling;
};

extern "C" __global__ __aicore__ void log_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelLogSoftmax op;
    op.Init(x, y,
            tilingData.outter, tilingData.inner,
            tilingData.tileM, tilingData.tmpBufferSize,
            tilingData.logSoftmaxTilingData);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}
