host_operator_src="""
#include <vector>
#include <numeric>
#include <algorithm>
#include "log_softmax_custom_tiling.h"
#include "register/op_def_registry.h"

namespace {
inline int64_t NormalizeAxis(int64_t axis, int64_t rank) {
    if (axis < 0) axis += rank;
    if (axis < 0) axis = 0;
    if (axis >= rank) axis = rank - 1;
    return axis;
}

inline uint32_t Prod(const std::vector<int64_t> &dims, int64_t start, int64_t end) {
    // product of dims in [start, end)
    uint64_t p = 1;
    for (int64_t i = start; i < end; ++i) {
        p *= static_cast<uint64_t>(dims[i] > 0 ? dims[i] : 1);
    }
    return static_cast<uint32_t>(p);
}
} // namespace

namespace optiling {
const uint32_t MAX_BLOCKS = 48;
const uint32_t DTYPE_SIZE_F32 = 4;

static ge::graphStatus TilingFunc(gert::TilingContext *context)
{
    LogSoftmaxCustomTilingData tiling;

    const gert::RuntimeAttrs *attrs = context->GetAttrs();
    const int64_t *dimAttr = attrs->GetAttrPointer<int64_t>(0); // dim

    // Get input shape dims
    const gert::Shape *in_shape = context->GetInputShape(0);
    const auto &origin = in_shape->GetOriginShape();
    std::vector<int64_t> dims = origin.GetDims();
    if (dims.empty()) {
        dims = {1};
    }
    int64_t rank = static_cast<int64_t>(dims.size());
    int64_t axis = NormalizeAxis(*dimAttr, rank);

    // Compute outter/inner according to axis
    uint32_t outter = Prod(dims, 0, axis);
    uint32_t inner = Prod(dims, axis, rank);
    if (outter == 0) outter = 1;
    if (inner == 0) inner = 1;

    // Decide block dim and rows per block
    uint32_t blockDim = std::min<uint32_t>(outter, MAX_BLOCKS);
    if (blockDim == 0) blockDim = 1;
    uint32_t rowsPerBlock = outter / blockDim;
    if (rowsPerBlock == 0) {
        rowsPerBlock = 1;
        blockDim = std::min<uint32_t>(outter, MAX_BLOCKS);
        if (blockDim == 0) blockDim = 1;
    }
    uint32_t lastRowsPerBlock = outter - rowsPerBlock * (blockDim - 1);
    if (lastRowsPerBlock == 0) lastRowsPerBlock = rowsPerBlock;

    // Prepare LogSoftMax tiling for a single row [1, inner]
    std::vector<int64_t> rowDims = {1, static_cast<int64_t>(inner)};
    ge::Shape rowShape(rowDims);
    const uint32_t tmpSizePerRow = AscendC::GetLogSoftMaxMaxTmpSize(rowShape, DTYPE_SIZE_F32, false);

    // Fill inner tiling for HLL API
    optiling::LogSoftMaxTiling logTiling;
    AscendC::LogSoftMaxTilingFunc(rowShape, DTYPE_SIZE_F32, tmpSizePerRow, logTiling);

    // Save tiling
    tiling.set_outter(outter);
    tiling.set_inner(inner);
    tiling.set_rowsPerBlock(rowsPerBlock);
    tiling.set_lastRowsPerBlock(lastRowsPerBlock);
    tiling.set_tmpSizePerRow(tmpSizePerRow);
    tiling.set_logSoftmaxTilingData(logTiling);

    context->SetBlockDim(blockDim);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // No GM workspace needed
    size_t *workspaces = context->GetWorkspaceSizes(1);
    workspaces[0] = 0;

    return ge::GRAPH_SUCCESS;
}

} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext *context)
{
    const gert::Shape *x_shape = context->GetInputShape(0);
    gert::Shape *y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const ge::DataType x_dtype = context->GetInputDataType(0);
    context->SetOutputDataType(0, x_dtype);
    return GRAPH_SUCCESS;
}
} // namespace ge

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "log_softmax_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelLogSoftmax {
public:
    __aicore__ inline KernelLogSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t outter, uint32_t inner,
                                uint32_t rowsPerBlock, uint32_t lastRowsPerBlock,
                                uint32_t tmpSizePerRow,
                                const optiling::LogSoftMaxTiling &logTiling) {
        this->inner = inner;
        this->outter = outter;
        this->tmpSizePerRow = tmpSizePerRow;
        this->logTiling = logTiling;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowsThisBlock = (blockIdx == blockNum - 1) ? lastRowsPerBlock : rowsPerBlock;
        this->baseRow = blockIdx * rowsPerBlock;

        if (this->rowsThisBlock == 0) return;

        uint64_t elemOffset = static_cast<uint64_t>(this->baseRow) * this->inner;
        xGm.SetGlobalBuffer((__gm__ float *)x + elemOffset, static_cast<uint64_t>(this->rowsThisBlock) * this->inner);
        yGm.SetGlobalBuffer((__gm__ float *)y + elemOffset, static_cast<uint64_t>(this->rowsThisBlock) * this->inner);

        // Each row length = inner
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->inner * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->inner * sizeof(float));
        pipe.InitBuffer(tmpQueue, BUFFER_NUM, this->tmpSizePerRow);
        pipe.InitBuffer(maxBuf, sizeof(float));
        pipe.InitBuffer(sumBuf, sizeof(float));
    }

    __aicore__ inline void Process() {
        if (this->rowsThisBlock == 0) return;

        for (uint32_t r = 0; r < this->rowsThisBlock; ++r) {
            CopyIn(r);
            Compute(r);
            CopyOut(r);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t r) {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[r * this->inner], this->inner);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*r*/) {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<uint8_t> tmpLocal = tmpQueue.AllocTensor<uint8_t>();

        AscendC::SoftMaxShapeInfo shapeInfo = {1, this->inner, 1, this->inner};
        AscendC::LogSoftMax<float, false>(yLocal, maxLocal, sumLocal, xLocal, tmpLocal, this->logTiling, shapeInfo);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
        tmpQueue.FreeTensor(tmpLocal);
    }

    __aicore__ inline void CopyOut(uint32_t r) {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[r * this->inner], yLocal, this->inner);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> tmpQueue;
    AscendC::TBuf<AscendC::QuePosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::QuePosition::VECCALC> sumBuf;
    AscendC::GlobalTensor<float> xGm, yGm;

    uint32_t inner;
    uint32_t outter;
    uint32_t baseRow;
    uint32_t rowsThisBlock;
    uint32_t tmpSizePerRow;
    optiling::LogSoftMaxTiling logTiling;
};

extern "C" __global__ __aicore__ void log_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    (void)workspace;
    GET_TILING_DATA(tiling_data, tiling);

    KernelLogSoftmax op;
    op.Init(x, y,
            tiling_data.outter(),
            tiling_data.inner(),
            tiling_data.rowsPerBlock(),
            tiling_data.lastRowsPerBlock(),
            tiling_data.tmpSizePerRow(),
            tiling_data.logSoftmaxTilingData());
    op.Process();
}
