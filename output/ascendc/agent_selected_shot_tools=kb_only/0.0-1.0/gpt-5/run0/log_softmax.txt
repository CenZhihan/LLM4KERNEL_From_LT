host_operator_src="""
#include "log_softmax_custom_tiling.h"
#include "register/op_def_registry.h"
#include "graph/utils/type_utils.h"
#include "tiling/tiling_api.h"

namespace {
static inline uint32_t u32_clamp(uint64_t v) { return static_cast<uint32_t>(v); }
}

namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext *context)
{
    LogSoftmaxCustomTilingData tiling;

    const gert::Shape *in_shape_ptr = context->GetInputShape(0);
    const ge::Shape &ori_shape = in_shape_ptr->GetOriginShape();
    std::vector<int64_t> dims = ori_shape.GetDims();
    if (dims.empty()) {
        return ge::GRAPH_FAILED;
    }

    const gert::RuntimeAttrs *attrs = context->GetAttrs();
    const int64_t *dimAttr = attrs->GetAttrPointer<int64_t>(0);
    int64_t dim_val = (dimAttr != nullptr) ? *dimAttr : 1;
    int64_t rank = static_cast<int64_t>(dims.size());
    if (dim_val < 0) dim_val += rank;
    // 当前实现仅支持沿最后一维进行LogSoftmax
    if (dim_val != rank - 1) {
        return ge::GRAPH_FAILED;
    }

    uint64_t inner_u64 = static_cast<uint64_t>(dims.back());
    uint64_t total_len_u64 = static_cast<uint64_t>(ori_shape.GetShapeSize());
    if (inner_u64 == 0 || total_len_u64 % inner_u64 != 0) {
        return ge::GRAPH_FAILED;
    }
    uint64_t outter_u64 = total_len_u64 / inner_u64;

    // 块并行划分（每个block处理若干行）
    const uint32_t MAX_BLOCK_DIM = 48;
    uint32_t blockDim = static_cast<uint32_t>(std::min<uint64_t>(MAX_BLOCK_DIM, std::max<uint64_t>(1, outter_u64)));
    uint32_t blockOutter = (blockDim == 0) ? 0 : static_cast<uint32_t>(outter_u64 / blockDim);
    if (blockOutter == 0) {
        blockDim = 1;
        blockOutter = static_cast<uint32_t>(outter_u64);
    }
    uint32_t lastBlockOutter = static_cast<uint32_t>(outter_u64 - static_cast<uint64_t>(blockOutter) * (blockDim - 1));

    // 我们在UB中按行处理：tileOutter = 1
    uint32_t tileOutter = 1;
    ge::Shape perTileShape({static_cast<int64_t>(tileOutter), static_cast<int64_t>(inner_u64)});
    const uint32_t dtypesize = 4; // float32

    // 计算LogSoftMax的UB临时空间
    const uint32_t tmpSize = AscendC::GetLogSoftMaxMaxTmpSize(perTileShape, dtypesize, false);

    // 生成LogSoftMax API的tiling参数（按单行形状）
    AscendC::LogSoftMaxTilingFunc(perTileShape, dtypesize, tmpSize, tiling.logSoftmaxTilingData);

    // 回填到tiling数据
    context->SetBlockDim(blockDim);
    tiling.set_inner(u32_clamp(inner_u64));
    tiling.set_blockOutter(blockOutter);
    tiling.set_lastBlockOutter(lastBlockOutter);
    tiling.set_tileOutter(tileOutter);
    tiling.set_tmpSize(tmpSize);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无需workspace（UB临时空间由kernel内TBuf分配）
    size_t *ws = context->GetWorkspaceSizes(1);
    ws[0] = 0;

    return ge::GRAPH_SUCCESS;
}
} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext *context)
{
    const gert::Shape *x_shape = context->GetInputShape(0);
    gert::Shape *y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const ge::DataType x_dtype = context->GetInputDataType(0);
    context->SetOutputDataType(0, x_dtype);
    return GRAPH_SUCCESS;
}
} // namespace ge

namespace ops {
class LogSoftmaxCustom : public OpDef {
public:
    explicit LogSoftmaxCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Attr("dim").AttrType(OPTIONAL).Int(1);
        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};
OP_ADD(LogSoftmaxCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "log_softmax_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelLogSoftmax {
public:
    __aicore__ inline KernelLogSoftmax() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t inner,
                                uint32_t blockOutter,
                                uint32_t lastBlockOutter,
                                uint32_t tileOutter,
                                uint32_t tmpSize,
                                const optiling::LogSoftMaxTiling &logSoftmaxTilingData)
    {
        this->inner = inner;
        this->tileOutter = tileOutter;
        this->tmpSize = tmpSize;
        this->tiling = logSoftmaxTilingData;

        // 当前block处理的行数
        if (AscendC::GetBlockIdx() == AscendC::GetBlockNum() - 1) {
            this->rowsThisBlock = lastBlockOutter;
        } else {
            this->rowsThisBlock = blockOutter;
        }
        if (this->rowsThisBlock == 0 || this->inner == 0) return;

        uint32_t blockStartRow = blockOutter * AscendC::GetBlockIdx();
        uint32_t offset = blockStartRow * this->inner;

        xGm.SetGlobalBuffer((__gm__ float *)x + offset, this->rowsThisBlock * this->inner);
        yGm.SetGlobalBuffer((__gm__ float *)y + offset, this->rowsThisBlock * this->inner);

        // 每行处理 inner 个元素
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->inner * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->inner * sizeof(float));

        // sum/max 均为每行一个值
        pipe.InitBuffer(sumBuf, this->tileOutter * sizeof(float));
        pipe.InitBuffer(maxBuf, this->tileOutter * sizeof(float));

        // LogSoftMax内部临时空间
        pipe.InitBuffer(tmpBuf, this->tmpSize);
    }

    __aicore__ inline void Process()
    {
        if (this->rowsThisBlock == 0 || this->inner == 0) return;

        // 单行形状信息
        AscendC::SoftMaxShapeInfo shapeInfo = {this->tileOutter, this->inner, this->tileOutter, this->inner};

        for (uint32_t r = 0; r < this->rowsThisBlock; ++r) {
            CopyIn(r, this->inner);
            Compute(shapeInfo, this->inner);
            CopyOut(r, this->inner);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx, uint32_t length)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->inner], length);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(const AscendC::SoftMaxShapeInfo &shapeInfo, uint32_t length)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>(); // [tileOutter] = [1]
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>(); // [tileOutter] = [1]
        AscendC::LocalTensor<uint8_t> tmpLocal = tmpBuf.Get<uint8_t>();

        // 调用高阶API：LogSoftMax
        AscendC::LogSoftMax<float, false>(yLocal, sumLocal, maxLocal, xLocal, tmpLocal, this->tiling, shapeInfo);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx, uint32_t length)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[rowIdx * this->inner], yLocal, length);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::QuePosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::QuePosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::QuePosition::VECCALC> tmpBuf;

    AscendC::GlobalTensor<float> xGm, yGm;

    uint32_t inner = 0;
    uint32_t tileOutter = 1;
    uint32_t rowsThisBlock = 0;
    uint32_t tmpSize = 0;
    optiling::LogSoftMaxTiling tiling;
};

extern "C" __global__ __aicore__ void log_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLogSoftmax op;
    op.Init(x, y,
            tiling_data.inner,
            tiling_data.blockOutter,
            tiling_data.lastBlockOutter,
            tiling_data.tileOutter,
            tiling_data.tmpSize,
            tiling_data.logSoftmaxTilingData);
    op.Process();
}
