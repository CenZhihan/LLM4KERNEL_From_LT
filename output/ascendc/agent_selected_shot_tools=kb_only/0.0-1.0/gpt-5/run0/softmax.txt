host_operator_src="""
#include "softmax_custom_tiling.h"
#include "register/op_def_registry.h"
#include "graph/utils/type_utils.h"

namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext *context)
{
    SoftmaxCustomTilingData tiling;

    // Input original shape [m, n]
    const gert::Shape *in_shape = context->GetInputShape(0);
    ge::Shape origin = in_shape->GetOriginShape();
    int64_t dim_num = origin.GetDimNum();
    if (dim_num < 1) {
        return ge::GRAPH_FAILED;
    }
    int64_t m = 1;
    for (int64_t i = 0; i < dim_num - 1; ++i) {
        m *= origin.GetDim(i);
    }
    int64_t n = origin.GetDim(dim_num - 1);

    // We perform row-wise softmax: per-row shape [1, n]
    std::vector<int64_t> rowShapeVec = {1, n};
    ge::Shape rowShape(rowShapeVec);

    // Get minimum temporary space for SoftMax and compute tiling for [1, n]
    const uint32_t localWorkSpaceSize =
        AscendC::GetSoftMaxMinTmpSize(rowShape, sizeof(float), false);
    AscendC::SoftMaxTilingFunc(rowShape, sizeof(float), localWorkSpaceSize, tiling.get_softmaxTilingData());

    // Set block dim and row partition
    // Split along rows (m). Keep it simple and robust.
    uint32_t BLOCK_DIM = static_cast<uint32_t>(m >= 32 ? 32 : (m > 0 ? m : 1));
    context->SetBlockDim(BLOCK_DIM);

    uint32_t rows = static_cast<uint32_t>(m);
    uint32_t cols = static_cast<uint32_t>(n);
    uint32_t rowsPerBlock = rows / BLOCK_DIM;
    uint32_t lastRows = rows - rowsPerBlock * (BLOCK_DIM - 1);

    tiling.set_rows(rows);
    tiling.set_cols(cols);
    tiling.set_rowsPerBlock(rowsPerBlock);
    tiling.set_lastRows(lastRows);
    tiling.set_localWorkSpaceSize(localWorkSpaceSize);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(),
                        context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // No GM workspace needed
    size_t *workspace = context->GetWorkspaceSizes(1);
    workspace[0] = 0;

    return ge::GRAPH_SUCCESS;
}
} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext *context)
{
    const gert::Shape *x_shape = context->GetInputShape(0);
    gert::Shape *y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const ge::DataType x_dtype = context->GetInputDataType(0);
    context->SetOutputDataType(0, x_dtype);
    return GRAPH_SUCCESS;
}
} // namespace ge

namespace ops {
class SoftmaxCustom : public OpDef {
public:
    explicit SoftmaxCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};
OP_ADD(SoftmaxCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "softmax_custom_tiling_data.h"

constexpr int32_t IN_QUEUE_NUM = 2;
constexpr int32_t OUT_QUEUE_NUM = 2;

class KernelSoftmax {
public:
    __aicore__ inline KernelSoftmax() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t rows, uint32_t cols,
                                uint32_t rowsPerBlock, uint32_t lastRows,
                                uint32_t localWorkSpaceSize,
                                const AscendC::SoftMaxTiling &softmaxTilingData)
    {
        this->cols = cols;
        this->rowsPerBlock = rowsPerBlock;
        this->localWorkSpaceSize = localWorkSpaceSize;
        this->softmaxTiling = softmaxTilingData;

        // Determine rows handled by this core
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        if (blockIdx == blockNum - 1) {
            this->rowsThisBlock = lastRows;
        } else {
            this->rowsThisBlock = rowsPerBlock;
        }

        uint64_t offset = static_cast<uint64_t>(blockIdx) * rowsPerBlock * cols;
        xGm.SetGlobalBuffer((__gm__ float *)x + offset, rowsThisBlock * cols);
        yGm.SetGlobalBuffer((__gm__ float *)y + offset, rowsThisBlock * cols);

        pipe.InitBuffer(inQueue, IN_QUEUE_NUM, cols * sizeof(float));
        pipe.InitBuffer(outQueue, OUT_QUEUE_NUM, cols * sizeof(float));
        pipe.InitBuffer(sharedTmpBuf, localWorkSpaceSize);
    }

    __aicore__ inline void Process()
    {
        if (rowsThisBlock == 0 || cols == 0) return;

        for (uint32_t r = 0; r < rowsThisBlock; ++r) {
            CopyIn(r);
            Compute(r);
            CopyOut(r);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * cols], cols);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*rowIdx*/)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueue.AllocTensor<float>();
        AscendC::LocalTensor<uint8_t> tmpLocal = sharedTmpBuf.Get<uint8_t>();

        AscendC::SoftMaxShapeInfo shapeInfo;
        shapeInfo.srcM = 1;               // per-row softmax
        shapeInfo.srcK = cols;            // last axis length (must be 32B aligned)
        shapeInfo.oriSrcM = 0;
        shapeInfo.oriSrcK = 0;

        // Use SoftMax high-level API with sharedTmpBuffer and host tiling
        AscendC::SoftMax(yLocal, xLocal, tmpLocal, softmaxTiling, shapeInfo);

        outQueue.EnQue<float>(yLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> yLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(yGm[rowIdx * cols], yLocal, cols);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, IN_QUEUE_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, OUT_QUEUE_NUM> outQueue;
    AscendC::TBuf<AscendC::QuePosition::VECCALC> sharedTmpBuf;

    AscendC::GlobalTensor<float> xGm, yGm;

    uint32_t cols;
    uint32_t rowsPerBlock;
    uint32_t rowsThisBlock;
    uint32_t localWorkSpaceSize;
    AscendC::SoftMaxTiling softmaxTiling;
};

extern "C" __global__ __aicore__ void softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSoftmax op;
    op.Init(x, y,
            tiling_data.rows, tiling_data.cols,
            tiling_data.rowsPerBlock, tiling_data.lastRows,
            tiling_data.localWorkSpaceSize,
            tiling_data.softmaxTilingData);
    op.Process();
}
