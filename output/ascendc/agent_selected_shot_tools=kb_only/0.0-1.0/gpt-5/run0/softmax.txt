host_operator_src="""
#include "softmax_custom_tiling.h"
#include "register/op_def_registry.h"
#include "graph/utils/type_utils.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext *context)
{
    SoftmaxCustomTilingData tiling;

    // 读取 attr: dim（仅支持按最后一维/指定为1的二维输入）
    const gert::RuntimeAttrs *attrs = context->GetAttrs();
    int64_t dim_attr = 1;
    if (attrs != nullptr && attrs->GetNumAttrs() > 0) {
        const int64_t *dim_ptr = attrs->GetAttrPointer<int64_t>(0);
        if (dim_ptr != nullptr) {
            dim_attr = *dim_ptr;
        }
    }

    // 获取输入形状
    const gert::Shape *in_shape = context->GetInputShape(0);
    const ge::Shape &ori = in_shape->GetOriginShape();
    const size_t rank = ori.GetDimNum();
    if (rank == 0) {
        return ge::GRAPH_FAILED;
    }

    // 展平到 [m, n]，n 为最后一维长度；本算子只支持沿最后一维softmax
    int64_t n = ori.GetDim(rank - 1);
    int64_t m = 1;
    for (size_t i = 0; i + 1 < rank; ++i) {
        m *= static_cast<int64_t>(ori.GetDim(i));
    }

    // 若传入 dim_attr != rank-1，直接报错（本示例仅支持 dim 为最后一维）
    if (dim_attr != static_cast<int64_t>(rank - 1) && !(rank == 2 && dim_attr == 1)) {
        return ge::GRAPH_FAILED;
    }

    // 设置并行核数
    context->SetBlockDim(BLOCK_DIM);

    // 每核分配的行数
    uint32_t m_u = static_cast<uint32_t>(m);
    uint32_t n_u = static_cast<uint32_t>(n);
    uint32_t rowsPerBlock = (m_u / BLOCK_DIM);
    uint32_t lastBlockRows = m_u - rowsPerBlock * (BLOCK_DIM - 1);
    if (rowsPerBlock == 0) {
        // 行数小于 BLOCK_DIM 时，除最后一个核外其余核分配0行
        rowsPerBlock = 0;
        lastBlockRows = m_u; // 全部放到最后一个核
    }

    // SoftMax 高阶API需要的tiling（以每次处理 [1, n] 为粒度）
    std::vector<int64_t> rowShape = {1, static_cast<int64_t>(n_u)};
    ge::Shape srcShape(rowShape);
    const uint32_t localWorkSpaceSize = AscendC::GetSoftMaxMinTmpSize(srcShape, sizeof(float), false);
    AscendC::SoftMaxTilingFunc(srcShape, sizeof(float), localWorkSpaceSize, tiling.softmaxTiling);

    tiling.set_rowsPerBlock(rowsPerBlock);
    tiling.set_lastBlockRows(lastBlockRows);
    tiling.set_kLen(n_u);

    // 保存tiling数据
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无需workspace
    size_t *workspaceSizes = context->GetWorkspaceSizes(1);
    workspaceSizes[0] = 0;

    return ge::GRAPH_SUCCESS;
}
} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext *context)
{
    const gert::Shape *x_shape = context->GetInputShape(0);
    gert::Shape *y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const ge::DataType x_dtype = context->GetInputDataType(0);
    context->SetOutputDataType(0, x_dtype);
    return GRAPH_SUCCESS;
}
} // namespace ge

namespace ops {
class SoftmaxCustom : public OpDef {
public:
    explicit SoftmaxCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Attr("dim").AttrType(OPTIONAL).Int(1);

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};
OP_ADD(SoftmaxCustom);
} // namespace ops
"""

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "softmax_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // 每个队列的张量个数

class KernelSoftmax {
public:
    __aicore__ inline KernelSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t rowsPerBlock, uint32_t lastBlockRows,
                                uint32_t kLen,
                                const AscendC::SoftMaxTiling &smTiling)
    {
        this->kLen = kLen;
        this->softmaxTiling = smTiling;

        // 当前核负责的行数
        if (AscendC::GetBlockIdx() == AscendC::GetBlockNum() - 1) {
            this->myRows = lastBlockRows;
        } else {
            this->myRows = rowsPerBlock;
        }

        if (this->myRows == 0 || this->kLen == 0) {
            return;
        }

        // 计算起始偏移（按行切分）
        uint32_t startRow = rowsPerBlock * AscendC::GetBlockIdx();
        uint32_t offset = startRow * this->kLen;

        xGm.SetGlobalBuffer((__gm__ float*)x + offset, this->myRows * this->kLen);
        yGm.SetGlobalBuffer((__gm__ float*)y + offset, this->myRows * this->kLen);

        // 初始化本地缓冲区（每次处理 1 行，长度 kLen）
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->kLen * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->kLen * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->myRows == 0 || this->kLen == 0) {
            return;
        }

        for (uint32_t r = 0; r < this->myRows; ++r) {
            CopyIn(r, this->kLen);
            Compute(r, this->kLen);
            CopyOut(r, this->kLen);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx, uint32_t length)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->kLen], length);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*rowIdx*/, uint32_t length)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // 使用SoftMax高阶API，按 [1, kLen] 行粒度计算
        AscendC::SoftMax<float>(yLocal, xLocal, this->softmaxTiling);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx, uint32_t length)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[rowIdx * this->kLen], yLocal, length);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<float> xGm, yGm;

    uint32_t myRows = 0;
    uint32_t kLen = 0;
    AscendC::SoftMaxTiling softmaxTiling;
};

extern "C" __global__ __aicore__ void softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelSoftmax op;
    op.Init(x, y,
            tiling_data.rowsPerBlock, tiling_data.lastBlockRows,
            tiling_data.kLen,
            tiling_data.softmaxTiling);
    op.Process();
}
