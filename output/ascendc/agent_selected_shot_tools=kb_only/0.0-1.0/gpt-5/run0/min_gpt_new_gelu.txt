kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "min_gpt_new_gelu_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

// FastGELU (minGPT): 0.5 * x * (1 + tanh( sqrt(2/pi) * (x + 0.044715 * x^3) ))
class KernelMinGptNewGelu {
public:
    __aicore__ inline KernelMinGptNewGelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t blockLength, uint32_t lastBlockLength, uint32_t tileLength)
    {
        this->tileLength = tileLength;

        if (AscendC::GetBlockIdx() == AscendC::GetBlockNum() - 1) {
            this->blockLength = lastBlockLength;
        } else {
            this->blockLength = blockLength;
        }

        if (this->blockLength == 0) return;

        this->tileNum = this->blockLength / this->tileLength;
        this->tailLength = this->blockLength % this->tileLength;

        uint32_t offset = blockLength * AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ float *)x + offset, this->blockLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + offset, this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        // three temporary buffers for intermediate results
        pipe.InitBuffer(tmpBuffer1, this->tileLength * sizeof(float)); // x^3
        pipe.InitBuffer(tmpBuffer2, this->tileLength * sizeof(float)); // inner linear
        pipe.InitBuffer(tmpBuffer3, this->tileLength * sizeof(float)); // tanh output and accumulations
    }

    __aicore__ inline void Process()
    {
        if (this->blockLength == 0) return;

        for (int32_t i = 0; i < this->tileNum; i++) {
            CopyIn(i, this->tileLength);
            Compute(i, this->tileLength);
            CopyOut(i, this->tileLength);
        }
        if (this->tailLength > 0) {
            CopyIn(this->tileNum, this->tailLength);
            Compute(this->tileNum, this->tailLength);
            CopyOut(this->tileNum, this->tailLength);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress, uint32_t length)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], length);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/, uint32_t length)
    {
        // constants
        const float kAlpha = 0.044715f;
        const float kC = 0.7978845608028654f; // sqrt(2/pi)
        const float kHalf = 0.5f;
        const float kOne  = 1.0f;

        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuffer1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuffer2.Get<float>();
        AscendC::LocalTensor<float> tmp3 = tmpBuffer3.Get<float>();

        // tmp1 = x^3
        AscendC::Mul(tmp1, xLocal, xLocal, length);     // x^2
        AscendC::Mul(tmp1, tmp1, xLocal, length);       // x^3

        // tmp2 = x + 0.044715 * x^3
        AscendC::Muls(tmp2, tmp1, kAlpha, length);
        AscendC::Add(tmp2, tmp2, xLocal, length);

        // tmp2 = sqrt(2/pi) * tmp2
        AscendC::Muls(tmp2, tmp2, kC, length);

        // tmp3 = tanh(tmp2)
        AscendC::Tanh(tmp3, tmp2, length);

        // tmp3 = 0.5 * (1 + tmp3)
        AscendC::Adds(tmp3, tmp3, kOne, length);
        AscendC::Muls(tmp3, tmp3, kHalf, length);

        // y = x * tmp3
        AscendC::Mul(yLocal, xLocal, tmp3, length);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress, uint32_t length)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, length);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::QuePosition::VECCALC> tmpBuffer1, tmpBuffer2, tmpBuffer3;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm, yGm;
    uint32_t blockLength = 0;
    uint32_t tileNum = 0;
    uint32_t tailLength = 0;
    uint32_t tileLength = 0;
};

extern "C" __global__ __aicore__ void min_gpt_new_gelu_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMinGptNewGelu op;
    op.Init(x, y, tiling_data.blockLength, tiling_data.lastBlockLength, tiling_data.tileLength);
    op.Process();
}
