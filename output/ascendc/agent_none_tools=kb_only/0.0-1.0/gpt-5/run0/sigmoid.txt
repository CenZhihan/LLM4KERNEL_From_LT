// SigmoidCustom: Tiling 结构与实现
// 参考：CANN Ascend C API Reference 6 Host API - 6.2.2 TilingData Structure Registration
//      以及 6.1.6.1/6.1.6.2 Input/Output 用法
#ifndef SIGMOID_CUSTOM_TILING_H_
#define SIGMOID_CUSTOM_TILING_H_

#include "register/tilingdata_base.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace optiling {

// 定义并注册 SigmoidCustom 的 TilingData 结构
BEGIN_TILING_DATA_DEF(SigmoidCustomTilingData)
TILING_DATA_FIELD_DEF(uint32_t, totalSize);        // 实际元素个数（按 ND 展开）
TILING_DATA_FIELD_DEF(uint32_t, calCountAligned);  // 32B 对齐后的计算元素个数
END_TILING_DATA_DEF;

REGISTER_TILING_DATA_CLASS(SigmoidCustom, SigmoidCustomTilingData)

// Tiling 实现：计算总元素数、对齐后的 calCount，并根据数据类型设置 TilingKey
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    SigmoidCustomTilingData tiling;

    // 获取输入形状，并计算元素总数（按存储形状 ND 展开）
    const gert::StorageShape* src_shape = context->GetInputShape(0);
    const auto& shape = src_shape->GetStorageShape();
    uint32_t dim_num = shape.GetDimNum();
    uint64_t total_elems = 1;
    for (uint32_t i = 0; i < dim_num; ++i) {
        total_elems *= static_cast<uint64_t>(shape.GetDim(i));
    }

    // 数据类型与对齐
    auto dt = context->GetInputTensor(0)->GetDataType();
    uint32_t typeSize = 0;
    if (dt == ge::DT_FLOAT16) {
        typeSize = 2;
        context->SetTilingKey(1);
    } else if (dt == ge::DT_FLOAT) {
        typeSize = 4;
        context->SetTilingKey(2);
    } else {
        // 不支持的数据类型
        return ge::GRAPH_FAILED;
    }

    uint64_t bytes = total_elems * typeSize;
    uint64_t aligned_bytes = (bytes + 32 - 1) / 32 * 32;
    uint64_t calCountAligned64 = aligned_bytes / typeSize;

    tiling.set_totalSize(static_cast<uint32_t>(total_elems));
    tiling.set_calCountAligned(static_cast<uint32_t>(calCountAligned64));

    // 配置并保存 TilingData
    context->SetBlockDim(1);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(),
                        context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无需 workspace
    size_t* currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    return ge::GRAPH_SUCCESS;
}

} // namespace optiling

#endif // SIGMOID_CUSTOM_TILING_H_
"""

kernel_src = r"""
// SigmoidCustom: AscendC Kernel 实现
// 参考：CANN Ascend C API Reference 5 High-level APIs
// 代码风格与缓冲区初始化参考 Sum/Floor 示例（DataCopy、TQue、TPipe 等）
#include "kernel_operator.h"

template <typename T>
class KernelSigmoidCustom
{
public:
    __aicore__ inline KernelSigmoidCustom() {}

    __aicore__ inline void Init(__gm__ uint8_t* xGm, __gm__ uint8_t* yGm,
                                uint32_t totalSize, uint32_t calCountAligned)
    {
        total_elems = totalSize;
        cal_elems_aligned = calCountAligned;

        xGlobal.SetGlobalBuffer((__gm__ T*)xGm);
        yGlobal.SetGlobalBuffer((__gm__ T*)yGm);

        // 输入/输出/临时计算队列缓冲区
        pipe.InitBuffer(inQueueX, 1, cal_elems_aligned * sizeof(T));
        pipe.InitBuffer(outQueueY, 1, cal_elems_aligned * sizeof(T));
        pipe.InitBuffer(tmpCalcQ, 1, cal_elems_aligned * sizeof(T));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<T> xLocal = inQueueX.AllocTensor<T>();
        AscendC::DataCopy(xLocal, xGlobal, total_elems);
        inQueueX.EnQue<T>(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<T> xLocal = inQueueX.DeQue<T>();
        AscendC::LocalTensor<T> yLocal = outQueueY.AllocTensor<T>();
        AscendC::LocalTensor<T> tmpLocal = tmpCalcQ.AllocTensor<T>();

        // Sigmoid(x) = 1 / (1 + exp(-x))
        T minusOne = static_cast<T>(-1.0f);
        T one = static_cast<T>(1.0f);

        // tmpLocal = -x
        AscendC::Muls(tmpLocal, xLocal, minusOne, total_elems);
        // tmpLocal = exp(-x)
        AscendC::Exp(tmpLocal, tmpLocal, total_elems);
        // tmpLocal = 1 + exp(-x)
        AscendC::Adds(tmpLocal, tmpLocal, one, total_elems);
        // yLocal = 1 / tmpLocal
        AscendC::Reciprocal(yLocal, tmpLocal, total_elems);

        outQueueY.EnQue<T>(yLocal);
        tmpCalcQ.FreeTensor(tmpLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<T> yLocal = outQueueY.DeQue<T>();
        AscendC::DataCopy(yGlobal, yLocal, total_elems);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueY;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpCalcQ;

    AscendC::GlobalTensor<T> xGlobal;
    AscendC::GlobalTensor<T> yGlobal;

    uint32_t total_elems = 0;
    uint32_t cal_elems_aligned = 0;
};

// kernel 入口，按 TilingKey 分派不同精度
extern "C" __global__ __aicore__ void sigmoid_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    if (TILING_KEY_IS(1)) { // float16
        KernelSigmoidCustom<half> op;
        op.Init(x, y, tiling_data.totalSize, tiling_data.calCountAligned);
        op.Process();
    } else if (TILING_KEY_IS(2)) { // float
        KernelSigmoidCustom<float> op;
        op.Init(x, y, tiling_data.totalSize, tiling_data.calCountAligned);
        op.Process();
    }
}

# Python 绑定：加载自定义算子并提供 Python 侧便捷调用接口
# 说明：推理/训练侧可在程序启动时调用 load_op() 完成 .so 加载
import os
import torch

_OP_NS = "ascend_custom"
_OP_NAME = "SigmoidCustom"

def load_op(so_path: str = None):
    """
    加载自定义算子动态库（.so）。
    优先使用传入路径，其次读取环境变量 CUSTOM_OP_SO。
    """
    lib_path = so_path or os.environ.get("CUSTOM_OP_SO", "")
    if lib_path and os.path.exists(lib_path):
        torch.ops.load_library(lib_path)

def sigmoid_custom(x: torch.Tensor) -> torch.Tensor:
    """
    Python 侧调用自定义算子，若不可用则回退至 PyTorch 实现。
    """
    try:
        op = getattr(torch.ops, _OP_NS).__getattr__(_OP_NAME)
        return op(x)
    except Exception:
        return torch.sigmoid(x)
