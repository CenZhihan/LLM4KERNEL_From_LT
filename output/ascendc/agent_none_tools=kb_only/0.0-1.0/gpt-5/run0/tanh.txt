kernel_src = r"""
// tanh_custom.cpp
// 内核实现基于 AscendC 高阶 API：Tanh，参考 5.1.1.1 Tanh 原型说明。
// 为简化并保证兼容性，这里使用由 API 框架分配临时空间的重载：Tanh(dstLocal, srcLocal)。
// 队列与拷贝方式参考 Sum/Mean 示例（DataCopy、TPipe/TQue），并按 32B 对齐申请本地缓冲。
#include "kernel_operator.h"

template <typename T>
class KernelTanhCustom {
public:
    __aicore__ inline KernelTanhCustom() {}
    __aicore__ inline void Init(__gm__ uint8_t *xGm, __gm__ uint8_t *yGm, uint32_t totalLen) {
        totalLength = totalLen;
        srcGlobal.SetGlobalBuffer((__gm__ T *)xGm);
        dstGlobal.SetGlobalBuffer((__gm__ T *)yGm);

        // 申请本地缓冲（按 32 字节对齐），参考文档示例中对齐方式
        uint32_t bytes = totalLength * sizeof(T);
        uint32_t alignedBytes = (bytes + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE;

        pipe.InitBuffer(inQueue, 1, alignedBytes);
        pipe.InitBuffer(outQueue, 1, alignedBytes);
    }

    __aicore__ inline void Process() {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn() {
        AscendC::LocalTensor<T> srcLocal = inQueue.AllocTensor<T>();
        // 按元素数进行拷贝
        AscendC::DataCopy(srcLocal, srcGlobal, totalLength);
        inQueue.EnQue(srcLocal);
    }

    __aicore__ inline void Compute() {
        AscendC::LocalTensor<T> srcLocal = inQueue.DeQue<T>();
        AscendC::LocalTensor<T> dstLocal = outQueue.AllocTensor<T>();

        // 使用 API 框架分配的临时空间重载（无需显式传入 sharedTmpBuffer）
        AscendC::Tanh<T>(dstLocal, srcLocal);

        outQueue.EnQue<T>(dstLocal);
        inQueue.FreeTensor(srcLocal);
    }

    __aicore__ inline void CopyOut() {
        AscendC::LocalTensor<T> dstLocal = outQueue.DeQue<T>();
        AscendC::DataCopy(dstGlobal, dstLocal, totalLength);
        outQueue.FreeTensor(dstLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue;
    AscendC::GlobalTensor<T> srcGlobal, dstGlobal;
    uint32_t totalLength = 0;
};

// 内核入口：根据 TilingKey 分支选择数据类型（参考 Mean 示例的 TILING_KEY_IS 用法）
extern "C" __global__ __aicore__ void tanh_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    if (TILING_KEY_IS(1)) { // float16
        KernelTanhCustom<half> op;
        op.Init(x, y, tiling_data.totalLength);
        op.Process();
    } else if (TILING_KEY_IS(2)) { // float32
        KernelTanhCustom<float> op;
        op.Init(x, y, tiling_data.totalLength);
        op.Process();
    } else { // 默认分支
        KernelTanhCustom<half> op;
        op.Init(x, y, tiling_data.totalLength);
        op.Process();
    }
}
"""

# 自定义算子 Python 绑定（占位），尝试加载已编译好的 AscendC 自定义算子共享库。
# 若加载失败，回退到 torch.tanh，以便在非昇腾环境或未编译时也可运行。
import os
import torch

def _load_library() -> bool:
    candidates = [
        "./build/libtanh_custom.so",
        "./out/libtanh_custom.so",
        "libtanh_custom.so"
    ]
    for path in candidates:
        if os.path.exists(path):
            try:
                torch.ops.load_library(path)
                return True
            except Exception:
                pass
    return False

_loaded = _load_library()

def tanh_custom(x: torch.Tensor) -> torch.Tensor:
    if _loaded:
        # 约定自定义域与算子名：ascendc_custom::tanh_custom
        # 实际域名以编译产物中注册为准，这里进行存在性检测并安全调用
        try:
            if hasattr(torch.ops, "ascendc_custom") and hasattr(torch.ops.ascendc_custom, "tanh_custom"):
                return torch.ops.ascendc_custom.tanh_custom(x)
        except Exception:
            pass
    return torch.tanh(x)
