// hardsigmoid_custom.cpp
// 参考资料中的 Sum/Mean 的宿主侧实现与注册流程，定义本算子的 Tiling 与 InferShape
#include "hardsigmoid_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace optiling {

// 简单按二维计算元素个数（参考示例中取 GetDim(0) 与 GetDim(1) 的写法）。
// 当前模型输入为 [batch, dim]，因此按二维即可满足需求。
static ge::graphStatus TilingFunc(gert::TilingContext *context)
{
    HardsigmoidCustomTilingData tiling;

    const gert::StorageShape *src_shape = context->GetInputShape(0);
    uint32_t outter = src_shape->GetStorageShape().GetDim(0);
    uint32_t inner  = src_shape->GetStorageShape().GetDim(1);
    uint32_t total  = outter * inner;

    tiling.set_totalLength(total);

    context->SetBlockDim(1); // 元素级逐点计算，单 block 即可
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(),
                        context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0; // 无需额外 workspace

    return ge::GRAPH_SUCCESS;
}

} // namespace optiling

namespace ge {

// 输出与输入同形状
static ge::graphStatus InferShape(gert::InferShapeContext *context)
{
    const gert::Shape *x_shape = context->GetInputShape(0);
    gert::Shape *y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}

} // namespace ge

kernel_src = r"""
// hardsigmoid_custom_kernel.cpp
// 参考 AscendC API 模板（Unary 示例）与 Sum/Mean 示例的管线和队列用法
// 实现 HardSigmoid: y = clamp(0.2 * x + 0.5, 0, 1)
// 注意：以下使用的 AscendC 向量标量接口（Muls/Adds/Maxs/Mins）为 AscendC 高层 API 常规用法
// 与参考中的 DataCopy、Duplicate、Sum/Mean 用法一致采用 kernel_operator.h 中的 API 风格。

#include "kernel_operator.h"

class KernelHardSigmoid {
public:
    __aicore__ inline KernelHardSigmoid() {}
    __aicore__ inline void Init(__gm__ uint8_t* xGm, __gm__ uint8_t* yGm, uint32_t totalLengthIn)
    {
        totalLength = totalLengthIn;
        xGlobal.SetGlobalBuffer((__gm__ half*)xGm);
        yGlobal.SetGlobalBuffer((__gm__ half*)yGm);

        // 为简化示例，直接按 totalLength 申请一次性缓冲，参考 Sum/Mean 示例的写法
        pipe.InitBuffer(inQueue, 1, totalLength * sizeof(half));
        pipe.InitBuffer(outQueue, 1, totalLength * sizeof(half));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<half> xLocal = inQueue.AllocTensor<half>();
        AscendC::DataCopy(xLocal, xGlobal, totalLength);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<half> xLocal = inQueue.DeQue<half>();
        AscendC::LocalTensor<half> yLocal = outQueue.AllocTensor<half>();

        // y = clamp(0.2 * x + 0.5, 0, 1)
        AscendC::Muls(yLocal, xLocal, (half)0.2f, totalLength);
        AscendC::Adds(yLocal, yLocal, (half)0.5f, totalLength);
        AscendC::Maxs(yLocal, yLocal, (half)0.0f, totalLength);
        AscendC::Mins(yLocal, yLocal, (half)1.0f, totalLength);

        outQueue.EnQue<half>(yLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<half> yLocal = outQueue.DeQue<half>();
        AscendC::DataCopy(yGlobal, yLocal, totalLength);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue;
    AscendC::GlobalTensor<half> xGlobal, yGlobal;

    uint32_t totalLength = 0;
};

extern "C" __global__ __aicore__ void hardsigmoid_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelHardSigmoid op;
    op.Init(x, y, tilingData.totalLength);
    op.Process();
}
"""
