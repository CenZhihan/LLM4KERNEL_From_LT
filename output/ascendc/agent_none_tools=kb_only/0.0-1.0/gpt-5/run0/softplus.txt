static ge::graphStatus TilingFunc(gert::TilingContext *context)
{
    // 从输入形状计算总元素个数
    const gert::Shape *x_shape = context->GetInputShape(0);
    std::vector<int64_t> dims;
    dims.reserve(x_shape->GetDimNum());
    for (size_t i = 0; i < x_shape->GetDimNum(); ++i) {
        dims.push_back(x_shape->GetDim(i));
    }

    uint32_t srcSize = 1;
    for (auto d : dims) {
        srcSize *= static_cast<uint32_t>(d);

    tiling.set_srcSize(srcSize);

    context->SetBlockDim(1);

kernel_src = r"""#include "kernel_operator.h"
#include "softplus_custom_tiling.h"

// 说明：本内核使用 AscendC 数学库组合实现 softplus(x) = log(1 + exp(x))，
// 参考了单输入模板与数学库示例（如 Exp/Asin 示例）的用法。
// 注意：具体 Add/Log API 的签名以实际 AscendC 版本为准。

template <typename T>
class KernelSoftplus {
public:
    __aicore__ inline KernelSoftplus() {}

    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR yGm, uint32_t size)
    {
        totalElems = size;
        xGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(xGm), totalElems);
        yGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(yGm), totalElems);

        // 输入、输出与中间缓冲区（用于临时存放 e^x 或 1+e^x）
        pipe.InitBuffer(inQueue, 1, totalElems * sizeof(T));
        pipe.InitBuffer(outQueue, 1, totalElems * sizeof(T));
        pipe.InitBuffer(tmpQueue, 1, totalElems * sizeof(T));
        pipe.InitBuffer(workQueue, 1, totalElems * sizeof(T));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<T> xLocal = inQueue.AllocTensor<T>();
        AscendC::DataCopy(xLocal, xGlobal, totalElems);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<T> xLocal = inQueue.DeQue<T>();
        AscendC::LocalTensor<T> tmpLocal = tmpQueue.AllocTensor<T>();   // 用于 e^x 与 1+e^x
        AscendC::LocalTensor<T> yLocal = outQueue.AllocTensor<T>();
        AscendC::LocalTensor<T> onesLocal = workQueue.AllocTensor<T>(); // 常量 1

        // 1) onesLocal <- 1
        T oneScalar(1);
        AscendC::Duplicate<T>(onesLocal, oneScalar, totalElems);

        // 2) tmpLocal <- exp(xLocal)
        // 示例中 Exp 的模板用法展示了 isReuseSource 的第三模板参数，这里不复用源
        // 模板参数与实际 API 版本可能存在差异，请以环境为准
        AscendC::Exp<T, 15, false>(tmpLocal, xLocal, totalElems);

        // 3) tmpLocal <- tmpLocal + onesLocal
        // 元素加法（Add）的接口以具体 AscendC 版本为准，这里按常见签名使用
        AscendC::Add<T>(tmpLocal, tmpLocal, onesLocal, totalElems);

        // 4) yLocal <- log(tmpLocal)
        AscendC::Log<T>(yLocal, tmpLocal, totalElems);

        outQueue.EnQue<T>(yLocal);
        // 释放本地张量
        workQueue.FreeTensor(onesLocal);
        tmpQueue.FreeTensor(tmpLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<T> yLocal = outQueue.DeQue<T>();
        AscendC::DataCopy(yGlobal, yLocal, totalElems);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::GlobalTensor<T> xGlobal;
    AscendC::GlobalTensor<T> yGlobal;
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> workQueue;
    uint32_t totalElems = 0;
};

extern "C" __global__ __aicore__ void softplus_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelSoftplus<half> op;
    op.Init(x, y, tilingData.srcSize);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}
"""

# 轻量级 Python 绑定：尝试加载自定义算子动态库并通过 torch.ops 调用
import os
import torch
import torch.nn.functional as F

def load_softplus_custom(lib_path: str = None):
    """
    加载已编译的自定义算子共享库（.so）
    """
    if lib_path is None:
        return
    if os.path.isfile(lib_path):
        try:
            torch.ops.load_library(lib_path)
        except Exception as e:
            # 忽略加载失败，后续调用将回退到 PyTorch 实现
            pass

def softplus_custom(x: torch.Tensor) -> torch.Tensor:
    """
    调用自定义 SoftplusCustom（若不可用则回退到 PyTorch 实现）
    """
    try:
        return torch.ops.SoftplusCustom(x)
    except Exception:
        return F.softplus(x)
