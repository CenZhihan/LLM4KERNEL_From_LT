// softmax_custom_tiling.h
// References:
// - 5.3.1.8 SoftMax Tiling
// - 5.3.1.9 SoftMax/SimpleSoftMax Tiling
#pragma once
#include "register/tilingdata_base.h"
#include "tiling/tiling_api.h"
#include "graph/tensor.h"
#include "graph/types.h"
#include "graph/shape.h"
#include <vector>

BEGIN_TILING_DATA_DEF(TilingData)
    TILING_DATA_FIELD_DEF(uint32_t, totalLength);
    TILING_DATA_FIELD_DEF(uint32_t, tileNum);
    TILING_DATA_FIELD_DEF(uint32_t, localWorkSpaceSize);
    TILING_DATA_FIELD_DEF_STRUCT(SoftMaxTiling, softmaxTilingData);
END_TILING_DATA_DEF;

namespace optiling {
static const uint32_t BLOCK_DIM = 8;
static const uint32_t TILE_NUM = 8;

// Tiling implementation following CANN C API Reference 5.3.1.8/5.3.1.9
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    TilingData tiling;
    context->SetBlockDim(BLOCK_DIM);

    auto in_desc = context->GetInputTensor(0);
    if (in_desc == nullptr) {
        return ge::GRAPH_FAILED;
    }

    uint64_t totalLength = in_desc->GetShapeSize();
    tiling.set_totalLength(static_cast<uint32_t>(totalLength));
    tiling.set_tileNum(TILE_NUM);

    ge::Shape srcShape = in_desc->GetShape();

    // Use minimum temporary space to ensure correctness (per docs).
    // Data type size: half = 2.
    const uint32_t dataTypeSize = 2U;
    const bool isReuseSource = false;
    const uint32_t localWorkSpaceSize = AscendC::GetSoftMaxMinTmpSize(srcShape, dataTypeSize, isReuseSource);
    tiling.set_localWorkSpaceSize(localWorkSpaceSize);

    // Obtain SoftMax tiling parameters
    AscendC::SoftMaxTilingFunc(srcShape, dataTypeSize, localWorkSpaceSize, tiling.softmaxTilingData());

    // Save tiling
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // Single tiling key
    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}
} // namespace optiling
"""

// softmax_custom_host.cpp
// Host-side operator skeleton that prepares tiling and launches kernel.
// References:
// - 5.3.1.8 SoftMax Tiling (Example workflow)

#include "softmax_custom_tiling.h"
#include "register/op_def_registry.h"
#include "kernel_operator.h"
#include "graph/tensor.h"
#include "graph/types.h"
#include "graph/shape.h"
#include "register/tilingdata_base.h"
#include "optiling/tilingdata.h"
#include "proto/op_proto.h"
#include "proto/op_attr.h"
#include "acl/acl.h"
#include "acl/acl_rt.h"

using namespace ge;

namespace {
static ge::graphStatus SoftmaxCustomInferShape(gert::InferShapeContext* context)
{
    auto x_desc = context->GetInputDesc(0);
    if (x_desc == nullptr) {
        return ge::GRAPH_FAILED;
    }
    // Output shape equals input shape
    context->SetOutputDesc(0, *x_desc);
    return ge::GRAPH_SUCCESS;
}

static ge::graphStatus SoftmaxCustomTiling(gert::TilingContext* context)
{
    return optiling::TilingFunc(context);
}

static ge::graphStatus SoftmaxCustomCompute(gert::KernelContext* context)
{
    // Retrieve GM addresses
    auto x_gm = reinterpret_cast<uint8_t*>(context->GetInput(0)->GetData());
    auto y_gm = reinterpret_cast<uint8_t*>(context->GetOutput(0)->GetData());
    auto tiling_gm = reinterpret_cast<uint8_t*>(context->GetRawTilingData()->GetData());

    // Block dim must match tiling setting
    const uint32_t blockDim = 8;

    // Launch kernel softmax_custom(x, y, tiling)
    rtStream_t stream = reinterpret_cast<rtStream_t>(context->GetStream());
    rtError_t rtRet = rtKernelLaunch((void*)softmax_custom,    // kernel func pointer (stub)
                                     blockDim,
                                     x_gm, y_gm, tiling_gm,   // args packed by framework
                                     nullptr, stream);
    if (rtRet != RT_ERROR_NONE) {
        return ge::GRAPH_FAILED;
    }
    return ge::GRAPH_SUCCESS;
}
} // namespace

extern "C" __attribute__((visibility("default")))
void RegisterSoftmaxCustom()
{
    static gert::OpDefBuilder builder("SoftmaxCustom");
    builder
        .Input(0, "x")
        .Output(0, "y")
        .SetInferShape(SoftmaxCustomInferShape)
        .SetTiling(SoftmaxCustomTiling)
        .SetCompute(SoftmaxCustomCompute)
        .Register();
}

kernel_src = r"""
// softmax_custom_kernel.cpp
// AscendC kernel implementing SoftMax over the last dimension.
// References:
// - 5.3.1.8 SoftMax Tiling (SoftMaxTiling, flow)
// - 5.3.1.9 SoftMax/SimpleSoftMax Tiling (GetSoftMaxMinTmpSize/SoftMaxTilingFunc)
// The SoftMax high-level API usage mirrors patterns from SoftmaxGrad/SoftmaxFlashV2 samples.

#include "kernel_operator.h"

template <typename T>
class KernelSoftmax {
public:
    __aicore__ inline KernelSoftmax() {}

    __aicore__ inline void Init(__gm__ uint8_t* xGm, __gm__ uint8_t* yGm, const SoftMaxTiling& tilingData)
    {
        xGlobal.SetGlobalBuffer((__gm__ T*)xGm);
        yGlobal.SetGlobalBuffer((__gm__ T*)yGm);
        tiling = tilingData;

        height = tiling.srcM;  // product of non-last axes
        width  = tiling.srcK;  // last axis length

        // Initialize queues with buffer sizes derived from shape
        pipe.InitBuffer(inQueueX,   1, height * width * sizeof(T));
        pipe.InitBuffer(outQueueY,  1, height * width * sizeof(T));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<T> xLocal = inQueueX.AllocTensor<T>();
        AscendC::DataCopy(xLocal, xGlobal, height * width);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<T> xLocal = inQueueX.DeQue<T>();
        AscendC::LocalTensor<T> yLocal = outQueueY.AllocTensor<T>();

        AscendC::SoftMaxShapeInfo srcShape = { height, width, height, width };
        // Call SoftMax high-level API with tiling info
        // Prototype referenced by SoftMax tiling docs flow.
        AscendC::SoftMax<T>(yLocal, xLocal, tiling, srcShape);

        outQueueY.EnQue<T>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<T> yLocal = outQueueY.DeQue<T>();
        AscendC::DataCopy(yGlobal, yLocal, height * width);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1>  inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueY;
    AscendC::GlobalTensor<T> xGlobal, yGlobal;
    uint32_t height = 0;
    uint32_t width  = 0;
    SoftMaxTiling   tiling;
};

// Kernel entry. Name must match project JSON and host launcher.
extern "C" __global__ __aicore__ void softmax_custom(__gm__ uint8_t* xGm, __gm__ uint8_t* yGm, __gm__ uint8_t* tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelSoftmax<half> op;
    op.Init(xGm, yGm, tilingData.softmaxTilingData);
    op.Process();
}
"""

# softmax_custom_pybind.py
# Lightweight Python registration to allow calling torch.ops.SoftmaxCustom.softmax_custom
# from Python while relying on backend implementation for NPU execution.
import torch

_lib = torch.library.Library("SoftmaxCustom", "DEF")
_lib.define("softmax_custom(Tensor x) -> Tensor")

# Provide a CPU fallback for functional testing; real execution will be backend-dispatched.
@torch.library.impl("SoftmaxCustom::softmax_custom", "CPU")
def _softmax_custom_cpu(x: torch.Tensor) -> torch.Tensor:
    return torch.softmax(x, dim=1)

# Meta implementation for shape inference
@torch.library.impl("SoftmaxCustom::softmax_custom", "Meta")
def _softmax_custom_meta(x: torch.Tensor) -> torch.Tensor:
    return x.new_empty(x.shape)
