host_tiling_src = r"""// log_softmax_custom_tiling.h
// 说明：基于参考手册的 LogSoftMax Tiling API 实现主机侧 Tiling，计算并下发 tiling 数据。
// 参考：5.3.2.2 LogSoftMax Tiling、5.1.31 More Examples
#ifndef LOG_SOFTMAX_CUSTOM_TILING_H_
#define LOG_SOFTMAX_CUSTOM_TILING_H_

#include <vector>
#include "register/op_def_registry.h"
#include "register/tilingdata_base.h"
#include "tiling/tiling_api.h"

namespace optiling {

// 自定义 TilingData，包含 shape 信息、临时缓冲区大小、类型大小以及 LogSoftMax 的 tiling 结构体
BEGIN_TILING_DATA_DEF(LogSoftmaxCustomTilingData)
TILING_DATA_FIELD_DEF(uint32_t, outter);
TILING_DATA_FIELD_DEF(uint32_t, inner);
TILING_DATA_FIELD_DEF(uint32_t, tmpBufferSize);
TILING_DATA_FIELD_DEF(uint32_t, typeSize);
TILING_DATA_FIELD_DEF(AscendC::LogSoftMaxTiling, logSoftmaxTilingData);
END_TILING_DATA_DEF;
REGISTER_TILING_DATA_CLASS(LogSoftmaxCustom, LogSoftmaxCustomTilingData)

// 主机侧 TilingFunc：将输入 shape 拉平成 [outter, inner]，按最后一维进行 LogSoftMax。
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 获取输入张量 shape
    const gert::StorageShape *src_shape = context->GetInputShape(0);
    auto &storageShape = src_shape->GetStorageShape();

    // 计算 outter = 除最后一维外所有维度元素数乘积，inner = 最后一维大小
    uint32_t dimCnt = storageShape.GetDimNum();
    if (dimCnt == 0) {
        return ge::GRAPH_FAILED;
    }
    uint32_t inner = static_cast<uint32_t>(storageShape.GetDim(dimCnt - 1));
    uint64_t outter64 = 1;
    for (uint32_t i = 0; i + 1 < dimCnt; ++i) {
        outter64 *= static_cast<uint64_t>(storageShape.GetDim(i));
    }
    if (outter64 == 0) {
        return ge::GRAPH_FAILED;
    }
    uint32_t outter = static_cast<uint32_t>(outter64);

    // 获取数据类型 size（字节）
    auto dt = context->GetInputTensor(0)->GetDataType();
    uint32_t typeSize = 0;
    if (dt == ge::DT_FLOAT16) {
        typeSize = 2;
    } else if (dt == ge::DT_FLOAT) {
        typeSize = 4;
    } else {
        // 不支持的数据类型
        return ge::GRAPH_FAILED;
    }

    // 生成用于 LogSoftMax Tiling 的二维 shape [outter, inner]
    std::vector<int64_t> srcDims = {static_cast<int64_t>(outter), static_cast<int64_t>(inner)};
    ge::Shape shape(srcDims);

    // 计算临时缓冲区大小（取最大值，以获得更好性能）
    // 参考：GetLogSoftMaxMaxTmpSize / LogSoftMaxTilingFunc
    const uint32_t tmpsize = AscendC::GetLogSoftMaxMaxTmpSize(shape, typeSize, false);

    LogSoftmaxCustomTilingData tiling;
    tiling.set_outter(outter);
    tiling.set_inner(inner);
    tiling.set_typeSize(typeSize);
    tiling.set_tmpBufferSize(tmpsize);

    // 调用 TilingFunc 生成 tiling 参数，写入 tiling 数据
    AscendC::LogSoftMaxTilingFunc(shape, typeSize, tmpsize, tiling.mutable_logSoftmaxTilingData());

    // 常规设置
    context->SetBlockDim(1);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());
    // 本算子不使用 GM workspace
    size_t *workspaceSizes = context->GetWorkspaceSizes(1);
    workspaceSizes[0] = 0;

    // 设置 TilingKey（可选）
    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}

} // namespace optiling

#endif // LOG_SOFTMAX_CUSTOM_TILING_H_
"""

kernel_src = r"""// log_softmax_custom.cpp
// 说明：AscendC 内核，调用高阶 API LogSoftMax 完成计算。
// 参考：5.3.2.1 LogSoftMax（算法与接口）、5.3.2.2 LogSoftMax Tiling、5.1.31 More Examples
#include "kernel_operator.h"

using namespace AscendC;

template <typename T>
class KernelLogSoftmax {
public:
    __aicore__ inline KernelLogSoftmax() {}
    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR yGm, uint32_t outterIn, uint32_t innerIn, uint32_t tmpBytes)
    {
        outter = outterIn;
        inner = innerIn;
        total = outter * inner;
        srcGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(xGm), total);
        dstGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(yGm), total);

        pipe.InitBuffer(inQueue, 1, total * sizeof(T));
        pipe.InitBuffer(outQueue, 1, total * sizeof(T));
        pipe.InitBuffer(tmpQueue, 1, tmpBytes);

        // max/sum 为按行归约的中间结果，每行一个元素，大小为 outter
        pipe.InitBuffer(maxQueue, 1, outter * sizeof(T));
        pipe.InitBuffer(sumQueue, 1, outter * sizeof(T));
    }

    __aicore__ inline void Process(const LogSoftMaxTiling &softmaxTiling)
    {
        CopyIn();
        Compute(softmaxTiling);
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        LocalTensor<T> srcLocal = inQueue.AllocTensor<T>();
        DataCopy(srcLocal, srcGlobal, total);
        inQueue.EnQue(srcLocal);
    }

    __aicore__ inline void Compute(const LogSoftMaxTiling &softmaxTiling)
    {
        LocalTensor<T> srcLocal = inQueue.DeQue<T>();
        LocalTensor<T> dstLocal = outQueue.AllocTensor<T>();
        LocalTensor<uint8_t> sharedTmpBuffer = tmpQueue.AllocTensor<uint8_t>();
        LocalTensor<T> maxLocal = maxQueue.AllocTensor<T>();
        LocalTensor<T> sumLocal = sumQueue.AllocTensor<T>();

        // 形状信息：按最后一维（inner）进行 LogSoftMax
        SoftMaxShapeInfo softmaxInfo = {outter, inner, outter, inner};

        // 调用高阶 API：LogSoftMax
        // 原型（参考文档示例调用）：LogSoftMax(dst, maxTensor, sumTensor, src, sharedTmpBuffer, tiling, softmaxShapeInfo)
        LogSoftMax<T, false>(dstLocal, maxLocal, sumLocal, srcLocal, sharedTmpBuffer, softmaxTiling, softmaxInfo);

        outQueue.EnQue<T>(dstLocal);
        inQueue.FreeTensor(srcLocal);
        tmpQueue.FreeTensor(sharedTmpBuffer);
        maxQueue.FreeTensor(maxLocal);
        sumQueue.FreeTensor(sumLocal);
    }

    __aicore__ inline void CopyOut()
    {
        LocalTensor<T> dstLocal = outQueue.DeQue<T>();
        DataCopy(dstGlobal, dstLocal, total);
        outQueue.FreeTensor(dstLocal);
    }

private:
    GlobalTensor<T> srcGlobal;
    GlobalTensor<T> dstGlobal;
    TPipe pipe;
    TQue<QuePosition::VECIN, 1> inQueue;
    TQue<QuePosition::VECOUT, 1> outQueue;
    TQue<QuePosition::VECCALC, 1> tmpQueue;
    TQue<QuePosition::VECCALC, 1> maxQueue;
    TQue<QuePosition::VECCALC, 1> sumQueue;
    uint32_t outter = 0;
    uint32_t inner = 0;
    uint32_t total = 0;
};

// 内核入口
extern "C" __global__ __aicore__ void log_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    // 读取 tiling 数据
    GET_TILING_DATA(tilingData, tiling);
    const uint32_t outter = tilingData.outter;
    const uint32_t inner = tilingData.inner;
    const uint32_t tmpSize = tilingData.tmpBufferSize;

    // 根据类型大小选择计算类型
    if (tilingData.typeSize == 2) {
        KernelLogSoftmax<half> op;
        op.Init(x, y, outter, inner, tmpSize);
        op.Process(tilingData.logSoftmaxTilingData);
    } else {
        KernelLogSoftmax<float> op;
        op.Init(x, y, outter, inner, tmpSize);
        op.Process(tilingData.logSoftmaxTilingData);
    }
}
"""
