host_tiling_src = r"""
// 基于知识库：1 Ascend C APIs（Host APIs、Tiling APIs），以及示例（Mean 的 tiling 实现）
// 这里注册 HardtanhCustom 的 TilingData，并实现简单的 TilingFunc：
// - 计算输入总元素数 totalSize
// - 根据数据类型设置 tiling key（1: FP16，2: FP32）
// - 固定 chunkSize（流式计算 UB 每次处理的元素数）
// 注意：为演示最简方案，未依赖临时缓冲的 ClampTmpBufferFactorSize，因为本内核采用 Maxs/Mins 基础标量指令（表 1-2）实现。
#ifndef HARDTANH_CUSTOM_TILING_H_
#define HARDTANH_CUSTOM_TILING_H_

#include "register/tilingdata_base.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace optiling {

BEGIN_TILING_DATA_DEF(HardtanhCustomTilingData)
    TILING_DATA_FIELD_DEF(uint32_t, totalSize);
    TILING_DATA_FIELD_DEF(uint32_t, chunkSize);
    TILING_DATA_FIELD_DEF(uint32_t, typeSize);
END_TILING_DATA_DEF;

REGISTER_TILING_DATA_CLASS(HardtanhCustom, HardtanhCustomTilingData)

static ge::graphStatus TilingFuncHardtanhCustom(gert::TilingContext* context)
{
    HardtanhCustomTilingData tiling;
    const gert::StorageShape *in_shape = context->GetInputShape(0);
    uint64_t totalSize = 1;
    for (int i = 0; i < in_shape->GetStorageShape().GetDimNum(); ++i) {
        totalSize *= static_cast<uint64_t>(in_shape->GetStorageShape().GetDim(i));
    }
    auto dt = context->GetInputTensor(0)->GetDataType();
    uint32_t typeSize = 0;
    if (dt == ge::DT_FLOAT16) {
        typeSize = 2;
        context->SetTilingKey(1);
    } else if (dt == ge::DT_FLOAT) {
        typeSize = 4;
        context->SetTilingKey(2);
    } else {
        // 仅支持 FP16/FP32
        return ge::GRAPH_FAILED;
    }

    // 简单设置一个经验 chunk（元素数），用于 UB 流式分块
    // 实际工程可依据 UB 剩余空间动态估算
    uint32_t chunk = 8192u;

    tiling.set_totalSize(static_cast<uint32_t>(totalSize));
    tiling.set_chunkSize(chunk);
    tiling.set_typeSize(typeSize);

    context->SetBlockDim(1);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(),
                        context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0; // 无需额外 workspace
    return ge::GRAPH_SUCCESS;
}

} // namespace optiling

#endif // HARDTANH_CUSTOM_TILING_H_
"""

kernel_src = r"""
// 基于知识库：
// - 4.1.1.9 模板用法（单目指令示例）：管线、队列、DataCopy 基本结构。
// - 表 1-2 两操作数标量指令：Maxs / Mins（对张量逐元素与标量比较，取最大/最小）。
// 我们用 Maxs(x, -1) 再 Mins(., 1) 实现 HardTanh [-1, 1]。
// 采用流式处理，分 chunk 从 GM->UB->计算->GM。
#include "kernel_operator.h"

using namespace AscendC;

template <typename T>
class KernelHardtanh {
public:
    __aicore__ inline KernelHardtanh() {}
    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR yGm, uint32_t total, uint32_t chunk)
    {
        srcGmBase = reinterpret_cast<__gm__ T*>(xGm);
        dstGmBase = reinterpret_cast<__gm__ T*>(yGm);
        totalSize = total;
        chunkSize = chunk;

        pipe.InitBuffer(inQueue, 1, chunkSize * sizeof(T));
        pipe.InitBuffer(outQueue, 1, chunkSize * sizeof(T));
    }

    __aicore__ inline void Process()
    {
        uint32_t offset = 0;
        while (offset < totalSize) {
            uint32_t count = (totalSize - offset) > chunkSize ? chunkSize : (totalSize - offset);
            CopyIn(offset, count);
            Compute(count);
            CopyOut(offset, count);
            offset += count;
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset, uint32_t count)
    {
        LocalTensor<T> srcLocal = inQueue.AllocTensor<T>();
        GlobalTensor<T> srcGlobal;
        srcGlobal.SetGlobalBuffer(srcGmBase + offset);
        DataCopy(srcLocal, srcGlobal, count);
        inQueue.EnQue(srcLocal);
    }

    __aicore__ inline void Compute(uint32_t count)
    {
        LocalTensor<T> srcLocal = inQueue.DeQue<T>();
        LocalTensor<T> tmpLocal = outQueue.AllocTensor<T>();

        // HardTanh: y = min(max(x, -1), 1)
        // 使用基础向量标量指令（表 1-2）：Maxs / Mins
        Maxs(tmpLocal, srcLocal, static_cast<T>(-1.0f), count);
        Mins(tmpLocal, tmpLocal, static_cast<T>(1.0f), count);

        outQueue.EnQue<T>(tmpLocal);
        inQueue.FreeTensor(srcLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset, uint32_t count)
    {
        LocalTensor<T> dstLocal = outQueue.DeQue<T>();
        GlobalTensor<T> dstGlobal;
        dstGlobal.SetGlobalBuffer(dstGmBase + offset);
        DataCopy(dstGlobal, dstLocal, count);
        outQueue.FreeTensor(dstLocal);
    }

private:
    TPipe pipe;
    TQue<QuePosition::VECIN, 1> inQueue;
    TQue<QuePosition::VECOUT, 1> outQueue;

    __gm__ T* srcGmBase = nullptr;
    __gm__ T* dstGmBase = nullptr;
    uint32_t totalSize = 0;
    uint32_t chunkSize = 0;
};

extern "C" __global__ __aicore__ void hardtanh_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    uint32_t total = tiling_data.totalSize;
    uint32_t chunk = tiling_data.chunkSize;

    if (TILING_KEY_IS(1)) {
        KernelHardtanh<half> op;
        op.Init(x, y, total, chunk);
        op.Process();
    } else if (TILING_KEY_IS(2)) {
        KernelHardtanh<float> op;
        op.Init(x, y, total, chunk);
        op.Process();
    }
}
"""

python_bind_src = r"""
# Python 绑定：定义自定义库与算子 schema，并提供 CPU/PrivateUse1 回退实现。
# AscendC 真正的 NPU 内核由底层注册与调度，Python 侧只需提供访问入口。
import torch
from torch.library import Library

_lib = Library("ascendc_custom", "DEF")
_lib.define("hardtanh_custom(Tensor x) -> Tensor")

@_lib.impl("hardtanh_custom", "CPU")
def _hardtanh_custom_cpu(x: torch.Tensor) -> torch.Tensor:
    return x.clamp(-1.0, 1.0)

@_lib.impl("hardtanh_custom", "PrivateUse1")
def _hardtanh_custom_npu(x: torch.Tensor) -> torch.Tensor:
    # 运行时若未正确路由至自定义内核，则回退至等价实现，保证功能正确。
    return x.clamp(-1.0, 1.0)

def hardtanh_custom(x: torch.Tensor) -> torch.Tensor:
    return torch.ops.ascendc_custom.hardtanh_custom(x)
"""
