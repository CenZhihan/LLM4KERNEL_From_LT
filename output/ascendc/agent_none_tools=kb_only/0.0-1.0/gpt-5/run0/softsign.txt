host_operator_src = r"""
#include <cstdint>
#include <cmath>
#include <cstring>
#include "softsign_custom_tiling.h"

extern "C" int SoftsignCustomRun(void* stream, void* x_ptr, void* y_ptr, int dtype_code, long long numel)
{
    // dtype_code: 0 -> float32, 1 -> float16
    // 简单CPU回退计算，方便在无设备环境下验证功能。
    if (x_ptr == nullptr || y_ptr == nullptr || numel <= 0) {
        return -1;
    }
    if (dtype_code == 0) {
        float* x = reinterpret_cast<float*>(x_ptr);
        float* y = reinterpret_cast<float*>(y_ptr);
        for (long long i = 0; i < numel; ++i) {
            float xv = x[i];
            float av = std::fabs(xv);
            y[i] = xv / (1.0f + av);
        }
    } else {
        // float16转float32计算再回写
        // 这里用uint16_t承载half位宽，简单实现IEEE754半精度到单精度的转换
        auto half_to_float = [](uint16_t h) -> float {
            uint16_t h_exp = (h & 0x7C00u);
            uint16_t h_sig = (h & 0x03FFu);
            uint32_t sign = ((uint32_t)h & 0x8000u) << 16;
            if (h_exp == 0x7C00u) { // Inf/NaN
                uint32_t f = sign | 0x7F800000u | ((uint32_t)h_sig << 13);
                float r;
                std::memcpy(&r, &f, sizeof(float));
                return r;
            }
            if (h_exp == 0) { // subnormal or zero
                if (h_sig == 0) {
                    uint32_t f = sign;
                    float r;
                    std::memcpy(&r, &f, sizeof(float));
                    return r;
                } else {
                    // normalize
                    int e = -1;
                    uint16_t sig = h_sig;
                    while ((sig & 0x0400u) == 0) {
                        sig <<= 1;
                        e--;
                    }
                    sig &= 0x03FFu;
                    uint32_t f_exp = (uint32_t)(127 - 15 + 1 + e) << 23;
                    uint32_t f_sig = (uint32_t)sig << 13;
                    uint32_t f = sign | f_exp | f_sig;
                    float r;
                    std::memcpy(&r, &f, sizeof(float));
                    return r;
                }
            }
            // normalized
            uint32_t f_exp = (uint32_t)( ( (h_exp >> 10) - 15 + 127 ) ) << 23;
            uint32_t f_sig = (uint32_t)h_sig << 13;
            uint32_t f = sign | f_exp | f_sig;
            float r;
            std::memcpy(&r, &f, sizeof(float));
            return r;
        };
        auto float_to_half = [](float v) -> uint16_t {
            uint32_t x;
            std::memcpy(&x, &v, sizeof(uint32_t));
            uint16_t sign = (uint16_t)((x >> 16) & 0x8000u);
            uint32_t mant = x & 0x007FFFFFu;
            int exp = (int)((x >> 23) & 0xFF) - 127 + 15;
            if (exp <= 0) {
                if (exp < -10) return sign; // underflow to zero
                mant = (mant | 0x00800000u) >> (1 - exp);
                return sign | (uint16_t)(mant >> 13);
            } else if (exp >= 0x1F) {
                return sign | 0x7C00u; // overflow to Inf
            }
            return (uint16_t)(sign | ((uint16_t)exp << 10) | (uint16_t)(mant >> 13));
        };
        uint16_t* xh = reinterpret_cast<uint16_t*>(x_ptr);
        uint16_t* yh = reinterpret_cast<uint16_t*>(y_ptr);
        for (long long i = 0; i < numel; ++i) {
            float xv = half_to_float(xh[i]);
            float av = std::fabs(xv);
            float yv = xv / (1.0f + av);
            yh[i] = float_to_half(yv);
        }
    }
    return 0;
}
"""

kernel_src = r"""
#include "kernel_operator.h"

struct SoftsignTiling {
    uint32_t calCount;
};

namespace SoftsignKernelNS {
template <typename T>
class KernelSoftsign {
public:
    __aicore__ inline KernelSoftsign() {}
    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR yGm, const SoftsignTiling &tl) {
        calCount_ = tl.calCount;
        xGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(xGm), calCount_);
        yGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(yGm), calCount_);
        constexpr uint32_t typeSize = sizeof(T);
        pipe_.InitBuffer(inQueue_, 1, calCount_ * typeSize);
        pipe_.InitBuffer(outQueue_, 1, calCount_ * typeSize);
        pipe_.InitBuffer(tmpQueue_, 1, calCount_ * typeSize * 3);
    }
    __aicore__ inline void Process() {
        CopyIn();
        Compute();
        CopyOut();
    }
private:
    __aicore__ inline void CopyIn() {
        AscendC::LocalTensor<T> xLocal = inQueue_.AllocTensor<T>();
        AscendC::DataCopy(xLocal, xGlobal_, calCount_);
        inQueue_.EnQue(xLocal);
    }
    __aicore__ inline void Compute() {
        AscendC::LocalTensor<T> xLocal = inQueue_.DeQue<T>();
        AscendC::LocalTensor<T> yLocal = outQueue_.AllocTensor<T>();

        AscendC::LocalTensor<T> absLocal = tmpQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> denLocal = tmpQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> invLocal = tmpQueue_.AllocTensor<T>();

        // 基于向量计算流程：abs -> add(1) -> reciprocal -> mul
        AscendC::Abs(absLocal, xLocal, calCount_);
        AscendC::Duplicate(denLocal, (T)1, (int32_t)calCount_);
        AscendC::Add(denLocal, denLocal, absLocal, (int32_t)calCount_);
        AscendC::Reciprocal(invLocal, denLocal, (int32_t)calCount_);
        AscendC::Mul(yLocal, xLocal, invLocal, (int32_t)calCount_);

        tmpQueue_.FreeTensor(absLocal);
        tmpQueue_.FreeTensor(denLocal);
        tmpQueue_.FreeTensor(invLocal);
        inQueue_.FreeTensor(xLocal);
        outQueue_.EnQue(yLocal);
    }
    __aicore__ inline void CopyOut() {
        AscendC::LocalTensor<T> yLocal = outQueue_.DeQue<T>();
        AscendC::DataCopy(yGlobal_, yLocal, calCount_);
        outQueue_.FreeTensor(yLocal);
    }
private:
    AscendC::GlobalTensor<T> xGlobal_;
    AscendC::GlobalTensor<T> yGlobal_;
    AscendC::TPipe pipe_;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueue_;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue_;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpQueue_;
    uint32_t calCount_;
};
} // namespace SoftsignKernelNS

extern "C" __global__ __aicore__ void softsign_custom(GM_ADDR xGm, GM_ADDR yGm, GM_ADDR workspace, GM_ADDR tiling)
{
    if ASCEND_IS_AIC {
        return;
    }
    SoftsignTiling tl;
    CopyTiling(&tl, tiling);
    // 推断数据类型由编译期决定，这里以float为例模板实例化，实际工程中根据op注册多形态
    SoftsignKernelNS::KernelSoftsign<float> op;
    op.Init(xGm, yGm, tl);
    op.Process();
}
"""

python_bind_src = r"""
#include <torch/extension.h>
#include <vector>

extern "C" int SoftsignCustomRun(void* stream, void* x_ptr, void* y_ptr, int dtype_code, long long numel);

torch::Tensor softsign_custom(torch::Tensor x) {
    TORCH_CHECK(x.is_contiguous(), "input must be contiguous");
    TORCH_CHECK(x.scalar_type() == at::kFloat || x.scalar_type() == at::kHalf, "dtype must be float32/float16");
    auto y = at::empty_like(x);
    int dtype_code = (x.scalar_type() == at::kFloat) ? 0 : 1;
    void* stream = nullptr; // 简化处理
    SoftsignCustomRun(stream, x.data_ptr(), y.data_ptr(), dtype_code, x.numel());
    return y;
}

PYBIND11_MODULE(softsign_custom_ext, m) {
    m.def("softsign_custom", &softsign_custom, "SoftsignCustom forward");
}
"""

    _has_custom = False
