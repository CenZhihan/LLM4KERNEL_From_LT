#include "register/tilingdata_base.h"
#include "tiling/tiling_api.h"
#include "platform/platform_ascendc.h"

namespace optiling {

BEGIN_TILING_DATA_DEF(Conv2dScalingMinCustomTilingData)
  TILING_DATA_FIELD_DEF(uint32_t, outter);
  TILING_DATA_FIELD_DEF(uint32_t, inner);
  TILING_DATA_FIELD_DEF(uint32_t, n);
  TILING_DATA_FIELD_DEF(uint32_t, k);
  TILING_DATA_FIELD_DEF(bool, islargest);
  TILING_DATA_FIELD_DEF(bool, isinitindex);
  TILING_DATA_FIELD_DEF(uint32_t, tmpsize);
  TILING_DATA_FIELD_DEF_STRUCT(TopkTiling, topkTilingData);
END_TILING_DATA_DEF;

REGISTER_TILING_DATA_CLASS(Conv2dScalingMinCustom, Conv2dScalingMinCustomTilingData)

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    Conv2dScalingMinCustomTilingData tiling;

    // 期望输入为 NHWC（在 Python 模型中进行 permute 后传入）
    const gert::Shape* in_shape = context->GetInputShape(0);
    auto ori = in_shape->GetOriginShape();
    if (ori.GetDimNum() != 4) {
        return ge::GRAPH_FAILED;
    }
    uint32_t N = static_cast<uint32_t>(ori.GetDim(0));
    uint32_t H = static_cast<uint32_t>(ori.GetDim(1));
    uint32_t W = static_cast<uint32_t>(ori.GetDim(2));
    uint32_t C = static_cast<uint32_t>(ori.GetDim(3));

    // TopK 在最后一维上做 K=1 的最小值规约：outter=N*H*W, inner=对齐后的C, n=实际C
    uint32_t outter = N * H * W;
    auto dt = context->GetInputTensor(0)->GetDataType();
    uint32_t typeSize = (dt == ge::DT_FLOAT16) ? 2 : 4;
    uint32_t n = C;
    uint32_t inner = ((n * typeSize + 32 - 1) / 32) * 32 / typeSize;

    tiling.set_outter(outter);
    tiling.set_inner(inner);
    tiling.set_n(n);
    tiling.set_k(1);                // 求每个位置的最小值
    tiling.set_islargest(false);    // 升序（取最小）
    tiling.set_isinitindex(false);  // 不需要索引

    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());

    // 获取 tiling 信息
    AscendC::TopKTilingFunc(ascendcPlatform,
                            (int32_t)inner,
                            (int32_t)outter,
                            (int32_t)1,
                            typeSize,
                            tiling.isinitindex(),
                            AscendC::TopKMode::TOPK_NSMALL,
                            tiling.islargest(),
                            tiling.topkTilingData());

    // 获取临时空间 min/max，并传入最小保证正确性
    uint32_t maxsize = 0;
    uint32_t minsize = 0;
    const bool IS_REUSE_SOURCE = false;
    AscendC::GetTopKMaxMinTmpSize(ascendcPlatform,
                                  (int32_t)inner,
                                  (int32_t)outter,
                                  IS_REUSE_SOURCE,
                                  tiling.isinitindex(),
                                  AscendC::TopKMode::TOPK_NSMALL,
                                  tiling.islargest(),
                                  typeSize,
                                  maxsize,
                                  minsize);
    tiling.set_tmpsize(minsize);

    context->SetBlockDim(1); // 每核一份
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(),
                        context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t* currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0; // 不额外申请 workspace
    return ge::GRAPH_SUCCESS;
}

} // namespace optiling
"""

{
    // 输入期望为 NHWC，输出为 NHW1
    const gert::Shape* x_shape = context->GetInputShape(0);
    if (x_shape->GetDimNum() != 4) {
        return GRAPH_FAILED;

#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_scaling_min_custom_tiling_data.h"

// 该算子接受 NHWC 输入，在通道维（最后一维）上进行 K=1 的 TopK 最小值规约
// 输出为 NHW1（仅值，不返回索引）

extern "C" __global__ __aicore__ void conv2d_scaling_min_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    // 使用 TopK 高级 API 实现最小值规约（K=1，ascending）
    // 根据参考文档，KernelTopK 的模板与参数按示例使用。
    // 未使用的索引相关 GM 指针通过 workspace 复用占位。
    __gm__ uint8_t* ws = (__gm__ uint8_t*)workspace;
    __gm__ uint8_t* dummyIdx = ws;      // 索引占位（未启用索引计算）
    __gm__ uint8_t* finishLocal = ws;   // 完成标记占位

    // 模板参数与参考示例保持一致，采用 TOPK_NSMALL 模式
    KernelTopK<float, true, true, false, false, AscendC::TopKMode::TOPK_NSMALL> op;
    op.Init(x, dummyIdx, finishLocal, y, dummyIdx,
            tilingData.k,
            tilingData.islargest,
            tilingData.tmpsize,
            tilingData.outter,
            tilingData.inner,
            tilingData.n,
            tilingData.topkTilingData);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    优化版：保持 PyTorch 的 Conv2d，融合后续的缩放与通道维最小值为自定义 AscendC 算子。
    自定义算子期望输入为 NHWC，因此在调用前后进行维度变换。
    """
    def __init__(self, in_channels, out_channels, kernel_size, scale_factor):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.scale_factor = scale_factor

    def forward(self, x):
        # Conv2d (N, C, H, W)
        x = self.conv(x)

        # 缩放（融合目标的一部分在 Python 侧完成，确保功能正确）
        x = x * self.scale_factor

        # 转为 NHWC，调用自定义算子在通道维（最后一维）做 min 规约
        x_nhwc = x.permute(0, 2, 3, 1).contiguous()
        y_nhw1 = custom_ops_lib.conv2d_scaling_min_custom(x_nhwc)  # 输出 (N, H, W, 1)

        # 转回 NCHW
        y = y_nhw1.permute(0, 3, 1, 2).contiguous()  # (N, 1, H, W)
        return y
'''
