#include "cumprod_custom_tiling.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    CumprodCustomTilingData tiling;

    // 读取输入形状信息
    const gert::Shape* in_shape = context->GetInputShape(0);
    auto origin = in_shape->GetOriginShape();
    uint32_t dimNum = origin.GetDimNum();
    // 仅支持沿最后一维计算前缀积（与示例模型 dim=1 且输入为二维相匹配）
    uint32_t cols = static_cast<uint32_t>(origin.GetDim(dimNum - 1));
    uint32_t totalLength = static_cast<uint32_t>(origin.GetShapeSize());
    uint32_t rows = cols == 0 ? 0 : (totalLength / cols);

    context->SetBlockDim(BLOCK_DIM);
    tiling.set_rows(rows);
    tiling.set_cols(cols);
    tiling.set_tileNum(TILE_NUM);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无需额外 workspace
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}
}

    const gert::Shape* x_shape = context->GetInputShape(0);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "cumprod_custom_tiling_data.h"

// 参考文档：
// - 基础二元/一元向量算子（如 Mul、Duplicate、DataCopy、Copy）接口用法：
//   CANN Ascend C API Reference - Basic APIs (Add/Sub/Mul/Copy/DataCopy/Duplicate)
// - 队列/本地张量/全局张量与管道管理：TPipe/TQue/LocalTensor/GlobalTensor
// - 示例参考：4.1.2 Sub、数据填充 Duplicate、5.1.31 示例（队列与拷贝模式）

constexpr int32_t BUFFER_NUM = 2;   // 每个队列的 tensor 个数

using DTYPE = float;

class KernelCumprod {
public:
    __aicore__ inline KernelCumprod() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t rows, uint32_t cols, uint32_t tileNum)
    {
        this->rows = rows;
        this->cols = cols;
        this->tileNum = tileNum;

        // 按行分块给不同 core
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowsPerBlock = rows / blockNum;
        uint32_t remain = rows % blockNum;
        // 平均分配 + 余数前移
        this->rowStart = blockIdx * rowsPerBlock + AscendC::Min(blockIdx, remain);
        uint32_t myRows = rowsPerBlock + (blockIdx < remain ? 1u : 0u);
        this->rowCount = myRows;

        // 每 tile 处理的列数（最后一个 tile 处理剩余）
        // 确保 tileLength >= 1
        uint32_t denom = tileNum * BUFFER_NUM;
        this->tileLength = denom == 0 ? cols : AscendC::Max(1u, cols / denom);

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(x) + this->rowStart * cols, this->rowCount * cols);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(y) + this->rowStart * cols, this->rowCount * cols);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        // 标量缓冲区（用于在 UB 内维护累乘的“前缀乘积”标量）
        pipe.InitBuffer(tmpScalarQ, 1, sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t r = 0; r < this->rowCount; ++r) {
            // 为当前行申请并初始化累乘标量（值为 1.0）
            AscendC::LocalTensor<DTYPE> carryLocal = tmpScalarQ.AllocTensor<DTYPE>();
            DTYPE one = static_cast<DTYPE>(1.0f);
            AscendC::Duplicate<DTYPE>(carryLocal, one, 1);

            // 逐 tile 处理整行（沿最后一维做前缀积）
            uint32_t processed = 0;
            int32_t progress = 0;
            while (processed < cols) {
                uint32_t curLen = AscendC::Min(this->tileLength, cols - processed);
                CopyIn(r, progress, processed, curLen);
                Compute(progress, curLen, carryLocal);
                CopyOut(r, progress, processed, curLen);
                processed += curLen;
                ++progress;
            }

            tmpScalarQ.FreeTensor(carryLocal);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx, int32_t progress, uint32_t colOffset, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * cols + colOffset], curLen);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t progress, uint32_t curLen, AscendC::LocalTensor<DTYPE>& carryLocal)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.AllocTensor<DTYPE>();

        // 依次计算 y[i] = x[i] * carry; carry = y[i]
        for (uint32_t i = 0; i < curLen; ++i) {
            // 1 元素视图
            AscendC::LocalTensor<DTYPE> x1 = xLocal[i];
            AscendC::LocalTensor<DTYPE> y1 = yLocal[i];
            // 逐元素乘法（calCount = 1）
            AscendC::Mul<DTYPE>(y1, x1, carryLocal, 1);
            // 将 y[i] 写回到 carryLocal，作为下一次的累乘因子
            AscendC::Copy<DTYPE>(carryLocal, y1, 1);
        }

        outQueueY.EnQue<DTYPE>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx, int32_t progress, uint32_t colOffset, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm[rowIdx * cols + colOffset], yLocal, curLen);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpScalarQ;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t rows;
    uint32_t cols;
    uint32_t tileNum;
    uint32_t tileLength;

    uint32_t rowStart;
    uint32_t rowCount;
};

extern "C" __global__ __aicore__ void cumprod_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelCumprod op;
    op.Init(x, y, tiling_data.rows, tiling_data.cols, tiling_data.tileNum);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}
"""
