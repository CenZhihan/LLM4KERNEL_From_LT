kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_gelu_global_avg_pool_custom_tiling_data.h"

// 本算子对输入特征图先执行GELU，再做全局平均池化(对H*W求均值)，输出形状[N, C]。
// 参考文档：Gelu 高级API (Ascend C API Reference 5.3.3.1)
// 参考文档：ReduceSum/Scalar/Muls 基础API (Ascend C API Reference Table 1-2)

#ifndef DTYPE_X
#define DTYPE_X float
#endif
#ifndef DTYPE_Y
#define DTYPE_Y float
#endif

constexpr int32_t BUFFER_NUM = 1;

class KernelGeluGap {
public:
    __aicore__ inline KernelGeluGap() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w, uint32_t hw)
    {
        this->n = n;
        this->c = c;
        this->h = h;
        this->w = w;
        this->hw = hw;
        this->groups = n * c;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, (uint64_t)n * c * h * w);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, (uint64_t)n * c);

        // 每次处理一个(N,C)组的H*W平面
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->hw * sizeof(DTYPE_X));
        pipe.InitBuffer(tmpQueue, BUFFER_NUM, this->hw * sizeof(DTYPE_X)); // 存放GELU后的中间结果
        pipe.InitBuffer(outQueueY, BUFFER_NUM, sizeof(DTYPE_Y));           // 存放均值标量
    }

    __aicore__ inline void Process()
    {
        for (uint32_t g = 0; g < groups; ++g) {
            CopyIn(g);
            Compute(g);
            CopyOut(g);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t groupIdx)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        uint64_t offset = (uint64_t)groupIdx * hw;
        AscendC::DataCopy(xLocal, xGm[offset], hw);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*groupIdx*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> geluLocal = tmpQueue.AllocTensor<DTYPE_X>();
        // GELU
        AscendC::Gelu(geluLocal, xLocal, hw);

        // 对H*W做ReduceSum得到一个标量，再除以(hw)得到均值
        AscendC::LocalTensor<DTYPE_Y> meanLocal = outQueueY.AllocTensor<DTYPE_Y>();
        // 将输出标量缓冲区清零以便累加（可选）
        // 这里直接调用ReduceSum，输出到单元素Tensor
        AscendC::ReduceSum(meanLocal, geluLocal, hw);
        DTYPE_Y inv = static_cast<DTYPE_Y>(1.0f) / static_cast<DTYPE_Y>(hw);
        AscendC::Muls(meanLocal, meanLocal, inv); // mean = sum / (H*W)

        outQueueY.EnQue(meanLocal);
        tmpQueue.FreeTensor(geluLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t groupIdx)
    {
        AscendC::LocalTensor<DTYPE_Y> meanLocal = outQueueY.DeQue<DTYPE_Y>();
        uint64_t yoff = (uint64_t)groupIdx;
        AscendC::DataCopy(yGm[yoff], meanLocal, 1);
        outQueueY.FreeTensor(meanLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> tmpQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t n;
    uint32_t c;
    uint32_t h;
    uint32_t w;
    uint32_t hw;
    uint32_t groups;
};

extern "C" __global__ __aicore__ void conv2d_gelu_global_avg_pool_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelGeluGap op;
    op.Init(x, y, tilingData.n, tilingData.c, tilingData.h, tilingData.w, tilingData.hw);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Conv2d -> GELU -> GlobalAvgPool(2D) fused with custom AscendC kernel (activation+pool).
    """
    def __init__(self, in_channels, out_channels, kernel_size):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)

    def forward(self, x):
        # 保留Conv2d由框架执行，后续GELU+GAP使用自定义算子
        x = self.conv(x)
        y = custom_ops_lib.conv2d_gelu_global_avg_pool_custom(x)
        return y
