#include "conv_standard_1d_dilated_strided_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace optiling {

static inline int64_t calc_out_len_1d(int64_t inLen, int64_t kernel, int64_t stride, int64_t dilation)
{
    // padding = 0
    // outLen = floor((inLen - dilation*(kernel - 1) - 1)/stride + 1)
    const int64_t eff_k = dilation * (kernel - 1) + 1;
    return (inLen - eff_k) / stride + 1;
}

const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM  = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    ConvStandard1dDilatedStridedCustomTilingData tiling;

    // inputs: x: [N, Cin, Lin], w: [Cout, Cin, K]
    const gert::Shape* xShape = context->GetInputShape(0);
    const gert::Shape* wShape = context->GetInputShape(1);

    int64_t N  = xShape->GetOriginShape().GetDim(0);
    int64_t Ci = xShape->GetOriginShape().GetDim(1);
    int64_t Lin = xShape->GetOriginShape().GetDim(2);

    int64_t Co = wShape->GetOriginShape().GetDim(0);
    int64_t CiW = wShape->GetOriginShape().GetDim(1);
    int64_t K  = wShape->GetOriginShape().GetDim(2);

    if (CiW != Ci) {
        return ge::GRAPH_FAILED;
    }

    const gert::RuntimeAttrs* attrs = context->GetAttrs();
    const uint32_t stride   = *(attrs->GetAttrPointer<uint32_t>(0));
    const uint32_t dilation = *(attrs->GetAttrPointer<uint32_t>(1));

    int64_t Lout = calc_out_len_1d(Lin, K, stride, dilation);
    if (Lout <= 0) {
        return ge::GRAPH_FAILED;
    }

    context->SetBlockDim(BLOCK_DIM);

    tiling.set_batch(static_cast<uint32_t>(N));
    tiling.set_inC(static_cast<uint32_t>(Ci));
    tiling.set_outC(static_cast<uint32_t>(Co));
    tiling.set_inLen(static_cast<uint32_t>(Lin));
    tiling.set_outLen(static_cast<uint32_t>(Lout));
    tiling.set_kernel(static_cast<uint32_t>(K));
    tiling.set_stride(stride);
    tiling.set_dilation(dilation);
    tiling.set_tileNum(TILE_NUM);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t* currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}
}

    int64_t Ci = xShape->GetOriginShape().GetDim(1);
    int64_t Lin = xShape->GetOriginShape().GetDim(2);

    int64_t Co = wShape->GetOriginShape().GetDim(0);

    int64_t K  = wShape->GetOriginShape().GetDim(2);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_standard_1d_dilated_strided_custom_tiling_data.h"

// 基于参考样例（如 Xor 与 Sum 的 TPipe/DataCopy 模式），此处给出按输出长度维分块的实现。
// 为简化说明与对齐样例风格，这里直接从GM读写完成计算。

class KernelConv1d {
public:
    __aicore__ inline KernelConv1d() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y,
                                uint32_t batch, uint32_t inC, uint32_t outC,
                                uint32_t inLen, uint32_t outLen,
                                uint32_t kernel, uint32_t stride, uint32_t dilation,
                                uint32_t tileNum)
    {
        this->N = batch;
        this->Ci = inC;
        this->Co = outC;
        this->Lin = inLen;
        this->Lout = outLen;
        this->K = kernel;
        this->S = stride;
        this->D = dilation;
        this->tileNum = tileNum;

        xGmPtr = reinterpret_cast<__gm__ DTYPE_X *>(x);
        wGmPtr = reinterpret_cast<__gm__ DTYPE_Y *>(w);
        yGmPtr = reinterpret_cast<__gm__ DTYPE_Z *>(y);

        // 按输出长度维在多核间均分
        this->blockOutLen = Lout / AscendC::GetBlockNum();
        this->blockStart = blockOutLen * AscendC::GetBlockIdx();
        this->tileLen = blockOutLen / tileNum;
        if (this->tileLen == 0) {
            this->tileLen = blockOutLen; // 兜底
        }
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = (blockOutLen + tileLen - 1) / tileLen;
        for (int32_t i = 0; i < loopCount; ++i) {
            ComputeTile(i);
        }
    }

private:
    __aicore__ inline void ComputeTile(int32_t tileIdx)
    {
        uint32_t tStart = blockStart + tileIdx * tileLen;
        if (tStart >= Lout) return;
        uint32_t curLen = (tStart + tileLen <= blockStart + blockOutLen) ? tileLen : (blockStart + blockOutLen - tStart);
        if (tStart + curLen > Lout) {
            curLen = Lout - tStart;
        }

        for (uint32_t n = 0; n < N; ++n) {
            for (uint32_t oc = 0; oc < Co; ++oc) {
                for (uint32_t p = 0; p < curLen; ++p) {
                    uint32_t outPos = tStart + p;
                    float acc = 0.f;
                    for (uint32_t ic = 0; ic < Ci; ++ic) {
                        // 输入窗口起点（无padding）
                        uint32_t inBase = n * (Ci * Lin) + ic * Lin;
                        uint32_t wBase = oc * (Ci * K) + ic * K;
                        for (uint32_t k = 0; k < K; ++k) {
                            uint32_t inPos = outPos * S + k * D;
                            float xv = static_cast<float>(xGmPtr[inBase + inPos]);
                            float ww = static_cast<float>(wGmPtr[wBase + k]);
                            acc += xv * ww;
                        }
                    }
                    uint32_t outIndex = n * (Co * Lout) + oc * Lout + outPos;
                    yGmPtr[outIndex] = static_cast<DTYPE_Z>(acc);
                }
            }
        }
    }

private:
    __gm__ DTYPE_X* xGmPtr;
    __gm__ DTYPE_Y* wGmPtr;
    __gm__ DTYPE_Z* yGmPtr;
    uint32_t N, Ci, Co, Lin, Lout, K, S, D;
    uint32_t tileNum;
    uint32_t blockOutLen;
    uint32_t blockStart;
    uint32_t tileLen;
};

extern "C" __global__ __aicore__ void conv_standard_1d_dilated_strided_custom(GM_ADDR x, GM_ADDR w, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv1d op;
    op.Init(x, w, y,
            tiling_data.batch, tiling_data.inC, tiling_data.outC,
            tiling_data.inLen, tiling_data.outLen,
            tiling_data.kernel, tiling_data.stride, tiling_data.dilation,
            tiling_data.tileNum);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor conv_standard_1d_dilated_strided_custom_impl_npu(const at::Tensor& x,
                                                            const at::Tensor& w,
                                                            int64_t stride,
                                                            int64_t dilation) {
    auto N = x.size(0);
    auto Ci = x.size(1);
    auto Lin = x.size(2);
    auto Co = w.size(0);
    auto K  = w.size(2);

    // outLen = floor((Lin - dilation*(K-1) - 1)/stride + 1)

    auto Lout = (Lin - eff_k) / stride + 1;

    EXEC_NPU_CMD(aclnnConvStandard1dDilatedStridedCustom, x, w, y, (int)stride, (int)dilation);
    return y;

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, stride: int = 1, dilation: int = 1, bias: bool = False) -> None:
        super().__init__()
        # 保留权重形状与PyTorch一致；bias此处不参与计算（自定义核当前未实现bias）
        self.conv1d = torch.nn.Conv1d(in_channels, out_channels, kernel_size, stride=stride, dilation=dilation, bias=bias)
        self.stride = stride
        self.dilation = dilation

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # 使用自定义AscendC算子，权重从self.conv1d中读取
        return custom_ops_lib.conv_standard_1d_dilated_strided_custom(x, self.conv1d.weight, self.stride, self.dilation)
