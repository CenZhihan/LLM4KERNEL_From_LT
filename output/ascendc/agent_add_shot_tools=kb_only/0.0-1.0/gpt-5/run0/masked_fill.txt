host_operator_src="""
#include "masked_fill_custom_tiling.h"
#include "register/op_def_registry.h"
#include <limits>

namespace optiling {
const uint32_t BLOCK_DIM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    MaskedFillCustomTilingData tiling;

    const gert::Shape& x_shape = context->GetInputShape(0)->GetOriginShape();
    const gert::Shape& m_shape = context->GetInputShape(1)->GetOriginShape();

    uint32_t dim_num = x_shape.GetDimNum();
    if (dim_num == 0) {
        return ge::GRAPH_PARAM_INVALID;
    }
    uint64_t total_elems = x_shape.GetShapeSize();
    uint32_t srcLastAxis = static_cast<uint32_t>(x_shape.GetDim(dim_num - 1));
    uint32_t firstAxis = static_cast<uint32_t>(total_elems / srcLastAxis);

    uint32_t maskLastAxis = static_cast<uint32_t>(m_shape.GetDim(m_shape.GetDimNum() - 1));

    // scalar is -inf for masked_fill
    tiling.set_scalarValue(-std::numeric_limits<float>::infinity());
    tiling.set_firstAxis(firstAxis);
    tiling.set_srcSize(srcLastAxis);
    tiling.set_maskSize(maskLastAxis);

    // Temporary buffer size (bytes). For safety, over-provision to 4 * firstAxis * srcLastAxis.
    uint64_t tmpBytes = static_cast<uint64_t>(firstAxis) * static_cast<uint64_t>(srcLastAxis) * 4u;
    tiling.set_tmpSize(static_cast<uint32_t>(tmpBytes));

    context->SetBlockDim(BLOCK_DIM);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *workspace = context->GetWorkspaceSizes(1);
    workspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto x_dtype = context->GetInputDataType(0);
    context->SetOutputDataType(0, x_dtype);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class MaskedFillCustom : public OpDef {
public:
    explicit MaskedFillCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("mask")
            .ParamType(REQUIRED)
            .DataType({ge::DT_BOOL})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);

        this->AICore()

        this->AICore().AddConfig("ascend910b");
    }

OP_ADD(MaskedFillCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "masked_fill_custom_tiling_data.h"

class KernelMaskedFill {
public:
    __aicore__ inline KernelMaskedFill() {}
    __aicore__ inline void Init(GM_ADDR xGmAddr, GM_ADDR maskGmAddr, GM_ADDR yGmAddr,
                                float scalarValue, uint32_t firstAxis, uint32_t srcLastAxis,
                                uint32_t maskLastAxis, uint32_t tmpSizeBytes)
    {
        info.firstAxis = firstAxis;
        info.srcLastAxis = srcLastAxis;
        info.maskLastAxis = maskLastAxis;
        scalar = scalarValue;

        uint32_t srcSize = firstAxis * srcLastAxis;   // element count
        uint32_t maskSize = firstAxis * maskLastAxis; // element count

        xGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(xGmAddr), srcSize);
        maskGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ bool *>(maskGmAddr), maskSize);
        yGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(yGmAddr), srcSize);

        pipe.InitBuffer(inQueueX, 1, srcSize * sizeof(float));
        pipe.InitBuffer(inQueueMask, 1, maskSize * sizeof(bool));
        pipe.InitBuffer(tmpQueue, 1, tmpSizeBytes);

        bufferSize = srcSize;
        maskBufferSize = maskSize;
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGlobal, bufferSize);
        inQueueX.EnQue(xLocal);

        AscendC::LocalTensor<bool> maskLocal = inQueueMask.AllocTensor<bool>();
        AscendC::DataCopy(maskLocal, maskGlobal, maskBufferSize);
        inQueueMask.EnQue(maskLocal);
    }

    __aicore__ inline void Compute()
    {
        xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<bool> maskLocal = inQueueMask.DeQue<bool>();
        AscendC::LocalTensor<uint8_t> tmpLocal = tmpQueue.AllocTensor<uint8_t>();

        // When mask == 0 -> pick tensor; otherwise pick scalar
        AscendC::SelectWithBytesMask(xLocal, xLocal, scalar, maskLocal, tmpLocal, info);

        inQueueMask.FreeTensor(maskLocal);
        tmpQueue.FreeTensor(tmpLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::DataCopy(yGlobal, xLocal, bufferSize);
        inQueueX.FreeTensor(xLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueMask;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> tmpQueue;

    AscendC::GlobalTensor<float> xGlobal;
    AscendC::GlobalTensor<bool>  maskGlobal;
    AscendC::GlobalTensor<float> yGlobal;

    AscendC::LocalTensor<float> xLocal;

    AscendC::SelectWithBytesMaskShapeInfo info;
    uint32_t bufferSize = 0;
    uint32_t maskBufferSize = 0;
    float scalar = 0.0f;
};

extern "C" __global__ __aicore__ void masked_fill_custom(GM_ADDR x, GM_ADDR mask, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelMaskedFill op;
    op.Init(x, mask, y,
            tilingData.scalarValue(),
            tilingData.firstAxis(),
            tilingData.srcSize(),
            tilingData.maskSize(),
            tilingData.tmpSize());
    op.Process();
}
