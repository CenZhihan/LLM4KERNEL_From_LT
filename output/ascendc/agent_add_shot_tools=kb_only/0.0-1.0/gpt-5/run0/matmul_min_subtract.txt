inline uint32_t GetDimSafe(const gert::Shape* shape, size_t idx) {
    if (shape->GetOriginShape().GetDimNum() > idx) {
        return static_cast<uint32_t>(shape->GetOriginShape().GetDim(idx));
    }
    return 1;
}

namespace optiling {

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());

    // 读取输入形状
    const gert::Shape* xShape = context->GetInputShape(0);      // [M, K]
    const gert::Shape* wShape = context->GetInputShape(1);      // [N, K]
    uint32_t M = GetDimSafe(xShape, 0);
    uint32_t K = GetDimSafe(xShape, 1);
    uint32_t N = GetDimSafe(wShape, 0);

    // 配置Matmul多核tiling
    matmul_tiling::MultiCoreMatmulTiling mmTiling(ascendcPlatform);
    mmTiling.SetDim(1);
    // A: float16, B: float16, C: float32, Bias: float32
    mmTiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT16);
    mmTiling.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT16);
    mmTiling.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    mmTiling.SetBiasType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    mmTiling.SetBias(true);
    mmTiling.SetShape(M, N, K);
    mmTiling.SetSingleShape(M, N, K);
    mmTiling.SetOrgShape(M, N, K);
    mmTiling.SetBufferSpace(-1, -1, -1);

    optiling::TCubeTiling cubeTiling;
    int ret = mmTiling.GetTiling(cubeTiling);
    if (ret == -1) {
        return ge::GRAPH_FAILED;
    }

    // 设置blockDim为tiling计算得到的核心数
    context->SetBlockDim(mmTiling.GetCoreNum());

    // 组织自定义TilingData
    MatmulMinSubtractCustomTilingData td;
    td.set_M(M);
    td.set_N(N);
    td.set_K(K);
    td.set_tileNum(8); // 简单设定一个Tile数量
    td.set_cubeTiling(cubeTiling);

    // 将tiling写入缓冲
    td.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(td.GetDataSize());

    // 工作空间（本算子不需额外workspace）
    size_t* ws = context->GetWorkspaceSizes(1);
    ws[0] = 0;

    return ge::GRAPH_SUCCESS;
}

{

    const gert::Shape* wShape = context->GetInputShape(1);

    gert::Shape* yShape = context->GetOutputShape(0);

    origin.SetDimNum(2);
    origin.SetDim(0, xShape->GetOriginShape().GetDim(0));
    origin.SetDim(1, wShape->GetOriginShape().GetDim(0));

}

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"
#define __NPU_TILING__
#include "matmul_min_subtract_custom_tiling_data.h"

using A_T = half;   // 输入A/权重B采用FP16以配合Matmul高性能路径
using B_T = half;
using C_T = float;  // C与Bias使用FP32

constexpr int32_t BUFFER_NUM = 2; // elementwise阶段的双缓冲

class KernelMatmulMinSubtract {
public:
    __aicore__ inline KernelMatmulMinSubtract() {}

    __aicore__ inline void Init(GM_ADDR aGM, GM_ADDR wGM, GM_ADDR bGM, GM_ADDR constGM,
                                GM_ADDR yGM, GM_ADDR workspaceGM,
                                const optiling::MatmulMinSubtractCustomTilingData& td)
    {
        this->M = td.M;
        this->N = td.N;
        this->K = td.K;
        this->tileNum = td.tileNum;
        this->cubeTiling = td.cubeTiling;

        // GM张量绑定
        aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(aGM), this->cubeTiling.ALayoutInfoB * this->cubeTiling.singleCoreM * this->cubeTiling.singleCoreK);
        bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(wGM), this->cubeTiling.BLayoutInfoB * this->cubeTiling.singleCoreK * this->cubeTiling.singleCoreN);
        cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(yGM), this->cubeTiling.CLayoutInfoB * this->cubeTiling.singleCoreM * this->cubeTiling.singleCoreN);
        biasGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(bGM), this->cubeTiling.CLayoutInfoB * this->cubeTiling.singleCoreN);
        constGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(constGM), 1);

        // elementwise阶段(对C做min和减常数)的分块
        this->totalLength = this->M * this->N;
        this->blockLength = this->totalLength / AscendC::GetBlockNum();
        if (this->blockLength == 0) {
            this->blockLength = this->totalLength; // 兜底
        }
        this->tileLength = this->blockLength / this->tileNum / BUFFER_NUM;
        if (this->tileLength == 0) {
            this->tileLength = this->blockLength; // 兜底
        }

        yGm.SetGlobalBuffer((__gm__ C_T*)yGM + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        // 初始化pipe与队列
        pipe.InitBuffer(inQueueC, BUFFER_NUM, this->tileLength * sizeof(C_T));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(C_T));
    }

    __aicore__ inline void Process()
    {
        // 1) Matmul + Bias
        DoMatmul();

        // 2) Elementwise: y = min(y, const) - const
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        C_T cVal = *(constGlobal.GetPhyAddr()); // 读取常数（示例写法）

        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i, cVal);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void DoMatmul()
    {
        // MatmulType定义（NORMAL布局）
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::NORMAL> aType;
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, B_T, true,  LayoutMode::NORMAL> bType; // B转置以适配Linear的weight形状[N, K]
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::NORMAL> cType;
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T> biasType;

        matmul::Matmul<aType, bType, cType, biasType> mm1;
        REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm1);

        // 复制TCubeTiling
        TCubeTiling tiling = this->cubeTiling;

        // 绑定GM
        AscendC::GlobalTensor<A_T> gm_a;
        AscendC::GlobalTensor<B_T> gm_b;
        AscendC::GlobalTensor<C_T> gm_c;
        AscendC::GlobalTensor<C_T> gm_bias;
        gm_a.SetGlobalBuffer(const_cast<__gm__ A_T*>(aGlobal[0].GetPhyAddr()), tiling.singleCoreM * tiling.singleCoreK);
        gm_b.SetGlobalBuffer(const_cast<__gm__ B_T*>(bGlobal[0].GetPhyAddr()), tiling.singleCoreK * tiling.singleCoreN);
        gm_c.SetGlobalBuffer(const_cast<__gm__ C_T*>(cGlobal[0].GetPhyAddr()), tiling.singleCoreM * tiling.singleCoreN);
        gm_bias.SetGlobalBuffer(const_cast<__gm__ C_T*>(biasGlobal[0].GetPhyAddr()), tiling.singleCoreN);

        // 初始化与执行
        mm1.Init(&tiling);
        mm1.SetTensorA(gm_a, false);
        mm1.SetTensorB(gm_b, true);   // 对weight进行转置以实现 A(M,K) x W^T(K,N)
        mm1.SetWorkspace(nullptr, 0);
        if (tiling.isBias) {
            mm1.SetBias(gm_bias);
        }
        mm1.IterateBatch(gm_c, 1, 1, false); // 单次BMM
    }

    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<C_T> cLocal = inQueueC.AllocTensor<C_T>();
        AscendC::DataCopy(cLocal, yGm[progress * this->tileLength], this->tileLength);
        inQueueC.EnQue(cLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/, C_T cVal)
    {
        AscendC::LocalTensor<C_T> cLocal = inQueueC.DeQue<C_T>();
        AscendC::LocalTensor<C_T> yLocal = outQueueY.AllocTensor<C_T>();
        AscendC::LocalTensor<C_T> kLocal = outQueueY.AllocTensor<C_T>(); // 复用队列缓存存放常数向量
        AscendC::Dup(kLocal, cVal, this->tileLength);
        AscendC::Min(yLocal, cLocal, kLocal, this->tileLength);
        AscendC::Sub(yLocal, yLocal, kLocal, this->tileLength);
        outQueueY.EnQue(yLocal);
        outQueueY.FreeTensor(kLocal);
        inQueueC.FreeTensor(cLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<C_T> yLocal = outQueueY.DeQue<C_T>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    // GM tensors for Matmul
    AscendC::GlobalTensor<A_T> aGlobal;
    AscendC::GlobalTensor<B_T> bGlobal;
    AscendC::GlobalTensor<C_T> cGlobal;
    AscendC::GlobalTensor<C_T> biasGlobal;
    AscendC::GlobalTensor<C_T> constGlobal;

    // Elementwise queues
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueC;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<C_T> yGm;

    // Shapes and tiling
    uint32_t M;
    uint32_t N;
    uint32_t K;
    uint32_t tileNum;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tileLength;
    TCubeTiling cubeTiling;
};

extern "C" __global__ __aicore__ void matmul_min_subtract_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR constant, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(td, tiling);
    KernelMatmulMinSubtract op;
    op.Init(x, weight, bias, constant, y, workspace, td);
    op.Process();
}
