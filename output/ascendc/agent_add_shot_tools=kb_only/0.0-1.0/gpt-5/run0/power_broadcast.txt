host_operator_src="""
#include "power_broadcast_custom_tiling.h"
#include "register/op_def_registry.h"
#include "kernel_operator.h"

namespace {
inline uint32_t CeilDiv(uint32_t a, uint32_t b) {
    return (a + b - 1) / b;
}
}

namespace optiling {
const uint32_t DEFAULT_BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;
constexpr uint32_t BUFFER_NUM = 2;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    PowerBroadcastCustomTilingData tiling;

    // input shapes
    const auto* in0 = context->GetInputShape(0); // x
    const auto* in1 = context->GetInputShape(1); // exponent (broadcast on first dim)
    uint32_t x_elems = static_cast<uint32_t>(in0->GetOriginShape().GetShapeSize());
    uint32_t exp_elems = static_cast<uint32_t>(in1->GetOriginShape().GetShapeSize());

    if (exp_elems == 0) {
        return ge::GRAPH_FAILED;
    }
    uint32_t rows = x_elems / exp_elems;    // assume row-wise broadcast
    uint32_t cols = exp_elems;

    // choose block dim
    uint32_t block_dim = DEFAULT_BLOCK_DIM;
    if (rows < block_dim) block_dim = rows == 0 ? 1 : rows;
    context->SetBlockDim(block_dim);

    // tile strategy: split columns
    uint32_t tileNum = TILE_NUM;
    if (tileNum == 0) tileNum = 1;
    uint32_t tileCols = CeilDiv(cols, tileNum * BUFFER_NUM);
    if (tileCols == 0) tileCols = 1;

    // temporary buffer size for Power with sharedTmpBuffer
    // Use tile-shape to query tmp size
    std::vector<int64_t> shape_vec = {static_cast<int64_t>(tileCols)};
    ge::Shape vec_shape(shape_vec);
    uint32_t maxValue = 0;
    uint32_t minValue = 0;
    AscendC::GetPowerMaxMinTmpSize(vec_shape, vec_shape, false, 4, false, maxValue, minValue);
    uint32_t tmpSize = maxValue;
    if (tmpSize < minValue) tmpSize = minValue;

    tiling.set_rows(rows);
    tiling.set_cols(cols);
    tiling.set_tileNum(tileNum);
    tiling.set_tileCols(tileCols);
    tiling.set_tmpSize(tmpSize);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t* workspaces = context->GetWorkspaceSizes(1);
    workspaces[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    // output adopts x's shape (row-wise broadcast of exponent)
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext* context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class PowerBroadcastCustom : public OpDef {
public:
    explicit PowerBroadcastCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("exponent")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("z")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);

        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(PowerBroadcastCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "power_broadcast_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // double buffer

class KernelPowerBroadcast {
public:
    __aicore__ inline KernelPowerBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t rows, uint32_t cols,
                                uint32_t tileNum, uint32_t tileCols,
                                uint32_t tmpSize)
    {
        this->rows = rows;
        this->cols = cols;
        this->tileNum = tileNum;
        this->tileCols = tileCols;
        this->tmpSize = tmpSize;

        // row range for this block
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowStart = (rows * blockIdx) / blockNum;
        this->rowEnd   = (rows * (blockIdx + 1)) / blockNum;
        if (this->rowEnd > rows) this->rowEnd = rows;

        // set global buffers
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->rowStart * this->cols,
                            (this->rowEnd - this->rowStart) * this->cols);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, this->cols); // broadcast source
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->rowStart * this->cols,
                            (this->rowEnd - this->rowStart) * this->cols);

        // init queues
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileCols * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, this->tileCols * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileCols * sizeof(DTYPE_Z));

        // tmp buffer for Power high-level API
        if (this->tmpSize > 0) {
            pipe.InitBuffer(tmpQue, 1, this->tmpSize);
            sharedTmpBuffer = tmpQue.AllocTensor<uint8_t>();
        }
    }

    __aicore__ inline void Process()
    {
        if (rowStart >= rowEnd || cols == 0) {
            return;
        }
        uint32_t numTileIters = (cols + tileCols - 1) / tileCols;

        for (uint32_t r = rowStart; r < rowEnd; ++r) {
            for (uint32_t t = 0; t < numTileIters; ++t) {
                uint32_t colOff = t * tileCols;
                uint32_t len = cols - colOff;
                if (len > tileCols) len = tileCols;

                // CopyIn
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.AllocTensor<DTYPE_Y>();
                AscendC::DataCopy(xLocal, xGm[(r - rowStart) * cols + colOff], len);
                AscendC::DataCopy(yLocal, yGm[colOff], len);
                inQueueX.EnQue(xLocal);
                inQueueY.EnQue(yLocal);

                // Compute
                AscendC::LocalTensor<DTYPE_X> xUse = inQueueX.DeQue<DTYPE_X>();
                AscendC::LocalTensor<DTYPE_Y> yUse = inQueueY.DeQue<DTYPE_Y>();
                AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
                if (this->tmpSize > 0) {
                    AscendC::Power(zLocal, xUse, yUse, sharedTmpBuffer, len);
                } else {
                    AscendC::Power(zLocal, xUse, yUse, len);
                }
                outQueueZ.EnQue<DTYPE_Z>(zLocal);
                inQueueX.FreeTensor(xUse);
                inQueueY.FreeTensor(yUse);

                // CopyOut
                AscendC::LocalTensor<DTYPE_Z> zUse = outQueueZ.DeQue<DTYPE_Z>();
                AscendC::DataCopy(zGm[(r - rowStart) * cols + colOff], zUse, len);
                outQueueZ.FreeTensor(zUse);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpQue;
    AscendC::LocalTensor<uint8_t> sharedTmpBuffer;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;

    uint32_t rows;
    uint32_t cols;
    uint32_t tileNum;
    uint32_t tileCols;
    uint32_t tmpSize;
    uint32_t rowStart;
    uint32_t rowEnd;
};

extern "C" __global__ __aicore__ void power_broadcast_custom(GM_ADDR x, GM_ADDR exponent, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelPowerBroadcast op;
    op.Init(x, exponent, z,
            tiling_data.rows, tiling_data.cols,
            tiling_data.tileNum, tiling_data.tileCols,
            tiling_data.tmpSize);
    op.Process();
}
