kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_subtract_subtract_mish_custom_tiling_data.h"

// 说明：该核将执行以下融合：y = mish(x - s1 - s2)；
// 这里使用可用API实现近似：mish(z) ≈ z * tanh(z)
// 依据文档：Tanh支持框架分配临时空间; Mul为基础算子; Duplicate用于填充常量

constexpr int32_t BUFFER_NUM = 2; // 每个队列的张量数

class KernelConv2dSubtractSubtractMish {
public:
    __aicore__ inline KernelConv2dSubtractSubtractMish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t tileNum, float subAddNeg)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->subAddNeg = subAddNeg;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        // 计算中间缓冲：常量、减法后结果、tanh结果
        pipe.InitBuffer(calcQueue, 3, this->tileLength * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        AscendC::LocalTensor<DTYPE_X> constLocal = calcQueue.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> subLocal   = calcQueue.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> tanhLocal  = calcQueue.AllocTensor<DTYPE_X>();

        // constLocal 填充为 -(s1 + s2)
        DTYPE_X c(this->subAddNeg);
        AscendC::Duplicate<DTYPE_X>(constLocal, c, this->tileLength);

        // subLocal = x + (-(s1+s2))  等价于 x - s1 - s2
        AscendC::Add(subLocal, xLocal, constLocal, this->tileLength);

        // tanhLocal = tanh(subLocal)
        // 文档：Tanh可以由框架分配临时空间，直接调用 calCount 版本
        AscendC::Tanh<DTYPE_X>(tanhLocal, subLocal, this->tileLength);

        // yLocal = subLocal * tanhLocal
        AscendC::Mul<DTYPE_X>(yLocal, subLocal, tanhLocal, this->tileLength);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
        calcQueue.FreeTensor(constLocal);
        calcQueue.FreeTensor(subLocal);
        calcQueue.FreeTensor(tanhLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECCALC, 3> calcQueue;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    DTYPE_X subAddNeg;
};

extern "C" __global__ __aicore__ void conv2d_subtract_subtract_mish_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dSubtractSubtractMish op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.tileNum, (float)tiling_data.subAddNeg);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    """
    使用自定义AscendC核，将 conv2d 之后的两次减法与Mish近似激活融合为一个算子：
    y ≈ (x - s1 - s2) * tanh(x - s1 - s2)
    """
    def __init__(self, in_channels, out_channels, kernel_size, subtract_value_1, subtract_value_2):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.subtract_value_1 = float(subtract_value_1)
        self.subtract_value_2 = float(subtract_value_2)

    def forward(self, x):
        x = self.conv(x)
        # 调用自定义融合算子（对conv输出进行两次标量减法 + Mish近似）
        x = custom_ops_lib.conv2d_subtract_subtract_mish_custom(x, self.subtract_value_1, self.subtract_value_2)
        return x
