kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transposed_3d_square_input_square_kernel_custom_tiling_data.h"

// 本示例内核演示：使用 AscendC 基础 API(DataCopy、Mul、Add、TPipe/TBuf 等)实现 stride=1、padding=0、groups=1、无bias 的 3D 反卷积
// 权重布局约定: w 的形状为 [Cin, Cout, K, K, K] (与 PyTorch ConvTranspose3d 权重一致)

class KernelConvTransposed3D {
public:
    __aicore__ inline KernelConvTransposed3D() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR z,
                                uint64_t totalOutLength,
                                int64_t n, int64_t cin, int64_t cout,
                                int64_t din, int64_t hin, int64_t win,
                                int64_t k, int64_t dout, int64_t hout, int64_t wout,
                                uint32_t tileNum)
    {
        this->N = n; this->Cin = cin; this->Cout = cout;
        this->Din = din; this->Hin = hin; this->Win = win;
        this->K = k;
        this->Dout = dout; this->Hout = hout; this->Wout = wout;
        this->totalOut = totalOutLength;
        this->tileNum = tileNum;

        // 输入/权重/输出总长度
        uint64_t xLen = (uint64_t)N * Cin * Din * Hin * Win;
        uint64_t wLen = (uint64_t)Cin * Cout * K * K * K;

        // 基于 block 进行输出空间划分
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->blockOutLen = (uint64_t)(totalOut / blockNum);
        uint64_t start = (uint64_t)blockIdx * this->blockOutLen;
        // 最后一个block吃尾
        if (blockIdx == blockNum - 1) {
            this->blockOutLen = totalOut - start;
        }

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(x), xLen);
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(w), wLen);
        zGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(z) + start, this->blockOutLen);

        // 临时本地缓冲：标量(长度=1)的 x、w、乘积prod、累加sum、输出标量
        pipe.InitBuffer(xBuf, 1, sizeof(float));
        pipe.InitBuffer(wBuf, 1, sizeof(float));
        pipe.InitBuffer(prodBuf, 1, sizeof(float));
        pipe.InitBuffer(sumBuf, 1, sizeof(float));
        pipe.InitBuffer(outBuf, 1, sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // 线性遍历当前 block 负责的输出区间 [0, blockOutLen)
        for (uint64_t localIdx = 0; localIdx < this->blockOutLen; ++localIdx) {
            ComputeOne(localIdx);
        }
    }

private:
    // 从线性下标恢复 (n, oc, od, oh, ow)
    __aicore__ inline void LinearIndex2Nd(uint64_t globalOutIndex, int64_t &n, int64_t &oc, int64_t &od, int64_t &oh, int64_t &ow)
    {
        // globalOutIndex 基于全局输出空间
        uint64_t idx = globalOutIndex;
        ow = (int64_t)(idx % Wout); idx /= Wout;
        oh = (int64_t)(idx % Hout); idx /= Hout;
        od = (int64_t)(idx % Dout); idx /= Dout;
        oc = (int64_t)(idx % Cout); idx /= Cout;
        n  = (int64_t) idx;
    }

    // 计算一个输出位置的值
    __aicore__ inline void ComputeOne(uint64_t localOutIndex)
    {
        // 当前 block 的全局输出起始位置
        uint64_t globalIdx = globalBase() + localOutIndex;

        int64_t n, oc, od, oh, ow;
        LinearIndex2Nd(globalIdx, n, oc, od, oh, ow);

        // sumLocal <- 0
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::Duplicate(sumLocal, 0.0f, 1);

        // 遍历 ic 和内核体素 (kd,kh,kw)
        for (int64_t ic = 0; ic < Cin; ++ic) {
            for (int64_t kd = 0; kd < K; ++kd) {
                int64_t id = od - kd;
                if (id < 0 || id >= Din) continue;
                for (int64_t kh = 0; kh < K; ++kh) {
                    int64_t ih = oh - kh;
                    if (ih < 0 || ih >= Hin) continue;
                    for (int64_t kw = 0; kw < K; ++kw) {
                        int64_t iw = ow - kw;
                        if (iw < 0 || iw >= Win) continue;

                        // x index: (((n*Cin+ic)*Din + id)*Hin + ih)*Win + iw
                        uint64_t xIndex =
                            (((uint64_t)n * Cin + (uint64_t)ic) * Din + (uint64_t)id) * Hin * Win +
                            ((uint64_t)ih * Win + (uint64_t)iw);
                        // w index: ((((ic)*Cout + oc)*K + kd)*K + kh)*K + kw
                        uint64_t wIndex =
                            ((((uint64_t)ic * Cout + (uint64_t)oc) * K + (uint64_t)kd) * K + (uint64_t)kh) * K +
                            (uint64_t)kw;

                        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
                        AscendC::DataCopy(xLocal, xGm[xIndex], 1);

                        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
                        AscendC::DataCopy(wLocal, wGm[wIndex], 1);

                        AscendC::LocalTensor<float> prodLocal = prodBuf.Get<float>();
                        AscendC::Mul(prodLocal, xLocal, wLocal, 1);
                        AscendC::Add(sumLocal, sumLocal, prodLocal, 1);
                    }
                }
            }
        }

        // 写回输出
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::DataCopy(outLocal, sumLocal, 1);
        AscendC::DataCopy(zGm[localOutIndex], outLocal, 1);
    }

    __aicore__ inline uint64_t globalBase() const
    {
        // 本 block 的全局起点 = 前面所有 block 的输出元素数量之和
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint64_t base = (uint64_t)blockIdx * (totalOut / blockNum);
        return base;
    }

private:
    AscendC::TPipe pipe;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> zGm;

    // 临时缓冲
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> prodBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;

    // dims
    int64_t N, Cin, Cout;
    int64_t Din, Hin, Win;
    int64_t K;
    int64_t Dout, Hout, Wout;

    // tiling
    uint64_t totalOut;
    uint64_t blockOutLen;
    uint32_t tileNum;
};

extern "C" __global__ __aicore__ void conv_transposed_3d_square_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelConvTransposed3D op;
    op.Init(x, w, z,
            tilingData.totalOutLength,
            tilingData.n, tilingData.cin, tilingData.cout,
            tilingData.din, tilingData.hin, tilingData.win,
            tilingData.k, tilingData.dout, tilingData.hout, tilingData.wout,
            tilingData.tileNum);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    自定义 AscendC 反卷积3D算子(方形输入/方形卷积核)，约束:
    - stride=1, padding=0, output_padding=0
    - groups=1, 无 bias
    - dtype=float32
    - 权重形状: [Cin, Cout, K, K, K]
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, stride: int = 1, padding: int = 0, output_padding: int = 0, groups: int = 1, bias: bool = False):
        super().__init__()
        assert stride == 1 and padding == 0 and output_padding == 0, "custom kernel only supports stride=1, padding=0, output_padding=0"
        assert groups == 1 and bias is False, "custom kernel only supports groups=1 and bias=False"
        K = kernel_size
        # 与 PyTorch ConvTranspose3d 权重一致: [Cin, Cout, K, K, K]
        w = torch.empty(in_channels, out_channels, K, K, K)
        # 简单初始化
        torch.nn.init.kaiming_uniform_(w.view(in_channels, out_channels, -1), a=5**0.5)
        self.weight = torch.nn.Parameter(w)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # 直接调用自定义 AscendC 算子
        return custom_ops_lib.conv_transposed_3d_square_input_square_kernel_custom(x, self.weight)
