kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_divide_leaky_relu_custom_tiling_data.h"

// 基于参考：4.1.3.2 Muls（标量与向量逐元素相乘）、4.1.3.7 LeakyRelu（负半区斜率）
// 这里将“除以常数”转化为“乘以其倒数”，随后执行LeakyReLU

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 2; // 每个队列中的张量块数

class KernelDivideLeakyRelu {
public:
    __aicore__ inline KernelDivideLeakyRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t tileNum, float divisor, float negativeSlope)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        // 计算乘法系数（除法转乘法）
        this->mulAlpha = 1.0f / divisor;
        this->negSlope = negativeSlope;

        inGm.SetGlobalBuffer((__gm__ DTYPE*)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outGm.SetGlobalBuffer((__gm__ DTYPE*)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(calcQueue, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueue.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, inGm[progress * this->tileLength], this->tileLength);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueue.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> tmpLocal = calcQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> yLocal = outQueue.AllocTensor<DTYPE>();

        // 先做逐元素乘以倒数（等价于除以常数）
        AscendC::Muls(tmpLocal, xLocal, this->mulAlpha, (int32_t)this->tileLength);
        // 再做LeakyReLU
        AscendC::LeakyRelu(yLocal, tmpLocal, this->negSlope, (int32_t)this->tileLength);

        outQueue.EnQue<DTYPE>(yLocal);
        calcQueue.FreeTensor(tmpLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> yLocal = outQueue.DeQue<DTYPE>();
        AscendC::DataCopy(outGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> calcQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;

    AscendC::GlobalTensor<DTYPE> inGm;
    AscendC::GlobalTensor<DTYPE> outGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    float mulAlpha;
    float negSlope;
};

extern "C" __global__ __aicore__ void conv2d_divide_leaky_relu_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelDivideLeakyRelu op;
    op.Init(x, y,
            tiling_data.totalLength,
            tiling_data.tileNum,
            tiling_data.divisor,
            tiling_data.negativeSlope);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    优化版：保留Conv2d，由自定义AscendC核替换“除以常数 + LeakyReLU”
    """
    def __init__(self, in_channels, out_channels, kernel_size, divisor):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.divisor = float(divisor)
        self.negative_slope = 0.01

    def forward(self, x):
        x = self.conv(x)
        # 使用自定义AscendC算子完成：x / divisor + LeakyReLU(negative_slope)
        x = custom_ops_lib.conv2d_divide_leaky_relu_custom(x, self.divisor, self.negative_slope)
        return x
