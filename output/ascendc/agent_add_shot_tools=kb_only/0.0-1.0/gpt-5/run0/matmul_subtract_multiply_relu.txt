static constexpr uint32_t BLOCK_DIM = 8;   // 可按需调整
static constexpr uint32_t TILE_NUM  = 8;   // 向量后处理分片数（UB流水）

// 将TCubeTiling + 自定义附加参数打包到RawTilingData：
// [ TCubeTiling | totalLength(uint32) | tileNum(uint32) | subtract_value(float) | multiply_value(float) ]
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 获取输入shape
    const gert::Shape* xShape = context->GetInputShape(0);     // [M, K]
    const gert::Shape* wShape = context->GetInputShape(1);     // [K, N]
    const gert::Shape* bShape = context->GetInputShape(2);     // [N]

    if (xShape == nullptr || wShape == nullptr || bShape == nullptr) {
        return ge::GRAPH_FAILED;
    }

    auto xOrigin = xShape->GetOriginShape();
    auto wOrigin = wShape->GetOriginShape();
    auto bOrigin = bShape->GetOriginShape();

    if (xOrigin.GetDimNum() != 2 || wOrigin.GetDimNum() != 2 || bOrigin.GetDimNum() != 1) {
        return ge::GRAPH_FAILED;
    }

    int64_t M = xOrigin.GetDim(0);
    int64_t K = xOrigin.GetDim(1);
    int64_t Kw = wOrigin.GetDim(0);
    int64_t N = wOrigin.GetDim(1);
    int64_t Nb = bOrigin.GetDim(0);
    if (K != Kw || N != Nb || M <= 0 || K <= 0 || N <= 0) {
        return ge::GRAPH_FAILED;
    }

    // 读取算子属性
    double subtractHost = 0.0;
    double multiplyHost = 1.0;
    (void)context->GetAttrs()->GetFloat("subtract_value", subtractHost);
    (void)context->GetAttrs()->GetFloat("multiply_value", multiplyHost);

    // 生成Matmul的TCubeTiling
    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    matmul_tiling::MatmulApiTiling mmTiling(ascendcPlatform);
    // 数据类型与存储位置/格式
    mmTiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    mmTiling.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    mmTiling.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);

    // 设置矩阵大小（A[M,K], B[K,N], C[M,N]）
    mmTiling.SetM(static_cast<uint32_t>(M));
    mmTiling.SetN(static_cast<uint32_t>(N));
    mmTiling.SetK(static_cast<uint32_t>(K));

    // 获取tiling数据
    optiling::TCubeTiling cubeTiling;
    int ret = mmTiling.GetTiling(cubeTiling);
    if (ret != 0) {
        return ge::GRAPH_FAILED;
    }

    // BlockDim设置（按需）
    context->SetBlockDim(BLOCK_DIM);

    // 打包raw tiling数据
    auto raw = context->GetRawTilingData();
    const size_t headSize = sizeof(optiling::TCubeTiling);
    const size_t extraSize = sizeof(uint32_t) + sizeof(uint32_t) + sizeof(float) + sizeof(float);
    if (raw->GetCapacity() < headSize + extraSize) {
        return ge::GRAPH_FAILED;
    }

    // 写入TCubeTiling
    std::memcpy(raw->GetData(), &cubeTiling, headSize);

    // 计算totalLength（向量后处理的数据总量）
    uint32_t totalLength = static_cast<uint32_t>(M * N);
    uint32_t tileNum = TILE_NUM;
    float subtractVal = static_cast<float>(subtractHost);
    float multiplyVal = static_cast<float>(multiplyHost);

    // 依次写入附加数据
    char* p = reinterpret_cast<char*>(raw->GetData()) + headSize;
    std::memcpy(p, &totalLength, sizeof(uint32_t)); p += sizeof(uint32_t);
    std::memcpy(p, &tileNum, sizeof(uint32_t));     p += sizeof(uint32_t);
    std::memcpy(p, &subtractVal, sizeof(float));    p += sizeof(float);
    std::memcpy(p, &multiplyVal, sizeof(float));    p += sizeof(float);

    raw->SetDataSize(headSize + extraSize);

    // 无需workspace
    size_t* workspaces = context->GetWorkspaceSizes(1);
    workspaces[0] = 0;

    return ge::GRAPH_SUCCESS;
}

REGISTER_TILING_FUNC(MatmulSubtractMultiplyReluCustom, TilingFunc);

} // namespace optiling

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

constexpr int32_t BUFFER_NUM = 2; // each queue buffer num for post vector ops

// Post-process: (x - subtract) * multiply -> ReLU
template <typename T>
class PostProcess {
public:
    __aicore__ inline PostProcess() {}
    __aicore__ inline void Init(GM_ADDR outGm, uint32_t totalLength, uint32_t tileNum, float subtractVal, float multiplyVal)
    {
        this->totalLength = totalLength;
        this->tileNum = tileNum;
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        outGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(outGm) + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->tileLength * sizeof(T));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(T));

        this->subtractVal = subtractVal;
        this->multiplyVal = multiplyVal;
    }

    __aicore__ inline void Run()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<T> buf = inQueue.AllocTensor<T>();
        AscendC::DataCopy(buf, outGlobal[progress * this->tileLength], this->tileLength);
        inQueue.EnQue(buf);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<T> inLocal = inQueue.DeQue<T>();
        AscendC::LocalTensor<T> outLocal = outQueue.AllocTensor<T>();

        // y = (in - subtractVal) * multiplyVal; y = relu(y)
        AscendC::Adds(outLocal, inLocal, static_cast<T>(-subtractVal), this->tileLength);
        AscendC::Muls(outLocal, outLocal, static_cast<T>(multiplyVal), this->tileLength);
        AscendC::Relu(outLocal, outLocal, this->tileLength);

        outQueue.EnQue<T>(outLocal);
        inQueue.FreeTensor(inLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<T> outLocal = outQueue.DeQue<T>();
        AscendC::DataCopy(outGlobal[progress * this->tileLength], outLocal, this->tileLength);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM>  inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::GlobalTensor<T> outGlobal;

    uint32_t totalLength = 0;
    uint32_t blockLength = 0;
    uint32_t tileNum = 0;
    uint32_t tileLength = 0;

    float subtractVal = 0.0f;
    float multiplyVal = 1.0f;
};

// Kernel入口：完成Matmul(A[M,K] * B[K,N] + bias) -> 写入y；随后对y做(sub, mul, relu)
extern "C" __global__ __aicore__ void matmul_subtract_multiply_relu_custom(
    GM_ADDR xGM, GM_ADDR wGM, GM_ADDR bGM, GM_ADDR yGM, GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    // 解析TCubeTiling
    optiling::TCubeTiling tiling;
    auto srcGM = reinterpret_cast<__gm__ uint32_t*>(tilingGM);
    auto dstLocal = reinterpret_cast<uint32_t*>(&tiling);
    for (int i = 0; i < static_cast<int>(sizeof(optiling::TCubeTiling) / sizeof(uint32_t)); ++i, ++srcGM, ++dstLocal) {
        *dstLocal = *srcGM;
    }

    // 读取附加参数
    uint32_t totalLength = 0;
    uint32_t tileNum = 0;
    float subtractVal = 0.0f;
    float multiplyVal = 1.0f;
    {
        __gm__ char* p = reinterpret_cast<__gm__ char*>(tilingGM) + sizeof(optiling::TCubeTiling);
        std::memcpy(&totalLength, p, sizeof(uint32_t)); p += sizeof(uint32_t);
        std::memcpy(&tileNum,     p, sizeof(uint32_t)); p += sizeof(uint32_t);
        std::memcpy(&subtractVal, p, sizeof(float));    p += sizeof(float);
        std::memcpy(&multiplyVal, p, sizeof(float));
    }

    // GM张量
    AscendC::GlobalTensor<float> aGlobal;
    AscendC::GlobalTensor<float> bGlobal;
    AscendC::GlobalTensor<float> cGlobal;
    AscendC::GlobalTensor<float> biasGlobal;

    // 正常布局（ND）
    int32_t sizeA = tiling.singleCoreM * tiling.singleCoreK;
    int32_t sizeB = tiling.singleCoreK * tiling.singleCoreN;
    int32_t sizeC = tiling.singleCoreM * tiling.singleCoreN;
    int32_t sizeBias = tiling.singleCoreN;

    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(xGM), sizeA);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(wGM), sizeB);
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(yGM), sizeC);
    biasGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(bGM), sizeBias);

    // Matmul类型定义（NORMAL, float）
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> aType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> bType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> cType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float> biasType;

    // 创建Matmul对象并初始化
    AscendC::TPipe pipe;
    matmul::Matmul<aType, bType, cType, biasType> mm1;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm1);
    mm1.Init(&tiling);
    mm1.SetTensorA(aGlobal, false);   // A: not transposed
    mm1.SetTensorB(bGlobal, false);   // B: not transposed (传入已为[K,N]布局)
    mm1.SetWorkspace(workspaceGM, 0);
    mm1.SetBias(biasGlobal);

    // 执行Matmul（单次Batch）
    mm1.IterateBatch(cGlobal, 1, 1, false);

    // 向量后处理： (y - subtract) * multiply -> ReLU
    PostProcess<float> post;
    post.Init(yGM, totalLength, tileNum, subtractVal, multiplyVal);
    post.Run();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, subtract_value, multiply_value):
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.subtract_value = float(subtract_value)
        self.multiply_value = float(multiply_value)
        # 按Linear语义定义参数：weight[out_features, in_features]，bias[out_features]
        self.weight = torch.nn.Parameter(torch.empty(out_features, in_features))
        self.bias = torch.nn.Parameter(torch.empty(out_features))
        torch.nn.init.kaiming_uniform_(self.weight, a=5**0.5)
        fan_in = in_features
        bound = 1 / (fan_in ** 0.5)
        torch.nn.init.uniform_(self.bias, -bound, bound)

    def forward(self, x):
        # 期望B为[K,N]，因此传入weight.t().contiguous()
        return custom_ops_lib.matmul_subtract_multiply_relu_custom(
            x, self.weight.t().contiguous(), self.bias, self.subtract_value, self.multiply_value
        )
