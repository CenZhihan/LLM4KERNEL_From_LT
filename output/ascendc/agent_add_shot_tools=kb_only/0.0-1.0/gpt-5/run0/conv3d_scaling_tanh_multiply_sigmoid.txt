kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv3d_scaling_tanh_multiply_sigmoid_custom_tiling_data.h"

// 简化演示：使用float类型
#define DTYPE_X float
#define DTYPE_S float
#define DTYPE_B float
#define DTYPE_Y float

constexpr int32_t BUFFER_NUM = 2; // 双缓冲

// 该算子对输入x执行：y = sigmoid( tanh(x * scale) * mulbias )
// 注：scale与mulbias在Python侧已expand为与x同形状，便于向量化计算。
// 参考文档中共享临时缓冲(sharedTmpBuffer)用法，供Tanh/Sigmoid等数学库API内部使用。
class KernelConv3dScalingTanhMultiplySigmoid {
public:
    __aicore__ inline KernelConv3dScalingTanhMultiplySigmoid() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR mulbias, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum, uint32_t tmpBufferSize)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        // 以BUFFER_NUM分块，保证每个tile大小
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        sGm.SetGlobalBuffer((__gm__ DTYPE_S *)scale + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)mulbias + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueS, BUFFER_NUM, this->tileLength * sizeof(DTYPE_S));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, this->tileLength * sizeof(DTYPE_B));
        pipe.InitBuffer(midQueue1, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y)); // 中间缓冲
        pipe.InitBuffer(midQueue2, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y)); // 中间缓冲
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        // 共享临时缓冲，用于数学库内部临时数据（参考高阶API示例）
        pipe.InitBuffer(tmpQueue, BUFFER_NUM, tmpBufferSize);
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_S> sLocal = inQueueS.AllocTensor<DTYPE_S>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();

        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(sLocal, sGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(bLocal, bGm[progress * this->tileLength], this->tileLength);

        inQueueX.EnQue(xLocal);
        inQueueS.EnQue(sLocal);
        inQueueB.EnQue(bLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_S> sLocal = inQueueS.DeQue<DTYPE_S>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.DeQue<DTYPE_B>();

        AscendC::LocalTensor<uint8_t> sharedTmpBuffer = tmpQueue.AllocTensor<uint8_t>();

        // mid1 = x * scale
        AscendC::LocalTensor<DTYPE_Y> mid1 = midQueue1.AllocTensor<DTYPE_Y>();
        AscendC::Mul(mid1, xLocal, sLocal, this->tileLength);
        AscendC::PipeBarrier<PIPE_V>();

        // mid2 = tanh(mid1)
        AscendC::LocalTensor<DTYPE_Y> mid2 = midQueue2.AllocTensor<DTYPE_Y>();
        AscendC::Tanh<DTYPE_Y, false>(mid2, mid1, sharedTmpBuffer, this->tileLength);
        AscendC::PipeBarrier<PIPE_V>();
        midQueue1.FreeTensor(mid1);

        // mid1(reuse) = mid2 * bLocal
        AscendC::LocalTensor<DTYPE_Y> mid3 = midQueue1.AllocTensor<DTYPE_Y>();
        AscendC::Mul(mid3, mid2, bLocal, this->tileLength);
        AscendC::PipeBarrier<PIPE_V>();
        midQueue2.FreeTensor(mid2);

        // y = sigmoid(mid3)
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::Sigmoid<DTYPE_Y, false>(yLocal, mid3, sharedTmpBuffer, this->tileLength);
        outQueueY.EnQue<DTYPE_Y>(yLocal);

        // 释放
        midQueue1.FreeTensor(mid3);
        inQueueX.FreeTensor(xLocal);
        inQueueS.FreeTensor(sLocal);
        inQueueB.FreeTensor(bLocal);
        tmpQueue.FreeTensor(sharedTmpBuffer);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueS;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueB;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> midQueue1;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> midQueue2;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_S> sGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv3d_scaling_tanh_multiply_sigmoid_custom(
    GM_ADDR x, GM_ADDR scale, GM_ADDR mulbias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dScalingTanhMultiplySigmoid op;
    op.Init(x, scale, mulbias, y, tiling_data.totalLength, tiling_data.tileNum, tiling_data.tmpBufferSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义AscendC算子优化：对Conv3d输出执行 y = sigmoid(tanh(x * scale) * mulbias)
    """
    def __init__(self, in_channels, out_channels, kernel_size, scaling_factor, bias_shape):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        # 与原模型一致：逐通道缩放与逐通道乘性“bias”（形状[C,1,1,1]）
        self.scaling_factor = torch.nn.Parameter(torch.randn(bias_shape))
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv(x)
        # 展开为与x同形状，便于核内向量化（广播在Python侧完成）
        scale_b = self.scaling_factor.expand_as(x)
        bias_b = self.bias.expand_as(x)
        return custom_ops_lib.conv3d_scaling_tanh_multiply_sigmoid_custom(x, scale_b, bias_b)
