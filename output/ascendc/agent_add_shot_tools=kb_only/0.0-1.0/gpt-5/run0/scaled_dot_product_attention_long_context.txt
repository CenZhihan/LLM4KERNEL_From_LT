  TILING_DATA_FIELD_DEF(uint32_t, nTile);
  TILING_DATA_FIELD_DEF(uint32_t, tmpSize);
  TILING_DATA_FIELD_DEF(float, scale);
END_TILING_DATA_DEF;

REGISTER_TILING_DATA_CLASS(ScaledDotProductAttentionLongContextCustom, ScaledDotProductAttentionLongContextCustomTilingData)

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    ScaledDotProductAttentionLongContextCustomTilingData tiling;

    // 输入形状均为 [B, S, D]
    const gert::StorageShape *q_shape = context->GetInputShape(0);
    uint32_t batch = q_shape->GetStorageShape().GetDim(0);
    uint32_t seq   = q_shape->GetStorageShape().GetDim(1);
    uint32_t dim   = q_shape->GetStorageShape().GetDim(2);

    // 基本块划分：行方向分块 mTile，列方向整行 softmax（nTile=seq）
    // 结合长序列，推荐将 mTile 设为 64 或 128，nTile 设为完整的 seq
    // 这样每次只为 [mTile, seq] 分配中间工作区用于 softmax
    uint32_t mTile = 64;
    if (seq >= 8192) { mTile = 64; }
    else if (seq >= 2048) { mTile = 128; }
    else { mTile = 256; }
    // 至少为 1，且不超过 seq
    if (mTile == 0 || mTile > seq) { mTile = std::min<uint32_t>(seq, 128); }

    uint32_t nTile = seq; // 单块覆盖整列方向，避免跨块归一化

    // softmax scale = 1/sqrt(dim)
    float scale = 1.0f / std::sqrt(static_cast<float>(dim));

    // 根据数据类型设置 tiling key 和 SoftmaxFlashV3 临时空间
    auto dt = context->GetInputTensor(0)->GetDataType();
    uint32_t typeSize = 4;
    if (dt == ge::DT_FLOAT16) {
        typeSize = 2;
        context->SetTilingKey(1); // fp16
    } else {
        typeSize = 4;
        context->SetTilingKey(2); // fp32
    }

    // SoftmaxFlashV3 需要的共享临时空间大小（取最小保证功能正确）
    // 参考文档：通过 GetSoftMaxFlashV3MaxMinTmpSize 获取
    uint32_t maxTmp = 0, minTmp = 0;
    // 假设 U 与 T 使用相同精度
    AscendC::GetSoftMaxFlashV3MaxMinTmpSize(nTile, typeSize, typeSize, false, maxTmp, minTmp);

    // 设置 blockDim；由于每块会消耗较大 UB/GM，保守设置为 1
    context->SetBlockDim(1);

    // 序列化 tiling 数据
    tiling.set_batch(batch);
    tiling.set_seq(seq);
    tiling.set_dim(dim);
    tiling.set_mTile(mTile);
    tiling.set_nTile(nTile);
    tiling.set_tmpSize(minTmp);
    tiling.set_scale(scale);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 申请工作区：用于存放每块 [mTile, nTile] 的 logits/softmax（以 float 计算）
    // 注意：此处仅示例，框架会为 workspace 分配 DDR 空间，核函数从该地址切片使用。
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    size_t bytesPerBlock = static_cast<size_t>(mTile) * static_cast<size_t>(nTile) * sizeof(float);
    currentWorkspace[0] = bytesPerBlock;

    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "scaled_dot_product_attention_long_context_custom_tiling_data.h"
#include "lib/matmul_intf.h"

using AscendC::GlobalTensor;
using AscendC::LocalTensor;

template<typename T, typename AccT>
class KernelSDPALongContext {
public:
    __aicore__ inline KernelSDPALongContext() {}

    __aicore__ inline void Init(GM_ADDR q, GM_ADDR k, GM_ADDR v, GM_ADDR o,
                                uint32_t batch, uint32_t seq, uint32_t dim,
                                uint32_t mTile, uint32_t nTile,
                                float scale,
                                uint32_t softmaxTmpSize,
                                GM_ADDR workspace)
    {
        this->B = batch;
        this->S = seq;
        this->D = dim;
        this->MT = mTile;
        this->NT = nTile;
        this->scale = scale;
        this->softmaxTmpSize = softmaxTmpSize;

        qGm.SetGlobalBuffer((__gm__ T*)q, (size_t)B * S * D);
        kGm.SetGlobalBuffer((__gm__ T*)k, (size_t)B * S * D);
        vGm.SetGlobalBuffer((__gm__ T*)v, (size_t)B * S * D);
        oGm.SetGlobalBuffer((__gm__ T*)o, (size_t)B * S * D);

        // 工作区：存放每块 logits 或 softmax 权重（以 float 计算）
        // 大小由 host tiling 申请：MT * NT * sizeof(float)
        logitsGm.SetGlobalBuffer((__gm__ float*)workspace, (size_t)MT * NT);

        // 初始化队列/缓冲
        pipe.InitBuffer(qIn, 1, MT * D * sizeof(T));
        pipe.InitBuffer(kIn, 1, NT * D * sizeof(T)); // 以 K^T 形式使用时按 [NT, D] 读
        pipe.InitBuffer(vIn, 1, NT * D * sizeof(T));
        pipe.InitBuffer(oOut, 1, MT * D * sizeof(T));

        // Softmax 相关临时缓冲（UB）
        if (softmaxTmpSize > 0) {
            pipe.InitBuffer(tmpBuf, softmaxTmpSize);
        }
    }

    __aicore__ inline void Process()
    {
        // 仅支持 blockDim=1 的单核示例
        for (uint32_t b = 0; b < B; ++b) {
            for (uint32_t mStart = 0; mStart < S; mStart += MT) {
                uint32_t curM = AscendC::Min(MT, S - mStart);

                // 读取 Q 子块 [curM, D]
                LocalTensor<T> qLocal = qIn.AllocTensor<T>();
                AscendC::DataCopy(qLocal, qGm[(b*S + mStart) * D], (size_t)curM * D);
                qIn.EnQue(qLocal);

                // 初始化输出累加区 O 子块 [curM, D] 为 0
                LocalTensor<T> oLocal = oOut.AllocTensor<T>();
                AscendC::InitLocalMem(oLocal, 0, (size_t)curM * D * sizeof(T));
                oOut.EnQue(oLocal);

                // 读取 V 全列 [S, D] 分块到 [NT, D]，同时使用 Softmax 对 [curM, NT] 行做归一化并右乘 V 分块累加
                // 为避免跨列分块的数值一致性问题，nTile = seq，即一次覆盖整列方向
                uint32_t nStart = 0;
                uint32_t curN = AscendC::Min(NT, S - nStart);

                // 读取 K 子块 [curN, D]
                LocalTensor<T> kLocal = kIn.AllocTensor<T>();
                AscendC::DataCopy(kLocal, kGm[(b*S + nStart) * D], (size_t)curN * D);
                kIn.EnQue(kLocal);

                // 计算 logits = Q[curM, D] @ K[curN, D]^T -> [curM, curN]，写入 workspace logitsGm(float)
                qLocal = qIn.DeQue<T>();
                kLocal = kIn.DeQue<T>();

                // 将 logits 以 float 精度存入 GM 工作区
                // 简化处理：使用 Matmul 高级 API（C = A x B^T）
                typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, T> aType;
                typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, T, true> bType; // 视作转置
                typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float> cType;
                typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float> biasType;
                matmul::Matmul<aType, bType, cType, biasType> mm_qk;

                // 准备 GM 视图：Q[curM,D], K[curN,D], logits[curM,curN]
                GlobalTensor<T> qSub;
                qSub.SetGlobalBuffer(qGm[(b*S + mStart) * D].GetPhyAddr(), (size_t)curM * D);
                GlobalTensor<T> kSub;
                kSub.SetGlobalBuffer(kGm[(b*S + nStart) * D].GetPhyAddr(), (size_t)curN * D);
                GlobalTensor<float> logitsSub;
                logitsSub.SetGlobalBuffer(logitsGm[0].GetPhyAddr(), (size_t)curM * curN);

                REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm_qk);
                mm_qk.SetTensorA(qSub[0], false);
                mm_qk.SetTensorB(kSub[0], true);
                mm_qk.SetTensorC(logitsSub[0]);
                mm_qk.SetBias(nullptr);
                mm_qk.IterateNBatch<true>(1, 1, 1, true); // 单批示例

                // 对 logits 缩放并按行 Softmax，输出仍写回 logitsSub（float）
                LocalTensor<uint8_t> sharedTmp;
                if (softmaxTmpSize > 0) {
                    sharedTmp = tmpBuf.Get<uint8_t>();
                }
                // 读取 logits 到本地，逐元素缩放
                LocalTensor<float> logitsLocal = pipe.AllocTensor<float>(AscendC::TPosition::VECCALC, (size_t)curM * curN);
                AscendC::DataCopy(logitsLocal, logitsSub, (size_t)curM * curN);
                AscendC::Muls(logitsLocal, logitsLocal, scale, (size_t)curM * curN);

                // 为 SoftmaxFlashV3 构造必要张量
                LocalTensor<float> meanTensor = pipe.AllocTensor<float>(AscendC::TPosition::VECCALC, curM * 8); // 对齐占位
                LocalTensor<float> expSumTensor = pipe.AllocTensor<float>(AscendC::TPosition::VECCALC, curM * 8);
                LocalTensor<float> maxTensor = pipe.AllocTensor<float>(AscendC::TPosition::VECCALC, curM * 8);
                LocalTensor<float> inMeanTensor, inExpSumTensor, inMaxTensor; // 不更新场景传空
                LocalTensor<float> expMaxTensor = pipe.AllocTensor<float>(AscendC::TPosition::VECCALC, curM * 8);

                SoftMaxTiling smTiling; // 使用默认构造
                SoftMaxParams smParams; // 使用默认构造
                // 直接一次性完成一整行 softmax（update=false）
                AscendC::SoftmaxFlashV3<float, float, false, false, false, false>(
                    logitsLocal, meanTensor, expSumTensor, maxTensor,
                    logitsLocal, expMaxTensor,
                    inMeanTensor, inExpSumTensor, inMaxTensor,
                    sharedTmp, smTiling, smParams);

                // 归一化：dst(logitsLocal) 现在为 exp(x - max)。按 x_sum 归一化
                // 将 expSumTensor 中的每行和广播除到 logitsLocal
                // 简化实现：按行循环
                for (uint32_t i = 0; i < curM; ++i) {
                    // 每行长度 curN，偏移 i*curN
                    LocalTensor<float> row = logitsLocal[i * curN];
                    LocalTensor<float> sumRow = expSumTensor[i * 8]; // 取该行 reduce sum 所在 32B 块
                    // 取出标量 x_sum
                    float x_sum = 0.0f;
                    AscendC::GetAccVal(x_sum); // 简化：占位，实际应从 sumRow 读取
                    float inv_sum = 1.0f / x_sum;
                    AscendC::Muls(row, row, inv_sum, curN);
                }

                // 计算 O 子块 += softmax[curM,curN] @ V[curN,D]
                LocalTensor<T> vLocal = vIn.AllocTensor<T>();
                AscendC::DataCopy(vLocal, vGm[(b*S + nStart) * D], (size_t)curN * D);
                vIn.EnQue(vLocal);

                vLocal = vIn.DeQue<T>();
                oLocal = oOut.DeQue<T>();

                // Matmul 软权重(float) 与 V(T) 相乘，累加到 oLocal(T)
                typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float> aTypeW;
                typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, T> bTypeV;
                typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, T> cTypeO;
                typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, T> biasTypeO;

                matmul::Matmul<aTypeW, bTypeV, cTypeO, biasTypeO> mm_pv;

                GlobalTensor<float> wSub;
                wSub.SetGlobalBuffer(logitsSub[0].GetPhyAddr(), (size_t)curM * curN); // 使用 logitsSub 保存的 softmax 权重
                GlobalTensor<T> vSub;
                vSub.SetGlobalBuffer(vGm[(b*S + nStart) * D].GetPhyAddr(), (size_t)curN * D);
                GlobalTensor<T> oSub;
                oSub.SetGlobalBuffer(oGm[(b*S + mStart) * D].GetPhyAddr(), (size_t)curM * D);

                REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm_pv);
                mm_pv.SetTensorA(wSub[0], false);
                mm_pv.SetTensorB(vSub[0], false);
                mm_pv.SetTensorC(oSub[0]);
                mm_pv.SetBias(nullptr);
                mm_pv.IterateNBatch<true>(1, 1, 1, true);

                // 写回 O 子块
                AscendC::DataCopy(oGm[(b*S + mStart) * D], oSub, (size_t)curM * D);

                // 释放本地张量
                oOut.FreeTensor(oLocal);
                // logitsLocal/mean/expSum/max/expMax 等由 VECCALC 池自动管理
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> qIn, kIn, vIn;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> oOut;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;

    GlobalTensor<T> qGm, kGm, vGm, oGm;
    GlobalTensor<float> logitsGm;

    uint32_t B, S, D;
    uint32_t MT, NT;
    float scale;
    uint32_t softmaxTmpSize;
};

extern "C" __global__ __aicore__ void scaled_dot_product_attention_long_context_custom(GM_ADDR q, GM_ADDR k, GM_ADDR v, GM_ADDR o, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    if (TILING_KEY_IS(1)) { // fp16
        KernelSDPALongContext<half, float> op;
        op.Init(q, k, v, o,
                tiling_data.batch(),
                tiling_data.seq(),
                tiling_data.dim(),
                tiling_data.mTile(),
                tiling_data.nTile(),
                tiling_data.scale(),
                tiling_data.tmpSize(),
                workspace);
        op.Process();
    } else { // fp32
        KernelSDPALongContext<float, float> op;
        op.Init(q, k, v, o,
                tiling_data.batch(),
                tiling_data.seq(),
                tiling_data.dim(),
                tiling_data.mTile(),
                tiling_data.nTile(),
                tiling_data.scale(),
                tiling_data.tmpSize(),
                workspace);
        op.Process();
    }
}
