kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "add_bias_four_dim_broadcast_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelAddBias4DBroadcast {
public:
    __aicore__ inline KernelAddBias4DBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w, uint32_t totalLength)
    {
        this->n = n;
        this->c = c;
        this->h = h;
        this->w = w;
        this->hw = h * w;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t groupsTotal = n * c;                  // number of (n,c) groups
        this->groupsPerBlock = groupsTotal / blockNum; // assume divisible as set by host
        this->tileLength = this->hw;                   // process one (H*W) plane per iteration
        this->loopCount = this->groupsPerBlock;

        // set GM buffers start at block offset
        uint32_t blockStartElem = (blockIdx * this->groupsPerBlock) * this->tileLength;
        xGm.SetGlobalBuffer((__gm__ float *)x + blockStartElem, this->groupsPerBlock * this->tileLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + blockStartElem, this->groupsPerBlock * this->tileLength);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, this->c);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->loopCount * BUFFER_NUM; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // compute channel index for this (H*W) tile
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t groupIdxGlobal = blockIdx * this->groupsPerBlock + progress; // [0, n*c)
        uint32_t cIdx = groupIdxGlobal % this->c;

        // load bias scalar
        // copy one element from GM to a local temp scalar buffer
        AscendC::LocalTensor<float> tmpBiasLocal = tmpBuf.Get<float>(AscendC::ROUND_UP_DIV(1, AscendC::ONE_BLK_SIZE) * AscendC::ONE_BLK_SIZE);
        AscendC::DataCopy(tmpBiasLocal, biasGm[cIdx], 1);
        float biasScalar = tmpBiasLocal.GetValue(0);

        // y = x + biasScalar (broadcast over H*W)
        AscendC::Adds(yLocal, xLocal, biasScalar, this->tileLength);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> biasGm;

    uint32_t n;
    uint32_t c;
    uint32_t h;
    uint32_t w;
    uint32_t hw;

    uint32_t groupsPerBlock;
    uint32_t tileLength;
    uint32_t loopCount;
};

extern "C" __global__ __aicore__ void add_bias_four_dim_broadcast_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelAddBias4DBroadcast op;
    op.Init(x, bias, y,
            tiling_data.n, tiling_data.c, tiling_data.h, tiling_data.w,
            tiling_data.totalLength);
    op.Process();
}
