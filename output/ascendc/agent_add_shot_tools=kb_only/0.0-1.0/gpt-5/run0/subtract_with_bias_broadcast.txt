kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "subtract_with_bias_broadcast_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 1; // 简化：每次处理一行（长度为H）

class KernelSubBiasBroadcast {
public:
    __aicore__ inline KernelSubBiasBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z, uint32_t bsLength, uint32_t hLength)
    {
        this->bsLength = bsLength;   // 行数（B*S）
        this->hLength  = hLength;    // 每行长度（H）

        // 每个核心均匀分配若干行
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = this->bsLength / blockNum;
        this->offsetRows   = blockIdx * this->rowsPerBlock;

        // 设置全局Tensor范围
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->offsetRows * this->hLength, this->rowsPerBlock * this->hLength);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_Y *)bias, this->hLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->offsetRows * this->hLength, this->rowsPerBlock * this->hLength);

        // 分配本地缓冲
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->hLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueBias, 1, this->hLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->hLength * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        // 预取并缓存bias一行（H）
        AscendC::LocalTensor<DTYPE_Y> biasLocal = inQueueBias.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(biasLocal, biasGm, this->hLength);
        inQueueBias.EnQue(biasLocal);
        AscendC::LocalTensor<DTYPE_Y> biasCached = inQueueBias.DeQue<DTYPE_Y>();

        // 逐行处理：每次取一行的x，与同一行的bias做逐元素相减
        for (uint32_t r = 0; r < this->rowsPerBlock; ++r) {
            CopyInRow(r);
            ComputeRow(biasCached);
            CopyOutRow(r);
        }

        inQueueBias.FreeTensor(biasCached);
    }

private:
    __aicore__ inline void CopyInRow(uint32_t rowIdx)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->hLength], this->hLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeRow(const AscendC::LocalTensor<DTYPE_Y>& biasLocal)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();

        // 逐元素：z = x - bias
        AscendC::Sub(zLocal, xLocal, biasLocal, this->hLength);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOutRow(uint32_t rowIdx)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[rowIdx * this->hLength], zLocal, this->hLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> biasGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;

    uint32_t bsLength;
    uint32_t hLength;
    uint32_t rowsPerBlock;
    uint32_t offsetRows;
};

extern "C" __global__ __aicore__ void subtract_with_bias_broadcast_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSubBiasBroadcast op;
    op.Init(x, bias, z, tiling_data.bsLength, tiling_data.hLength);
    op.Process();
}
