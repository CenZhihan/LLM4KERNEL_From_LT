kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose3d_softmax_sigmoid_custom_tiling_data.h"

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 2; // 双缓冲

class KernelSoftmaxSigmoid3D {
public:
    __aicore__ inline KernelSoftmaxSigmoid3D() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum,
                                const SoftMaxTiling &softmaxTiling)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->softmaxTiling = softmaxTiling;
        this->kSize = softmaxTiling.srcK; // 软最大化的最后一维长度

        // 计算每个tile包含的完整行数，确保为kSize的整数倍
        uint32_t baseTile = this->blockLength / this->tileNum / BUFFER_NUM;
        uint32_t rows = baseTile / this->kSize;
        if (rows == 0) { rows = 1; }
        this->tileLength = rows * this->kSize;

        xGm.SetGlobalBuffer((__gm__ DTYPE*)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE*)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        // 为单tile申请本地缓冲
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        uint32_t loopCount = (this->blockLength + this->tileLength - 1) / this->tileLength;
        for (uint32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        uint32_t offset = progress * this->tileLength;
        uint32_t remaining = this->blockLength - offset;
        currElems = remaining > this->tileLength ? this->tileLength : remaining;

        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm[offset], currElems);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE> srcLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> dstLocal = outQueueY.AllocTensor<DTYPE>();

        // 当前tile的行数（每行长度为kSize）
        uint32_t height = currElems / this->kSize;
        uint32_t width = this->kSize;

        // 配置SoftMax形状信息
        AscendC::SoftMaxShapeInfo srcShape = {height, width, height, width};
        // 先进行SoftMax（沿最后一维width）
        AscendC::SoftMax<DTYPE>(dstLocal, srcLocal, this->softmaxTiling, srcShape);
        // 再进行Sigmoid逐元素变换
        AscendC::Sigmoid(dstLocal, dstLocal, height * width);

        outQueueY.EnQue<DTYPE>(dstLocal);
        inQueueX.FreeTensor(srcLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        uint32_t offset = progress * this->tileLength;
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm[offset], yLocal, currElems);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t blockLength = 0;
    uint32_t tileNum = 0;
    uint32_t tileLength = 0;
    uint32_t currElems = 0;
    uint32_t kSize = 0;
    SoftMaxTiling softmaxTiling;
};

extern "C" __global__ __aicore__ void conv_transpose3d_softmax_sigmoid_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSoftmaxSigmoid3D op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.tileNum, tiling_data.softmaxTilingData);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
import torch.nn as nn

class ModelNew(torch.nn.Module):
    """
    使用AscendC自定义算子，将Softmax(dim=1)+Sigmoid融合为单个核函数。
    通过先将通道维移到最后（N, D, H, W, C），在最后一维做Softmax和Sigmoid，再转回（N, C, D, H, W）。
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, bias=True):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size,
                                                 stride=stride, padding=padding,
                                                 output_padding=output_padding, bias=bias)

    def forward(self, x):
        x = self.conv_transpose(x)                  # (N, C, D, H, W)
        x = x.permute(0, 2, 3, 4, 1).contiguous()   # (N, D, H, W, C) 使通道为最后一维
        x = custom_ops_lib.conv_transpose3d_softmax_sigmoid_custom(x)  # 在最后一维做Softmax+Sigmoid
        x = x.permute(0, 4, 1, 2, 3).contiguous()   # (N, C, D, H, W)
        return x
