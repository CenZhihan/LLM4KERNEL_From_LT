kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "unet_softmax_custom_tiling_data.h"

// 参考文档 5.3.1.1/5.3.1.8/5.3.1.9：SoftMax 高级 API + Tiling 用法
// 本核对输入按最后一维（width/最后轴）执行逐行 softmax：将 ND 折叠为 [m, n]，其中 n=lastAxisK

constexpr int32_t BUFFER_NUM = 2; // pipeline 队列深度

class KernelUnetSoftmax {
public:
    __aicore__ inline KernelUnetSoftmax() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalLength, uint32_t lastAxisK)
    {
        this->totalLength = totalLength;
        this->K = lastAxisK;
        this->M = (K == 0) ? 0 : (totalLength / K);

        // 按行分片到各个核：每核处理部分 M
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        // 简单均分（最后一核可能少一些）
        uint32_t rowsPerBlock = (M + blockNum - 1) / blockNum;
        rowStart = rowsPerBlock * blockIdx;
        rowEnd = AscendC::min(rowStart + rowsPerBlock, M);

        xGm.SetGlobalBuffer((__gm__ float*)x, totalLength);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalLength);

        // 每次处理 1 行（为保证 UB 与对齐简单），可按需扩大行为多行以提升吞吐
        tileRows = 1;
        tileElems = tileRows * K;

        pipe.InitBuffer(inQueue,  BUFFER_NUM, tileElems * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, tileElems * sizeof(float));
    }

    __aicore__ inline void Process(const optiling::SoftMaxTiling &softmaxTilingData)
    {
        if (rowStart >= rowEnd || K == 0) {
            return;
        }
        int32_t loopCount = (rowEnd - rowStart) * BUFFER_NUM;
        int32_t progress = 0;
        for (uint32_t r = rowStart; r < rowEnd; ++r) {
            CopyIn(progress, r);
            Compute(progress, softmaxTilingData);
            CopyOut(progress, r);
            progress++;
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t /*progress*/, uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        // 每行连续 K 元素
        AscendC::DataCopy(xLocal, xGm[rowIdx * K], K);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/, const optiling::SoftMaxTiling &softmaxTilingData)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueue.AllocTensor<float>();

        // 使用 SoftMax 高级 API，按最后轴（长度 K）做逐行 softmax
        // isReuseSource=false（与 Host 侧 GetSoftMaxMinTmpSize 一致）
        AscendC::SoftMax<float, false>(yLocal, xLocal, softmaxTilingData);

        outQueue.EnQue<float>(yLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t /*progress*/, uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> yLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(yGm[rowIdx * K], yLocal, K);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t totalLength = 0;
    uint32_t K = 0;                 // 最后轴长度
    uint32_t M = 0;                 // 折叠后的行数
    uint32_t rowStart = 0, rowEnd = 0;

    uint32_t tileRows = 1;          // 每次处理的行数
    uint32_t tileElems = 0;         // tileRows * K
};

extern "C" __global__ __aicore__ void unet_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelUnetSoftmax op;
    op.Init(x, y, tilingData.totalLength, tilingData.lastAxisK);
    if (TILING_KEY_IS(1)) {
        op.Process(tilingData.softmaxTilingData);
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
import torch.nn as nn

# 用自定义 AscendC Softmax（最后一维）替换 nn.Softmax(dim=-1)
def unet_softmax_lastdim(x: torch.Tensor) -> torch.Tensor:
    return custom_ops_lib.unet_softmax_custom(x)

class DoubleConvNew(nn.Module):
    def __init__(self, in_channels, out_channels):
        super().__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1)
        self.bn1 = nn.BatchNorm2d(out_channels)
        self.conv2 = nn.Conv2d(out_channels, out_channels, kernel_size=3, padding=1)
        self.bn2 = nn.BatchNorm2d(out_channels)

    def forward(self, x):
        x = self.conv1(x)
        x = self.bn1(x)
        x = unet_softmax_lastdim(x)  # 自定义算子：沿最后一维 softmax

        x = self.conv2(x)
        x = self.bn2(x)
        x = unet_softmax_lastdim(x)  # 自定义算子：沿最后一维 softmax
        return x

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, features):
        super(ModelNew, self).__init__()
        self.encoder1 = DoubleConvNew(in_channels, features)
        self.pool1 = nn.MaxPool2d(kernel_size=2, stride=2)
        self.encoder2 = DoubleConvNew(features, features * 2)
        self.pool2 = nn.MaxPool2d(kernel_size=2, stride=2)
        self.encoder3 = DoubleConvNew(features * 2, features * 4)
        self.pool3 = nn.MaxPool2d(kernel_size=2, stride=2)
        self.encoder4 = DoubleConvNew(features * 4, features * 8)
        self.pool4 = nn.MaxPool2d(kernel_size=2, stride=2)

        self.bottleneck = DoubleConvNew(features * 8, features * 16)

        self.upconv4 = nn.ConvTranspose2d(features * 16, features * 8, kernel_size=2, stride=2)
        self.decoder4 = DoubleConvNew(features * 16, features * 8)
        self.upconv3 = nn.ConvTranspose2d(features * 8, features * 4, kernel_size=2, stride=2)
        self.decoder3 = DoubleConvNew(features * 8, features * 4)
        self.upconv2 = nn.ConvTranspose2d(features * 4, features * 2, kernel_size=2, stride=2)
        self.decoder2 = DoubleConvNew(features * 4, features * 2)
        self.upconv1 = nn.ConvTranspose2d(features * 2, features, kernel_size=2, stride=2)
        self.decoder1 = DoubleConvNew(features * 2, features)

        self.final_conv = nn.Conv2d(features, out_channels, kernel_size=1)

    def forward(self, x):
        enc1 = self.encoder1(x)
        enc2 = self.encoder2(self.pool1(enc1))
        enc3 = self.encoder3(self.pool2(enc2))
        enc4 = self.encoder4(self.pool3(enc3))

        bottleneck = self.bottleneck(self.pool4(enc4))

        dec4 = self.upconv4(bottleneck)
        dec4 = torch.cat((dec4, enc4), dim=1)
        dec4 = self.decoder4(dec4)
        dec3 = self.upconv3(dec4)
        dec3 = torch.cat((dec3, enc3), dim=1)
        dec3 = self.decoder3(dec3)
        dec2 = self.upconv2(dec3)
        dec2 = torch.cat((dec2, enc2), dim=1)
        dec2 = self.decoder2(dec2)
        dec1 = self.upconv1(dec2)
        dec1 = torch.cat((dec1, enc1), dim=1)
        dec1 = self.decoder1(dec1)

        return self.final_conv(dec1)

# 输入生成接口与原模型保持一致
batch_size = 8
in_channels = 8
out_channels = 4
height = 64
width = 512
features = 64

def get_inputs():
    return [torch.rand(batch_size, in_channels, height, width)]

def get_init_inputs():
    return [in_channels, out_channels, features]
