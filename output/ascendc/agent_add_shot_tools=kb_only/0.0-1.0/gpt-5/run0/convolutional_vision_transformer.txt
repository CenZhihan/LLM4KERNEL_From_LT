kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "convolutional_vision_transformer_custom_tiling_data.h"

template <typename T>
class KernelCvtTokenConcat {
public:
    __aicore__ inline KernelCvtTokenConcat() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR cls, GM_ADDR out, uint32_t batch, uint32_t embedDim)
    {
        this->batch = batch;
        this->embedDim = embedDim;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(x), static_cast<uint32_t>(batch * embedDim));
        clsGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(cls), embedDim);
        outGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(out), static_cast<uint32_t>(batch * 2 * embedDim));

        pipe.InitBuffer(inQueueX, 1, embedDim * sizeof(T));
        pipe.InitBuffer(inQueueCls, 1, embedDim * sizeof(T));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t b = 0; b < batch; ++b) {
            // Load cls once per batch and write to output at [b, 0, :]
            AscendC::LocalTensor<T> clsLocal = inQueueCls.AllocTensor<T>();
            AscendC::DataCopy(clsLocal, clsGm, embedDim);
            AscendC::DataCopy(outGm[(b * 2) * embedDim], clsLocal, embedDim);
            inQueueCls.FreeTensor(clsLocal);

            // Load x[b, :] and write to output at [b, 1, :]
            AscendC::LocalTensor<T> xLocal = inQueueX.AllocTensor<T>();
            AscendC::DataCopy(xLocal, xGm[b * embedDim], embedDim);
            AscendC::DataCopy(outGm[(b * 2 + 1) * embedDim], xLocal, embedDim);
            inQueueX.FreeTensor(xLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueCls;

    AscendC::GlobalTensor<T> xGm;
    AscendC::GlobalTensor<T> clsGm;
    AscendC::GlobalTensor<T> outGm;

    uint32_t batch = 0;
    uint32_t embedDim = 0;
};

extern "C" __global__ __aicore__ void convolutional_vision_transformer_custom(
    GM_ADDR x, GM_ADDR cls, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelCvtTokenConcat<float> op;
    op.Init(x, cls, out, tiling_data.batch, tiling_data.embedDim);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, num_classes, embed_dim=512, num_heads=8, num_layers=6, 
                 mlp_ratio=4.0, patch_size=4, in_channels=3, image_size=32):
        super(ModelNew, self).__init__()

        self.patch_size = patch_size
        self.image_size = image_size
        self.embed_dim = embed_dim

        self.conv1 = torch.nn.Conv2d(in_channels, embed_dim, kernel_size=patch_size, stride=patch_size)
        num_patches = (image_size // patch_size) ** 2
        self.linear_proj = torch.nn.Linear(embed_dim * num_patches, embed_dim)

        self.transformer_layers = torch.nn.ModuleList([
            torch.nn.TransformerEncoderLayer(
                d_model=embed_dim,
                nhead=num_heads,
                dim_feedforward=int(embed_dim * mlp_ratio),
                dropout=0.0,
                batch_first=True
            ) for _ in range(num_layers)
        ])

        self.cls_token = torch.nn.Parameter(torch.zeros(1, 1, embed_dim))
        self.fc_out = torch.nn.Linear(embed_dim, num_classes)

    def forward(self, x):
        B = x.size(0)
        x = self.conv1(x)                  # (B, E, H/ps, W/ps)
        x = x.flatten(start_dim=1)         # (B, E * num_patches)
        x = self.linear_proj(x)            # (B, E)

        # Use custom AscendC kernel to create [CLS, token] sequence -> (B, 2, E)
        cls_flat = self.cls_token.view(1, self.embed_dim).contiguous()  # (1, E)
        x = custom_ops_lib.convolutional_vision_transformer_custom(x, cls_flat)

        for layer in self.transformer_layers:
            x = layer(x)                   # (B, 2, E)

        return self.fc_out(x[:, 0])
'''
