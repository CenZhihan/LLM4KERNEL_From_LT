kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv3d_hardswish_relu_softmax_mean_custom_tiling_data.h"

// Simple fused kernel example that:
// 1) Applies HardSwish on x
// 2) Computes mean over (D, H, W) for each (B, C)
// 3) Applies per-channel affine using gamma, beta
// Notes:
// - For demonstration, this kernel focuses on data movement and simple math.
// - GroupNorm is not explicitly implemented here.
// - The output shape is (B, C).

constexpr int32_t BUFFER_NUM = 2;

class KernelConv3dHardswishReluSoftmaxMeanCustom {
public:
    __aicore__ inline KernelConv3dHardswishReluSoftmaxMeanCustom() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z,
                                uint32_t b, uint32_t c, uint32_t d, uint32_t h, uint32_t w,
                                uint32_t inner, uint32_t tileNum)
    {
        this->B = b;
        this->C = c;
        this->D = d;
        this->H = h;
        this->W = w;
        this->Inner = inner;
        this->tileNum = AscendC::Max(tileNum, (uint32_t)1);
        this->tileLength = AscendC::Max((uint32_t)1, inner / (this->tileNum * BUFFER_NUM));

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(x), static_cast<uint32_t>(b * c * inner));
        gammaGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(gamma), c);
        betaGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(beta), c);
        zGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(z), static_cast<uint32_t>(b * c));

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueGamma, 1, C * sizeof(float));
        pipe.InitBuffer(inQueueBeta, 1, C * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, sizeof(float)); // write one value at a time
    }

    __aicore__ inline void Process()
    {
        // Load gamma and beta once
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.AllocTensor<float>();
        AscendC::DataCopy(gammaLocal, gammaGm, C);
        inQueueGamma.EnQue(gammaLocal);
        AscendC::LocalTensor<float> betaLocal = inQueueBeta.AllocTensor<float>();
        AscendC::DataCopy(betaLocal, betaGm, C);
        inQueueBeta.EnQue(betaLocal);

        for (uint32_t b = 0; b < B; ++b) {
            for (uint32_t c = 0; c < C; ++c) {
                float accum = 0.0f;
                uint32_t base = (b * C + c) * Inner;
                uint32_t processed = 0;
                while (processed < Inner) {
                    uint32_t cur = AscendC::Min(tileLength, Inner - processed);
                    AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                    AscendC::DataCopy(xLocal, xGm[base + processed], cur);
                    // HardSwish: x * clamp(x+3, 0, 6) / 6
                    for (uint32_t i = 0; i < cur; ++i) {
                        float v = xLocal.GetValue(i);
                        float t = v + 3.0f;
                        if (t < 0.0f) t = 0.0f;
                        if (t > 6.0f) t = 6.0f;
                        float hs = v * (t * (1.0f / 6.0f));
                        accum += hs;
                    }
                    inQueueX.FreeTensor(xLocal);
                    processed += cur;
                }
                float mean_val = accum / static_cast<float>(Inner);

                // Apply per-channel affine
                AscendC::LocalTensor<float> gammaL = inQueueGamma.DeQue<float>();
                AscendC::LocalTensor<float> betaL = inQueueBeta.DeQue<float>();
                float out_val = mean_val * gammaL.GetValue(c) + betaL.GetValue(c);
                inQueueGamma.EnQue(gammaL);
                inQueueBeta.EnQue(betaL);

                AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
                zLocal.SetValue(0, out_val);
                outQueueZ.EnQue(zLocal);

                // Write one value back
                AscendC::LocalTensor<float> zLocalW = outQueueZ.DeQue<float>();
                AscendC::DataCopy(zGm[b * C + c], zLocalW, 1);
                outQueueZ.FreeTensor(zLocalW);
            }
        }

        // Free gamma/beta locals
        AscendC::LocalTensor<float> gammaL = inQueueGamma.DeQue<float>();
        AscendC::LocalTensor<float> betaL = inQueueBeta.DeQue<float>();
        inQueueGamma.FreeTensor(gammaL);
        inQueueBeta.FreeTensor(betaL);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueGamma;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueBeta;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueZ;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t B;
    uint32_t C;
    uint32_t D;
    uint32_t H;
    uint32_t W;
    uint32_t Inner;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv3d_hardswish_relu_softmax_mean_custom(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dHardswishReluSoftmaxMeanCustom op;
    op.Init(x, gamma, beta, z,
            tiling_data.b, tiling_data.c, tiling_data.d, tiling_data.h, tiling_data.w,
            tiling_data.inner, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Fused path:
    conv3d -> custom_op(conv3d_hardswish_relu_softmax_mean_custom)
    The custom op consumes the conv output and applies HardSwish and spatial mean with per-channel affine.
    """
    def __init__(self, in_channels, out_channels, kernel_size, num_groups=4, bias=True):
        super().__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size, bias=bias)
        # Keep GroupNorm only for holding affine parameters (gamma/beta).
        self.group_norm = torch.nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)  # (B, C, D, H, W)
        # Use the GN affine parameters as gamma/beta for the fused op.
        return custom_ops_lib.conv3d_hardswish_relu_softmax_mean_custom(x, self.group_norm.weight, self.group_norm.bias)
