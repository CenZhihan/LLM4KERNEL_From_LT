kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose2d_gelu_group_norm_custom_tiling_data.h"

// 本算子在Kernel侧融合GELU + GroupNorm
// 参考文档：Gelu、GroupNorm 高阶API与示例（Ascend C API Reference）

using dataType = float;

constexpr int32_t ONE_BLK_SIZE = 32;

class KernelConvTranspose2dGeluGroupNorm {
public:
    __aicore__ inline KernelConvTranspose2dGeluGroupNorm() {}

    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR gammaGm, GM_ADDR betaGm, GM_ADDR yGm,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w,
                                uint32_t numGroups, float epsilon) {
        this->n = n; this->c = c; this->h = h; this->w = w;
        this->numGroups = numGroups;
        this->epsilon = epsilon;
        bshw = n * c * h * w;
        gnStatLen = numGroups * n;

        xGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ dataType*>(xGm), bshw);
        gammaGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ dataType*>(gammaGm), c);
        betaGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ dataType*>(betaGm), c);
        yGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ dataType*>(yGm), bshw);

        // UB/Queue分配：对齐到32B
        uint32_t hwAlignSize = ((sizeof(dataType) * h * w + ONE_BLK_SIZE - 1) / ONE_BLK_SIZE * ONE_BLK_SIZE) / sizeof(dataType);
        this->hwAlignSize = hwAlignSize;

        pipe.InitBuffer(inQueueX, 1, sizeof(dataType) * n * c * hwAlignSize);
        pipe.InitBuffer(inQueueGamma, 1, ((uint32_t)sizeof(dataType) * c + 31) / 32 * 32);
        pipe.InitBuffer(inQueueBeta, 1, ((uint32_t)sizeof(dataType) * c + 31) / 32 * 32);
        pipe.InitBuffer(outQueueY, 1, sizeof(dataType) * n * c * hwAlignSize);
        pipe.InitBuffer(meanBuf, ((uint32_t)sizeof(dataType) * gnStatLen + 31) / 32 * 32);
        pipe.InitBuffer(varBuf, ((uint32_t)sizeof(dataType) * gnStatLen + 31) / 32 * 32);
    }

    __aicore__ inline void Process() {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn() {
        inputXLocal = inQueueX.AllocTensor<dataType>();
        gammaLocal = inQueueGamma.AllocTensor<dataType>();
        betaLocal  = inQueueBeta.AllocTensor<dataType>();

        AscendC::DataCopyParams copyParamsX{static_cast<uint16_t>(n*c), static_cast<uint16_t>(h*w*sizeof(dataType)), 0, 0};
        AscendC::DataCopyPadParams padParamsX{true, 0, static_cast<uint8_t>(hwAlignSize - h * w), 0};
        AscendC::DataCopyPad(inputXLocal, xGlobal, copyParamsX, padParamsX);

        AscendC::DataCopyParams copyParamsC{1, static_cast<uint16_t>(c*sizeof(dataType)), 0, 0};
        AscendC::DataCopyPadParams padNo{false, 0, 0, 0};
        AscendC::DataCopyPad(gammaLocal, gammaGlobal, copyParamsC, padNo);
        AscendC::DataCopyPad(betaLocal,  betaGlobal,  copyParamsC, padNo);
    }

    __aicore__ inline void Compute() {
        // GELU: 支持src/dst同一缓冲区复用
        uint32_t geluSize = n * c * hwAlignSize;
        AscendC::Gelu(inputXLocal, inputXLocal, geluSize);

        // GroupNorm: 使用高阶API，按文档构造tiling
        LocalTensor<dataType> yLocal = outQueueY.AllocTensor<dataType>();
        LocalTensor<dataType> meanLocal = meanBuf.Get<dataType>();
        LocalTensor<dataType> varLocal  = varBuf.Get<dataType>();

        // 获取stackBuffer大小并计算GroupNormTiling
        uint32_t stackBufferSize = 0;
        {
            LocalTensor<float> stackBuffer;
            bool ok = AscendC::PopStackBuffer<float, AscendC::TPosition::LCM>(stackBuffer);
            (void)ok;
            stackBufferSize = stackBuffer.GetSize();
        }

        AscendC::GroupNormTiling groupNormTiling;
        uint32_t inputShape[4] = {n, c, h, w};
        AscendC::ShapeInfo shapeInfo{ (uint8_t)4, inputShape, (uint8_t)4, inputShape, AscendC::DataFormat::ND };
        AscendC::GetGroupNormNDTillingInfo(shapeInfo, stackBufferSize, sizeof(dataType), false, numGroups, groupNormTiling);

        AscendC::GroupNorm<dataType, false>(yLocal, meanLocal, varLocal,
            inputXLocal, gammaLocal, betaLocal, (dataType)epsilon, groupNormTiling);

        outQueueY.EnQue<dataType>(yLocal);
    }

    __aicore__ inline void CopyOut() {
        LocalTensor<dataType> yLocal = outQueueY.DeQue<dataType>();
        AscendC::DataCopyParams copyParamsX{static_cast<uint16_t>(n*c), static_cast<uint16_t>(h*w*sizeof(dataType)), 0, 0};
        AscendC::DataCopyPadParams padParamsX{true, 0, static_cast<uint8_t>(hwAlignSize - h * w), 0};
        AscendC::DataCopyPad(yGlobal, yLocal, copyParamsX, padParamsX);

        inQueueX.FreeTensor(inputXLocal);
        inQueueGamma.FreeTensor(gammaLocal);
        inQueueBeta.FreeTensor(betaLocal);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  inQueueGamma;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  inQueueBeta;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> meanBuf, varBuf;

    AscendC::GlobalTensor<dataType> xGlobal;
    AscendC::GlobalTensor<dataType> gammaGlobal;
    AscendC::GlobalTensor<dataType> betaGlobal;
    AscendC::GlobalTensor<dataType> yGlobal;

    AscendC::LocalTensor<dataType> inputXLocal;
    AscendC::LocalTensor<dataType> gammaLocal;
    AscendC::LocalTensor<dataType> betaLocal;

    uint32_t n, c, h, w;
    uint32_t numGroups;
    uint32_t bshw;
    uint32_t gnStatLen;
    uint32_t hwAlignSize;
    float epsilon;
};

extern "C" __global__ __aicore__ void conv_transpose2d_gelu_group_norm_custom(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tilingData, tiling);
    KernelConvTranspose2dGeluGroupNorm op;
    op.Init(x, gamma, beta, y,
            tilingData.n, tilingData.c, tilingData.h, tilingData.w,
            tilingData.num_groups, tilingData.epsilon);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    ConvTranspose2d -> fused(GELU + GroupNorm on NPU)
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, groups, num_groups, eps: float = 1e-5):
        super().__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride)
        self.group_norm = torch.nn.GroupNorm(num_groups=num_groups, num_channels=out_channels, eps=eps)

    def forward(self, x):
        x = self.conv_transpose(x)
        # 使用NPU自定义算子在ConvTranspose2d输出上执行GELU + GroupNorm
        x = custom_ops_lib.conv_transpose2d_gelu_group_norm_custom(
            x, self.group_norm.weight, self.group_norm.bias, self.group_norm.num_groups, self.group_norm.eps
        )
        return x
