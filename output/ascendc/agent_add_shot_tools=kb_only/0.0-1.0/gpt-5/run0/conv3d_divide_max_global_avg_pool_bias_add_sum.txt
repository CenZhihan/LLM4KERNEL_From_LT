kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv3d_divide_max_global_avg_pool_bias_add_sum_custom_tiling_data.h"

// 说明：该内核完成以下计算：
// 输入x形状[N,C,D,H,W]，执行：
// 1) 对每个(N,C)在DHW维度做全局平均 -> [N,C,1,1,1]
// 2) 加上bias[C]（按通道广播到N批次）
// 3) 沿通道维求和 -> 输出[N,1,1,1]
//
// 注意：与模型中Conv3d/除法/MaxPool3d保持在PyTorch侧，
// 本自定义核从MaxPool3d之后的张量开始处理（全局平均+加bias+按通道求和）。

constexpr int32_t BUFFER_NUM = 1;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                uint32_t outterMean, uint32_t innerMean, uint32_t nMean,
                                uint32_t outterSum, uint32_t innerSum, uint32_t nSum,
                                uint32_t cLen) {
        this->outterMean = outterMean;
        this->innerMean = innerMean;
        this->nMean = nMean;

        this->outterSum = outterSum;
        this->innerSum = innerSum;
        this->nSum = nSum;

        this->cLen = cLen;

        // 全量搬运，示例代码不做分块
        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x, outterMean * innerMean);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_Y*)bias, cLen);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Z*)y, outterSum);

        // 输入x本地缓存（N*C*innerMean）
        pipe.InitBuffer(inQueueX, BUFFER_NUM, outterMean * innerMean * sizeof(DTYPE_X));
        // bias本地缓存（C）
        pipe.InitBuffer(inQueueB, BUFFER_NUM, cLen * sizeof(DTYPE_Y));
        // 中间(mean后/加bias后)与输出缓存
        // mean输出[N*C]（32B对齐到outterMean*sizeof(T)的上取整）
        uint32_t meanOutAlignedElems = (outterMean * sizeof(DTYPE_Z) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(DTYPE_Z);
        pipe.InitBuffer(midQueue, BUFFER_NUM, meanOutAlignedElems * sizeof(DTYPE_Z));
        // 最终输出[N]（32B对齐）
        uint32_t outAlignedElems = (outterSum * sizeof(DTYPE_Z) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(DTYPE_Z);
        pipe.InitBuffer(outQueueY, BUFFER_NUM, outAlignedElems * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process() {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn() {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm, outterMean * innerMean);
        inQueueX.EnQue(xLocal);

        AscendC::LocalTensor<DTYPE_Y> bLocal = inQueueB.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(bLocal, biasGm, cLen);
        inQueueB.EnQue(bLocal);
    }

    __aicore__ inline void Compute() {
        // 取回输入与bias
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> bLocal = inQueueB.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> meanLocal = midQueue.AllocTensor<DTYPE_Z>();

        // 1) 全局平均池化（按DHW）
        AscendC::MeanParams meanParams;
        meanParams.outter = outterMean;   // N*C
        meanParams.inner  = innerMean;    // align32(D*H*W)
        meanParams.n      = nMean;        // D*H*W
        // 使用框架分配临时空间的变体
        AscendC::Mean<DTYPE_Z, DTYPE_Z>(meanLocal, xLocal, meanParams);

        // 2) 加bias（对每个batch的C通道加同一个bias[C]）
        // meanLocal逻辑形状为[N, CAligned]，仅前C个元素有效
        for (uint32_t nIdx = 0; nIdx < outterSum; ++nIdx) {
            // 起始偏移：nIdx * innerSum
            uint32_t rowOffset = nIdx * innerSum;
            // 对有效的C元素执行按位加
            // 仅对前nSum=C个元素参与计算
            AscendC::Add(meanLocal[rowOffset], meanLocal[rowOffset], bLocal, nSum);
        }

        // 3) 沿通道维求和：将[N, CAligned] -> [N]
        AscendC::LocalTensor<DTYPE_Z> yLocal = outQueueY.AllocTensor<DTYPE_Z>();
        AscendC::SumParams sumParams;
        sumParams.outter = outterSum;  // N
        sumParams.inner  = innerSum;   // align32(C)
        sumParams.n      = nSum;       // C
        AscendC::Sum<DTYPE_Z>(yLocal, meanLocal, sumParams);

        // 入队输出
        outQueueY.EnQue<DTYPE_Z>(yLocal);

        // 释放
        midQueue.FreeTensor(meanLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueB.FreeTensor(bLocal);
    }

    __aicore__ inline void CopyOut() {
        AscendC::LocalTensor<DTYPE_Z> yLocal = outQueueY.DeQue<DTYPE_Z>();
        // 拷贝N个元素（32B对齐长度）
        uint32_t outAlignedElems = (outterSum * sizeof(DTYPE_Z) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(DTYPE_Z);
        AscendC::DataCopy(yGm, yLocal, outAlignedElems);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueB;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> midQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> biasGm;
    AscendC::GlobalTensor<DTYPE_Z> yGm;

    uint32_t outterMean;
    uint32_t innerMean;
    uint32_t nMean;

    uint32_t outterSum;
    uint32_t innerSum;
    uint32_t nSum;

    uint32_t cLen;
};

extern "C" __global__ __aicore__ void conv3d_divide_max_global_avg_pool_bias_add_sum_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, y,
            tiling_data.outterMean, tiling_data.innerMean, tiling_data.nMean,
            tiling_data.outterSum, tiling_data.innerSum, tiling_data.nSum,
            tiling_data.cLen);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    保持Conv3d/除法/MaxPool3d在PyTorch侧执行，
    使用AscendC自定义算子完成：全局平均池化(到1x1x1) + 加bias + 沿通道求和。
    """
    def __init__(self, in_channels, out_channels, kernel_size, divisor, pool_size, bias_shape, sum_dim):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.divisor = divisor
        self.max_pool = torch.nn.MaxPool3d(pool_size)
        # 自定义核内部会做GAP + bias add + channel sum
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.sum_dim = sum_dim  # 期望为1（通道维）

    def forward(self, x):
        x = self.conv(x)
        x = x / self.divisor
        x = self.max_pool(x)
        # bias形状[C,1,1,1]，展开为[C]传入算子
        y = custom_ops_lib.conv3d_divide_max_global_avg_pool_bias_add_sum_custom(x, self.bias.view(-1))
