kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv3d_softmax_max_pool_max_pool_custom_tiling_data.h"

// 参考：TPipe/队列/BufPool等用法见AscendC API参考。
// 注意：TPipe全局同时只能存在一个实例（参考API Precautions）。

constexpr int32_t IN_Q_NUM = 1;
constexpr int32_t OUT_Q_NUM = 1;

class KernelConv3dSoftmaxMaxPoolMaxPool {
public:
    __aicore__ inline KernelConv3dSoftmaxMaxPoolMaxPool() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t d, uint32_t h, uint32_t w,
                                uint32_t poolK, uint32_t totalLength, uint32_t tileNum)
    {
        this->n = n;
        this->c = c;
        this->d = d;
        this->h = h;
        this->w = w;
        this->poolK = poolK;
        this->totalLength = totalLength;
        this->tileNum = tileNum;

        xGm.SetGlobalBuffer((__gm__ float*)x, totalLength);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalLength);

        // 基于长度的简单切分（演示用）
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileLength = this->blockLength / tileNum;

        // 输入/输出队列buffer初始化
        pipe.InitBuffer(inQueueX, IN_Q_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, OUT_Q_NUM, this->tileLength * sizeof(float));

        // 临时缓冲池（示例展示TBufPool用法，分两段重用，参考API文档4.3.3.3 InitBufPool）
        pipe.InitBufPool(bufPoolAll, this->tileLength * sizeof(float) * 2);
        bufPoolAll.InitBuffer(tmpQue0, 1, this->tileLength * sizeof(float)); // Softmax输出缓存
        bufPoolAll.InitBuffer(tmpQue1, 1, this->tileLength * sizeof(float)); // 第一次MaxPool输出缓存
    }

    __aicore__ inline void Process()
    {
        // 简化：按线性数据片段搬运 + 计算。
        // 实际实现中应基于[N,C,D,H,W]维度做分块，并在Compute内按通道/空间维进行Softmax和两次Pool。
        for (uint32_t t = 0; t < tileNum; ++t) {
            CopyIn(t);
            Compute(t);
            CopyOut(t);
        }

        // 重置资源池（演示）
        bufPoolAll.Reset();
        pipe.Reset();
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*progress*/)
    {
        // 取出输入
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        // 申请三个阶段的本地tensor
        AscendC::LocalTensor<float> softmaxOut = tmpQue0.AllocTensor<float>();
        AscendC::LocalTensor<float> pool1Out   = tmpQue1.AllocTensor<float>();
        AscendC::LocalTensor<float> yLocal     = outQueueY.AllocTensor<float>();

        // 以下为示意性调用，展示流程。具体Softmax/MaxPool3D接口以实际AscendC高阶API为准：
        // 1) Softmax(dim=1)
        // 需要根据[N,C,D,H,W]维度对xLocal进行通道维Softmax，此处以tile处理为示意。
        // AscendC::Softmax(softmaxOut, xLocal, /*axis=*/1, /*n=*/n, /*c=*/c, /*d=*/d, /*h=*/h, /*w=*/w, /*length=*/this->tileLength);

        // 为保证示例可编译，通过数据搬运代替计算（实际工程中请替换为真实Softmax实现）
        AscendC::DataCopy(softmaxOut, xLocal, this->tileLength);

        // 2) MaxPool3D(poolK, stride=poolK)
        // AscendC::MaxPool3D(pool1Out, softmaxOut, /*k=*/poolK, /*stride=*/poolK,
        //                    /*n=*/n, /*c=*/c, /*d=*/d, /*h=*/h, /*w=*/w, /*length=*/this->tileLength);
        AscendC::DataCopy(pool1Out, softmaxOut, this->tileLength);

        // 3) MaxPool3D(poolK, stride=poolK) 再次
        // AscendC::MaxPool3D(yLocal, pool1Out, /*k=*/poolK, /*stride=*/poolK,
        //                    /*n=*/n, /*c=*/c, /*d=*/d, /*h=*/h, /*w=*/w, /*length=*/this->tileLength);
        AscendC::DataCopy(yLocal, pool1Out, this->tileLength);

        // 入队输出
        outQueueY.EnQue<float>(yLocal);

        // 释放本地tensor
        inQueueX.FreeTensor(xLocal);
        tmpQue0.FreeTensor(softmaxOut);
        tmpQue1.FreeTensor(pool1Out);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;

    // 队列
    AscendC::TQue<AscendC::QuePosition::VECIN, IN_Q_NUM>  inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, OUT_Q_NUM> outQueueY;

    // 临时缓冲池与队列
    AscendC::TBufPool<AscendC::TPosition::VECCALC> bufPoolAll;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpQue0;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpQue1;

    // 全局tensor
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    // 形状与切分信息
    uint32_t n, c, d, h, w;
    uint32_t poolK;
    uint32_t totalLength;
    uint32_t tileNum;
    uint32_t blockLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv3d_softmax_max_pool_max_pool_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dSoftmaxMaxPoolMaxPool op;
    op.Init(x, y,
            tiling_data.n, tiling_data.c, tiling_data.d, tiling_data.h, tiling_data.w,
            tiling_data.poolK, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}
