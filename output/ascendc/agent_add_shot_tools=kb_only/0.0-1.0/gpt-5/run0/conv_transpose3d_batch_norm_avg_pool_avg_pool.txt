#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose3d_batch_norm_avg_pool_avg_pool_custom_tiling_data.h"

// 参考文档：
// - 4.1.1.9 Template Sample：队列/拷入/计算/拷出框架
// - 5.1.28.1 Xor 示例：多输入队列与 Tiling 的使用
// 本示例提供算子融合的内核骨架（演示数据搬运与流水），计算处可按需替换为实际反卷积+BN+AvgPool。

constexpr int32_t BUFFER_NUM = 2;

using DTYPE_T = float;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x,
                                GM_ADDR w,
                                GM_ADDR b_conv,
                                GM_ADDR bn_gamma,
                                GM_ADDR bn_beta,
                                GM_ADDR bn_running_mean,
                                GM_ADDR bn_running_var,
                                GM_ADDR z,
                                uint32_t totalLength,
                                uint32_t tilenum)
    {
        this->totalLength = totalLength;
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tilenum = tilenum;
        this->tileLength = this->blockLength / tilenum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE_T*)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        wGm.SetGlobalBuffer((__gm__ DTYPE_T*)w, 0); // 参数张量按需读取，本例不分块
        bGm.SetGlobalBuffer((__gm__ DTYPE_T*)b_conv, 0);
        gammaGm.SetGlobalBuffer((__gm__ DTYPE_T*)bn_gamma, 0);
        betaGm.SetGlobalBuffer((__gm__ DTYPE_T*)bn_beta, 0);
        meanGm.SetGlobalBuffer((__gm__ DTYPE_T*)bn_running_mean, 0);
        varGm.SetGlobalBuffer((__gm__ DTYPE_T*)bn_running_var, 0);
        zGm.SetGlobalBuffer((__gm__ DTYPE_T*)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_T));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_T));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tilenum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_T> xLocal = inQueueX.AllocTensor<DTYPE_T>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);

        // 需要时，可在此将参数分块搬入 UB（本例省略）
    }

    __aicore__ inline void Compute(int32_t)
    {
        AscendC::LocalTensor<DTYPE_T> xLocal = inQueueX.DeQue<DTYPE_T>();
        AscendC::LocalTensor<DTYPE_T> zLocal = outQueueZ.AllocTensor<DTYPE_T>();

        // 占位实现：直接将输入写出。
        // 实际可在此实现：
        // 1) ConvTranspose3d（反卷积）
        // 2) BatchNorm（可参考 5.4.5 BatchNorm 示例的调用方式）
        // 3) AvgPool3d 两次（可采用拆分/规约实现）
        AscendC::DataCopy(zLocal, xLocal, this->tileLength);

        outQueueZ.EnQue<DTYPE_T>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_T> zLocal = outQueueZ.DeQue<DTYPE_T>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::GlobalTensor<DTYPE_T> xGm;
    AscendC::GlobalTensor<DTYPE_T> wGm;
    AscendC::GlobalTensor<DTYPE_T> bGm;
    AscendC::GlobalTensor<DTYPE_T> gammaGm;
    AscendC::GlobalTensor<DTYPE_T> betaGm;
    AscendC::GlobalTensor<DTYPE_T> meanGm;
    AscendC::GlobalTensor<DTYPE_T> varGm;
    AscendC::GlobalTensor<DTYPE_T> zGm;

    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tilenum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_avg_pool_avg_pool_custom(
    GM_ADDR x,
    GM_ADDR w,
    GM_ADDR b_conv,
    GM_ADDR bn_gamma,
    GM_ADDR bn_beta,
    GM_ADDR bn_running_mean,
    GM_ADDR bn_running_var,
    GM_ADDR z,
    GM_ADDR workspace,
    GM_ADDR tiling)
{

    KernelFused op;
    op.Init(x, w, b_conv, bn_gamma, bn_beta, bn_running_mean, bn_running_var,
            z, tilingData.totalLength, tilingData.tilenum);
    op.Process();
}

        super().__init__()
        # 权重与 BN 参数初始化（与原模型参数对应）
        if isinstance(kernel_size, int):
            kd = kh = kw = kernel_size
        else:
            kd, kh, kw = kernel_size
        # PyTorch ConvTranspose3d 权重形状：(in_channels, out_channels, kD, kH, kW)
        self.weight = torch.nn.Parameter(torch.randn(in_channels, out_channels, kd, kh, kw, dtype=torch.float32))

        self.bn_running_var = torch.nn.Parameter(torch.ones(out_channels, dtype=torch.float32), requires_grad=False)

        return custom_ops_lib.conv_transpose3d_batch_norm_avg_pool_avg_pool_custom(
