kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transposed_2d_square_input_square_kernel_custom_tiling_data.h"

// 仅支持 float, ND，stride=1, padding=0, output_padding=0, groups=1
// 参考文档：LocalTensor 管理、DataCopy、TQue/TPipe的使用（CANN Ascend C API Reference）

using DTYPE = float;
constexpr int32_t IN_BUF_NUM = 1;
constexpr int32_t OUT_BUF_NUM = 1;

class KernelConvTransposed2D {
public:
    __aicore__ inline KernelConvTransposed2D() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y,
                                uint32_t N, uint32_t Cin, uint32_t Co,
                                uint32_t H, uint32_t W, uint32_t K,
                                uint32_t OH, uint32_t OW,
                                uint32_t tileOH, uint32_t tileOW)
    {
        this->N_ = N; this->Cin_ = Cin; this->Co_ = Co;
        this->H_ = H; this->W_ = W; this->K_ = K;
        this->OH_ = OH; this->OW_ = OW;
        this->tileOH_ = tileOH; this->tileOW_ = tileOW;

        xGm_.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(x), (uint64_t)N_*Cin_*H_*W_);
        wGm_.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(w), (uint64_t)Cin_*Co_*K_*K_);
        yGm_.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(y), (uint64_t)N_*Co_*OH_*OW_);

        // 最大patch: (tileOH + K - 1) * (tileOW + K - 1)
        maxPatchH_ = tileOH_ + K_ - 1;
        maxPatchW_ = tileOW_ + K_ - 1;
        maxPatchSize_ = maxPatchH_ * maxPatchW_;
        maxYTileSize_ = tileOH_ * tileOW_;

        // 分配本地buffer
        pipe_.InitBuffer(inQueueX_, IN_BUF_NUM, maxPatchSize_ * sizeof(DTYPE));
        pipe_.InitBuffer(inQueueW_, IN_BUF_NUM, K_ * K_ * sizeof(DTYPE));
        pipe_.InitBuffer(outQueueY_, OUT_BUF_NUM, maxYTileSize_ * sizeof(DTYPE));

        // 仅单Block执行，避免输出写冲突
        // 如果需要并行，需要对输出空间做无重叠分块
    }

    __aicore__ inline void Process()
    {
        if (AscendC::GetBlockIdx() != 0) {
            return;
        }
        for (uint32_t n = 0; n < N_; ++n) {
            for (uint32_t co = 0; co < Co_; ++co) {
                for (uint32_t oh0 = 0; oh0 < OH_; oh0 += tileOH_) {
                    for (uint32_t ow0 = 0; ow0 < OW_; ow0 += tileOW_) {
                        uint32_t curTileH = AscendC::Min(tileOH_, OH_ - oh0);
                        uint32_t curTileW = AscendC::Min(tileOW_, OW_ - ow0);

                        // 分配y tile并清零
                        AscendC::LocalTensor<DTYPE> yTile = outQueueY_.AllocTensor<DTYPE>();
                        for (uint32_t i = 0; i < curTileH * curTileW; ++i) {
                            yTile[i] = static_cast<DTYPE>(0.0f);
                        }

                        // 遍历输入通道，累加
                        for (uint32_t ci = 0; ci < Cin_; ++ci) {
                            // 拷贝权重 [ci, co, :, :]
                            AscendC::LocalTensor<DTYPE> wLocal = inQueueW_.AllocTensor<DTYPE>();
                            uint64_t wBase = ((uint64_t)ci * Co_ + co) * (uint64_t)K_ * (uint64_t)K_;
                            AscendC::DataCopy(wLocal, wGm_[wBase], K_ * K_);
                            inQueueW_.EnQue(wLocal);
                            wLocal = inQueueW_.DeQue<DTYPE>();

                            // 针对该输出tile需要的输入patch范围
                            // 输出oh in [oh0, oh0+curTileH-1], ow in [ow0, ow0+curTileW-1]
                            // 对应输入位置ih = oh - kh, iw = ow - kw
                            int32_t ihStart = static_cast<int32_t>(oh0) - static_cast<int32_t>(K_) + 1;
                            int32_t iwStart = static_cast<int32_t>(ow0) - static_cast<int32_t>(K_) + 1;
                            int32_t ihEnd   = static_cast<int32_t>(oh0 + curTileH - 1);
                            int32_t iwEnd   = static_cast<int32_t>(ow0 + curTileW - 1);

                            // 有效输入边界裁剪到 [0, H_-1]/[0, W_-1]
                            int32_t ihCopyBeg = AscendC::Max(0, ihStart);
                            int32_t iwCopyBeg = AscendC::Max(0, iwStart);
                            int32_t ihCopyEnd = AscendC::Min((int32_t)H_ - 1, ihEnd);
                            int32_t iwCopyEnd = AscendC::Min((int32_t)W_ - 1, iwEnd);

                            uint32_t patchH = (ihCopyEnd >= ihCopyBeg) ? (ihCopyEnd - ihCopyBeg + 1) : 0;
                            uint32_t patchW = (iwCopyEnd >= iwCopyBeg) ? (iwCopyEnd - iwCopyBeg + 1) : 0;

                            AscendC::LocalTensor<DTYPE> xPatch = inQueueX_.AllocTensor<DTYPE>();
                            // 先清零patch，方便边界外当0处理
                            for (uint32_t i = 0; i < maxPatchSize_; ++i) {
                                xPatch[i] = static_cast<DTYPE>(0.0f);
                            }

                            // 将有效区域复制到patch对应偏移
                            if (patchH > 0 && patchW > 0) {
                                uint32_t rowOffset = (uint32_t)(ihCopyBeg - ihStart); // patch中起始行偏移
                                uint32_t colOffset = (uint32_t)(iwCopyBeg - iwStart); // patch中起始列偏移
                                for (uint32_t r = 0; r < patchH; ++r) {
                                    uint64_t xRowBase = ((((uint64_t)n * Cin_ + ci) * H_) + (uint64_t)(ihCopyBeg + (int32_t)r)) * (uint64_t)W_;
                                    AscendC::DataCopy(
                                        xPatch[(rowOffset + r) * maxPatchW_ + colOffset],
                                        xGm_[xRowBase + (uint64_t)iwCopyBeg],
                                        patchW
                                    );
                                }
                            }
                            inQueueX_.EnQue(xPatch);
                            xPatch = inQueueX_.DeQue<DTYPE>();

                            // 计算：yTile += convT(xPatch, wLocal)（stride=1, padding=0）
                            // 对于输出tile内每个像素 (th, tw)
                            for (uint32_t th = 0; th < curTileH; ++th) {
                                for (uint32_t tw = 0; tw < curTileW; ++tw) {
                                    DTYPE acc = 0.0f;
                                    // 卷积反卷积对应：ih = th - (K-1) + kh  -> 已经将x按补零方式放入patch
                                    // 直接遍历权重并从patch读取
                                    for (uint32_t kh = 0; kh < K_; ++kh) {
                                        for (uint32_t kw = 0; kw < K_; ++kw) {
                                            uint32_t pr = th + kh; // patch中的行位置
                                            uint32_t pc = tw + kw; // patch中的列位置
                                            DTYPE xv = xPatch[pr * maxPatchW_ + pc];
                                            DTYPE wv = wLocal[kh * K_ + kw];
                                            acc += xv * wv;
                                        }
                                    }
                                    yTile[th * curTileW + tw] += acc;
                                }
                            }

                            inQueueW_.FreeTensor(wLocal);
                            inQueueX_.FreeTensor(xPatch);
                        } // for ci

                        // 写回y tile -> GM
                        for (uint32_t th = 0; th < curTileH; ++th) {
                            uint64_t yRowBase = ((((uint64_t)n * Co_) + co) * OH_ + (oh0 + th)) * (uint64_t)OW_;
                            AscendC::DataCopy(yGm_[yRowBase + ow0], yTile[th * curTileW], curTileW);
                        }
                        outQueueY_.FreeTensor(yTile);
                    }
                }
            }
        }
    }

private:
    AscendC::TPipe pipe_;
    AscendC::TQue<AscendC::QuePosition::VECIN, IN_BUF_NUM>  inQueueX_, inQueueW_;
    AscendC::TQue<AscendC::QuePosition::VECOUT, OUT_BUF_NUM> outQueueY_;
    AscendC::GlobalTensor<DTYPE> xGm_, wGm_, yGm_;

    uint32_t N_, Cin_, Co_;
    uint32_t H_, W_, K_;
    uint32_t OH_, OW_;
    uint32_t tileOH_, tileOW_;
    uint32_t maxPatchH_, maxPatchW_, maxPatchSize_, maxYTileSize_;
};

extern "C" __global__ __aicore__ void conv_transposed_2d_square_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTransposed2D op;
    op.Init(x, weight, y,
            tiling_data.N, tiling_data.Cin, tiling_data.Co,
            tiling_data.H, tiling_data.W, tiling_data.K,
            tiling_data.OH, tiling_data.OW,
            tiling_data.tileOH, tiling_data.tileOW);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用AscendC自定义算子替换 ConvTranspose2d：
    约束：
      - 仅支持 float32
      - 输入为 NCHW，方形输入 (H==W) 与方形kernel (K==K)
      - stride=1, padding=0, output_padding=0
      - groups=1
      - bias=False（如需bias，可在外部额外加到输出上）
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, stride: int = 1, padding: int = 0, output_padding: int = 0, groups: int = 1, bias: bool = False):
        super().__init__()
        # 仅用于权重和参数校验/管理
        assert stride == 1 and padding == 0 and output_padding == 0 and groups == 1, "Only stride=1, padding=0, output_padding=0, groups=1 are supported"
        assert not bias, "bias=False is required"
        assert isinstance(kernel_size, int), "Only square kernel size supported"
        self.conv_transpose2d = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding, groups=groups, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        N, C, H, W = x.shape
        K = self.conv_transpose2d.kernel_size[0]
        assert H == W, "Only square input supported"
        assert self.conv_transpose2d.kernel_size[0] == self.conv_transpose2d.kernel_size[1], "Only square kernel supported"
        # 权重布局为 [Cin, Co, K, K]
        weight = self.conv_transpose2d.weight
        y = custom_ops_lib.conv_transposed_2d_square_input_square_kernel_custom(x, weight)
        return y
