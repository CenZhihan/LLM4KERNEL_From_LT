kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"
#include "op_tiling/tiling.h"
#include <cstring>

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t TILE_NUM = 8;

using A_T = half;   // x
using B_T = half;   // weight
using C_T = float;  // y (and bias type)
using BiasT = float;

class KernelMatmulSwishScaling {
public:
    __aicore__ inline KernelMatmulSwishScaling() {}

    __aicore__ inline void Init(GM_ADDR aGM, GM_ADDR bGM, GM_ADDR biasGM, GM_ADDR scaleGM,
                                GM_ADDR cGM, GM_ADDR tilingGM, GM_ADDR workspaceGM)
    {
        // Rebuild TCubeTiling from tilingGM (as per Matmul example in API reference)
        auto tempTilingGM = reinterpret_cast<__gm__ uint32_t*>(tilingGM);
        auto tempTiling = reinterpret_cast<uint32_t*>(&tiling_);
        for (int i = 0; i < (int)(sizeof(TCubeTiling) / sizeof(uint32_t)); ++i, ++tempTilingGM, ++tempTiling) {
            *tempTiling = *tempTilingGM;
        }

        // Init GM tensors for full-core chunk
        aGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(aGM), 
                                 tiling_.ALayoutInfoB * tiling_.singleCoreM * tiling_.singleCoreK * sizeof(A_T));
        bGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(bGM), 
                                 tiling_.BLayoutInfoB * tiling_.singleCoreK * tiling_.singleCoreN * sizeof(B_T));
        cGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(cGM), 
                                 tiling_.CLayoutInfoB * tiling_.singleCoreM * tiling_.singleCoreN * sizeof(C_T));
        biasGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ BiasT*>(biasGM), 
                                    tiling_.CLayoutInfoB * tiling_.singleCoreN * sizeof(BiasT));
        scaleGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(scaleGM), sizeof(float));

        // Per-core views
        int offset_a = 0, offset_b = 0, offset_c = 0, offset_bias = 0;
        gm_a_.SetGlobalBuffer(const_cast<__gm__ A_T*>(aGlobal_[offset_a].GetPhyAddr()),
                              tiling_.singleCoreM * tiling_.singleCoreK);
        gm_b_.SetGlobalBuffer(const_cast<__gm__ B_T*>(bGlobal_[offset_b].GetPhyAddr()),
                              tiling_.singleCoreK * tiling_.singleCoreN);
        gm_c_.SetGlobalBuffer(const_cast<__gm__ C_T*>(cGlobal_[offset_c].GetPhyAddr()),
                              tiling_.singleCoreM * tiling_.singleCoreN);
        gm_bias_.SetGlobalBuffer(const_cast<__gm__ BiasT*>(biasGlobal_[offset_bias].GetPhyAddr()),
                                 tiling_.singleCoreN);

        // Workspace pointer (size 0 set on host)
        workspaceGM_ = workspaceGM;

        // Prepare UB buffers for post-processing
        blockLength_ = tiling_.singleCoreM * tiling_.singleCoreN;
        tileLength_ = blockLength_ / (TILE_NUM * BUFFER_NUM);
        if (tileLength_ == 0 && blockLength_ > 0) {
            tileLength_ = blockLength_; // small blocks
        }

        pipe_.InitBuffer(inQueue_, BUFFER_NUM, tileLength_ * sizeof(C_T));
        pipe_.InitBuffer(outQueue_, BUFFER_NUM, tileLength_ * sizeof(C_T));
        pipe_.InitBuffer(scaleQueue_, 1, sizeof(float)); // for scalar scale
    }

    __aicore__ inline void Process()
    {
        // Configure and run Matmul (reference: REGIST_MATMUL_OBJ and Init usage)
        constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);
        matmul::Matmul<
            matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::NORMAL>,
            matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, B_T, true,  LayoutMode::NORMAL>,
            matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::NORMAL>,
            matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, BiasT>
        , MM_CFG> mm;

        REGIST_MATMUL_OBJ(&pipe_, GetSysWorkSpacePtr(), mm);
        // Share sizes setting as example suggests
        tiling_.shareMode = 0;
        tiling_.shareL1Size = 512 * 1024;
        tiling_.shareL0CSize = 128 * 1024;
        tiling_.shareUbSize = 0;

        mm.Init(&tiling_);
        mm.SetTensorA(gm_a_, 0);   // not transposed
        mm.SetTensorB(gm_b_, 1);   // weight [N,K], use transpose flag
        mm.SetWorkspace(workspaceGM_, 0);
        if (tiling_.isBias) {
            mm.SetBias(gm_bias_);
        }
        // Non-batch compute
        mm.IterateBatch(gm_c_, 1, 1, false);

        // Load scaling factor (single float) into UB
        AscendC::LocalTensor<float> scaleLocal = scaleQueue_.AllocTensor<float>();
        AscendC::DataCopy(scaleLocal, scaleGlobal_, 1);
        scaleQueue_.EnQue(scaleLocal);

        // Swish + scaling on C in-place: y = swish(C) * scale
        int32_t loopCount = (tileLength_ == 0) ? 0 : (blockLength_ / tileLength_);
        int32_t tail = (tileLength_ == 0) ? 0 : (blockLength_ - loopCount * tileLength_);

        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i, tileLength_);
            Compute(i, tileLength_);
            CopyOut(i, tileLength_);
        }
        if (tail > 0) {
            CopyIn(loopCount, tail);
            Compute(loopCount, tail);
            CopyOut(loopCount, tail);
        }

        // free scale local
        scaleLocal = scaleQueue_.DeQue<float>();
        scaleQueue_.FreeTensor(scaleLocal);
    }

private:
    __aicore__ inline void CopyIn(int32_t idx, uint32_t len)
    {
        AscendC::LocalTensor<C_T> cLocal = inQueue_.AllocTensor<C_T>();
        AscendC::DataCopy(cLocal, gm_c_[idx * len], len);
        inQueue_.EnQue(cLocal);
    }

    __aicore__ inline void Compute(int32_t /*idx*/, uint32_t len)
    {
        AscendC::LocalTensor<C_T> inLocal = inQueue_.DeQue<C_T>();
        AscendC::LocalTensor<C_T> outLocal = outQueue_.AllocTensor<C_T>();

        // Swish with beta=1.0f: y = x * sigmoid(x)
        Swish(outLocal, inLocal, len, static_cast<C_T>(1.0f));

        // Multiply by scalar scale
        AscendC::LocalTensor<float> scaleLocal = scaleQueue_.DeQue<float>();
        float s = scaleLocal.GetValue(0);
        scaleQueue_.EnQue(scaleLocal);
        AscendC::Muls(outLocal, outLocal, s, len);

        outQueue_.EnQue<C_T>(outLocal);
        inQueue_.FreeTensor(inLocal);
    }

    __aicore__ inline void CopyOut(int32_t idx, uint32_t len)
    {
        AscendC::LocalTensor<C_T> outLocal = outQueue_.DeQue<C_T>();
        AscendC::DataCopy(gm_c_[idx * len], outLocal, len);
        outQueue_.FreeTensor(outLocal);
    }

private:
    // GM tensors (full, then per-core views)
    AscendC::GlobalTensor<A_T> aGlobal_;
    AscendC::GlobalTensor<B_T> bGlobal_;
    AscendC::GlobalTensor<C_T> cGlobal_;
    AscendC::GlobalTensor<BiasT> biasGlobal_;
    AscendC::GlobalTensor<float> scaleGlobal_;

    AscendC::GlobalTensor<A_T> gm_a_;
    AscendC::GlobalTensor<B_T> gm_b_;
    AscendC::GlobalTensor<C_T> gm_c_;
    AscendC::GlobalTensor<BiasT> gm_bias_;

    AscendC::TPipe pipe_;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueue_;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue_;
    AscendC::TQue<AscendC::TPosition::VECIN,  1>          scaleQueue_;

    GM_ADDR workspaceGM_ = nullptr;
    TCubeTiling tiling_;
    uint32_t blockLength_ = 0;
    uint32_t tileLength_ = 0;
};

extern "C" __global__ __aicore__ void matmul_swish_scaling_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
                                                                  GM_ADDR scale, GM_ADDR y,
                                                                  GM_ADDR workspace, GM_ADDR tiling)
{
    KernelMatmulSwishScaling op;
    op.Init(x, weight, bias, scale, y, tiling, workspace);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model using custom AscendC kernel:
      y = (x @ W^T + b) -> Swish -> scale
      Swish(x) = x * sigmoid(x), with beta = 1.0
    Input x and weight are computed in FP16; bias/output in FP32.
    """
    def __init__(self, in_features, out_features, scaling_factor):
        super().__init__()
        self.matmul = torch.nn.Linear(in_features, out_features)
        self.scaling_factor = float(scaling_factor)

    def forward(self, x):
        # Prepare tensors for custom kernel
        x_half = x.to(torch.float16)
        w_half = self.matmul.weight.to(torch.float16)  # [N, K]
        b_float = self.matmul.bias.to(torch.float32)   # [N]
        scale_tensor = torch.tensor(self.scaling_factor, device=x.device, dtype=torch.float32)

        y = custom_ops_lib.matmul_swish_scaling_custom(x_half, w_half, b_float, scale_tensor)
        return y.to(x.dtype)
