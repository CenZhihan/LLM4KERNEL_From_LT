host_operator_src="""
#include "four_dim_tensor_matrix_multiplication_custom_tiling.h"
#include "register/op_def_registry.h"

namespace {
inline uint64_t GetDimSafe(const gert::Shape& s, size_t idx) {
    if (idx < s.GetDimNum()) {
        return static_cast<uint64_t>(s.GetDim(idx));
    }
    return 1ULL;
}
}

namespace optiling {
const uint32_t BLOCK_DIM = 8;

// Tiling：将(b,i,j,l) x (l,k) 转换为 GEMM 维度 M=b*i*j, K=l, N=k
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    FourDimTensorMatrixMultiplicationCustomTilingData tiling;

    const gert::Shape& aShape = context->GetInputShape(0)->GetOriginShape();
    const gert::Shape& bShape = context->GetInputShape(1)->GetOriginShape();

    uint64_t b = GetDimSafe(aShape, 0);
    uint64_t i = GetDimSafe(aShape, 1);
    uint64_t j = GetDimSafe(aShape, 2);
    uint64_t l = GetDimSafe(aShape, 3);

    uint64_t kdim = GetDimSafe(bShape, 1);

    uint64_t M = b * i * j;
    uint64_t N = kdim;
    uint64_t K = l;

    context->SetBlockDim(BLOCK_DIM);

    tiling.set_m(static_cast<uint32_t>(M));
    tiling.set_n(static_cast<uint32_t>(N));
    tiling.set_k(static_cast<uint32_t>(K));

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
// 推理输出形状：输入A为(b,i,j,l)，输入B为(l,k)，输出为(b,i,j,k)
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* a_shape = context->GetInputShape(0);
    const gert::Shape* b_shape = context->GetInputShape(1);
    if (a_shape == nullptr || b_shape == nullptr) {
        return ge::GRAPH_FAILED;
    }
    gert::Shape* y_shape = context->GetOutputShape(0);
    uint64_t b = a_shape->GetOriginShape().GetDim(0);
    uint64_t i = a_shape->GetOriginShape().GetDim(1);
    uint64_t j = a_shape->GetOriginShape().GetDim(2);
    uint64_t k = b_shape->GetOriginShape().GetDim(1);
    y_shape->SetDimNum(4);
    y_shape->SetDim(0, static_cast<int64_t>(b));
    y_shape->SetDim(1, static_cast<int64_t>(i));
    y_shape->SetDim(2, static_cast<int64_t>(j));
    y_shape->SetDim(3, static_cast<int64_t>(k));
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {

public:
    explicit FourDimTensorMatrixMultiplicationCustom(const char* name) : OpDef(name)
    {
        this->Input("A")

OP_ADD(FourDimTensorMatrixMultiplicationCustom);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "four_dim_tensor_matrix_multiplication_custom_tiling_data.h"

// 该示例以内核端简化实现为主：先将(b,i,j,l)和(l,k)展平为GEMM [M,K]x[K,N]，再分块到不同block计算。
// 实际性能优化可进一步改造为AscendC高阶Matmul API配合TCubeTiling（参考文档 5.2.1.11 REGIST_MATMUL_OBJ 与 5.2.1.12 Init）。

extern "C" __global__ __aicore__ void four_dim_tensor_matrix_multiplication_custom(GM_ADDR A, GM_ADDR B, GM_ADDR C, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tilingData, tiling);

    const uint32_t M = tilingData.m(); // b*i*j
    const uint32_t N = tilingData.n(); // k
    const uint32_t K = tilingData.k(); // l

    // 全局内存指针
    __gm__ float* aGm = (__gm__ float*)A;  // [M, K]
    __gm__ float* bGm = (__gm__ float*)B;  // [K, N]
    __gm__ float* cGm = (__gm__ float*)C;  // [M, N]

    uint32_t blockNum = AscendC::GetBlockNum();
    uint32_t blockIdx = AscendC::GetBlockIdx();
    uint32_t rowsPerBlock = (M + blockNum - 1) / blockNum; // 向上取整分配行
    uint32_t rowStart = blockIdx * rowsPerBlock;
    uint32_t rowEnd = AscendC::Min(rowStart + rowsPerBlock, M);

    // 朴素三重循环：C = A x B
    for (uint32_t m = rowStart; m < rowEnd; ++m) {
        // C[m, 0:N] 初始化为0
        for (uint32_t n = 0; n < N; ++n) {
            cGm[m * N + n] = 0.0f;
        }
        // 累加 K 维度
        for (uint32_t k = 0; k < K; ++k) {
            float aVal = aGm[m * K + k];
            const __gm__ float* bRow = bGm + k * N;
            for (uint32_t n = 0; n < N; ++n) {
                cGm[m * N + n] += aVal * bRow[n];
            }
        }
    }
}
