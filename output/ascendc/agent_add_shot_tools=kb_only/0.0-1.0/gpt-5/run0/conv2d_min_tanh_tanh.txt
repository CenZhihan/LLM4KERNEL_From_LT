kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_min_tanh_tanh_custom_tiling_data.h"

// 说明：实现对输入[N, C, H, W]在通道维度(C)做逐元素最小值归约，随后两次Tanh。
// 计算流程：按样本N逐个处理；对每个样本，将HW平面切成tile；每个tile上遍历所有通道，使用元素级Min累积最小值；最后对最小值执行两次Tanh并写回。
// 使用到的API依据参考文档：
// - DataCopy：数据搬移
// - Min（两操作数元素级最小）：表1-2 Two-Operand Instructions
// - Tanh（高阶数学库）：表1-10 Math library APIs
// - TPipe/TQue/Data structures：用于本地缓冲与队列管理
// - GetBlockNum/GetBlockIdx：系统变量访问API

constexpr int32_t BUFFER_NUM = 2; // 简单双缓冲示例

class KernelConv2dMinTanhTanh {
public:
    __aicore__ inline KernelConv2dMinTanhTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t nVal, uint32_t cVal, uint32_t hwVal, uint32_t tileNum)
    {
        this->N = nVal;
        this->C = cVal;
        this->HW = hwVal;
        this->tileNum = tileNum;

        // 每个样本的HW平面按照 tileNum*BUFFER_NUM 等份上限切分
        uint32_t loops = this->tileNum * BUFFER_NUM;
        this->tileLen = (this->HW + loops - 1) / loops;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z*)z);

        // 分配本地缓冲
        pipe.InitBuffer(workQueue, BUFFER_NUM, this->tileLen * sizeof(DTYPE_X));   // 累积min的工作缓冲
        pipe.InitBuffer(tmpQueue, 1, this->tileLen * sizeof(DTYPE_X));             // 临时通道切片
        pipe.InitBuffer(outQueue, 1, this->tileLen * sizeof(DTYPE_Z));             // 输出tile缓冲
    }

    __aicore__ inline void Process()
    {
        uint32_t loops = this->tileNum * BUFFER_NUM;
        for (uint32_t n = 0; n < this->N; ++n) {
            for (uint32_t i = 0; i < loops; ++i) {
                uint32_t basePos = i * this->tileLen;
                if (basePos >= this->HW) {
                    break;
                }
                uint32_t curLen = this->HW - basePos;
                if (curLen > this->tileLen) curLen = this->tileLen;

                // 读取第0个通道作为初始min
                AscendC::LocalTensor<DTYPE_X> acc = workQueue.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(acc, xGm[(n * this->C + 0) * this->HW + basePos], curLen);

                // 遍历其余通道，逐元素最小
                for (uint32_t c = 1; c < this->C; ++c) {
                    AscendC::LocalTensor<DTYPE_X> tmp = tmpQueue.AllocTensor<DTYPE_X>();
                    AscendC::DataCopy(tmp, xGm[(n * this->C + c) * this->HW + basePos], curLen);

                    AscendC::LocalTensor<DTYPE_X> accOut = workQueue.AllocTensor<DTYPE_X>();
                    AscendC::Min(accOut, acc, tmp, curLen);

                    workQueue.FreeTensor(acc);
                    tmpQueue.FreeTensor(tmp);
                    acc = accOut;
                }

                // 两次Tanh
                AscendC::LocalTensor<DTYPE_Z> out = outQueue.AllocTensor<DTYPE_Z>();
                AscendC::Tanh(out, acc, curLen);
                AscendC::Tanh(out, out, curLen);

                // 写回输出[N, 1, H, W]
                AscendC::DataCopy(zGm[(n * 1 + 0) * this->HW + basePos], out, curLen);

                outQueue.FreeTensor(out);
                workQueue.FreeTensor(acc);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> workQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> tmpQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;

    uint32_t N;
    uint32_t C;
    uint32_t HW;
    uint32_t tileNum;
    uint32_t tileLen;
};

extern "C" __global__ __aicore__ void conv2d_min_tanh_tanh_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dMinTanhTanh op;
    op.Init(x, z, tiling_data.nVal, tiling_data.cVal, tiling_data.hwVal, tiling_data.tileNum);
    op.Process();
}
