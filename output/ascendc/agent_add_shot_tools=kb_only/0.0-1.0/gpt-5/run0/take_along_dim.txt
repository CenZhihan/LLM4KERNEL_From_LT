namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    TakeAlongDimCustomTilingData tiling;

    // 输入0: x -> [B, H]
    // 输入1: idx -> [B, S]，输出形状等同于idx
    const auto* xShapeInfo = context->GetInputShape(0);
    const auto* idxShapeInfo = context->GetInputShape(1);
    const auto xShape = xShapeInfo->GetOriginShape();
    const auto idxShape = idxShapeInfo->GetOriginShape();

    // 读取维度
    uint32_t bLength = static_cast<uint32_t>(xShape.GetDim(0));
    uint32_t hLength = static_cast<uint32_t>(xShape.GetDim(1));
    uint32_t sLength = static_cast<uint32_t>(idxShape.GetDim(1));

    uint32_t totalLength = static_cast<uint32_t>(idxShape.GetShapeSize()); // B * S

    context->SetBlockDim(BLOCK_DIM);

    tiling.set_totalLength(totalLength);
    tiling.set_tileNum(TILE_NUM);
    tiling.set_bLength(bLength);
    tiling.set_sLength(sLength);
    tiling.set_hLength(hLength);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无需workspace
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)

    // 输出形状等同于idx
    const gert::Shape* idx_shape = context->GetInputShape(1);

    // 输出数据类型与x一致
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "take_along_dim_custom_tiling_data.h"

// 参考官方文档中LayerNorm/DropOut/GroupNorm等算子的tiling/GET_TILING_DATA用法，
// 采用totalLength+tileNum划分任务，并按block分片。

constexpr int32_t BUFFER_NUM = 2;

class KernelTakeAlongDim {
public:
    __aicore__ inline KernelTakeAlongDim() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR idx, GM_ADDR z,
                                uint32_t totalLength, uint32_t tileNum,
                                uint32_t bLength, uint32_t sLength, uint32_t hLength)
    {
        // 保存维度信息：x:[B,H]，idx:[B,S]，输出z:[B,S]，沿dim=1（H轴）采样
        this->totalLength = totalLength; // B * S
        this->tileNum = tileNum;
        this->bLength = bLength;
        this->sLength = sLength;
        this->hLength = hLength;

        // 每个block处理的输出元素数
        this->blockLength = (totalLength + AscendC::GetBlockNum() - 1) / AscendC::GetBlockNum();
        this->blockStart = AscendC::GetBlockIdx() * this->blockLength;
        // 对最后一个block做边界保护
        uint32_t remain = (this->blockStart + this->blockLength <= totalLength) ? this->blockLength : (totalLength - this->blockStart);
        this->blockLength = remain;

        // tile长度
        uint32_t baseTile = (this->blockLength + (tileNum * BUFFER_NUM) - 1) / (tileNum * BUFFER_NUM);
        this->tileLength = baseTile;

        // 绑定GM
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, bLength * hLength);
        idxGm.SetGlobalBuffer((__gm__ DTYPE_Y *)idx + this->blockStart, this->blockLength); // int32 idx
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->blockStart, this->blockLength);

        // 队列buffer，仅传输idx和输出z，x按行读入
        pipe.InitBuffer(inQueueIdx, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));

        // 行缓存：最大一行H
        pipe.InitBuffer(inQueueXRow, 1, this->hLength * sizeof(DTYPE_X));
        pipe.InitBuffer(tmpIdxSeg, 1, this->tileLength * sizeof(int32_t));
    }

    __aicore__ inline void Process()
    {
        // 循环处理每个tile
        uint32_t tiles = (this->blockLength + this->tileLength - 1) / this->tileLength;
        for (uint32_t t = 0; t < tiles; ++t) {
            uint32_t curStart = t * this->tileLength;
            uint32_t curLen = AscendC::min(this->tileLength, this->blockLength - curStart);

            CopyIn(curStart, curLen);
            Compute(curStart, curLen);
            CopyOut(curStart, curLen);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t curStart, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_Y> idxLocal = inQueueIdx.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(idxLocal, idxGm[curStart], curLen);
        inQueueIdx.EnQue(idxLocal);
    }

    __aicore__ inline void Compute(uint32_t curStart, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_Y> idxLocal = inQueueIdx.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();

        // 以扁平索引gid = blockStart + curStart + i，求row/col
        // row = gid / S, col = gid % S; val = x[row, idx[row,col]]
        uint32_t gidStart = this->blockStart + curStart;
        uint32_t i = 0;
        while (i < curLen) {
            uint32_t gid = gidStart + i;
            uint32_t row = gid / this->sLength;
            uint32_t col = gid - row * this->sLength;

            // 将该行x整行搬入本地（避免随机GM访问）
            AscendC::LocalTensor<DTYPE_X> xRowLocal = inQueueXRow.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xRowLocal, xGm[row * this->hLength], this->hLength);

            // 在当前行中可以连续处理的一段长度
            uint32_t canLen = AscendC::min(this->sLength - col, curLen - i);

            // 拷贝该段idx到临时local（便于逐元素索引）
            AscendC::LocalTensor<int32_t> idxSegLocal = tmpIdxSeg.AllocTensor<int32_t>();
            // idxLocal是tile局部，段起点为i，长度canLen
            // 这里通过DataCopy完成本地->本地的段拷贝（允许）
            AscendC::DataCopy(idxSegLocal, idxLocal[i], canLen);

            // 逐元素gather：zLocal[i + k] = xRowLocal[idxSegLocal[k]]
            // 假设LocalTensor支持指针访问（与示例相同的本地张量用法）
            DTYPE_Z* zPtr = reinterpret_cast<DTYPE_Z*>(zLocal.GetData());
            DTYPE_X* xRowPtr = reinterpret_cast<DTYPE_X*>(xRowLocal.GetData());
            int32_t* idxPtr = reinterpret_cast<int32_t*>(idxSegLocal.GetData());

            for (uint32_t k = 0; k < canLen; ++k) {
                int32_t j = idxPtr[k];
                j = j < 0 ? 0 : (j >= (int32_t)this->hLength ? ((int32_t)this->hLength - 1) : j);
                zPtr[i + k] = static_cast<DTYPE_Z>(xRowPtr[j]);
            }

            tmpIdxSeg.FreeTensor(idxSegLocal);
            inQueueXRow.FreeTensor(xRowLocal);

            i += canLen;
        }

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueIdx.FreeTensor(idxLocal);
    }

    __aicore__ inline void CopyOut(uint32_t curStart, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[curStart], zLocal, curLen);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;

    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueIdx;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;

    // 行缓冲与idx临时段
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueXRow;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpIdxSeg;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> idxGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;

    uint32_t totalLength;
    uint32_t tileNum;
    uint32_t bLength;
    uint32_t sLength;
    uint32_t hLength;

    uint32_t blockStart;
    uint32_t blockLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void take_along_dim_custom(GM_ADDR x, GM_ADDR idx, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelTakeAlongDim op;
    op.Init(x, idx, z,
            tiling_data.totalLength,
            tiling_data.tileNum,
            tiling_data.bLength,
            tiling_data.sLength,
            tiling_data.hLength);
    op.Process();
}
