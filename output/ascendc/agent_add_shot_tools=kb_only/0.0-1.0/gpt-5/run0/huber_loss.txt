kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "huber_loss_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelHuberLoss {
public:
    __aicore__ inline KernelHuberLoss() {}
    __aicore__ inline void Init(GM_ADDR pred, GM_ADDR target, GM_ADDR loss, uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        predGm.SetGlobalBuffer((__gm__ float *)pred + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        targetGm.SetGlobalBuffer((__gm__ float *)target + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        lossGm.SetGlobalBuffer((__gm__ float *)loss + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueuePred, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueTarget, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueLoss, BUFFER_NUM, this->tileLength * sizeof(float));

        // two temporary local buffers for intermediate results
        pipe.InitBuffer(tmpQueueA, BUFFER_NUM, this->tileLength * sizeof(uint8_t)); // used as float buffer
        pipe.InitBuffer(tmpQueueB, BUFFER_NUM, this->tileLength * sizeof(uint8_t)); // used as float buffer
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> predLocal = inQueuePred.AllocTensor<float>();
        AscendC::LocalTensor<float> targetLocal = inQueueTarget.AllocTensor<float>();
        AscendC::DataCopy(predLocal, predGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(targetLocal, targetGm[progress * this->tileLength], this->tileLength);
        inQueuePred.EnQue(predLocal);
        inQueueTarget.EnQue(targetLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> predLocal = inQueuePred.DeQue<float>();
        AscendC::LocalTensor<float> targetLocal = inQueueTarget.DeQue<float>();
        AscendC::LocalTensor<float> lossLocal = outQueueLoss.AllocTensor<float>();

        AscendC::LocalTensor<uint8_t> tmpALocalU8 = tmpQueueA.AllocTensor<uint8_t>();
        AscendC::LocalTensor<uint8_t> tmpBLocalU8 = tmpQueueB.AllocTensor<uint8_t>();
        AscendC::LocalTensor<float> tmpALocal = tmpALocalU8.ReinterpretCast<float>();
        AscendC::LocalTensor<float> tmpBLocal = tmpBLocalU8.ReinterpretCast<float>();

        // diff = pred - target => store in lossLocal
        AscendC::Sub(lossLocal, predLocal, targetLocal, this->tileLength);

        // absd = Abs(diff) => in place in lossLocal
        AscendC::Abs(lossLocal, lossLocal, this->tileLength);

        // tmpBLocal = 1.0 (broadcast)
        float one = 1.0f;
        AscendC::Duplicate(tmpBLocal, one, this->tileLength);

        // tmpALocal = min(absd, 1.0) => t
        AscendC::Minimum(tmpALocal, lossLocal, tmpBLocal, this->tileLength);

        // term2: lossLocal = absd - t
        AscendC::Sub(lossLocal, lossLocal, tmpALocal, this->tileLength);

        // term1: tmpBLocal = 0.5 * t * t
        AscendC::Mul(tmpBLocal, tmpALocal, tmpALocal, this->tileLength);
        AscendC::Muls(tmpBLocal, tmpBLocal, 0.5f, this->tileLength);

        // lossLocal = term1 + term2
        AscendC::Add(lossLocal, lossLocal, tmpBLocal, this->tileLength);

        outQueueLoss.EnQue<float>(lossLocal);

        tmpQueueA.FreeTensor(tmpALocalU8);
        tmpQueueB.FreeTensor(tmpBLocalU8);
        inQueuePred.FreeTensor(predLocal);
        inQueueTarget.FreeTensor(targetLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> lossLocal = outQueueLoss.DeQue<float>();
        AscendC::DataCopy(lossGm[progress * this->tileLength], lossLocal, this->tileLength);
        outQueueLoss.FreeTensor(lossLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueuePred, inQueueTarget;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpQueueA, tmpQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueLoss;

    AscendC::GlobalTensor<float> predGm;
    AscendC::GlobalTensor<float> targetGm;
    AscendC::GlobalTensor<float> lossGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void huber_loss_custom(GM_ADDR pred, GM_ADDR target, GM_ADDR loss, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelHuberLoss op;
    op.Init(pred, target, loss, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}
"""
