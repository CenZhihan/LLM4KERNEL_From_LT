kernel_src="""
#include \"kernel_operator.h\"
#define __NPU_TILING__
#include \"matmul_divide_gelu_custom_tiling_data.h\"

// 参考：
// - AscendC::Gelu API（CANN 5.3.3.1）
// - Matmul 回调SetSelfDefineData示例（CANN 5.2.1.40）
// 说明：本内核完成 y = GELU((x @ w + b) / divisor)
//       其中 x:[M,K], w:[K,N], b:[N], divisor:[1]。
//       使用多核按M维拆分，随后以向量方式执行除法与GELU。

using DTYPE_X = float;
using DTYPE_W = float;
using DTYPE_B = float;
using DTYPE_Y = float;

constexpr int32_t BUFFER_NUM = 2;

// 简化：按M维均匀划分到各个core
class KernelMatmulDivideGelu {
public:
    __aicore__ inline KernelMatmulDivideGelu() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR divisor, GM_ADDR y,
                                uint32_t m, uint32_t n, uint32_t k, uint32_t tileNum)
    {
        this->M = m;
        this->N = n;
        this->K = k;
        this->tileNum = tileNum;

        // 每个block负责处理的行数
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->mPerBlock = (m + blockNum - 1) / blockNum;
        // 起止行
        this->rowStart = blockIdx * this->mPerBlock;
        this->rowEnd = AscendC::Min<uint32_t>(this->rowStart + this->mPerBlock, m);
        if (this->rowStart >= this->rowEnd) {
            this->mPerBlock = 0;
        } else {
            this->mPerBlock = this->rowEnd - this->rowStart;
        }

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_X*>(x) + static_cast<uint64_t>(this->rowStart) * K,
                            static_cast<uint64_t>(this->mPerBlock) * K);
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_W*>(w), static_cast<uint64_t>(K) * N);
        bGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_B*>(b), N);
        divGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_Y*>(divisor), 1);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_Y*>(y) + static_cast<uint64_t>(this->rowStart) * N,
                            static_cast<uint64_t>(this->mPerBlock) * N);

        // 简单向量化tile，按行输出
        // tileLength 表示每次处理的输出元素数量（按N方向切分）
        this->tileLength = (this->N == 0 ? 0 : (this->N / tileNum));
        if (this->tileLength == 0 && this->N > 0) this->tileLength = this->N; // 边界处理

        // 队列缓冲
        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y)); // 用于后处理阶段的中间空间
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        if (this->mPerBlock == 0 || this->N == 0) {
            return;
        }

        // 读取divisor标量
        DTYPE_Y divisor = 1.0f;
        {
            AscendC::LocalTensor<DTYPE_Y> divLocal = inQueueA.AllocTensor<DTYPE_Y>();
            AscendC::DataCopy(divLocal, divGm, 1);
            divisor = static_cast<DTYPE_Y>(divLocal.GetValue(0));
            inQueueA.FreeTensor(divLocal);
        }
        DTYPE_Y invDiv = 1.0f / divisor;

        // 逐行计算：C[row, :] = x[row, :] @ w + b
        for (uint32_t r = 0; r < this->mPerBlock; ++r) {
            // 对整行输出按N方向分块
            uint32_t processed = 0;
            while (processed < this->N) {
                uint32_t curTile = AscendC::Min<uint32_t>(this->tileLength, this->N - processed);

                // 分配本地输出片段
                AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

                // 初始化 yLocal 为bias片段
                AscendC::DataCopy(yLocal, bGm[processed], curTile);

                // 计算 yLocal += xRow(1xK) * wPart(KxcurTile)
                // 朴素实现：对K聚合（此处作为演示，真实优化需使用官方Matmul高阶API及tiling）
                for (uint32_t kk = 0; kk < this->K; ++kk) {
                    // 读取 x[row, kk]
                    DTYPE_X aVal;
                    {
                        // 直接从GM读标量（示例写法，性能上更优做法应批量搬运）
                        aVal = static_cast<DTYPE_X>(xGm[r * this->K + kk]);
                    }

                    // 将 w[kk, processed:processed+curTile] 读入本地
                    AscendC::LocalTensor<DTYPE_W> wLocal = inQueueA.AllocTensor<DTYPE_W>();
                    AscendC::DataCopy(wLocal, wGm[static_cast<uint64_t>(kk) * this->N + processed], curTile);

                    // yLocal += aVal * wLocal（逐元素FMA）
                    for (uint32_t i = 0; i < curTile; ++i) {
                        DTYPE_Y acc = yLocal.GetValue(i);
                        acc = acc + static_cast<DTYPE_Y>(aVal) * static_cast<DTYPE_Y>(wLocal.GetValue(i));
                        yLocal.SetValue(i, acc);
                    }
                    inQueueA.FreeTensor(wLocal);
                }

                // yLocal /= divisor
                for (uint32_t i = 0; i < curTile; ++i) {
                    DTYPE_Y v = yLocal.GetValue(i);
                    yLocal.SetValue(i, v * invDiv);
                }

                // GELU激活（使用AscendC::Gelu高阶API）
                {
                    AscendC::LocalTensor<DTYPE_Y> geluLocal = outQueueY.AllocTensor<DTYPE_Y>();
                    AscendC::Gelu(geluLocal, yLocal, curTile);
                    // 覆写 yLocal 为gelu结果
                    for (uint32_t i = 0; i < curTile; ++i) {
                        yLocal.SetValue(i, geluLocal.GetValue(i));
                    }
                    outQueueY.FreeTensor(geluLocal);
                }

                // 写回到全局输出
                AscendC::DataCopy(yGm[static_cast<uint64_t>(r) * this->N + processed], yLocal, curTile);
                outQueueY.FreeTensor(yLocal);

                processed += curTile;
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_W> wGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_Y> divGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t M, N, K;
    uint32_t tileNum;
    uint32_t tileLength;

    uint32_t rowStart;
    uint32_t rowEnd;
    uint32_t mPerBlock;
};

extern \"C\" __global__ __aicore__ void matmul_divide_gelu_custom(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR divisor,
                                                                  GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulDivideGelu op;
    op.Init(x, w, b, divisor, y,
            tiling_data.m(), tiling_data.n(), tiling_data.k(), tiling_data.tileNum());
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

# 说明：
# - 本自定义算子期望输入为 x:[M,K], w:[K,N]（即将Linear权重转置后传入）, b:[N], divisor:[1]
# - 输出为 y:[M,N]
class ModelNew(nn.Module):
    def __init__(self, input_size, output_size, divisor):
        super(ModelNew, self).__init__()
        self.linear = nn.Linear(input_size, output_size)
        self.divisor = float(divisor)

    def forward(self, x: torch.Tensor):
        # 将权重转置为[K, N]
        w_t = self.linear.weight.t().contiguous()
        b = self.linear.bias if self.linear.bias is not None else torch.zeros(w_t.size(1), dtype=x.dtype, device=x.device)
        divisor_tensor = torch.tensor([self.divisor], dtype=x.dtype, device=x.device)
        return custom_ops_lib.matmul_divide_gelu_custom(x, w_t, b, divisor_tensor)

def get_inputs():
    batch_size = 1024
    input_size = 8192
    return [torch.rand(batch_size, input_size)]

def get_init_inputs():
    input_size = 8192
    output_size = 8192
    divisor = 10.0
    return [input_size, output_size, divisor]
