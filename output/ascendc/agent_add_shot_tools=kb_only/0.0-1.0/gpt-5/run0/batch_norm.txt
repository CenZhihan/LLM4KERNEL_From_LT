host_operator_src="""
#include "batch_norm_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    BatchNormCustomTilingData tiling;

    const gert::Shape* inShape = context->GetInputShape(0);
    // Assume NCHW
    uint32_t n = static_cast<uint32_t>(inShape->GetDim(0));
    uint32_t c = static_cast<uint32_t>(inShape->GetDim(1));
    uint32_t h = static_cast<uint32_t>(inShape->GetDim(2));
    uint32_t w = static_cast<uint32_t>(inShape->GetDim(3));

    context->SetBlockDim(BLOCK_DIM);

    tiling.set_n(n);
    tiling.set_c(c);
    tiling.set_h(h);
    tiling.set_w(w);
    tiling.set_tileNum(TILE_NUM);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;

    // out_mean/out_var are per-channel (C)
    const int64_t c = context->GetInputShape(0)->GetDim(1);
    gert::Shape* mean_shape = context->GetOutputShape(1);
    gert::Shape* var_shape  = context->GetOutputShape(2);
    mean_shape->SetDimNum(1);
    mean_shape->SetDim(0, c);
    var_shape->SetDimNum(1);
    var_shape->SetDim(0, c);
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    context->SetOutputDataType(1, inputDataType);
    context->SetOutputDataType(2, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class BatchNormCustom : public OpDef {
public:
    explicit BatchNormCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("gamma")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("beta")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("out_mean")
            .ParamType(OPTIONAL)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("out_var")
            .ParamType(OPTIONAL)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);

        this->AICore()
            .AutoContiguous()   // auto make inputs contiguous if needed
            .SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(BatchNormCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "batch_norm_custom_tiling_data.h"

extern "C" __global__ __aicore__ void batch_norm_custom(GM_ADDR inputX_gm,
                                                        GM_ADDR gamma_gm,
                                                        GM_ADDR beta_gm,
                                                        GM_ADDR output_gm,
                                                        GM_ADDR outputMean_gm,
                                                        GM_ADDR outputVariance_gm,
                                                        GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    // Use high-level BatchNorm API per CANN reference
    KernelBatchnorm<float, false, false> op;
    op.Init(inputX_gm, gamma_gm, beta_gm, output_gm, outputMean_gm, outputVariance_gm,
            tilingData.batchNormTilingData);
    if (TILING_KEY_IS(1)) {
        op.Process();
    } else {
        op.Process();
    }
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor batch_norm_custom_impl_npu(const at::Tensor& x,
                                      const at::Tensor& gamma,
                                      const at::Tensor& beta) {
    auto y = at::empty_like(x);
    auto c = gamma.numel();
    at::Tensor out_mean = at::empty({c}, x.options());
    at::Tensor out_var  = at::empty({c}, x.options());
    EXEC_NPU_CMD(aclnnBatchNormCustom, x, gamma, beta, y, out_mean, out_var);
    return y;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("batch_norm_custom", &batch_norm_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("batch_norm_custom", &batch_norm_custom_impl_npu, "BatchNorm 2D (custom AscendC)");
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model using custom AscendC BatchNorm kernel.
    """
    def __init__(self, num_features: int):
        super().__init__()
        # Learnable affine parameters, same as nn.BatchNorm2d affine=True
        self.weight = torch.nn.Parameter(torch.ones(num_features))
        self.bias = torch.nn.Parameter(torch.zeros(num_features))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Use custom operator: computes per-channel BN with current batch stats internally.
        return custom_ops_lib.batch_norm_custom(x, self.weight, self.bias)
