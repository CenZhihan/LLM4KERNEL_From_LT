kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "multi_query_attention_custom_tiling_data.h"

#include <math.h>

extern "C" __global__ __aicore__ void multi_query_attention_custom(GM_ADDR q, GM_ADDR k, GM_ADDR v, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    const uint32_t B = tiling_data.b;
    const uint32_t N = tiling_data.n;
    const uint32_t H = tiling_data.h;
    const uint32_t D = tiling_data.d;

    const uint32_t totalItems = B * N * H; // number of (b,n,h) triplets
    const uint32_t blockNum = AscendC::GetBlockNum();
    const uint32_t blockIdx = AscendC::GetBlockIdx();
    const uint32_t perBlock = (totalItems + blockNum - 1) / blockNum;
    const uint32_t startIdx = perBlock * blockIdx;
    const uint32_t endIdx = (startIdx + perBlock) > totalItems ? totalItems : (startIdx + perBlock);

    __gm__ float* qPtr = (__gm__ float*)q;
    __gm__ float* kPtr = (__gm__ float*)k;
    __gm__ float* vPtr = (__gm__ float*)v;
    __gm__ float* oPtr = (__gm__ float*)out;

    const float scale = 1.0f / sqrtf((float)D);

    for (uint32_t idx = startIdx; idx < endIdx; ++idx) {
        uint32_t b = idx / (N * H);
        uint32_t rem = idx % (N * H);
        uint32_t n = rem / H;
        uint32_t h = rem % H;

        uint64_t qBase = ((((uint64_t)b * N + n) * H + h) * D);
        uint64_t oBase = qBase;

        // Pass 1: compute max score for numerical stability
        float maxScore = -3.402823e+38F; // -FLT_MAX
        for (uint32_t kk = 0; kk < N; ++kk) {
            uint64_t kBase = ((((uint64_t)b * N + kk) * 1u + 0u) * D);
            float dot = 0.0f;
            for (uint32_t d = 0; d < D; ++d) {
                float qv = qPtr[qBase + d];
                float kv = kPtr[kBase + d];
                dot += qv * kv;
            }
            dot = dot * scale;
            if (dot > maxScore) {
                maxScore = dot;
            }
        }

        // Pass 2: compute denominator sumExp
        float sumExp = 0.0f;
        for (uint32_t kk = 0; kk < N; ++kk) {
            uint64_t kBase = ((((uint64_t)b * N + kk) * 1u + 0u) * D);
            float dot = 0.0f;
            for (uint32_t d = 0; d < D; ++d) {
                float qv = qPtr[qBase + d];
                float kv = kPtr[kBase + d];
                dot += qv * kv;
            }
            dot = dot * scale;
            float e = expf(dot - maxScore);
            sumExp += e;
        }
        float invSum = sumExp > 0.0f ? (1.0f / sumExp) : 0.0f;

        // Pass 3: accumulate weighted V into output vector
        for (uint32_t d = 0; d < D; ++d) {
            oPtr[oBase + d] = 0.0f;
        }
        for (uint32_t kk = 0; kk < N; ++kk) {
            uint64_t kBase = ((((uint64_t)b * N + kk) * 1u + 0u) * D);
            uint64_t vBase = kBase; // V has same layout [B, N, 1, D]
            float dot = 0.0f;
            for (uint32_t d = 0; d < D; ++d) {
                float qv = qPtr[qBase + d];
                float kv = kPtr[kBase + d];
                dot += qv * kv;
            }
            float weight = expf(dot * scale - maxScore) * invSum;
            for (uint32_t d = 0; d < D; ++d) {
                float vv = vPtr[vBase + d];
                oPtr[oBase + d] += weight * vv;
            }
        }
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
import math
import torch.nn as nn

class ModelNew(torch.nn.Module):
    """
    Multi-Query Attention (MQA) optimized with custom AscendC kernel for attention core.
    - Q, K, V projections in PyTorch
    - Custom kernel computes softmax(Q K^T / sqrt(d)) @ V with shared K/V head
    """
    def __init__(self, d_model=4096, num_heads=32):
        super().__init__()
        self.q_proj = nn.Linear(d_model, d_model)
        self.k_proj = nn.Linear(d_model, d_model // num_heads)  # shared across heads
        self.v_proj = nn.Linear(d_model, d_model // num_heads)
        self.out_proj = nn.Linear(d_model, d_model)
        self.num_heads = num_heads

    def forward(self, x):
        B, N, D = x.shape
        H = self.num_heads
        d_h = D // H

        Q = self.q_proj(x).view(B, N, H, d_h)
        K = self.k_proj(x).unsqueeze(2)  # [B, N, 1, d_h]
        V = self.v_proj(x).unsqueeze(2)  # [B, N, 1, d_h]

        # Custom AscendC kernel replaces einsum-softmax-einsum
        ctx = custom_ops_lib.multi_query_attention_custom(Q, K, V)  # [B, N, H, d_h]
        out = ctx.reshape(B, N, D)
        return self.out_proj(out)
'''
