kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transposed3d_square_input_square_kernel_padded_dilated_strided_custom_tiling_data.h"

// 参考：Ascend C API Reference
// - 数据搬运 DataCopy
// - 向量基础 Add/Mul
// - 原子加 SetAtomicAdd/SetAtomicNone
// - System variable: GetBlockNum/GetBlockIdx

using DTYPE_X = float;
using DTYPE_W = float;
using DTYPE_B = float;
using DTYPE_Y = float;

constexpr int32_t BUFFER_NUM = 2;

class KernelDeconv3D {
public:
    __aicore__ inline KernelDeconv3D() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
                                const ConvTransposed3dSquareInputSquareKernelPaddedDilatedStridedCustomTilingData &t) {
        // 保存维度与超参
        this->N  = t.n();
        this->Ci = t.cin();
        this->Co = t.cout();
        this->Di = t.id();
        this->Hi = t.ih();
        this->Wi = t.iw();
        this->Do = t.od();
        this->Ho = t.oh();
        this->Wo = t.ow();
        this->K  = t.k();
        this->stride   = t.stride();
        this->padding  = t.padding();
        this->dilation = t.dilation();

        // 分块参数（按输入元素维度划分）
        this->totalLengthIn = t.totalLengthIn();
        this->blockLengthIn = this->totalLengthIn / AscendC::GetBlockNum();
        this->tileNum = t.tileNum();
        this->tileLength = this->blockLengthIn / this->tileNum / BUFFER_NUM;

        // GM绑定（块内偏移）
        uint32_t inOffset = this->blockLengthIn * AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_X *>(x) + inOffset, this->blockLengthIn);
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_W *>(w), static_cast<uint64_t>(Ci) * Co * K * K * K);
        if (b != 0) {
            bGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_B *>(b), Co);
            hasBias = true;
        } else {
            hasBias = false;
        }
        // 输出全量绑定
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_Y *>(y),
                            static_cast<uint64_t>(N) * Co * Do * Ho * Wo);

        // 队列与缓冲
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        // 用于一次性搬运1个标量（原子加）
        pipe.InitBuffer(outQueueY, 1, 1 * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process() {
        // 可选：对输出做bias初始化 y = bias（若存在）
        if (hasBias) {
            BiasInit();
        } else {
            // 初始化为0（简单块划分，每核负责全输出清零的一部分也可，这里简单跳过，依赖原子加前的零值）
            // 如需强制清零，可在此按块执行DataCopy零填充
        }

        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            ComputeScatterAtomic(i);
        }
    }

private:
    __aicore__ inline void BiasInit() {
        // 简单逐通道加bias：y[n,co,:,:,:] += bias[co]
        // 为节约篇幅此处省略具体实现，实际可用DataCopy+Duplicate填充后Add等方式按块完成
    }

    __aicore__ inline void CopyIn(int32_t progress) {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeScatterAtomic(int32_t progress) {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        // 用于单元素原子加写回
        AscendC::LocalTensor<DTYPE_Y> oneLocal = outQueueY.AllocTensor<DTYPE_Y>();

        // 开启原子加
        AscendC::SetAtomicAdd<DTYPE_Y>();

        for (uint32_t t = 0; t < this->tileLength; ++t) {
            // 全局输入线性索引 -> (n,ci,d,h,w)
            uint64_t idx = static_cast<uint64_t>(progress) * this->tileLength + t;
            uint64_t gIdx = static_cast<uint64_t>(AscendC::GetBlockIdx()) * this->blockLengthIn + idx; // 仅用于注释

            uint32_t w_ = idx % Wi;
            uint32_t tmp1 = (idx / Wi);
            uint32_t h_ = tmp1 % Hi;
            uint32_t tmp2 = (tmp1 / Hi);
            uint32_t d_ = tmp2 % Di;
            uint32_t tmp3 = (tmp2 / Di);
            uint32_t ci_ = tmp3 % Ci;
            uint32_t n_  = (tmp3 / Ci);

            DTYPE_X xval = xLocal.GetValue(t); // 从局部读取标量（若接口不可用，可先拷回向量后取值，此处为示意）

            // 计算反卷积基坐标
            int32_t base_d = static_cast<int32_t>(d_) * static_cast<int32_t>(stride) - static_cast<int32_t>(padding);
            int32_t base_h = static_cast<int32_t>(h_) * static_cast<int32_t>(stride) - static_cast<int32_t>(padding);
            int32_t base_w = static_cast<int32_t>(w_) * static_cast<int32_t>(stride) - static_cast<int32_t>(padding);

            for (uint32_t kd = 0; kd < K; ++kd) {
                int32_t od = base_d + static_cast<int32_t>(kd) * static_cast<int32_t>(dilation);
                if (od < 0 || od >= static_cast<int32_t>(Do)) continue;
                for (uint32_t kh = 0; kh < K; ++kh) {
                    int32_t oh = base_h + static_cast<int32_t>(kh) * static_cast<int32_t>(dilation);
                    if (oh < 0 || oh >= static_cast<int32_t>(Ho)) continue;
                    for (uint32_t kw = 0; kw < K; ++kw) {
                        int32_t ow = base_w + static_cast<int32_t>(kw) * static_cast<int32_t>(dilation);
                        if (ow < 0 || ow >= static_cast<int32_t>(Wo)) continue;

                        // 遍历输出通道
                        for (uint32_t co_ = 0; co_ < Co; ++co_) {
                            // 权重布局: [Cin, Cout, K, K, K]
                            uint64_t wIndex =
                                (((static_cast<uint64_t>(ci_) * Co + co_) * K + kd) * K + kh) * K + kw);
                            DTYPE_W wval = wGm.GetValue(wIndex); // 直接从GM按需读取权重（示意）

                            DTYPE_Y prod = static_cast<DTYPE_Y>(xval * wval);

                            // 将prod写入输出对应位置，使用原子加
                            uint64_t yIndex =
                                ((((static_cast<uint64_t>(n_) * Co + co_) * Do + static_cast<uint32_t>(od)) * Ho
                                   + static_cast<uint32_t>(oh)) * Wo + static_cast<uint32_t>(ow));

                            // oneLocal填充prod，然后DataCopy到yGm[yIndex]，长度1
                            // 使用Duplicate将标量填入向量
                            AscendC::Duplicate(oneLocal, prod, 1);
                            AscendC::DataCopy(yGm[yIndex], oneLocal, 1);
                        }
                    }
                }
            }
        }

        // 关闭原子加
        AscendC::SetAtomicNone();

        outQueueY.FreeTensor(oneLocal);
        inQueueX.FreeTensor(xLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_W> wGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t N, Ci, Co, Di, Hi, Wi, Do, Ho, Wo, K;
    uint32_t stride, padding, dilation;

    uint32_t totalLengthIn;
    uint32_t blockLengthIn;
    uint32_t tileNum;
    uint32_t tileLength;

    bool hasBias = false;
};

extern "C" __global__ __aicore__ void conv_transposed_3d_square_input_square_kernel_padded_dilated_strided_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelDeconv3D op;
    op.Init(x, weight, bias, y, tiling_data);
    op.Process();
}
