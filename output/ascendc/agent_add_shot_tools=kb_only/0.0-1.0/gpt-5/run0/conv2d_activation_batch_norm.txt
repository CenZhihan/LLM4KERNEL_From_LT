kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_activation_batch_norm_custom_tiling_data.h"

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 2;

class KernelConvActBn {
public:
    __aicore__ inline KernelConvActBn() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR running_mean, GM_ADDR running_var,
                                GM_ADDR eps, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w)
    {
        this->n = n;
        this->c = c;
        this->h = h;
        this->w = w;
        this->hw = h * w;
        this->tilesTotal = n * c;

        xGm.SetGlobalBuffer((__gm__ DTYPE*)x);
        yGm.SetGlobalBuffer((__gm__ DTYPE*)y);
        gammaGm.SetGlobalBuffer((__gm__ DTYPE*)gamma);
        betaGm.SetGlobalBuffer((__gm__ DTYPE*)beta);
        meanGm.SetGlobalBuffer((__gm__ DTYPE*)running_mean);
        varGm.SetGlobalBuffer((__gm__ DTYPE*)running_var);
        epsGm.SetGlobalBuffer((__gm__ DTYPE*)eps);

        // tile = one (n,c) feature map of size h*w
        tileLength = this->hw;
        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(DTYPE));

        // parameter locals
        pipe.InitBuffer(inGamma, 1, this->c * sizeof(DTYPE));
        pipe.InitBuffer(inBeta, 1, this->c * sizeof(DTYPE));
        pipe.InitBuffer(inMean, 1, this->c * sizeof(DTYPE));
        pipe.InitBuffer(inVar, 1, this->c * sizeof(DTYPE));
        pipe.InitBuffer(workScale, 1, this->c * sizeof(DTYPE));
        pipe.InitBuffer(workBias, 1, this->c * sizeof(DTYPE));
        pipe.InitBuffer(workTmp, 1, this->c * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        // Preload eps scalar
        DTYPE epsScalar;
        {
            AscendC::LocalTensor<DTYPE> epsLocal = workTmp.AllocTensor<DTYPE>();
            AscendC::DataCopy(epsLocal, epsGm, 1);
            epsScalar = epsLocal.GetValue(0);
            workTmp.FreeTensor(epsLocal);
        }

        // Load BN params and compute per-channel scale and bias:
        // scale = gamma / sqrt(var + eps), bias = beta - mean * scale
        AscendC::LocalTensor<DTYPE> gammaLocal = inGamma.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> betaLocal  = inBeta.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> meanLocal  = inMean.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> varLocal   = inVar.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> scaleLocal = workScale.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> biasLocal  = workBias.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> tmpLocal   = workTmp.AllocTensor<DTYPE>();

        AscendC::DataCopy(gammaLocal, gammaGm, this->c);
        AscendC::DataCopy(betaLocal,  betaGm,  this->c);
        AscendC::DataCopy(meanLocal,  meanGm,  this->c);
        AscendC::DataCopy(varLocal,   varGm,   this->c);

        // var + eps
        AscendC::Adds(tmpLocal, varLocal, epsScalar, this->c);
        // sqrt(var + eps)
        AscendC::Sqrt(tmpLocal, tmpLocal, this->c);
        // scale = gamma / sqrt(...)
        AscendC::Div(scaleLocal, gammaLocal, tmpLocal, this->c);
        // tmp = mean * scale
        AscendC::Mul(tmpLocal, meanLocal, scaleLocal, this->c);
        // bias = beta - tmp
        AscendC::Sub(biasLocal, betaLocal, tmpLocal, this->c);

        inGamma.EnQue(gammaLocal);
        inBeta.EnQue(betaLocal);
        inMean.EnQue(meanLocal);
        inVar.EnQue(varLocal);
        workScale.EnQue(scaleLocal);
        workBias.EnQue(biasLocal);
        workTmp.EnQue(tmpLocal);

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t tilesPerBlock = (tilesTotal + blockNum - 1) / blockNum;
        uint32_t startTile = tilesPerBlock * blockIdx;
        uint32_t endTile = AscendC::Min<uint32_t>(startTile + tilesPerBlock, tilesTotal);

        for (uint32_t tile = startTile; tile < endTile; ++tile) {
            uint32_t cIdx = tile % this->c;
            uint64_t baseOffset = static_cast<uint64_t>(tile) * this->hw;

            CopyIn(baseOffset);
            Compute(cIdx);
            CopyOut(baseOffset);
        }

        // Free param locals
        gammaLocal = inGamma.DeQue<DTYPE>();
        betaLocal  = inBeta.DeQue<DTYPE>();
        meanLocal  = inMean.DeQue<DTYPE>();
        varLocal   = inVar.DeQue<DTYPE>();
        scaleLocal = workScale.DeQue<DTYPE>();
        biasLocal  = workBias.DeQue<DTYPE>();
        tmpLocal   = workTmp.DeQue<DTYPE>();

        inGamma.FreeTensor(gammaLocal);
        inBeta.FreeTensor(betaLocal);
        inMean.FreeTensor(meanLocal);
        inVar.FreeTensor(varLocal);
        workScale.FreeTensor(scaleLocal);
        workBias.FreeTensor(biasLocal);
        workTmp.FreeTensor(tmpLocal);
    }

private:
    __aicore__ inline void CopyIn(uint64_t offset)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm[offset], tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t cIdx)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.AllocTensor<DTYPE>();

        // yLocal = tanh(softplus(x)) * x
        // softplus(x) = ln(1 + exp(x))
        AscendC::LocalTensor<DTYPE> tmp0 = workTmp.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> tmp1 = workTmp.AllocTensor<DTYPE>();

        // tmp0 = exp(x)
        AscendC::Exp(tmp0, xLocal, tileLength);
        // tmp0 = tmp0 + 1
        AscendC::Adds(tmp0, tmp0, (DTYPE)1.0f, tileLength);
        // tmp0 = ln(tmp0)
        AscendC::Ln(tmp0, tmp0, tileLength);
        // tmp1 = tanh(tmp0)
        AscendC::Tanh(tmp1, tmp0, tileLength);
        // yLocal = tmp1 * xLocal
        AscendC::Mul(yLocal, tmp1, xLocal, tileLength);

        // Load per-channel scale and bias
        AscendC::LocalTensor<DTYPE> scaleLocal = workScale.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> biasLocal  = workBias.DeQue<DTYPE>();
        DTYPE scale = scaleLocal.GetValue(cIdx);
        DTYPE bias  = biasLocal.GetValue(cIdx);
        workScale.EnQue(scaleLocal);
        workBias.EnQue(biasLocal);

        // yLocal = yLocal * scale + bias
        AscendC::Muls(yLocal, yLocal, scale, tileLength);
        AscendC::Adds(yLocal, yLocal, bias, tileLength);

        outQueueY.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);
        workTmp.FreeTensor(tmp0);
        workTmp.FreeTensor(tmp1);
    }

    __aicore__ inline void CopyOut(uint64_t offset)
    {
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm[offset], yLocal, tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    // Param queues/buffers
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inGamma, inBeta, inMean, inVar;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> workScale, workBias, workTmp;

    AscendC::GlobalTensor<DTYPE> xGm, yGm;
    AscendC::GlobalTensor<DTYPE> gammaGm, betaGm, meanGm, varGm, epsGm;

    uint32_t n, c, h, w, hw;
    uint32_t tilesTotal;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv2d_activation_batch_norm_custom(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta,
                                                                          GM_ADDR running_mean, GM_ADDR running_var,
                                                                          GM_ADDR eps, GM_ADDR y,
                                                                          GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvActBn op;
    op.Init(x, gamma, beta, running_mean, running_var, eps, y,
            tiling_data.n, tiling_data.c, tiling_data.h, tiling_data.w);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model:
    - Keep nn.Conv2d as-is.
    - Fuse activation (tanh(softplus(x)) * x) + BatchNorm into a custom AscendC kernel.
    """
    def __init__(self, in_channels, out_channels, kernel_size, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.bn = nn.BatchNorm2d(out_channels, eps=eps, momentum=momentum)

    def forward(self, x):
        x = self.conv(x)
        eps_tensor = x.new_tensor([self.bn.eps])
        # Pass BN params and eps to custom fused op
        y = custom_ops_lib.conv2d_activation_batch_norm_custom(
            x,
            self.bn.weight,       # gamma
            self.bn.bias,         # beta
            self.bn.running_mean,
            self.bn.running_var,
            eps_tensor
        )
        return y
