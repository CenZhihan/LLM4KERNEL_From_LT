static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // Input0: x [B, M], Input1: weight [N, M], Input2: bias [N]
    const auto *xShape = context->GetInputShape(0);
    const auto *wShape = context->GetInputShape(1);
    if (xShape == nullptr || wShape == nullptr) {
        return ge::GRAPH_FAILED;
    }

    // ND layout assumed: dims [B, M] and [N, M]
    const int64_t B = (xShape->GetOriginShape().GetDimNum() >= 1) ? xShape->GetOriginShape().GetDim(0) : 1;
    const int64_t M = (xShape->GetOriginShape().GetDimNum() >= 2) ? xShape->GetOriginShape().GetDim(1) : 1;
    const int64_t N = (wShape->GetOriginShape().GetDimNum() >= 1) ? wShape->GetOriginShape().GetDim(0) : 1;

    // Infer output shape: [B, 1]
    auto *yShape = context->GetOutputShape(0);
    if (yShape != nullptr) {
        gert::Shape y;
        y.SetDimNum(2);
        y.SetDim(0, B);
        y.SetDim(1, 1);
        *yShape = y;
    }

    // Prepare Matmul tiling for NORMAL layout
    auto platform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    matmul_tiling::MatmulApiTiling mmTiling(platform);
    mmTiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    // B is transposed at compute time, weight is [N, M], so use transposeB = true
    mmTiling.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    mmTiling.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    mmTiling.SetBiasType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    mmTiling.EnableBias(true);
    // single M, N, K
    mmTiling.SetShape(static_cast<int32_t>(M), static_cast<int32_t>(N), static_cast<int32_t>(M));
    // org M, N, Ka, Kb
    mmTiling.SetOrgShape(static_cast<int32_t>(M), static_cast<int32_t>(N), static_cast<int32_t>(M));
    // Use full AI Core memory
    mmTiling.SetBufferSpace(-1, -1, -1);

    optiling::TCubeTiling cubeTiling;
    int ret = mmTiling.GetTiling(cubeTiling);
    if (ret != 0) {
        return ge::GRAPH_FAILED;
    }

    // Fill raw tiling buffer: [B, M, N, reserved] + TCubeTiling bytes
    // Header: 4 x uint32_t
    const uint32_t headerCount = 4;
    const uint32_t headerBytes = headerCount * sizeof(uint32_t);
    const uint32_t cubeBytes = sizeof(optiling::TCubeTiling);
    const uint32_t totalBytes = headerBytes + cubeBytes;

    auto rawBuf = context->GetRawTilingData();
    if (rawBuf->GetCapacity() < totalBytes) {
        return ge::GRAPH_FAILED;
    }
    uint8_t *buf = reinterpret_cast<uint8_t*>(rawBuf->GetData());
    uint32_t *hdr = reinterpret_cast<uint32_t*>(buf);
    hdr[0] = static_cast<uint32_t>(B);
    hdr[1] = static_cast<uint32_t>(M);
    hdr[2] = static_cast<uint32_t>(N);
    hdr[3] = 1; // isTransposeB = true

    // Copy TCubeTiling after header
    std::memcpy(buf + headerBytes, reinterpret_cast<void*>(&cubeTiling), cubeBytes);
    rawBuf->SetDataSize(totalBytes);

    // Set block dim (can be tuned). Use min(B, 8)
    const uint32_t blockDim = static_cast<uint32_t>((B > 8) ? 8 : (B > 0 ? B : 1));
    context->SetBlockDim(blockDim);

    // Workspace requirement:
    // 1) temp C buffer for matmul result: B * N * sizeof(float)
    size_t *wk = context->GetWorkspaceSizes(1);
    wk[0] = static_cast<size_t>(B) * static_cast<size_t>(N) * sizeof(float);

    return ge::GRAPH_SUCCESS;

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using A_T = float;
using B_T = float;
using C_T = float;

// A: x [B, M], B: weight [N, M], compute with transposeB(true) so effective B^T is [M, N].
// Output temp C: [B, N], then apply Sigmoid and reduce-sum over N to [B,1].

extern "C" __global__ __aicore__ void matmul_sigmoid_sum_custom(
    GM_ADDR xGM, GM_ADDR weightGM, GM_ADDR biasGM, GM_ADDR yGM, GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    // Read header [B, M, N, isTransposeB]
    auto hdrGM = (__gm__ uint32_t*)tilingGM;
    uint32_t B = hdrGM[0];
    uint32_t M = hdrGM[1];
    uint32_t N = hdrGM[2];
    uint32_t isTransposeBIn = hdrGM[3];

    // Copy TCubeTiling from GM after header
    optiling::TCubeTiling tiling;
    auto srcTiling = reinterpret_cast<__gm__ uint32_t*>(hdrGM + 4);
    auto dstTiling = reinterpret_cast<uint32_t*>(&tiling);
    for (uint32_t i = 0; i < sizeof(optiling::TCubeTiling) / sizeof(uint32_t); ++i) {
        dstTiling[i] = srcTiling[i];
    }

    // Prepare GM tensors for inputs
    AscendC::GlobalTensor<A_T> aGlobal;
    AscendC::GlobalTensor<B_T> bGlobal;
    AscendC::GlobalTensor<C_T> cTempGlobal; // temp C buffer at workspace
    AscendC::GlobalTensor<C_T> biasGlobal;
    AscendC::GlobalTensor<C_T> yGlobal;

    // Set buffers
    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(xGM), B * M);
    bGlobal.SetGlobalBuffer(reinterpret_cast__gm__ B_T*>(weightGM), N * M);
    biasGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(biasGM), N);
    cTempGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(workspaceGM), B * N);
    yGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(yGM), B * 1);

    // Define Matmul types for NORMAL layout
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::NORMAL> aType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, B_T, true,  LayoutMode::NORMAL> bType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::NORMAL> cType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T> biasType;

    constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);
    matmul::Matmul<aType, bType, cType, biasType, MM_CFG> mm;
    AscendC::TPipe pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm);
    mm.Init(&tiling);

    // Set A/B/C/Bias tensors
    // For NORMAL: A [B, M], B [N, M] with transposeBIn=true
    mm.SetTensorA(aGlobal, false);
    mm.SetTensorB(bGlobal, isTransposeBIn);
    mm.SetWorkspace(workspaceGM, 0);
    mm.SetBias(biasGlobal);

    // Execute multi-batch matmul over batch dimension: batchA = B, batchB = 1
    mm.IterateBatch(cTempGlobal, B, 1, false);

    // Post-processing: y = sum(sigmoid(C), dim=1, keepdim=True)
    // Process per row
    // We assume N is 32-byte aligned in elements for float (multiple of 8), as in docs.
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQ;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQ;
    pipe.InitBuffer(inQ, 1, N * sizeof(C_T));
    pipe.InitBuffer(outQ, 1, 1 * sizeof(C_T));

    for (uint32_t b = 0; b < B; ++b) {
        AscendC::LocalTensor<C_T> rowLocal = inQ.AllocTensor<C_T>();
        AscendC::DataCopy(rowLocal, cTempGlobal[b * N], N);
        inQ.EnQue(rowLocal);

        // Compute sigmoid and reduce-sum
        AscendC::LocalTensor<C_T> rowLocalIn = inQ.DeQue<C_T>();
        // In-place sigmoid
        AscendC::Sigmoid(rowLocalIn, rowLocalIn, N);

        AscendC::LocalTensor<C_T> sumLocal = outQ.AllocTensor<C_T>();
        AscendC::ReduceSum(sumLocal, rowLocalIn, N);

        inQ.FreeTensor(rowLocalIn);
        outQ.EnQue(sumLocal);

        AscendC::LocalTensor<C_T> sumOut = outQ.DeQue<C_T>();
        AscendC::DataCopy(yGlobal[b], sumOut, 1);
        outQ.FreeTensor(sumOut);
    }
}
