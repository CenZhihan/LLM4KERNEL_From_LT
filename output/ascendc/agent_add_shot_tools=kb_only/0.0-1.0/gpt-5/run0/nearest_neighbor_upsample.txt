namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    NearestNeighborUpsampleCustomTilingData tiling;

    const gert::Shape* inShape = context->GetInputShape(0);
    // Expect ND with 4 dims: NCHW
    uint32_t dims = inShape->GetOriginShape().GetDimNum();
    if (dims != 4) {
        return ge::GRAPH_FAILED;
    }
    uint32_t n = inShape->GetOriginShape().GetDim(0);
    uint32_t c = inShape->GetOriginShape().GetDim(1);
    uint32_t hIn = inShape->GetOriginShape().GetDim(2);
    uint32_t wIn = inShape->GetOriginShape().GetDim(3);

    const uint32_t scale = 4;
    uint32_t hOut = hIn * scale;
    uint32_t wOut = wIn * scale;

    // Each block handles one (n,c) plane.
    uint32_t blockDim = n * c;
    if (blockDim == 0) {
        blockDim = 1;
    }
    context->SetBlockDim(blockDim);

    // Choose a small stripe height to fit local memory. Ensure >=1.
    uint32_t tileH = (hIn >= 8) ? 8 : ((hIn >= 4) ? 4 : 1);

    tiling.set_n(n);
    tiling.set_c(c);
    tiling.set_hIn(hIn);
    tiling.set_wIn(wIn);
    tiling.set_hOut(hOut);
    tiling.set_wOut(wOut);
    tiling.set_scale(scale);
    tiling.set_tileH(tileH);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *workspaceSizes = context->GetWorkspaceSizes(1);
    workspaceSizes[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    if (x_shape->GetDimNum() != 4) {
        return GRAPH_FAILED;
    }
    uint32_t n = x_shape->GetDim(0);
    uint32_t c = x_shape->GetDim(1);
    uint32_t h = x_shape->GetDim(2);
    uint32_t w = x_shape->GetDim(3);
    uint32_t scale = 4;

    gert::Shape* y_shape = context->GetOutputShape(0);
    y_shape->SetDimNum(4);
    y_shape->SetDim(0, n);
    y_shape->SetDim(1, c);
    y_shape->SetDim(2, h * scale);
    y_shape->SetDim(3, w * scale);
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "nearest_neighbor_upsample_custom_tiling_data.h"

constexpr int32_t IN_QUEUE_NUM = 2;
constexpr int32_t OUT_QUEUE_NUM = 2;

class KernelNearestNeighborUpsample {
public:
    __aicore__ inline KernelNearestNeighborUpsample() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t n, uint32_t c,
                                uint32_t hIn, uint32_t wIn,
                                uint32_t hOut, uint32_t wOut,
                                uint32_t scale, uint32_t tileH)
    {
        this->n = n;
        this->c = c;
        this->hIn = hIn;
        this->wIn = wIn;
        this->hOut = hOut;
        this->wOut = wOut;
        this->scale = scale;
        this->tileH = tileH;

        uint32_t planeInSize = hIn * wIn;
        uint32_t planeOutSize = hOut * wOut;

        // One (n,c) plane per block
        uint32_t planeIdx = AscendC::GetBlockIdx();
        __gm__ float* xBase = reinterpret_cast<__gm__ float*>(x);
        __gm__ float* yBase = reinterpret_cast<__gm__ float*>(y);
        xGm.SetGlobalBuffer(xBase + planeIdx * planeInSize, planeInSize);
        yGm.SetGlobalBuffer(yBase + planeIdx * planeOutSize, planeOutSize);

        // Local buffer sizes for one stripe
        uint32_t inStripeElems = tileH * wIn;
        uint32_t midStripeElems = tileH * wOut;        // after width broadcast
        uint32_t outStripeElems = tileH * scale * wOut; // after height broadcast

        pipe.InitBuffer(inQueueX, IN_QUEUE_NUM, inStripeElems * sizeof(float));
        pipe.InitBuffer(outQueueY, OUT_QUEUE_NUM, outStripeElems * sizeof(float));
        pipe.InitBuffer(tmpQueue, 1, midStripeElems * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t hStart = 0; hStart < hIn; hStart += tileH) {
            uint32_t curH = AscendC::min(tileH, hIn - hStart);
            CopyIn(hStart, curH);
            Compute(hStart, curH);
            CopyOut(hStart, curH);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t hStart, uint32_t curH)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        uint32_t inElems = curH * wIn;
        AscendC::DataCopy(xLocal, xGm[hStart * wIn], inElems);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*hStart*/, uint32_t curH)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpQueue.AllocTensor<float>();     // [curH, wOut]
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();      // [curH*scale, wOut]

        // 1) Broadcast along width: [curH, wIn] -> [curH, wOut]
        uint32_t srcShapeW[2] = { curH, wIn };
        uint32_t dstShapeW[2] = { curH, wOut };
        AscendC::BroadCast<float, 2, 1>(tmpLocal, xLocal, dstShapeW, srcShapeW);

        // 2) Broadcast along height: [curH, wOut] -> [curH * scale, wOut]
        uint32_t srcShapeH[2] = { curH, wOut };
        uint32_t dstShapeH[2] = { curH * scale, wOut };
        AscendC::BroadCast<float, 2, 0>(yLocal, tmpLocal, dstShapeH, srcShapeH);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
        tmpQueue.FreeTensor(tmpLocal);
    }

    __aicore__ inline void CopyOut(uint32_t hStart, uint32_t curH)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        uint32_t outElems = curH * scale * wOut;
        // Output row start after height scaling
        AscendC::DataCopy(yGm[(hStart * scale) * wOut], yLocal, outElems);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, IN_QUEUE_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, OUT_QUEUE_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpQueue;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t n, c;
    uint32_t hIn, wIn, hOut, wOut;
    uint32_t scale;
    uint32_t tileH;
};

extern "C" __global__ __aicore__ void nearest_neighbor_upsample_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelNearestNeighborUpsample op;
    op.Init(x, y,
            tiling_data.n, tiling_data.c,
            tiling_data.hIn, tiling_data.wIn,
            tiling_data.hOut, tiling_data.wOut,
            tiling_data.scale, tiling_data.tileH);
    op.Process();
}
