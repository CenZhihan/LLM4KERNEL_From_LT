kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom_tiling_data.h"

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 1;

class KernelGemmSubAvgLseGeluResidualAdd {
public:
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR residual, GM_ADDR sub, GM_ADDR y,
                                uint32_t batch, uint32_t inFeat, uint32_t outFeat)
    {
        this->batch = batch;
        this->inFeatures = inFeat;
        this->outFeatures = outFeat;

        // Block-level batch split
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (batch + blockNum - 1) / blockNum;
        this->rowStart = blockIdx * rowsPerBlock;
        this->rowEnd = AscendC::Min(batch, rowStart + rowsPerBlock);

        // Set global buffers ranged for this block
        if (rowStart < rowEnd) {
            xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(x) + static_cast<uint64_t>(rowStart) * outFeatures,
                                static_cast<uint64_t>(rowEnd - rowStart) * outFeatures);
            residualGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(residual) + static_cast<uint64_t>(rowStart) * inFeatures,
                                       static_cast<uint64_t>(rowEnd - rowStart) * inFeatures);
            yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(y) + static_cast<uint64_t>(rowStart) * inFeatures,
                                static_cast<uint64_t>(rowEnd - rowStart) * inFeatures);
        } else {
            xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(x), 0);
            residualGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(residual), 0);
            yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(y), 0);
        }
        subGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(sub), outFeatures);

        // Queues for per-row processing
        pipe.InitBuffer(inQueueX, BUFFER_NUM, outFeatures * sizeof(DTYPE));
        pipe.InitBuffer(inQueueResidual, BUFFER_NUM, inFeatures * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, inFeatures * sizeof(DTYPE));

        // Buffer to hold sub vector locally
        pipe.InitBuffer(bufSub, 1, outFeatures * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        if (rowStart >= rowEnd) {
            return;
        }
        // Load sub vector once
        AscendC::LocalTensor<DTYPE> subLocal = bufSub.AllocTensor<DTYPE>();
        AscendC::DataCopy(subLocal, subGm, outFeatures);

        for (uint32_t r = rowStart; r < rowEnd; ++r) {
            // Copy one row of x (gemm output) and residual
            AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
            AscendC::LocalTensor<DTYPE> residualLocal = inQueueResidual.AllocTensor<DTYPE>();
            AscendC::DataCopy(xLocal, xGm[(r - rowStart) * outFeatures], outFeatures);
            AscendC::DataCopy(residualLocal, residualGm[(r - rowStart) * inFeatures], inFeatures);
            inQueueX.EnQue(xLocal);
            inQueueResidual.EnQue(residualLocal);

            ComputeRow(subLocal);

            // Write back
            AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
            AscendC::DataCopy(yGm[(r - rowStart) * inFeatures], yLocal, inFeatures);
            outQueueY.FreeTensor(yLocal);
        }
        bufSub.FreeTensor(subLocal);
    }

private:
    __aicore__ inline void ComputeRow(const AscendC::LocalTensor<DTYPE>& subLocal)
    {
        // Dequeue inputs
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> residualLocal = inQueueResidual.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.AllocTensor<DTYPE>();

        // Step 1: Subtract and compute mean over features (GlobalAvgPool on feature dim)
        DTYPE acc = (DTYPE)0;
        // Loop over outFeatures for mean(x - sub)
        for (uint32_t i = 0; i < outFeatures; ++i) {
            acc += (xLocal.GetValue(i) - subLocal.GetValue(i));
        }
        DTYPE meanVal = acc / static_cast<DTYPE>(outFeatures);

        // Step 2: LogSumExp over dim=1 with keepdim=True.
        // Since the tensor is [B,1] after mean, LSE over size-1 dim is the value itself.
        DTYPE lseVal = meanVal;

        // Step 3: GELU activation (approx) on scalar
        // tanh-based approximation
        const DTYPE k0 = (DTYPE)0.7978845608028654;      // sqrt(2/pi)
        const DTYPE k1 = (DTYPE)0.044715;
        DTYPE x3 = lseVal * lseVal * lseVal;
        DTYPE tanhArg = k0 * (lseVal + k1 * x3);
        // device math tanhf
        DTYPE t = (DTYPE)tanhf((float)tanhArg);
        DTYPE geluVal = (DTYPE)0.5 * lseVal * ((DTYPE)1.0 + t);

        // Step 4: Residual add with broadcast over inFeatures
        for (uint32_t j = 0; j < inFeatures; ++j) {
            yLocal.SetValue(j, residualLocal.GetValue(j) + geluVal);
        }

        // Enqueue output and free inputs
        outQueueY.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueResidual.FreeTensor(residualLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueResidual;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::TBuf<AscendC::TPosition::VECIN> bufSub;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> residualGm;
    AscendC::GlobalTensor<DTYPE> subGm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t batch;
    uint32_t inFeatures;
    uint32_t outFeatures;

    uint32_t rowsPerBlock;
    uint32_t rowStart;
    uint32_t rowEnd;
};

extern "C" __global__ __aicore__ void gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom(
    GM_ADDR x, GM_ADDR residual, GM_ADDR sub, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelGemmSubAvgLseGeluResidualAdd op;
    op.Init(x, residual, sub, y,
            tilingData.batch, tilingData.inFeatures, tilingData.outFeatures);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Fused kernel replaces: Subtract -> GlobalAvgPool (dim=1, keepdim=True) -> LogSumExp (dim=1, keepdim=True) -> GELU -> ResidualAdd
    GEMM (Linear) remains as nn.Linear to reuse highly-optimized GEMM on NPU.
    """
    def __init__(self, in_features, out_features, bias=True):
        super().__init__()
        self.gemm = torch.nn.Linear(in_features, out_features, bias=bias)
        self.subtract = torch.nn.Parameter(torch.randn(out_features))

    def forward(self, x):
        original_x = x
        # GEMM
        x = self.gemm(x)
        # Fused custom AscendC op
        y = custom_ops_lib.gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom(x, original_x, self.subtract)
        return y
