kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose2d_bias_add_clamp_scaling_clamp_divide_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num per queue

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR scale, GM_ADDR y, uint32_t totalLength, uint32_t tileNum, uint32_t tmpSize)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        bGm.SetGlobalBuffer((__gm__ float *)bias + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        sGm.SetGlobalBuffer((__gm__ float *)scale + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueS, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));

        this->tmpSize = tmpSize;
        if (this->tmpSize != 0) {
            pipe.InitBuffer(tmpQueue, BUFFER_NUM, this->tmpSize);
        }
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute();
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
        AscendC::LocalTensor<float> sLocal = inQueueS.AllocTensor<float>();

        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(bLocal, bGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(sLocal, sGm[progress * this->tileLength], this->tileLength);

        inQueueX.EnQue(xLocal);
        inQueueB.EnQue(bLocal);
        inQueueS.EnQue(sLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> bLocal = inQueueB.DeQue<float>();
        AscendC::LocalTensor<float> sLocal = inQueueS.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // y = x + bias
        AscendC::Add(yLocal, xLocal, bLocal, this->tileLength);

        // clamp to [0, 1]
        if (this->tmpSize != 0) {
            AscendC::LocalTensor<uint8_t> tmpLocal = tmpQueue.AllocTensor<uint8_t>();
            AscendC::ClampMin<float>(yLocal, yLocal, tmpLocal, static_cast<float>(0.0f), this->tileLength);
            AscendC::ClampMax<float>(yLocal, yLocal, tmpLocal, static_cast<float>(1.0f), this->tileLength);
            tmpQueue.FreeTensor(tmpLocal);
        } else {
            // Use high-level API that allocates tmp internally
            AscendC::ClampMin<float>(yLocal, yLocal, static_cast<float>(0.0f), this->tileLength);
            AscendC::ClampMax<float>(yLocal, yLocal, static_cast<float>(1.0f), this->tileLength);
        }

        // y = y * scale (element-wise)
        AscendC::Mul(yLocal, yLocal, sLocal, this->tileLength);

        // clamp again to [0, 1]
        if (this->tmpSize != 0) {
            AscendC::LocalTensor<uint8_t> tmpLocal2 = tmpQueue.AllocTensor<uint8_t>();
            AscendC::ClampMin<float>(yLocal, yLocal, tmpLocal2, static_cast<float>(0.0f), this->tileLength);
            AscendC::ClampMax<float>(yLocal, yLocal, tmpLocal2, static_cast<float>(1.0f), this->tileLength);
            tmpQueue.FreeTensor(tmpLocal2);
        } else {
            AscendC::ClampMin<float>(yLocal, yLocal, static_cast<float>(0.0f), this->tileLength);
            AscendC::ClampMax<float>(yLocal, yLocal, static_cast<float>(1.0f), this->tileLength);
        }

        // y = y / scale (element-wise)
        AscendC::Div(yLocal, yLocal, sLocal, this->tileLength);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueB.FreeTensor(bLocal);
        inQueueS.FreeTensor(sLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueB;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueS;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpQueue;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> sGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t tmpSize;
};

extern "C" __global__ __aicore__ void conv_transpose2d_bias_add_clamp_scaling_clamp_divide_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR scale, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, scale, y, tiling_data.totalLength, tiling_data.tileNum, tiling_data.tmpSize);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized: keep ConvTranspose2d, fuse bias add + clamp + scale + clamp + divide into a custom AscendC op.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, bias_shape, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        y = self.conv_transpose(x)
        # Prepare bias and scale tensors to match y's shape for efficient elementwise fusion
        bias_expanded = self.bias.view(1, -1, 1, 1).expand_as(y)
        scale_tensor = torch.full_like(y, self.scaling_factor)
        return custom_ops_lib.conv_transpose2d_bias_add_clamp_scaling_clamp_divide_custom(y, bias_expanded, scale_tensor)
