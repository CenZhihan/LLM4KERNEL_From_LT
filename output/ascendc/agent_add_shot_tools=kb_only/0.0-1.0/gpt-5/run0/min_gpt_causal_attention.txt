host_operator_src="""
#include <cstdint>
#include <cstring>
#include "register/op_def_registry.h"
#include "min_gpt_causal_attention_custom_tiling.h"

// AscendC Matmul tiling APIs
#include "tiling/platform/platform_ascendc.h"
#include "tiling/matmul_tiling.h"

namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 输入 q, k 形状：[B, nh, T, hs]
    const gert::Shape* qShape = context->GetInputShape(0);
    // 读取原始维度
    int64_t B   = qShape->GetStorageShape().GetDim(0);
    int64_t nh  = qShape->GetStorageShape().GetDim(1);
    int64_t T   = qShape->GetStorageShape().GetDim(2);
    int64_t hs  = qShape->GetStorageShape().GetDim(3);

    int32_t M = static_cast<int32_t>(T);
    int32_t N = static_cast<int32_t>(T);
    int32_t K = static_cast<int32_t>(hs);
    int32_t batch = static_cast<int32_t>(B * nh);

    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    matmul_tiling::MultiCoreMatmulTiling tiling(ascendcPlatform);

    // 配置多核数量（示例值，可由工具自动给出）
    tiling.SetDim(1);

    // 设置A/B/C类型（ND，float），与KB一致
    tiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                    matmul_tiling::DataType::DT_FLOAT);
    tiling.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                    matmul_tiling::DataType::DT_FLOAT);
    tiling.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                    matmul_tiling::DataType::DT_FLOAT);
    tiling.SetBiasType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                    matmul_tiling::DataType::DT_FLOAT);

    // 单次Matmul形状：A[M,K] * B[K,N] -> C[M,N]
    tiling.SetShape(M, N, K);
    tiling.SetOrgShape(M, N, K);
    tiling.SetBias(false);

    // NORMAL布局下配置批次信息，参照KB SetBatchInfoForNormal
    tiling.SetBatchInfoForNormal(batch, batch, M, N, K);
    tiling.SetBufferSpace(-1, -1, -1);

    optiling::TCubeTiling tilingData;
    int ret = tiling.GetTiling(tilingData);
    if (ret != 0) {
        return ge::GRAPH_FAILED;
    }

    // 设置BlockDim（来自多核tiling）
    context->SetBlockDim(tiling.GetCoreNum());

    // 保存TCubeTiling到RawTilingData
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(sizeof(optiling::TCubeTiling));

    // 追加自定义Extra信息（供Device侧计算矩阵步长与批次）
    MinGptCausalAttentionCustomExtra extra;
    extra.M = static_cast<uint32_t>(M);
    extra.N = static_cast<uint32_t>(N);
    extra.K = static_cast<uint32_t>(K);
    extra.batch   = static_cast<uint32_t>(batch);
    extra.strideA = static_cast<uint32_t>(M * K);
    extra.strideB = static_cast<uint32_t>(K * N);
    extra.strideC = static_cast<uint32_t>(M * N);
    WriteExtraToRawTilingData(context, extra);

    // workspace
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

    // 输入q形状：[B, nh, T, hs] -> 输出att形状：[B, nh, T, T]
    const gert::Shape* qShape = context->GetInputShape(0);
    int64_t B   = qShape->GetStorageShape().GetDim(0);
    int64_t nh  = qShape->GetStorageShape().GetDim(1);
    int64_t T   = qShape->GetStorageShape().GetDim(2);

    gert::Shape* yShape = context->GetOutputShape(0);
    gert::Shape result;
    result.SetDimNum(4);
    result.SetDim(0, B);
    result.SetDim(1, nh);
    result.SetDim(2, T);

    return GRAPH_SUCCESS;

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"
#define __NPU_TILING__
#include "min_gpt_causal_attention_custom_tiling.h"

// 使用AscendC Matmul高阶API（参考KB 5.2.1与IterateNBatch说明）
// 计算 att = q @ k^T
// 输入q/k形状：[B, nh, T, hs]（NORMAL布局），输出att形状：[B, nh, T, T]

extern "C" __global__ __aicore__ void min_gpt_causal_attention_custom(GM_ADDR qGM, GM_ADDR kGM, GM_ADDR attGM,
                                                                      GM_ADDR workspaceGM, GM_ADDR tilingGM) {
    // 1) 从GM中读取TCubeTiling（参考KB示例）
    optiling::TCubeTiling tiling;
    auto tGM = (__gm__ uint32_t*)tilingGM;
    auto tLocal = (uint32_t*)&tiling;
    for (int i = 0; i < (int)(sizeof(optiling::TCubeTiling) / sizeof(uint32_t)); ++i) {
        *tLocal++ = *tGM++;
    }

    // 2) 读取追加的自定义Extra（host侧追加）
    optiling::MinGptCausalAttentionCustomExtra extra;
    auto eLocal = (uint32_t*)&extra;
    for (int i = 0; i < (int)(sizeof(optiling::MinGptCausalAttentionCustomExtra) / sizeof(uint32_t)); ++i) {
        *eLocal++ = *tGM++;
    }

    const uint32_t M = extra.M;
    const uint32_t N = extra.N;
    const uint32_t K = extra.K;
    const uint32_t batch = extra.batch;
    const uint32_t strideA = extra.strideA; // M*K
    const uint32_t strideB = extra.strideB; // K*N
    const uint32_t strideC = extra.strideC; // M*N

    // 3) 准备GM Tensor
    AscendC::GlobalTensor<float> qGlobal;
    AscendC::GlobalTensor<float> kGlobal;
    AscendC::GlobalTensor<float> attGlobal;
    uint64_t sizeABytes = (uint64_t)batch * (uint64_t)strideA * sizeof(float);
    uint64_t sizeBBytes = (uint64_t)batch * (uint64_t)strideB * sizeof(float);
    uint64_t sizeCBytes = (uint64_t)batch * (uint64_t)strideC * sizeof(float);
    qGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(qGM), sizeABytes);
    kGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(kGM), sizeBBytes);
    attGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(attGM), sizeCBytes);

    // 4) 创建Matmul对象，NORMAL布局；B矩阵使能转置（参考KB：ISTRANS与SetTensorB传参）
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> aType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, true,  LayoutMode::NORMAL> bType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> cType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float> biasType;

    matmul::Matmul<aType, bType, cType, biasType> mm;
    AscendC::TPipe pipe;
    g_cubeTPipePtr = &pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm);

    // 5) 初始化与设置Tensor
    mm.Init(&tiling);
    mm.SetTensorA(qGlobal[0], false);
    mm.SetTensorB(kGlobal[0], true);
    mm.SetWorkspace(workspaceGM, 0);

    // 6) 执行批次Matmul
    // NORMAL布局下，批次信息通过host侧SetBatchInfoForNormal设置
    // 这里一次性处理所有batch，设置batchLoop=1，batchA=batch，batchB=batch
    // 并提供每个矩阵之间的步长（元素数）
    const bool enSequentialWrite = true;
    mm.IterateNBatch<true /*sync*/, false /*wait*/>(1, batch, batch, enSequentialWrite,
                                                    strideA, strideB, strideC);

    // 将结果映射到attGM（C），按顺序写入
    // IterateNBatch内部已完成写出
}

model_src='''
import math
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

# 使用自定义AscendC算子计算注意力得分 att = q @ k^T（批量多头）
class ModelNew(torch.nn.Module):
    def __init__(self, n_embd, n_head, attn_pdrop, resid_pdrop, max_seqlen):
        super().__init__()
        assert n_embd % n_head == 0
        self.c_attn = nn.Linear(n_embd, 3 * n_embd)
        self.c_proj = nn.Linear(n_embd, n_embd)
        self.attn_dropout = nn.Dropout(attn_pdrop)
        self.resid_dropout = nn.Dropout(resid_pdrop)
        self.register_buffer("bias", torch.tril(torch.ones(max_seqlen, max_seqlen))
                                     .view(1, 1, max_seqlen, max_seqlen))
        self.n_head = n_head
        self.n_embd = n_embd

    def forward(self, x):
        B, T, C = x.size()
        nh = self.n_head
        hs = C // nh

        # 线性映射得到q,k,v
        q, k, v = self.c_attn(x).split(self.n_embd, dim=2)
        q = q.view(B, T, nh, hs).transpose(1, 2).contiguous()  # (B, nh, T, hs)
        k = k.view(B, T, nh, hs).transpose(1, 2).contiguous()  # (B, nh, T, hs)
        v = v.view(B, T, nh, hs).transpose(1, 2).contiguous()  # (B, nh, T, hs)

        # 使用自定义AscendC算子计算 att_pre = q @ k^T
        att_pre = custom_ops_lib.min_gpt_causal_attention_custom(q, k)

        # 缩放 + 掩码 + softmax + dropout
        att = att_pre * (1.0 / math.sqrt(hs))
        att = att.masked_fill(self.bias[:, :, :T, :T] == 0, float("-inf"))
        att = F.softmax(att, dim=-1)
        att = self.attn_dropout(att)

        # y = att @ v
        y = torch.matmul(att, v)  # (B, nh, T, hs)
        y = y.transpose(1, 2).contiguous().view(B, T, C)

        # 输出投影
        y = self.resid_dropout(self.c_proj(y))
        return y
'''
