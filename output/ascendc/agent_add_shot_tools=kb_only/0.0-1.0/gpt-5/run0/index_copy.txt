#include "index_copy_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {

const uint32_t BLOCK_DIM = 1;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    IndexCopyCustomTilingData tiling;

    auto* xShape = context->GetInputShape(0);
    auto* idxShape = context->GetInputShape(1);
    auto* srcShape = context->GetInputShape(2);

    uint64_t xSize = xShape->GetOriginShape().GetShapeSize();      // N * C
    uint64_t mSize = idxShape->GetOriginShape().GetShapeSize();    // M
    uint64_t srcSize = srcShape->GetOriginShape().GetShapeSize();  // M * C

    uint32_t M = static_cast<uint32_t>(mSize);
    uint32_t C = static_cast<uint32_t>(srcSize / (mSize == 0 ? 1 : mSize));
    uint32_t N = static_cast<uint32_t>(xSize / (C == 0 ? 1 : C));

    context->SetBlockDim(BLOCK_DIM);
    tiling.set_rows(N);
    tiling.set_cols(C);
    tiling.set_indicesNum(M);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* z_shape = context->GetOutputShape(0);

    return GRAPH_SUCCESS;

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "index_copy_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelIndexCopy {
public:
    __aicore__ inline KernelIndexCopy() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR indices, GM_ADDR src, GM_ADDR z,
                                uint32_t rows, uint32_t cols, uint32_t indicesNum)
    {
        this->rows = rows;
        this->cols = cols;
        this->indicesNum = indicesNum;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(x), rows * cols);
        idxGm.SetGlobalBuffer(reinterpret_cast<__gm__ int32_t*>(indices), indicesNum);
        srcGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(src), indicesNum * cols);
        zGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(z), rows * cols);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, cols * sizeof(float));
        pipe.InitBuffer(inQueueSrc, BUFFER_NUM, cols * sizeof(float));
        pipe.InitBuffer(inQueueIdx, BUFFER_NUM, 1 * sizeof(int32_t));
    }

    __aicore__ inline void Process()
    {
        CopyBaseToOutput();
        ScatterCopy();
    }

private:
    __aicore__ inline void CopyBaseToOutput()
    {
        for (uint32_t r = 0; r < rows; ++r) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[r * cols], cols);
            inQueueX.EnQue(xLocal);

            AscendC::LocalTensor<float> xLocalDeq = inQueueX.DeQue<float>();
            AscendC::DataCopy(zGm[r * cols], xLocalDeq, cols);
            inQueueX.FreeTensor(xLocalDeq);
        }
    }

    __aicore__ inline void ScatterCopy()
    {
        for (uint32_t i = 0; i < indicesNum; ++i) {
            // load one index
            AscendC::LocalTensor<int32_t> idxLocal = inQueueIdx.AllocTensor<int32_t>();
            AscendC::DataCopy(idxLocal, idxGm[i], 1);
            inQueueIdx.EnQue(idxLocal);

            // load one source row
            AscendC::LocalTensor<float> srcLocal = inQueueSrc.AllocTensor<float>();
            AscendC::DataCopy(srcLocal, srcGm[i * cols], cols);
            inQueueSrc.EnQue(srcLocal);

            AscendC::LocalTensor<int32_t> idxLocalDeq = inQueueIdx.DeQue<int32_t>();
            AscendC::LocalTensor<float> srcLocalDeq = inQueueSrc.DeQue<float>();

            // read index value from local tensor
            int32_t dstRow = idxLocalDeq.GetValue(0); // obtain scalar from LocalTensor

            // write to destination row in z
            AscendC::DataCopy(zGm[static_cast<uint32_t>(dstRow) * cols], srcLocalDeq, cols);

            inQueueIdx.FreeTensor(idxLocalDeq);
            inQueueSrc.FreeTensor(srcLocalDeq);
        }
    }

private:
    uint32_t rows;
    uint32_t cols;
    uint32_t indicesNum;

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueSrc;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueIdx;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<int32_t> idxGm;
    AscendC::GlobalTensor<float> srcGm;
    AscendC::GlobalTensor<float> zGm;
};

extern "C" __global__ __aicore__ void index_copy_custom(GM_ADDR x, GM_ADDR indices, GM_ADDR src, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelIndexCopy op;
    op.Init(x, indices, src, z, tiling_data.rows, tiling_data.cols, tiling_data.indicesNum);
    op.Process();
}
