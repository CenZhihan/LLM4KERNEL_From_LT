    tiling.set_c(c);
    tiling.set_h(h);

    tiling.set_outW(outW);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "upsample_grid_sample_custom_tiling_data.h"

// 参考 AscendC API 模板用法：使用 LocalTensor 队列、DataCopy、手写计算流程。
// 本示例实现按批次和通道的两阶段计算：先双线性上采样，再按仿射网格执行 grid_sample（均为 align_corners=false）。

constexpr int32_t BUFFER_NUM = 1;

struct UpsampleGridSampleVecTiling {
  uint32_t n;
  uint32_t c;
  uint32_t h;
  uint32_t w;
  uint32_t outH;
  uint32_t outW;
  uint32_t totalLength;
  uint32_t tileNum;
};

class KernelUpsampleGridSample {
public:
    __aicore__ inline KernelUpsampleGridSample() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR theta, GM_ADDR y, GM_ADDR tiling)
    {
        GET_TILING_DATA(tilingData, tiling);
        this->n = tilingData.n();
        this->c = tilingData.c();
        this->h = tilingData.h();
        this->w = tilingData.w();
        this->outH = tilingData.outH();
        this->outW = tilingData.outW();
        // 扁平长度
        this->inSize = this->n * this->c * this->h * this->w;
        this->thetaSize = this->n * 6; // 每个 batch 2x3
        this->outSize = this->n * this->c * this->outH * this->outW;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(x), this->inSize);
        thetaGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(theta), this->thetaSize);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(y), this->outSize);

        const uint32_t inBytes = this->inSize * sizeof(float);
        const uint32_t thetaBytes = this->thetaSize * sizeof(float);
        const uint32_t outBytes = this->outSize * sizeof(float);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, inBytes);
        pipe.InitBuffer(inQueueTheta, BUFFER_NUM, thetaBytes);
        pipe.InitBuffer(outQueueY, BUFFER_NUM, outBytes);
        // 临时 UB 缓冲，用于上采样后的中间结果
        pipe.InitBuffer(tmpQueueUp, BUFFER_NUM, outBytes);
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline uint32_t Index4d(uint32_t n_, uint32_t c_, uint32_t h_, uint32_t w_,
                                       uint32_t C, uint32_t H, uint32_t W) {
        return ((n_ * C + c_) * H + h_) * W + w_;
    }

    __aicore__ inline float SafeRead(const AscendC::LocalTensor<float>& buf, int32_t n_, int32_t c_, int32_t hh, int32_t ww,
                                     uint32_t N, uint32_t C, uint32_t H, uint32_t W) {
        if (hh < 0 || ww < 0 || hh >= (int32_t)H || ww >= (int32_t)W) {
            return 0.0f;
        }
        uint32_t idx = Index4d((uint32_t)n_, (uint32_t)c_, (uint32_t)hh, (uint32_t)ww, C, H, W);
        return buf.GetValue(idx);
    }

    __aicore__ inline float BilinearSample(const AscendC::LocalTensor<float>& buf, int32_t n_, int32_t c_,
                                           float y, float x, uint32_t N, uint32_t C, uint32_t H, uint32_t W) {
        int32_t x0 = static_cast<int32_t>(AscendC::Floor(x));
        int32_t y0 = static_cast<int32_t>(AscendC::Floor(y));
        int32_t x1 = x0 + 1;
        int32_t y1 = y0 + 1;
        float dx = x - static_cast<float>(x0);
        float dy = y - static_cast<float>(y0);
        float v00 = SafeRead(buf, n_, c_, y0, x0, N, C, H, W);
        float v01 = SafeRead(buf, n_, c_, y0, x1, N, C, H, W);
        float v10 = SafeRead(buf, n_, c_, y1, x0, N, C, H, W);
        float v11 = SafeRead(buf, n_, c_, y1, x1, N, C, H, W);
        float v0 = v00 * (1.f - dx) + v01 * dx;
        float v1 = v10 * (1.f - dx) + v11 * dx;
        return v0 * (1.f - dy) + v1 * dy;
    }

    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> thetaLocal = inQueueTheta.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm, this->inSize);
        AscendC::DataCopy(thetaLocal, thetaGm, this->thetaSize);
        inQueueX.EnQue(xLocal);
        inQueueTheta.EnQue(thetaLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> thetaLocal = inQueueTheta.DeQue<float>();
        AscendC::LocalTensor<float> upLocal = tmpQueueUp.AllocTensor<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // 1) 先对 x 做双线性上采样到 upLocal（scale=2, align_corners=false）
        for (uint32_t nIdx = 0; nIdx < n; ++nIdx) {
            for (uint32_t cIdx = 0; cIdx < c; ++cIdx) {
                for (uint32_t oh = 0; oh < outH; ++oh) {
                    // 映射到输入坐标（align_corners=false）
                    float inY = (static_cast<float>(oh) + 0.5f) / 2.0f - 0.5f; // = (oh - 0.5)/2
                    for (uint32_t ow = 0; ow < outW; ++ow) {
                        float inX = (static_cast<float>(ow) + 0.5f) / 2.0f - 0.5f;
                        float v = BilinearSample(xLocal, (int32_t)nIdx, (int32_t)cIdx, inY, inX, n, c, h, w);
                        uint32_t upIdx = Index4d(nIdx, cIdx, oh, ow, c, outH, outW);
                        upLocal.SetValue(upIdx, v);
                    }
                }
            }
        }

        AscendC::PipeBarrier<PIPE_V>();

        // 2) 生成与输出同尺寸的采样网格并对 upLocal 执行 grid_sample（bilinear + zeros padding, align_corners=false）
        for (uint32_t nIdx = 0; nIdx < n; ++nIdx) {
            // 读取该 batch 的 theta 2x3
            uint32_t baseT = nIdx * 6;
            float a00 = thetaLocal.GetValue(baseT + 0);
            float a01 = thetaLocal.GetValue(baseT + 1);
            float a02 = thetaLocal.GetValue(baseT + 2);
            float a10 = thetaLocal.GetValue(baseT + 3);
            float a11 = thetaLocal.GetValue(baseT + 4);
            float a12 = thetaLocal.GetValue(baseT + 5);
            for (uint32_t cIdx = 0; cIdx < c; ++cIdx) {
                for (uint32_t oh = 0; oh < outH; ++oh) {
                    // 目标归一化坐标（align_corners=false）
                    float yn = (static_cast<float>(oh) + 0.5f) / static_cast<float>(outH) * 2.f - 1.f;
                    for (uint32_t ow = 0; ow < outW; ++ow) {
                        float xn = (static_cast<float>(ow) + 0.5f) / static_cast<float>(outW) * 2.f - 1.f;
                        // 仿射变换
                        float xsN = a00 * xn + a01 * yn + a02;
                        float ysN = a10 * xn + a11 * yn + a12;
                        // 归一化 -> 上采样域真实坐标（align_corners=false）
                        float xs = (xsN + 1.f) * 0.5f * static_cast<float>(outW) - 0.5f;
                        float ys = (ysN + 1.f) * 0.5f * static_cast<float>(outH) - 0.5f;

                        float val = BilinearSample(upLocal, (int32_t)nIdx, (int32_t)cIdx, ys, xs, n, c, outH, outW);
                        uint32_t yIdx = Index4d(nIdx, cIdx, oh, ow, c, outH, outW);
                        yLocal.SetValue(yIdx, val);
                    }
                }
            }
        }

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueTheta.FreeTensor(thetaLocal);
        tmpQueueUp.FreeTensor(upLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm, yLocal, this->outSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueTheta;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> tmpQueueUp;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> thetaGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t n, c, h, w, outH, outW;
    uint32_t inSize, thetaSize, outSize;
};

extern "C" __global__ __aicore__ void upsample_grid_sample_custom(GM_ADDR x, GM_ADDR theta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    KernelUpsampleGridSample op;
    op.Init(x, theta, y, tiling);
    op.Process();
}
