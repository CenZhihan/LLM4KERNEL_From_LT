host_operator_src="""
#include "vanilla_rnn_custom_tiling.h"
#include "register/op_def_registry.h"

namespace {
constexpr uint32_t kDefaultBlockDim = 8;
constexpr uint32_t kTileNum = 8;

inline uint64_t GetDimSafe(const gert::Shape& s, int idx) {
    if (idx < 0 || idx >= (int)s.GetOriginShape().GetDimNum()) return 0;
    return s.GetOriginShape().GetDim(idx);
}
}

namespace optiling {

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 获取输入形状
    const gert::Shape* x_shape = context->GetInputShape(0);
    const gert::Shape* h_shape = context->GetInputShape(1);

    uint64_t rows  = GetDimSafe(*x_shape, 0);
    uint64_t colsX = GetDimSafe(*x_shape, 1);
    uint64_t colsH = GetDimSafe(*h_shape, 1);

    // 基本防御
    if (rows == 0 || colsX == 0 || colsH == 0) {
        return ge::GRAPH_FAILED;
    }

    // 配置 BlockDim
    context->SetBlockDim(kDefaultBlockDim);

    // 填充 tiling 数据
    VanillaRnnCustomTilingData tiling;
    tiling.set_rows(static_cast<uint32_t>(rows));
    tiling.set_colsX(static_cast<uint32_t>(colsX));
    tiling.set_colsH(static_cast<uint32_t>(colsH));
    tiling.set_tileNum(kTileNum);

    // 写入 RawTilingData
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 本算子无额外 workspace
    size_t *workspace = context->GetWorkspaceSizes(1);
    workspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}

} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    const gert::Shape* h_shape = context->GetInputShape(1);
    if (x_shape == nullptr || h_shape == nullptr) {
        return GRAPH_FAILED;
    }

    // 输出形状: [B, I + H]
    uint64_t rows  = x_shape->GetOriginShape().GetDim(0);
    uint64_t colsX = x_shape->GetOriginShape().GetDim(1);
    uint64_t colsH = h_shape->GetOriginShape().GetDim(1);

    gert::Shape* y_shape = context->GetOutputShape(0);
    gert::Shape::OriginShape out;
    out.SetDimNum(2);
    out.SetDim(0, rows);
    out.SetDim(1, colsX + colsH);
    y_shape->SetOriginShape(out);
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
} // namespace ge

namespace ops {
class VanillaRnnCustom : public OpDef {
public:
    explicit VanillaRnnCustom(const char* name) : OpDef(name)

        this->Input("x")

            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(VanillaRnnCustom);
} // namespace ops

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "vanilla_rnn_custom_tiling_data.h"

// 将 x:[B,I] 与 h:[B,H] 按行拼接为 y:[B, I+H]
// 简单按行 + 分块复制实现
constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t TILE_ELEMS = 1024; // 行内拷贝分块大小(元素数)，可根据实际调优

using DTYPE = float;

class KernelVanillaRnnConcat {
public:
    __aicore__ inline KernelVanillaRnnConcat() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR h, GM_ADDR y,
                                uint32_t rows, uint32_t colsX, uint32_t colsH, uint32_t tileNum)
    {
        this->rows  = rows;
        this->colsX = colsX;
        this->colsH = colsH;
        this->colsY = colsX + colsH;
        this->tileNum = tileNum;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t bid = AscendC::GetBlockIdx();
        // 均匀划分行到每个 block
        uint32_t rowsPerBlock = (rows + blockNum - 1) / blockNum;
        rowStart = bid * rowsPerBlock;
        rowEnd   = AscendC::min(rows, rowStart + rowsPerBlock);
        localRows = (rowEnd > rowStart) ? (rowEnd - rowStart) : 0;

        // 设置 GM 指针范围
        xGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(x) + static_cast<uint64_t>(rowStart) * colsX,
                                static_cast<uint64_t>(localRows) * colsX);
        hGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(h) + static_cast<uint64_t>(rowStart) * colsH,
                                static_cast<uint64_t>(localRows) * colsH);
        yGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(y) + static_cast<uint64_t>(rowStart) * colsY,
                                static_cast<uint64_t>(localRows) * colsY);

        // 队列缓冲
        pipe.InitBuffer(inQueue,  BUFFER_NUM, TILE_ELEMS * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        if (localRows == 0) {
            return;
        }
        for (uint32_t r = 0; r < localRows; ++r) {
            // 1) 复制 x 的一行到 y 的前半部分
            RowCopy(xGlobal, colsX, r, /*yOffset=*/0);

            // 2) 复制 h 的一行到 y 的后半部分
            RowCopy(hGlobal, colsH, r, /*yOffset=*/colsX);
        }
    }

private:
    __aicore__ inline void RowCopy(AscendC::GlobalTensor<DTYPE>& srcGlobal, uint32_t srcCols,
                                   uint32_t rLocal, uint32_t yOffset)
    {
        // 行内按 TILE_ELEMS 分块复制
        for (uint32_t offset = 0; offset < srcCols; offset += TILE_ELEMS) {
            uint32_t len = AscendC::min(TILE_ELEMS, srcCols - offset);

            // 从源行取一段到本地
            AscendC::LocalTensor<DTYPE> buf = inQueue.AllocTensor<DTYPE>();
            AscendC::DataCopy(buf, srcGlobal[static_cast<uint64_t>(rLocal) * srcCols + offset], len);

            // 将本地数据写到 y 的对应位置
            AscendC::DataCopy(
                yGlobal[static_cast<uint64_t>(rLocal) * colsY + yOffset + offset],
                buf, len);

            inQueue.FreeTensor(buf);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;

    AscendC::GlobalTensor<DTYPE> xGlobal;
    AscendC::GlobalTensor<DTYPE> hGlobal;
    AscendC::GlobalTensor<DTYPE> yGlobal;

    uint32_t rows;
    uint32_t colsX;
    uint32_t colsH;
    uint32_t colsY;
    uint32_t tileNum;

    uint32_t rowStart;
    uint32_t rowEnd;
    uint32_t localRows;
};

extern "C" __global__ __aicore__ void vanilla_rnn_custom(GM_ADDR x, GM_ADDR h, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelVanillaRnnConcat op;
    op.Init(x, h, y,
            tiling_data.rows,
            tiling_data.colsX,
            tiling_data.colsH,
            tiling_data.tileNum);
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor vanilla_rnn_custom_impl_npu(const at::Tensor& x, const at::Tensor& h) {
    TORCH_CHECK(x.dim() == 2 && h.dim() == 2, "Inputs must be 2D tensors");
    TORCH_CHECK(x.size(0) == h.size(0), "Batch size must match");
    auto B = x.size(0);
    auto I = x.size(1);
    auto H = h.size(1);
    at::Tensor y = at::empty({B, I + H}, x.options());
    EXEC_NPU_CMD(aclnnVanillaRnnCustom, x, h, y);
    return y;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("vanilla_rnn_custom", &vanilla_rnn_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("vanilla_rnn_custom", &vanilla_rnn_custom_impl_npu, "Concatenate x and h on last dim");
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size: int, hidden_size: int, output_size: int):
        super(ModelNew, self).__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.hidden = torch.randn((batch_size, hidden_size))

        self.i2h = torch.nn.Linear(input_size + hidden_size, hidden_size)
        self.h2o = torch.nn.Linear(hidden_size, output_size)
        self.tanh = torch.nn.Tanh()

    def forward(self, x: torch.Tensor, initial_hidden=None) -> torch.Tensor:
        if initial_hidden is not None:
            self.hidden.copy_(initial_hidden)
        self.hidden = self.hidden.to(x.device)

        # 使用自定义 AscendC 内核完成拼接
        combined = custom_ops_lib.vanilla_rnn_custom(x, self.hidden)

        # 后续与原模型一致：线性 + tanh + 线性
        self.hidden = self.tanh(self.i2h(combined))
        output = self.h2o(self.hidden)
        return output
