#include "shufflenet_custom_tiling.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 1; // one whole plane per tile for simplicity

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    const gert::Shape* in_shape = context->GetInputShape(0);
    std::vector<int64_t> dims = in_shape->GetOriginShape().GetDims();
    // Expect NCHW
    uint32_t n = dims.size() > 0 ? static_cast<uint32_t>(dims[0]) : 1;
    uint32_t c = dims.size() > 1 ? static_cast<uint32_t>(dims[1]) : 1;
    uint32_t h = dims.size() > 2 ? static_cast<uint32_t>(dims[2]) : 1;
    uint32_t w = dims.size() > 3 ? static_cast<uint32_t>(dims[3]) : 1;

    uint32_t planeSize = h * w;
    uint32_t totalPlanes = n * c;

    ShufflenetCustomTilingData tiling;
    tiling.set_n(n);
    tiling.set_c(c);
    tiling.set_h(h);
    tiling.set_w(w);
    tiling.set_planeSize(planeSize);
    tiling.set_totalPlanes(totalPlanes);
    tiling.set_tileNum(TILE_NUM);

    // Default groups = 3 if not provided by runtime attribute; keep conservative default
    uint32_t groups = 3;
    tiling.set_groups(groups);

    context->SetBlockDim(BLOCK_DIM);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0; // no extra workspace
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {

{
    const gert::Shape* in_shape = context->GetInputShape(0);
    gert::Shape* out_shape = context->GetOutputShape(0);
    *out_shape = *in_shape; // channel shuffle keeps shape
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext* context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "shufflenet_custom_tiling_data.h"
#include "tiling/platform/platform_ascendc.h"

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 1; // single-buffer per plane

class KernelChannelShuffle {
public:
    __aicore__ inline KernelChannelShuffle() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w,
                                uint32_t groups, uint32_t tileNum)
    {
        this->n = n;
        this->c = c;
        this->h = h;
        this->w = w;
        this->groups = groups;
        this->cpg = c / groups;
        this->planeSize = h * w;
        this->totalPlanes = n * c;
        this->tileNum = tileNum;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(x), static_cast<uint32_t>(n * c * h * w));
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(y), static_cast<uint32_t>(n * c * h * w));

        // One whole plane per buffer to simplify
        pipe.InitBuffer(inQueue, BUFFER_NUM, this->planeSize * sizeof(DTYPE));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->planeSize * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        // Static partition by block across planes (N*C)
        uint32_t blkNum = AscendC::GetBlockNum();
        uint32_t blkId = AscendC::GetBlockIdx();

        uint32_t base = (this->totalPlanes / blkNum) * blkId;
        uint32_t tail = (blkId == blkNum - 1) ? (this->totalPlanes - (this->totalPlanes / blkNum) * (blkNum - 1)) : (this->totalPlanes / blkNum);
        uint32_t startPlane = base;
        uint32_t planeCount = tail;

        for (uint32_t p = 0; p < planeCount; ++p) {
            uint32_t planeIdx = startPlane + p;   // [0, N*C)
            uint32_t nIdx = planeIdx / this->c;
            uint32_t cIn  = planeIdx % this->c;

            uint32_t g = cIn / this->cpg;
            uint32_t k = cIn % this->cpg;
            uint32_t cOut = k * this->groups + g;

            uint32_t outPlaneIdx = nIdx * this->c + cOut;

            CopyPlane(planeIdx, outPlaneIdx);
        }
    }

private:
    __aicore__ inline void CopyPlane(uint32_t inPlaneIdx, uint32_t outPlaneIdx)
    {
        // Copy entire H*W plane from input to local
        AscendC::LocalTensor<DTYPE> srcLocal = inQueue.AllocTensor<DTYPE>();
        AscendC::DataCopy(srcLocal, xGm[inPlaneIdx * this->planeSize], this->planeSize);
        inQueue.EnQue(srcLocal);

        // Move through pipe
        AscendC::LocalTensor<DTYPE> inLocal = inQueue.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> dstLocal = outQueue.AllocTensor<DTYPE>();
        AscendC::DataCopy(dstLocal, inLocal, this->planeSize);
        outQueue.EnQue<DTYPE>(dstLocal);
        inQueue.FreeTensor(inLocal);

        // Copy to output plane
        AscendC::LocalTensor<DTYPE> outLocal = outQueue.DeQue<DTYPE>();
        AscendC::DataCopy(yGm[outPlaneIdx * this->planeSize], outLocal, this->planeSize);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM>  inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t n, c, h, w;
    uint32_t groups, cpg;
    uint32_t planeSize;
    uint32_t totalPlanes;
    uint32_t tileNum;
};

extern "C" __global__ __aicore__ void shufflenet_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    KernelChannelShuffle op;
    op.Init(x, y,
            tiling_data.n, tiling_data.c, tiling_data.h, tiling_data.w,
            tiling_data.groups, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class ChannelShuffleCustom(nn.Module):
    def __init__(self, groups: int):
        super().__init__()
        self.groups = groups
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return custom_ops_lib.shufflenet_custom(x, self.groups)

class ShuffleNetUnit(nn.Module):
    def __init__(self, in_channels, out_channels, groups=3):
        super(ShuffleNetUnit, self).__init__()
        assert out_channels % 4 == 0
        mid_channels = out_channels // 4

        self.conv1 = nn.Conv2d(in_channels, mid_channels, kernel_size=1, stride=1, padding=0, groups=groups, bias=False)
        self.bn1 = nn.BatchNorm2d(mid_channels)

        self.conv2 = nn.Conv2d(mid_channels, mid_channels, kernel_size=3, stride=1, padding=1, groups=mid_channels, bias=False)
        self.bn2 = nn.BatchNorm2d(mid_channels)

        self.conv3 = nn.Conv2d(mid_channels, out_channels, kernel_size=1, stride=1, padding=0, groups=groups, bias=False)
        self.bn3 = nn.BatchNorm2d(out_channels)

        # Replace with custom AscendC kernel
        self.shuffle = ChannelShuffleCustom(groups)

        if in_channels == out_channels:
            self.shortcut = nn.Sequential()
        else:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=1, padding=0, bias=False),
                nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out = self.shuffle(out)
        out = F.relu(self.bn3(self.conv3(out)))
        out += self.shortcut(x)
        return out

class ModelNew(nn.Module):
    def __init__(self, num_classes=1000, groups=3, stages_repeats=[3, 7, 3], stages_out_channels=[24, 240, 480, 960]):
        super(ModelNew, self).__init__()
        self.conv1 = nn.Conv2d(3, stages_out_channels[0], kernel_size=3, stride=2, padding=1, bias=False)
        self.bn1 = nn.BatchNorm2d(stages_out_channels[0])
        self.maxpool = nn.MaxPool2d(kernel_size=3, stride=2, padding=1)
        self.stage2 = self._make_stage(stages_out_channels[0], stages_out_channels[1], stages_repeats[0], groups)
        self.stage3 = self._make_stage(stages_out_channels[1], stages_out_channels[2], stages_repeats[1], groups)
        self.stage4 = self._make_stage(stages_out_channels[2], stages_out_channels[3], stages_repeats[2], groups)
        self.conv5 = nn.Conv2d(stages_out_channels[3], 1024, kernel_size=1, stride=1, padding=0, bias=False)
        self.bn5 = nn.BatchNorm2d(1024)
        self.fc = nn.Linear(1024, num_classes)

    def _make_stage(self, in_channels, out_channels, repeats, groups):
        layers = []
        layers.append(ShuffleNetUnit(in_channels, out_channels, groups))
        for _ in range(1, repeats):
            layers.append(ShuffleNetUnit(out_channels, out_channels, groups))
        return nn.Sequential(*layers)

    def forward(self, x):
        x = F.relu(self.bn1(self.conv1(x)))
        x = self.maxpool(x)
        x = self.stage2(x)
        x = self.stage3(x)
        x = self.stage4(x)
        x = F.relu(self.bn5(self.conv5(x)))
        x = F.adaptive_avg_pool2d(x, (1, 1))
        x = x.view(x.size(0), -1)
        x = self.fc(x)
        return x
'''
