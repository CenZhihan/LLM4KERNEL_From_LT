kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_avg_pool_sigmoid_sum_custom_tiling_data.h"

// This kernel reduces the input tensor x of shape [N, C, H, W] along the last axis (CHW) per batch,
// using AscendC::Sum per official API. Output y shape: [N].

template <typename T>
class KernelConv2dAvgPoolSigmoidSumCustom {
public:
    __aicore__ inline KernelConv2dAvgPoolSigmoidSumCustom() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t tmpSize, uint32_t outter, uint32_t inner, uint32_t n)
    {
        height = outter;    // N
        width = inner;      // aligned CHW
        nNum = n;           // actual CHW
        // Bind GM
        srcGlobal.SetGlobalBuffer((__gm__ T *)x);
        dstGlobal.SetGlobalBuffer((__gm__ T *)y);
        // Queues and buffers
        pipe.InitBuffer(inQueueSrc, 1, static_cast<uint64_t>(height) * width * sizeof(T));
        // Output buffer needs to be 32B aligned per API note
        uint32_t outAligned = (height * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE;
        pipe.InitBuffer(outQueueDst, 1, outAligned);
        // tmp buffer for Sum API
        pipe.InitBuffer(workQueue, 1, tmpSize);
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<T> srcLocal = inQueueSrc.AllocTensor<T>();
        AscendC::DataCopy(srcLocal, srcGlobal, height * width);
        inQueueSrc.EnQue(srcLocal);
    }
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<T> srcLocal = inQueueSrc.DeQue<T>();
        AscendC::LocalTensor<uint8_t> workLocal = workQueue.AllocTensor<uint8_t>();
        AscendC::LocalTensor<T> dstLocal = outQueueDst.AllocTensor<T>();
        AscendC::SumParams params{height, width, nNum}; // outter, inner, n
        // Initialize dstLocal buffer (aligned length)
        uint32_t outAlignedElems = (height * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(T);
        T zero(0);
        AscendC::Duplicate<T>(dstLocal, zero, outAlignedElems);
        // Run Sum
        AscendC::Sum(dstLocal, srcLocal, params);
        outQueueDst.EnQue<T>(dstLocal);
        workQueue.FreeTensor(workLocal);
        inQueueSrc.FreeTensor(srcLocal);
    }
    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<T> dstLocal = outQueueDst.DeQue<T>();
        uint32_t outAlignedElems = (height * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(T);
        AscendC::DataCopy(dstGlobal, dstLocal, outAlignedElems);
        outQueueDst.FreeTensor(dstLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueSrc;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> workQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueDst;

    AscendC::GlobalTensor<T> srcGlobal, dstGlobal;
    uint32_t width = 0;   // inner (aligned CHW)
    uint32_t height = 0;  // outter (N)
    uint32_t nNum = 0;    // actual CHW
};

extern "C" __global__ __aicore__ void conv2d_avg_pool_sigmoid_sum_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
#ifdef DTYPE_X
    using DataType = DTYPE_X;
#else
    using DataType = float;
#endif
    KernelConv2dAvgPoolSigmoidSumCustom<DataType> op;
    op.Init(x, y, tiling_data.size(), tiling_data.outter(), tiling_data.inner(), tiling_data.n());
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    This model keeps Conv2d and AvgPool in PyTorch, applies sigmoid,
    then calls a custom AscendC kernel to sum over [C,H,W] per batch.
    """
    def __init__(self, in_channels, out_channels, kernel_size, pool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.avg_pool = torch.nn.AvgPool2d(pool_kernel_size)

    def forward(self, x):
        x = self.conv(x)
        x = self.avg_pool(x)
        x = torch.sigmoid(x)
        # custom kernel reduces along dims [1,2,3] => returns [N]
        x = custom_ops_lib.conv2d_avg_pool_sigmoid_sum_custom(x)
        return x
'''
