kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv3d_leaky_relu_sum_clamp_gelu_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // number of tiles buffered per queue

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR sum, GM_ADDR minv, GM_ADDR maxv, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        sumGm.SetGlobalBuffer((__gm__ DTYPE_X *)sum + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        minGm.SetGlobalBuffer((__gm__ DTYPE_X *)minv + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        maxGm.SetGlobalBuffer((__gm__ DTYPE_X *)maxv + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_X *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueSum, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueMin, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueMax, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> sumLocal = inQueueSum.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> minLocal = inQueueMin.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> maxLocal = inQueueMax.AllocTensor<DTYPE_X>();

        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(sumLocal, sumGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(minLocal, minGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(maxLocal, maxGm[progress * this->tileLength], this->tileLength);

        inQueueX.EnQue(xLocal);
        inQueueSum.EnQue(sumLocal);
        inQueueMin.EnQue(minLocal);
        inQueueMax.EnQue(maxLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> sumLocal = inQueueSum.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> minLocal = inQueueMin.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> maxLocal = inQueueMax.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> yLocal = outQueueY.AllocTensor<DTYPE_X>();

        // Step 1: LeakyReLU with negative_slope=0.2
        const DTYPE_X negSlope = static_cast<DTYPE_X>(0.2f);
        AscendC::LeakyRelu<DTYPE_X>(yLocal, xLocal, negSlope, (int32_t)this->tileLength);

        // Step 2: Element-wise add with broadcasted sum tensor
        AscendC::Add(yLocal, yLocal, sumLocal, this->tileLength);

        // Step 3: Clamp to [min_tensor, max_tensor] using element-wise max then min
        AscendC::Maximum(yLocal, yLocal, minLocal, this->tileLength);
        AscendC::Minimum(yLocal, yLocal, maxLocal, this->tileLength);

        // Step 4: GELU activation
        AscendC::Gelu(yLocal, yLocal, this->tileLength);

        outQueueY.EnQue<DTYPE_X>(yLocal);

        inQueueX.FreeTensor(xLocal);
        inQueueSum.FreeTensor(sumLocal);
        inQueueMin.FreeTensor(minLocal);
        inQueueMax.FreeTensor(maxLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> yLocal = outQueueY.DeQue<DTYPE_X>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueSum;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueMin;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueMax;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_X> sumGm;
    AscendC::GlobalTensor<DTYPE_X> minGm;
    AscendC::GlobalTensor<DTYPE_X> maxGm;
    AscendC::GlobalTensor<DTYPE_X> yGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv3d_leaky_relu_sum_clamp_gelu_custom(GM_ADDR x, GM_ADDR sum, GM_ADDR minv, GM_ADDR maxv, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, sum, minv, maxv, y, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model: Conv3d (PyTorch) + fused AscendC kernel for LeakyReLU, sum, clamp, GELU.
    """
    def __init__(self, in_channels, out_channels, kernel_size, sum_tensor_shape):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.sum_tensor = torch.nn.Parameter(torch.randn(sum_tensor_shape))

    def forward(self, x):
        # Conv3d using existing high-performance implementation
        x = self.conv(x)

        # Prepare broadcasted tensors for fused kernel
        # sum_tensor: (C,1,1,1) -> (1,C,1,1,1) then expand to x.shape
        sum_b = self.sum_tensor.view(1, self.sum_tensor.shape[0], 1, 1, 1).expand_as(x).contiguous()

        # clamp bounds as tensors expanded to x.shape
        min_b = torch.full_like(x, -1.0)
        max_b = torch.full_like(x,  1.0)

        # Fused: LeakyReLU(0.2) -> + sum_b -> clamp[-1,1] -> GELU
        y = custom_ops_lib.conv3d_leaky_relu_sum_clamp_gelu_custom(x, sum_b, min_b, max_b)
        return y
