kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"
#define __NPU_TILING__
#include "gemm_bias_add_hardtanh_mish_group_norm_custom_tiling_data.h"

// 参考文档：
// - 5.2.1.11 REGIST_MATMUL_OBJ
// - 5.2.1.12 Init
// - 5.2.1.24 IterateNBatch / IterateBatch
// - 5.2.2.1.1 Matmul Tiling 获取并传入 TCubeTiling

using A_T = half;    // x, w
using B_T = half;
using C_T = float;   // accumulation/output
using BiasT = float;

constexpr int32_t VEC_BUFFER_NUM = 2;

class PostProcessor {
public:
    __aicore__ inline PostProcessor() {}
    __aicore__ inline void Init(uint32_t M, uint32_t N, AscendC::GlobalTensor<C_T> cGm,
                                AscendC::GlobalTensor<C_T> yGm,
                                AscendC::GlobalTensor<C_T> biasExtra,
                                AscendC::GlobalTensor<C_T> gnGamma,
                                AscendC::GlobalTensor<C_T> gnBeta,
                                float ht_min, float ht_max, int32_t numGroups, float eps) {
        this->M = M;
        this->N = N;
        this->cGm = cGm;
        this->yGm = yGm;
        this->biasExtra = biasExtra;
        this->gnGamma = gnGamma;
        this->gnBeta  = gnBeta;
        this->ht_min = ht_min;
        this->ht_max = ht_max;
        this->numGroups = numGroups;
        this->eps = eps;
        this->channelsPerGroup = (numGroups == 0) ? N : (N / numGroups);

        uint32_t tileElems = GetTileElems();
        pipe.InitBuffer(inQueue,  VEC_BUFFER_NUM, tileElems * sizeof(C_T));
        pipe.InitBuffer(outQueue, VEC_BUFFER_NUM, tileElems * sizeof(C_T));
        pipe.InitBuffer(biasQ,    VEC_BUFFER_NUM, tileElems * sizeof(C_T));
    }

    __aicore__ inline void Run()
    {
        // 行优先处理，每次搬运一个 tile 的列片段
        uint32_t tileElems = GetTileElems();
        uint32_t tilesPerRow = N / tileElems;
        for (uint32_t row = 0; row < M; ++row) {
            // 两段式：第一段统计每组的 mean/sum、sumsq；第二段做归一化与仿射
            // 为节省UB空间，分两次遍历同一行
            ComputeGroupStatsForRow(row, tileElems, tilesPerRow);
            ApplyGroupNormForRow(row, tileElems, tilesPerRow);
        }
    }

private:
    __aicore__ inline uint32_t GetTileElems() const
    {
        // 简单按N维切分，每个tile大小取128对齐
        const uint32_t base = 128;
        uint32_t t = (N >= base) ? base : N;
        if (N % t != 0) {
            t = 1u << AscendC::FloorLog2(N);
        }
        if (t == 0) t = N;
        return t;
    }

    __aicore__ inline void ComputeGroupStatsForRow(uint32_t row, uint32_t tileElems, uint32_t tilesPerRow)
    {
        // 清零累加器
        for (uint32_t g = 0; g < (uint32_t)numGroups; ++g) {
            sumBuf[g] = 0.f;
            sum2Buf[g] = 0.f;
        }

        for (uint32_t t = 0; t < tilesPerRow; ++t) {
            uint32_t offset = row * N + t * tileElems;

            // 搬入数据
            AscendC::LocalTensor<C_T> inLocal = inQueue.AllocTensor<C_T>();
            AscendC::DataCopy(inLocal, cGm[offset], tileElems);
            inQueue.EnQue(inLocal);

            // 取出并计算统计量（同时做BiasAdd、Hardtanh、Mish，中间结果不落GM）
            AscendC::LocalTensor<C_T> xLocal = inQueue.DeQue<C_T>();
            AscendC::LocalTensor<C_T> zLocal = outQueue.AllocTensor<C_T>();

            // 加上额外bias
            AscendC::LocalTensor<C_T> bLocal = biasQ.AllocTensor<C_T>();
            AscendC::DataCopy(bLocal, biasExtra[t * tileElems], tileElems);

            for (uint32_t i = 0; i < tileElems; ++i) {
                C_T v = xLocal.GetValue(i) + bLocal.GetValue(i);
                // Hardtanh
                if (v < ht_min) v = ht_min;
                if (v > ht_max) v = ht_max;
                // Mish: v * tanh(softplus(v))
                // 这里假设存在基础math支持；若无，可替换为近似实现
                C_T sp = AscendC::Log1p(AscendC::Exp(v));
                C_T m = v * AscendC::Tanh(sp);
                zLocal.SetValue(i, m);

                // 统计到对应group
                uint32_t c = t * tileElems + i;
                uint32_t g = (channelsPerGroup == 0) ? 0 : (c / channelsPerGroup);
                if (g >= (uint32_t)numGroups) g = (uint32_t)numGroups - 1;
                sumBuf[g]  += m;
                sum2Buf[g] += m * m;
            }

            outQueue.EnQue(zLocal);
            inQueue.FreeTensor(xLocal);
            biasQ.FreeTensor(bLocal);
        }

        // 计算每组 mean 和 rstd
        for (uint32_t g = 0; g < (uint32_t)numGroups; ++g) {
            float elems = static_cast<float>(channelsPerGroup);
            float mean = sumBuf[g] / elems;
            float var  = sum2Buf[g] / elems - mean * mean;
            if (var < 0.f) var = 0.f;
            meanBuf[g] = mean;
            rstdBuf[g] = AscendC::Rsqrt(var + eps);
        }
    }

    __aicore__ inline void ApplyGroupNormForRow(uint32_t row, uint32_t tileElems, uint32_t tilesPerRow)
    {
        for (uint32_t t = 0; t < tilesPerRow; ++t) {
            AscendC::LocalTensor<C_T> zLocal = outQueue.DeQue<C_T>(); // 上一步的mish结果
            AscendC::LocalTensor<C_T> yLocal = outQueue.AllocTensor<C_T>();

            // gn的gamma、beta
            AscendC::LocalTensor<C_T> gLocal = biasQ.AllocTensor<C_T>();
            AscendC::LocalTensor<C_T> btLocal = biasQ.AllocTensor<C_T>();
            AscendC::DataCopy(gLocal,  gnGamma[t * tileElems], tileElems);
            AscendC::DataCopy(btLocal, gnBeta[t * tileElems],  tileElems);

            for (uint32_t i = 0; i < tileElems; ++i) {
                uint32_t c = t * tileElems + i;
                uint32_t g = (channelsPerGroup == 0) ? 0 : (c / channelsPerGroup);
                if (g >= (uint32_t)numGroups) g = (uint32_t)numGroups - 1;

                C_T m = zLocal.GetValue(i);
                C_T nrm = (m - (C_T)meanBuf[g]) * (C_T)rstdBuf[g];
                C_T yv = nrm * gLocal.GetValue(i) + btLocal.GetValue(i);
                yLocal.SetValue(i, yv);
            }

            // 写回
            uint32_t offset = row * N + t * tileElems;
            AscendC::DataCopy(yGm[offset], yLocal, tileElems);

            outQueue.FreeTensor(zLocal);
            outQueue.FreeTensor(yLocal);
            biasQ.FreeTensor(gLocal);
            biasQ.FreeTensor(btLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  VEC_BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, VEC_BUFFER_NUM> outQueue;
    AscendC::TQue<AscendC::TPosition::VECIN,  VEC_BUFFER_NUM> biasQ;

    AscendC::GlobalTensor<C_T> cGm;
    AscendC::GlobalTensor<C_T> yGm;
    AscendC::GlobalTensor<C_T> biasExtra;
    AscendC::GlobalTensor<C_T> gnGamma;
    AscendC::GlobalTensor<C_T> gnBeta;

    uint32_t M;
    uint32_t N;
    int32_t  numGroups;
    uint32_t channelsPerGroup;
    float ht_min;
    float ht_max;
    float eps;

    // 简单使用本地标量缓冲保存每组统计
    float sumBuf[1024];   // 假设最多1024组
    float sum2Buf[1024];
    float meanBuf[1024];
    float rstdBuf[1024];
};

    const uint32_t N = tiling_data.N();

    __gm__ float* htMaxPtr = reinterpret_cast<__gm__ float*>(hardtanh_max);

    AscendC::GlobalTensor<B_T> bGlobal;  // w
    AscendC::GlobalTensor<C_T> cGlobal;  // 临时c
    AscendC::GlobalTensor<C_T> yGlobal;  // 输出y

    AscendC::GlobalTensor<C_T> biasExtraGlobal;
    AscendC::GlobalTensor<C_T> gnGammaGlobal;
    AscendC::GlobalTensor<C_T> gnBetaGlobal;

    int32_t sizeA = M * K * sizeof(A_T);
    int32_t sizeB = N * K * sizeof(B_T);
    int32_t sizeC = M * N * sizeof(C_T);
    int32_t sizeBias = N * sizeof(C_T);

    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(xGM), sizeA);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(wGM), sizeB);

    biasExtraGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(biasExtraGM), sizeBias);
    gnGammaGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(gnGammaGM), sizeBias);
    gnBetaGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(gnBetaGM), sizeBias);

    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::NORMAL> aType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, B_T, true,  LayoutMode::NORMAL> bType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::NORMAL> cType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, BiasT> biasType;

    constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);
    matmul::Matmul<aType, bType, cType, biasType, MM_CFG> mm;
    AscendC::TPipe pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm); // 见5.2.1.11
    mm.Init(&cubeTiling);                                 // 见5.2.1.12

    AscendC::GlobalTensor<A_T> gm_a;
    AscendC::GlobalTensor<B_T> gm_b;
    AscendC::GlobalTensor<C_T> gm_c;
    AscendC::GlobalTensor<BiasT> gm_bias1;

    gm_b.SetGlobalBuffer(const_cast<__gm__ B_T*>(bGlobal[0].GetPhyAddr()), K * N);

    gm_bias1.SetGlobalBuffer(const_cast<__gm__ BiasT*>(biasLinearGlobal[0].GetPhyAddr()), N);

    mm.SetTensorB(gm_b, true);
    mm.SetWorkspace(workspaceGM, 0);
    mm.SetBias(gm_bias1);

    // 执行GEMM（单批次）
    mm.IterateBatch(gm_c, 1, 1, false);

    // Post-ops: BiasAdd(extra) + Hardtanh + Mish + GroupNorm
    PostProcessor post;
    post.Init(M, N, cGlobal, yGlobal, biasExtraGlobal, gnGammaGlobal, gnBetaGlobal, htMin, htMax, groups, gnEps);
    post.Run();
}
