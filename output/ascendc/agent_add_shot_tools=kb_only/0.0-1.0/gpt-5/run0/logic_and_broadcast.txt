const uint32_t BLOCK_DIM = 8;

    LogicAndBroadcastCustomTilingData tiling;
    const auto *in0 = context->GetInputShape(0);

    uint32_t rank = in0->GetOriginShape().GetDimNum();
    if (rank != 3) {

    uint32_t bLen = static_cast<uint32_t>(in0->GetOriginShape().GetDim(0));
    uint32_t sLen = static_cast<uint32_t>(in0->GetOriginShape().GetDim(1));
    uint32_t hLen = static_cast<uint32_t>(in0->GetOriginShape().GetDim(2));

    tiling.set_sLen(sLen);
    tiling.set_hLen(hLen);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "logic_and_broadcast_custom_tiling_data.h"

// 将 bool 当作 0/1 的 uint8 进行计算，使用标量乘实现按位与（0/1）
using DTYPE = uint8_t;

constexpr int32_t BUFFER_NUM = 1;

class KernelLogicAndBroadcast {
public:
    __aicore__ inline KernelLogicAndBroadcast() {}

    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR out,
                                uint32_t bLen, uint32_t sLen, uint32_t hLen)
    {
        this->bLen = bLen;
        this->sLen = sLen;
        this->hLen = hLen;

        uint32_t totalBS = bLen * sLen;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // 平均划分 [B*S] 维度上的块
        uint32_t bsPerCoreBase = totalBS / blockNum;
        uint32_t bsRemain = totalBS % blockNum;

        // 前 bsRemain 个核多处理 1 个块
        if (blockIdx < bsRemain) {
            this->bsStart = blockIdx * (bsPerCoreBase + 1);
            this->bsCount = bsPerCoreBase + 1;
        } else {
            this->bsStart = bsRemain * (bsPerCoreBase + 1) + (blockIdx - bsRemain) * bsPerCoreBase;
            this->bsCount = bsPerCoreBase;
        }

        // 为 a/out 设置对应核的起始地址和长度（单位：元素个数）
        uint32_t elemOffset = this->bsStart * this->hLen;
        uint32_t elemCount  = this->bsCount * this->hLen;

        aGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(a) + elemOffset, elemCount);
        outGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(out) + elemOffset, elemCount);

        // b 仅在 S 维度上有长度（形状 [1, S, 1] 展平成 S）
        bGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(b), this->sLen);

        // 队列缓冲：每次处理一整段 H
        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->hLen * sizeof(DTYPE));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->hLen * sizeof(DTYPE));
        pipe.InitBuffer(inQueueBScalar, 1, sizeof(DTYPE)); // 用于载入单个标量 b[s]
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->bsCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t i)
    {
        // 载入 a 的一个 H 段
        AscendC::LocalTensor<DTYPE> aLocal = inQueueA.AllocTensor<DTYPE>();
        AscendC::DataCopy(aLocal, aGm[i * this->hLen], this->hLen);
        inQueueA.EnQue(aLocal);

        // 载入对应该段的 b 标量（按 [B*S] 展开后 i 映射到 s = (global_bs % sLen)）
        uint32_t globalBs = this->bsStart + i;
        uint32_t sIdx = globalBs % this->sLen;

        AscendC::LocalTensor<DTYPE> bScalarLocal = inQueueBScalar.AllocTensor<DTYPE>();
        AscendC::DataCopy(bScalarLocal, bGm[sIdx], 1);
        inQueueBScalar.EnQue(bScalarLocal);
    }

    __aicore__ inline void Compute(uint32_t /*i*/)
    {
        AscendC::LocalTensor<DTYPE> aLocal = inQueueA.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> bScalarLocal = inQueueBScalar.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.AllocTensor<DTYPE>();

        // 将 b 标量取出（0/1），用标量乘实现按位与
        DTYPE b_scalar = bScalarLocal.GetValue(0);
        AscendC::Muls(zLocal, aLocal, b_scalar, this->hLen);

        outQueueZ.EnQue<DTYPE>(zLocal);
        inQueueA.FreeTensor(aLocal);
        inQueueBScalar.FreeTensor(bScalarLocal);
    }

    __aicore__ inline void CopyOut(uint32_t i)
    {
        AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.DeQue<DTYPE>();
        AscendC::DataCopy(outGm[i * this->hLen], zLocal, this->hLen);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueA;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueBScalar;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::GlobalTensor<DTYPE> aGm;
    AscendC::GlobalTensor<DTYPE> bGm;
    AscendC::GlobalTensor<DTYPE> outGm;

    uint32_t bLen;
    uint32_t sLen;
    uint32_t hLen;

    uint32_t bsStart;
    uint32_t bsCount;
};

extern "C" __global__ __aicore__ void logic_and_broadcast_custom(GM_ADDR a, GM_ADDR b, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelLogicAndBroadcast op;
    op.Init(a, b, out, tiling_data.bLen, tiling_data.sLen, tiling_data.hLen);
    op.Process();
}
"""
