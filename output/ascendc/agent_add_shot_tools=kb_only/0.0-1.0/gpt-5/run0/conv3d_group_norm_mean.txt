kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv3d_group_norm_mean_custom_tiling_data.h"

// 本内核：对输入 x (B, C, D, H, W) 的每个 batch 进行求和（随后在 Python 绑定中除以 C*D*H*W 得到均值）
// 参考：
// - 归约算子 PairReduceSum（用于两两求和）：CANN Ascend C API Reference 4.1.8.10 PairReduceSum
// - 队列/拷贝/向量加法用法参考示例代码

constexpr int32_t BUFFER_NUM = 1;  // 单批处理

class KernelConv3dGroupNormMean {
public:
    __aicore__ inline KernelConv3dGroupNormMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batch, uint32_t c, uint32_t d, uint32_t h, uint32_t w,
                                uint32_t rLen, uint32_t rLenPad)
    {
        this->batch = batch;
        this->c = c;
        this->d = d;
        this->h = h;
        this->w = w;
        this->rLength = rLen;
        this->rLengthWithPadding = rLenPad;

        // 将输入视为 [B, R] 的扁平化布局（按 batch 连续）
        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_X *>(x), static_cast<uint64_t>(batch) * rLen);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_Y *>(y), batch);

        // 为每个 batch 的 R 元素申请本地缓冲
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->rLengthWithPadding * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, sizeof(DTYPE_Y) * batch);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t b = AscendC::GetBlockIdx(); b < batch; b += AscendC::GetBlockNum()) {
            CopyIn(b);
            Compute(b);
            CopyOut(b);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t bIdx)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        // 将该 batch 的 R 个元素复制到本地缓冲
        AscendC::DataCopy(xLocal, xGm[bIdx * rLength], rLength);
        // 对 padding 部分填 0，避免参与计算
        if (rLengthWithPadding > rLength) {
            AscendC::LocalTensor<DTYPE_X> padLocal = xLocal[rLength];
            DTYPE_X zero = static_cast<DTYPE_X>(0);
            AscendC::Duplicate(padLocal, zero, rLengthWithPadding - rLength);
        }
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*bIdx*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();

        // 通过迭代“前半段 += 后半段”的方式进行树形归约求和
        int32_t len = static_cast<int32_t>(rLengthWithPadding);
        while (len > 1) {
            int32_t half = len >> 1;
            // 将前 half 个元素与后 half 个元素逐元素相加，结果存回前 half
            AscendC::LocalTensor<DTYPE_X> left  = xLocal;          // 从 0 开始
            AscendC::LocalTensor<DTYPE_X> right = xLocal[half];    // 从 half 偏移开始
            AscendC::Add(left, left, right, half);
            len = half;
        }

        // xLocal[0] 现为该 batch 的和，暂存到 y 缓冲（在 CopyOut 中写回）
        // 为了与队列模型匹配，分配一个 yLocal 元素并复制
        if (!outCached) {
            yLocalCache = outQueueY.AllocTensor<DTYPE_Y>();
            // 预填充为 0
            AscendC::Duplicate(yLocalCache, static_cast<DTYPE_Y>(0), batch);
            outCached = true;
        }
        // 将 sum 写入缓存位置（此处直接写入偏移 b 的位置）
        // 注意：由于不支持随机写 GM，这里在 CopyOut 中按 b 索引写回 GM
        // 暂把值存放在 class 成员 tempSum，随 b 一起输出
        tempSum = static_cast<DTYPE_Y>(xLocal.GetValue(0));
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t bIdx)
    {
        // 直接将该 batch 的 sum 写回 GM（每次一个标量）
        // 为简化，单独申请标量局部缓冲进行回写
        DTYPE_Y sumVal = tempSum;
        AscendC::DataCopy(yGm[bIdx], &sumVal, 1);
        // 若是最后一个 batch，可释放 yLocalCache（非必须）
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t batch;
    uint32_t c, d, h, w;
    uint32_t rLength;
    uint32_t rLengthWithPadding;

    // 简化用的缓存变量
    bool outCached = false;
    AscendC::LocalTensor<DTYPE_Y> yLocalCache;
    DTYPE_Y tempSum;
};

extern "C" __global__ __aicore__ void conv3d_group_norm_mean_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dGroupNormMean op;
    op.Init(x, y,
            tiling_data.batch(), tiling_data.c(), tiling_data.d(), tiling_data.h(), tiling_data.w(),
            tiling_data.rLength(), tiling_data.rLengthWithPadding());
    op.Process();
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

// 内核输出每个 batch 的和，绑定中再除以 C*D*H*W 得到均值
at::Tensor conv3d_group_norm_mean_custom_impl_npu(const at::Tensor& x) {
    TORCH_CHECK(x.dim() == 5, \"x must be 5D: [B, C, D, H, W]\");
    auto B = x.size(0);
    auto C = x.size(1);
    auto D = x.size(2);
    auto H = x.size(3);
    auto W = x.size(4);

    at::Tensor y_sum = at::empty({B}, x.options());
    EXEC_NPU_CMD(aclnnConv3dGroupNormMeanCustom, x, y_sum);

    // 转为均值
    const double denom = static_cast<double>(C) * static_cast<double>(D) * static_cast<double>(H) * static_cast<double>(W);
    at::Tensor y_mean = y_sum / denom;
    return y_mean;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("conv3d_group_norm_mean_custom", &conv3d_group_norm_mean_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("conv3d_group_norm_mean_custom", &conv3d_group_norm_mean_custom_impl_npu, "Mean over [C,D,H,W] after GroupNorm");
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    优化版：保留 PyTorch 的 Conv3d 与 GroupNorm，实现最终的按 [C,D,H,W] 维度均值计算为自定义 AscendC 内核
    """
    def __init__(self, in_channels, out_channels, kernel_size, num_groups):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.group_norm = torch.nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)
        x = self.group_norm(x)
        # 使用自定义 AscendC 内核在 NPU 上计算每个 batch 的均值
        y = custom_ops_lib.conv3d_group_norm_mean_custom(x)
        return y
