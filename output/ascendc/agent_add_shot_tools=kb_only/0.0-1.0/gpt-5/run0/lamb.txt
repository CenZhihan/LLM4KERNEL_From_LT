namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    LambCustomTilingData tiling;

    // 输入总元素个数
    uint32_t totalLength = context->GetInputShape(0)->GetOriginShape().GetShapeSize();
    context->SetBlockDim(BLOCK_DIM);
    tiling.set_totalLength(totalLength);
    tiling.set_tileNum(TILE_NUM);

    // 从运行时属性读取 alpha(= lr * trust_ratio) 与 eps
    const gert::RuntimeAttrs* attrs = context->GetAttrs();
    float alpha = 0.f;
    float eps = 0.f;
    if (attrs != nullptr) {
        alpha = *(attrs->GetAttrPointer<float>(0));  // alpha
        eps   = *(attrs->GetAttrPointer<float>(1));  // eps
    }
    tiling.set_alpha(alpha);
    tiling.set_eps(eps);

    // 保存tiling数据
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 不需要workspace
    size_t* currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    return ge::GRAPH_SUCCESS;
}
}

namespace ge {

    const gert::Shape* in_shape = context->GetInputShape(0);

    *out_shape = *in_shape;
    return GRAPH_SUCCESS;

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "lamb_custom_tiling_data.h"

// 基于Ascend C基础向量API：Sqrt、Adds、Div、Muls、Sub、DataCopy
// 参考向量与两操作数标量指令：Adds/Muls，以及数据搬运DataCopy。
// 队列/流水线组织参考官方样例。

constexpr int32_t BUFFER_NUM = 2; // 每个队列的张量数量

class KernelLamb {
public:
    __aicore__ inline KernelLamb() {}
    __aicore__ inline void Init(GM_ADDR param, GM_ADDR m, GM_ADDR v, GM_ADDR out,
                                uint32_t totalLength, uint32_t tileNum, float alpha, float eps)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->alpha = alpha;
        this->eps = eps;

        paramGm.SetGlobalBuffer((__gm__ DTYPE_X*)param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        mGm.SetGlobalBuffer((__gm__ DTYPE_Y*)m + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        vGm.SetGlobalBuffer((__gm__ DTYPE_Z*)v + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outGm.SetGlobalBuffer((__gm__ DTYPE_Z*)out + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueParam, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueM, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(inQueueV, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(calcQueue, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z)); // 计算中间缓冲
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> paramLocal = inQueueParam.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> mLocal = inQueueM.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> vLocal = inQueueV.AllocTensor<DTYPE_Z>();

        AscendC::DataCopy(paramLocal, paramGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(mLocal, mGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(vLocal, vGm[progress * this->tileLength], this->tileLength);

        inQueueParam.EnQue(paramLocal);
        inQueueM.EnQue(mLocal);
        inQueueV.EnQue(vLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        // 取出局部张量
        AscendC::LocalTensor<DTYPE_X> paramLocal = inQueueParam.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> mLocal = inQueueM.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> vLocal = inQueueV.DeQue<DTYPE_Z>(); // 将在此复用为中间r与缩放后的r
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueue.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_Z> calcLocal = calcQueue.AllocTensor<DTYPE_Z>();

        // denom = sqrt(v) + eps
        AscendC::Sqrt(calcLocal, vLocal, this->tileLength);
        AscendC::Adds(calcLocal, calcLocal, (DTYPE_Z)this->eps, this->tileLength);

        // vLocal = r = m / denom
        AscendC::Div(vLocal, mLocal, calcLocal, this->tileLength);

        // vLocal = r * alpha (标量缩放)
        AscendC::Muls(vLocal, vLocal, (DTYPE_Z)this->alpha, this->tileLength);

        // out = param - vLocal
        AscendC::Sub(outLocal, paramLocal, vLocal, this->tileLength);

        // 回收入队张量
        outQueue.EnQue<DTYPE_Z>(outLocal);
        inQueueParam.FreeTensor(paramLocal);
        inQueueM.FreeTensor(mLocal);
        inQueueV.FreeTensor(vLocal);
        calcQueue.FreeTensor(calcLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueue.DeQue<DTYPE_Z>();
        AscendC::DataCopy(outGm[progress * this->tileLength], outLocal, this->tileLength);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueParam, inQueueM, inQueueV;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> calcQueue;
    AscendC::GlobalTensor<DTYPE_X> paramGm;
    AscendC::GlobalTensor<DTYPE_Y> mGm;
    AscendC::GlobalTensor<DTYPE_Z> vGm;
    AscendC::GlobalTensor<DTYPE_Z> outGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    float alpha;
    float eps;
};

extern "C" __global__ __aicore__ void lamb_custom(GM_ADDR param, GM_ADDR m, GM_ADDR v, GM_ADDR out,
                                                   GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelLamb op;
    op.Init(param, m, v, out, tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.alpha, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, lr=1e-3, eps=1e-6):
        super().__init__()
        self.lr = lr
        self.eps = eps

    def forward(self, param, m, v):
        # 先在Python侧计算全局trust_ratio（避免在核内做跨块归约）
        r = m / (v.sqrt() + self.eps)
        trust_ratio = param.norm(p=2) / (r.norm(p=2) + self.eps)
        alpha = float(self.lr * trust_ratio.item())
        # 调用AscendC自定义融合核：完成 r 计算与参数更新 param - alpha * r
        return custom_ops_lib.lamb_custom(param, m, v, alpha, float(self.eps))
'''
