kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matmul_max_pool_sum_scale_custom_tiling_data.h"

// 该自定义核实现以下功能的融合（针对输入已是线性层后的输出 [B, L]）：
// 1) MaxPool1d(kernel_size=K, stride=K) 沿最后一维
// 2) ReduceSum 沿最后一维（使用高阶API Sum，参考文档）
// 3) 标量缩放

class KernelMatmulMaxPoolSumScale {
public:
    __aicore__ inline KernelMatmulMaxPoolSumScale() {}

    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR yGm,
                                uint32_t outter, uint32_t featureLen,
                                uint32_t pooledLen, uint32_t inner, uint32_t K,
                                uint32_t tmpSize, float scale)
    {
        B = outter;
        L = featureLen;
        P = pooledLen;
        alignedInner = inner;
        kernelSize = K;
        tmpBufferSize = tmpSize;
        scaleFactor = scale;

        xGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(xGm), B * L);
        yGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(yGm), B);

        pipe.InitBuffer(inQueue, 1, B * L * sizeof(float));
        pipe.InitBuffer(poolQueue, 1, B * P * sizeof(float));
        // 输出对齐到完整block倍数（参考Sum示例的对齐写回）
        uint32_t outAlignedSize = ((B * sizeof(float) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE) * AscendC::ONE_BLK_SIZE;
        pipe.InitBuffer(outQueue, 1, outAlignedSize);
        pipe.InitBuffer(tmpQueue, 1, tmpBufferSize);
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGlobal, B * L);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        // 取出输入
        AscendC::LocalTensor<float> xLocal = inQueue.DeQue<float>();

        // 分配池化后的buffer
        AscendC::LocalTensor<float> pooledLocal = poolQueue.AllocTensor<float>();

        // 执行MaxPool1d(kernel=K, stride=K) —— 简化实现：逐元素比较K个元素
        for (uint32_t b = 0; b < B; ++b) {
            uint32_t baseIn  = b * L;
            uint32_t baseOut = b * P;
            for (uint32_t i = 0; i < P; ++i) {
                float maxv = xLocal[baseIn + i * kernelSize];
                for (uint32_t k = 1; k < kernelSize; ++k) {
                    float v = xLocal[baseIn + i * kernelSize + k];
                    maxv = maxv > v ? maxv : v;
                }
                pooledLocal[baseOut + i] = maxv;
            }
        }

        // 分配输出
        AscendC::LocalTensor<float> yLocal = outQueue.AllocTensor<float>();
        // 分配Sum临时缓冲区（参考文档Sum）
        AscendC::LocalTensor<uint8_t> workLocal = tmpQueue.AllocTensor<uint8_t>();

        // ReduceSum沿最后一维
        AscendC::SumParams params;
        params.outter = B;         // 外维 = batch
        params.inner  = alignedInner; // 对齐后的inner
        params.n      = P;         // 实际长度

        // 将yLocal预置为0的对齐大小（参考Sum示例）
        uint32_t outAlignedElem = ((B * sizeof(float) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE) * AscendC::ONE_BLK_SIZE / sizeof(float);
        float zero = 0.0f;
        AscendC::Duplicate<float>(yLocal, zero, outAlignedElem);

        // 执行Sum（框架分配临时空间版本）
        AscendC::Sum<float>(yLocal, pooledLocal, params);

        // 标量缩放
        AscendC::Muls<float>(yLocal, yLocal, scaleFactor, B);

        // 入队/释放
        outQueue.EnQue<float>(yLocal);
        poolQueue.FreeTensor(pooledLocal);
        inQueue.FreeTensor(xLocal);
        tmpQueue.FreeTensor(workLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<float> yLocal = outQueue.DeQue<float>();
        // 只回写B个元素（其余对齐填充数据不回写）
        AscendC::DataCopy(yGlobal, yLocal, B);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> poolQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> tmpQueue;

    AscendC::GlobalTensor<float> xGlobal;
    AscendC::GlobalTensor<float> yGlobal;

    uint32_t B = 0;               // batch size
    uint32_t L = 0;               // 输入特征长度（池化前）
    uint32_t P = 0;               // 池化后长度
    uint32_t alignedInner = 0;    // Sum inner对齐
    uint32_t kernelSize = 0;
    uint32_t tmpBufferSize = 0;
    float    scaleFactor = 1.0f;
};

extern "C" __global__ __aicore__ void matmul_max_pool_sum_scale_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulMaxPoolSumScale op;
    op.Init(x, y,
            tiling_data.outter, tiling_data.featureLen,
            tiling_data.n, tiling_data.inner, tiling_data.kernelSize,
            tiling_data.size, tiling_data.scale);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    与原始模型等价：Linear -> (MaxPool1d(k, stride=k) on last-dim) -> ReduceSum(last-dim) -> scale
    其中线性层仍由PyTorch执行，后续池化+求和+缩放由自定义AscendC算子实现。
    """
    def __init__(self, in_features, out_features, kernel_size, scale_factor):
        super().__init__()
        self.matmul = torch.nn.Linear(in_features, out_features, bias=True)
        self.kernel_size = int(kernel_size)
        self.scale_factor = float(scale_factor)
        self.out_features = int(out_features)

    def forward(self, x):
        # 线性层
        x = self.matmul(x)                         # [B, out_features]
        # 自定义AscendC算子：在最后一维做MaxPool1d(k, stride=k)，再ReduceSum到[B]并缩放
        y = custom_ops_lib.matmul_max_pool_sum_scale_custom(
            x, self.out_features, self.kernel_size, self.scale_factor
        )
        return y
