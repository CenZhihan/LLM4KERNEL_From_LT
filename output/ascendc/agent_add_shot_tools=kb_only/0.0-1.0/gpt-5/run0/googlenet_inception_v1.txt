kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "googlenet_inception_v1_custom_tiling_data.h"

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 2; 
constexpr int32_t TILE_ELEMS = 4096;

class KernelConcatCh {
public:
    __aicore__ inline KernelConcatCh() {}
    __aicore__ inline void Init(GM_ADDR x1, GM_ADDR x2, GM_ADDR x3, GM_ADDR x4,
                                GM_ADDR y, uint32_t totalLength,
                                uint32_t n, uint32_t h, uint32_t w,
                                uint32_t c0, uint32_t c1, uint32_t c2, uint32_t c3, uint32_t cout)
    {
        this->n = n;
        this->h = h;
        this->w = w;
        this->c0 = c0;
        this->c1 = c1;
        this->c2 = c2;
        this->c3 = c3;
        this->cout = cout;
        this->plane = h * w;

        x1Gm.SetGlobalBuffer((__gm__ DTYPE *)x1, n * c0 * plane);
        x2Gm.SetGlobalBuffer((__gm__ DTYPE *)x2, n * c1 * plane);
        x3Gm.SetGlobalBuffer((__gm__ DTYPE *)x3, n * c2 * plane);
        x4Gm.SetGlobalBuffer((__gm__ DTYPE *)x4, n * c3 * plane);
        yGm.SetGlobalBuffer((__gm__ DTYPE *)y,  totalLength);

        pipe.InitBuffer(inQueue, BUFFER_NUM, TILE_ELEMS * sizeof(DTYPE));
        pipe.InitBuffer(outQueue, BUFFER_NUM, TILE_ELEMS * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        uint32_t prefix = 0;
        CopyBranch(x1Gm, c0, prefix);
        prefix += c0;
        CopyBranch(x2Gm, c1, prefix);
        prefix += c1;
        CopyBranch(x3Gm, c2, prefix);
        prefix += c2;
        CopyBranch(x4Gm, c3, prefix);
    }

private:
    __aicore__ inline void CopyRange(__gm__ DTYPE* srcBase, uint32_t srcOffset,
                                     __gm__ DTYPE* dstBase, uint32_t dstOffset,
                                     uint32_t len)
    {
        uint32_t done = 0;
        while (done < len) {
            uint32_t curr = AscendC::Min((uint32_t)TILE_ELEMS, len - done);
            AscendC::LocalTensor<DTYPE> inLocal = inQueue.AllocTensor<DTYPE>();
            AscendC::LocalTensor<DTYPE> outLocal = outQueue.AllocTensor<DTYPE>();
            AscendC::DataCopy(inLocal, srcBase + srcOffset + done, curr);
            // local to local copy
            AscendC::DataCopy(outLocal, inLocal, curr);
            inQueue.EnQue(inLocal);
            outQueue.EnQue(outLocal);

            inLocal = inQueue.DeQue<DTYPE>();
            outLocal = outQueue.DeQue<DTYPE>();
            AscendC::DataCopy(dstBase + dstOffset + done, outLocal, curr);
            inQueue.FreeTensor(inLocal);
            outQueue.FreeTensor(outLocal);

            done += curr;
        }
    }

    __aicore__ inline void CopyBranch(AscendC::GlobalTensor<DTYPE>& src, uint32_t ci, uint32_t cprefix)
    {
        if (ci == 0) { return; }
        uint32_t perBatchSrc = ci * plane;
        uint32_t perBatchDst = cout * plane;
        for (uint32_t bn = 0; bn < n; ++bn) {
            uint32_t srcOffset = bn * perBatchSrc;
            uint32_t dstOffset = bn * perBatchDst + cprefix * plane;
            CopyRange((__gm__ DTYPE*)src.GetTensor(), srcOffset,
                      (__gm__ DTYPE*)yGm.GetTensor(), dstOffset,
                      perBatchSrc);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;

    AscendC::GlobalTensor<DTYPE> x1Gm;
    AscendC::GlobalTensor<DTYPE> x2Gm;
    AscendC::GlobalTensor<DTYPE> x3Gm;
    AscendC::GlobalTensor<DTYPE> x4Gm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t n;
    uint32_t h;
    uint32_t w;
    uint32_t plane;
    uint32_t c0, c1, c2, c3;
    uint32_t cout;
};

extern "C" __global__ __aicore__ void googlenet_inception_v1_custom(GM_ADDR x1, GM_ADDR x2, GM_ADDR x3, GM_ADDR x4,
                                                                    GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConcatCh op;
    op.Init(x1, x2, x3, x4, y,
            tiling_data.totalLength,
            tiling_data.n, tiling_data.h, tiling_data.w,
            tiling_data.c0, tiling_data.c1, tiling_data.c2, tiling_data.c3, tiling_data.cout);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
import torch.nn as nn
import torch.nn.functional as F

class InceptionModule(nn.Module):
    def __init__(self, in_channels, out_1x1, reduce_3x3, out_3x3, reduce_5x5, out_5x5, pool_proj):
        super(InceptionModule, self).__init__()
        self.branch1x1 = nn.Conv2d(in_channels, out_1x1, kernel_size=1)
        self.branch3x3 = nn.Sequential(
            nn.Conv2d(in_channels, reduce_3x3, kernel_size=1),
            nn.Conv2d(reduce_3x3, out_3x3, kernel_size=3, padding=1)
        )
        self.branch5x5 = nn.Sequential(
            nn.Conv2d(in_channels, reduce_5x5, kernel_size=1),
            nn.Conv2d(reduce_5x5, out_5x5, kernel_size=5, padding=2)
        )
        self.branch_pool = nn.Sequential(
            nn.MaxPool2d(kernel_size=3, stride=1, padding=1),
            nn.Conv2d(in_channels, pool_proj, kernel_size=1)
        )

    def forward(self, x):
        b1 = self.branch1x1(x)
        b3 = self.branch3x3(x)
        b5 = self.branch5x5(x)
        bp = self.branch_pool(x)
        # Custom AscendC concat along channel dimension
        return custom_ops_lib.googlenet_inception_v1_custom(b1, b3, b5, bp)

class ModelNew(nn.Module):
    def __init__(self, num_classes=1000):
        super(ModelNew, self).__init__()
        self.conv1 = nn.Conv2d(3, 64, kernel_size=7, stride=2, padding=3)
        self.maxpool1 = nn.MaxPool2d(3, stride=2, padding=1)
        self.conv2 = nn.Conv2d(64, 64, kernel_size=1)
        self.conv3 = nn.Conv2d(64, 192, kernel_size=3, padding=1)
        self.maxpool2 = nn.MaxPool2d(3, stride=2, padding=1)

        self.inception3a = InceptionModule(192, 64, 96, 128, 16, 32, 32)
        self.inception3b = InceptionModule(256, 128, 128, 192, 32, 96, 64)
        self.maxpool3 = nn.MaxPool2d(3, stride=2, padding=1)

        self.inception4a = InceptionModule(480, 192, 96, 208, 16, 48, 64)
        self.inception4b = InceptionModule(512, 160, 112, 224, 24, 64, 64)
        self.inception4c = InceptionModule(512, 128, 128, 256, 24, 64, 64)
        self.inception4d = InceptionModule(512, 112, 144, 288, 32, 64, 64)
        self.inception4e = InceptionModule(528, 256, 160, 320, 32, 128, 128)
        self.maxpool4 = nn.MaxPool2d(3, stride=2, padding=1)

        self.inception5a = InceptionModule(832, 256, 160, 320, 32, 128, 128)
        self.inception5b = InceptionModule(832, 384, 192, 384, 48, 128, 128)

        self.avgpool = nn.AdaptiveAvgPool2d((1, 1))
        self.dropout = nn.Dropout(0.0)
        self.fc = nn.Linear(1024, num_classes)

    def forward(self, x):
        x = self.maxpool1(F.relu(self.conv1(x)))
        x = F.relu(self.conv2(x))
        x = self.maxpool2(F.relu(self.conv3(x)))

        x = self.inception3a(x)
        x = self.inception3b(x)
        x = self.maxpool3(x)

        x = self.inception4a(x)
        x = self.inception4b(x)
        x = self.inception4c(x)
        x = self.inception4d(x)
        x = self.inception4e(x)
        x = self.maxpool4(x)

        x = self.inception5a(x)
        x = self.inception5b(x)

        x = self.avgpool(x)
        x = torch.flatten(x, 1)
        x = self.dropout(x)
        x = self.fc(x)
        return x
'''
