kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "rmsprop_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // 每个队列双缓冲

class KernelRmsprop {
public:
    __aicore__ inline KernelRmsprop() {}
    __aicore__ inline void Init(GM_ADDR param, GM_ADDR grad, GM_ADDR v, GM_ADDR out,
                                uint32_t totalLength, uint32_t tileNum, float lr, float alpha, float eps)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        this->lr = lr;
        this->alpha = alpha;
        this->eps = eps;

        paramGm.SetGlobalBuffer((__gm__ float *)param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        gradGm.SetGlobalBuffer((__gm__ float *)grad + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        vGm.SetGlobalBuffer((__gm__ float *)v + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outGm.SetGlobalBuffer((__gm__ float *)out + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueParam, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueGrad,  BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueV,     BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueOut,  BUFFER_NUM, this->tileLength * sizeof(float));

        // 中间计算需要的临时缓冲：grad^2、v_new、denom
        pipe.InitBuffer(tmpQue1, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpQue2, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpQue3, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> paramLocal = inQueueParam.AllocTensor<float>();
        AscendC::LocalTensor<float> gradLocal  = inQueueGrad.AllocTensor<float>();
        AscendC::LocalTensor<float> vLocal     = inQueueV.AllocTensor<float>();

        AscendC::DataCopy(paramLocal, paramGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(gradLocal,  gradGm[progress * this->tileLength],  this->tileLength);
        AscendC::DataCopy(vLocal,     vGm[progress * this->tileLength],     this->tileLength);

        inQueueParam.EnQue(paramLocal);
        inQueueGrad.EnQue(gradLocal);
        inQueueV.EnQue(vLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<float> paramLocal = inQueueParam.DeQue<float>();
        AscendC::LocalTensor<float> gradLocal  = inQueueGrad.DeQue<float>();
        AscendC::LocalTensor<float> vLocal     = inQueueV.DeQue<float>();

        AscendC::LocalTensor<float> grad2Local = tmpQue1.AllocTensor<float>();
        AscendC::LocalTensor<float> vnewLocal  = tmpQue2.AllocTensor<float>();
        AscendC::LocalTensor<float> denomLocal = tmpQue3.AllocTensor<float>();
        AscendC::LocalTensor<float> outLocal   = outQueueOut.AllocTensor<float>();

        // grad2 = grad * grad
        AscendC::Mul(grad2Local, gradLocal, gradLocal, this->tileLength);

        // v_new = alpha * v + (1 - alpha) * grad2
        AscendC::Muls(vnewLocal, vLocal, this->alpha, this->tileLength);                  // alpha * v
        AscendC::Muls(denomLocal, grad2Local, (1.0f - this->alpha), this->tileLength);    // (1-alpha) * grad2
        AscendC::Add(vnewLocal, vnewLocal, denomLocal, this->tileLength);                 // vnew = ...

        // denom = sqrt(v_new) + eps
        AscendC::Sqrt(denomLocal, vnewLocal, this->tileLength);
        AscendC::Adds(denomLocal, denomLocal, this->eps, this->tileLength);

        // update = lr * grad / denom
        AscendC::Div(outLocal, gradLocal, denomLocal, this->tileLength);
        AscendC::Muls(outLocal, outLocal, this->lr, this->tileLength);

        // out = param - update
        AscendC::Sub(outLocal, paramLocal, outLocal, this->tileLength);

        outQueueOut.EnQue<float>(outLocal);

        // 释放队列中的局部张量
        inQueueParam.FreeTensor(paramLocal);
        inQueueGrad.FreeTensor(gradLocal);
        inQueueV.FreeTensor(vLocal);
        tmpQue1.FreeTensor(grad2Local);
        tmpQue2.FreeTensor(vnewLocal);
        tmpQue3.FreeTensor(denomLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> outLocal = outQueueOut.DeQue<float>();
        AscendC::DataCopy(outGm[progress * this->tileLength], outLocal, this->tileLength);
        outQueueOut.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueParam, inQueueGrad, inQueueV;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueOut;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpQue1, tmpQue2, tmpQue3;

    AscendC::GlobalTensor<float> paramGm;
    AscendC::GlobalTensor<float> gradGm;
    AscendC::GlobalTensor<float> vGm;
    AscendC::GlobalTensor<float> outGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;

    float lr;
    float alpha;
    float eps;
};

extern "C" __global__ __aicore__ void rmsprop_custom(GM_ADDR param, GM_ADDR grad, GM_ADDR v, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelRmsprop op;
    op.Init(param, grad, v, out,
            tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.lr, tiling_data.alpha, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, lr=1e-3, alpha=0.99, eps=1e-8):
        super().__init__()
        # 内核当前使用默认超参（lr/alpha/eps），与该初始化参数保持一致
        self.lr = lr
        self.alpha = alpha
        self.eps = eps

    def forward(self, param, grad, v):
        # 使用自定义AscendC算子完成：
        # v_new = alpha * v + (1 - alpha) * grad^2
        # param_new = param - lr * grad / (sqrt(v_new) + eps)
        # 当前接口返回更新后的param
        return custom_ops_lib.rmsprop_custom(param, grad, v)
