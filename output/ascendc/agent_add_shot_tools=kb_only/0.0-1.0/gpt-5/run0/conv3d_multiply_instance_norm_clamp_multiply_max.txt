kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv3d_multiply_instance_norm_clamp_multiply_max_custom_tiling_data.h"

// Kernel does: y = max_c( clamp( x * multiplier[c], min, max ) * multiplier[c] ), where
// x: [N,C,D,H,W], multiplier: [C,1,1,1], output y: [N,D,H,W].
// Note: This kernel focuses on fused elementwise ops and channel-wise max reduction.

extern "C" __global__ __aicore__ void conv3d_multiply_instance_norm_clamp_multiply_max_custom(
    GM_ADDR x, GM_ADDR multiplier, GM_ADDR clamp_min, GM_ADDR clamp_max, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    const uint32_t N = tilingData.n();
    const uint32_t C = tilingData.c();
    const uint32_t D = tilingData.d();
    const uint32_t H = tilingData.h();
    const uint32_t W = tilingData.w();
    const uint32_t S = tilingData.spatialSize(); // D*H*W

    const uint32_t blockNum = AscendC::GetBlockNum();
    const uint32_t blockIdx = AscendC::GetBlockIdx();

    __gm__ float* xGm = reinterpret_cast<__gm__ float*>(x);
    __gm__ float* mGm = reinterpret_cast<__gm__ float*>(multiplier);
    __gm__ float* minGm = reinterpret_cast<__gm__ float*>(clamp_min);
    __gm__ float* maxGm = reinterpret_cast<__gm__ float*>(clamp_max);
    __gm__ float* yGm = reinterpret_cast<__gm__ float*>(y);

    const float vmin = *minGm;
    const float vmax = *maxGm;

    // Work partition: split by batch N along blocks.
    uint32_t nPerBlock = (N + blockNum - 1) / blockNum;
    uint32_t nStart = blockIdx * nPerBlock;
    uint32_t nEnd = AscendC::Min(nStart + nPerBlock, N);

    for (uint32_t n = nStart; n < nEnd; ++n) {
        // base offsets
        uint64_t baseInN = static_cast<uint64_t>(n) * C * S;
        uint64_t baseOutN = static_cast<uint64_t>(n) * S;
        for (uint32_t s = 0; s < S; ++s) {
            float maxVal = -3.402823466e+38F; // -FLT_MAX
            for (uint32_t c = 0; c < C; ++c) {
                uint64_t idx = baseInN + static_cast<uint64_t>(c) * S + s;
                float mul_c = mGm[c]; // multiplier per channel
                float v = xGm[idx] * mul_c;
                // clamp
                if (v < vmin) v = vmin;
                if (v > vmax) v = vmax;
                // multiply again
                v = v * mul_c;
                if (v > maxVal) maxVal = v;
            }
            yGm[baseOutN + s] = maxVal;
        }
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Conv3d -> multiply/instance_norm/clamp/multiply/max fused call.
    """
    def __init__(self, in_channels, out_channels, kernel_size, multiplier_shape, clamp_min, clamp_max):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.multiplier = torch.nn.Parameter(torch.randn(multiplier_shape))
        self.instance_norm = torch.nn.InstanceNorm3d(out_channels)
        # clamp range as tensors to pass into custom op
        self.register_buffer("clamp_min_tensor", torch.tensor([clamp_min], dtype=torch.float32))
        self.register_buffer("clamp_max_tensor", torch.tensor([clamp_max], dtype=torch.float32))

    def forward(self, x):
        x = self.conv(x)  # [N,C,D,H,W]
        # fuse elementwise ops and channel-wise max in custom kernel
        # multiplier expects shape [C,1,1,1]
        return custom_ops_lib.conv3d_multiply_instance_norm_clamp_multiply_max_custom(
            x, self.multiplier, self.clamp_min_tensor, self.clamp_max_tensor
        )
