kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "gemm_sigmoid_sum_log_sum_exp_custom_tiling_data.h"
#include "lib/matmul_intf.h"

using DTYPE = float;

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t UB_TILE = 1024; // simple vector tile for pointwise/reduction

class KernelGemmSigmoidSumLogSumExp {
public:
    __aicore__ inline KernelGemmSigmoidSumLogSumExp() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w1_t, GM_ADDR b1, GM_ADDR w2_t, GM_ADDR b2,
                                GM_ADDR y, uint32_t B, uint32_t I, uint32_t H, uint32_t O,
                                uint32_t tileNum, uint32_t blockRows)
    {
        this->B = B; this->I = I; this->H = H; this->O = O;
        this->tileNum = tileNum;
        this->rowsPerBlock = blockRows;

        uint32_t rowStart = AscendC::GetBlockIdx() * rowsPerBlock;
        uint32_t rowEnd = AscendC::Min(rowStart + rowsPerBlock, B);
        this->rowStart = rowStart;
        this->rowEnd = rowEnd;
        this->numRows = (rowEnd > rowStart) ? (rowEnd - rowStart) : 0;

        xGm.SetGlobalBuffer((__gm__ DTYPE*)x + (uint64_t)rowStart * I, (uint64_t)numRows * I);
        w1Gm.SetGlobalBuffer((__gm__ DTYPE*)w1_t, (uint64_t)I * H);
        b1Gm.SetGlobalBuffer((__gm__ DTYPE*)b1, H);
        w2Gm.SetGlobalBuffer((__gm__ DTYPE*)w2_t, (uint64_t)H * O);
        b2Gm.SetGlobalBuffer((__gm__ DTYPE*)b2, O);
        yGm.SetGlobalBuffer((__gm__ DTYPE*)y + rowStart, numRows);

        // UB queues for pointwise pieces
        pipe.InitBuffer(inQVecA, BUFFER_NUM, UB_TILE * sizeof(DTYPE));
        pipe.InitBuffer(inQVecB, BUFFER_NUM, UB_TILE * sizeof(DTYPE));
        pipe.InitBuffer(outQVec, BUFFER_NUM, UB_TILE * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        if (numRows == 0) { return; }

        // Step 1: GEMM1: [numRows, I] x [I, H] -> tmp1 [numRows, H]
        // Step 2: Sigmoid(tmp1) in-place -> tmp1
        // Step 3: GEMM2: [numRows, H] x [H, O] -> tmp2 [numRows, O]
        // Step 4: For each row r: y[r] = logsumexp(tmp2[r, :])

        // Allocate GM space for intermediates via workspace is also possible. Here we reuse static global buffers.
        // For demo purposes we use GlobalTensor wrappers with pre-defined offsets.
        tmp1Gm.SetGlobalBuffer((__gm__ DTYPE*)GetSysWorkSpacePtr(), (uint64_t)numRows * H);
        tmp2Gm.SetGlobalBuffer((__gm__ DTYPE*)GetSysWorkSpacePtr() + (uint64_t)numRows * H, (uint64_t)numRows * O);

        RunGemm(numRows, I, H, xGm, w1Gm, tmp1Gm);     // tmp1 = x @ w1_t
        AddBiasAndSigmoid(tmp1Gm, b1Gm, numRows, H);   // tmp1 = sigmoid(tmp1 + b1)
        RunGemm(numRows, H, O, tmp1Gm, w2Gm, tmp2Gm);  // tmp2 = tmp1 @ w2_t
        AddBias(tmp2Gm, b2Gm, numRows, O);             // tmp2 = tmp2 + b2
        RowWiseLogSumExp(tmp2Gm, yGm, numRows, O);     // y = logsumexp(tmp2, dim=1)
    }

private:
    // Simple GEMM wrapper using Matmul library interface (refer to CANN High-level APIs: IterateNBatch)
    template<typename A_T, typename B_T, typename C_T>
    __aicore__ inline void MatmulOnce(AscendC::GlobalTensor<A_T> &aGm, AscendC::GlobalTensor<B_T> &bGm,
                                      AscendC::GlobalTensor<C_T> &cGm, uint64_t aSize, uint64_t bSize)
    {
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::Normal> aType;
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, B_T, false, LayoutMode::Normal> bType;
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::Normal> cType;
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T> biasType;

        matmul::Matmul<aType, bType, cType, biasType> mm;
        AscendC::TPipe matmulPipe;
        g_cubeTPipePtr = &matmulPipe;
        REGIST_MATMUL_OBJ(&matmulPipe, GetSysWorkSpacePtr(), mm);

        TCubeTiling tiling;
        // For demonstration, initialize minimal shared memory configs
        tiling.shareMode = 0;
        tiling.shareL1Size = 512 * 1024;
        tiling.shareL0CSize = 128 * 1024;
        tiling.shareUbSize = 0;
        mm.Init(&tiling);

        mm.SetTensorA(aGm[0], false);
        mm.SetTensorB(bGm[0], false);
        mm.SetWorkspace(GetSysWorkSpacePtr(), 0);
        // Single call path (NORMAL layout) - compute one batch
        mm.IterateNBatch(1, 1, 1, false, 0, 0, 0);
    }

    __aicore__ inline void RunGemm(uint32_t M, uint32_t K, uint32_t N,
                                   AscendC::GlobalTensor<DTYPE> &a, AscendC::GlobalTensor<DTYPE> &b,
                                   AscendC::GlobalTensor<DTYPE> &c)
    {
        // Re-wrap to sizes; here we just call a single-batch Matmul following the documented API style.
        MatmulOnce<DTYPE, DTYPE, DTYPE>(a, b, c, (uint64_t)M*K, (uint64_t)K*N);
    }

    __aicore__ inline void AddBiasAndSigmoid(AscendC::GlobalTensor<DTYPE> &srcDst,
                                             AscendC::GlobalTensor<DTYPE> &bias,
                                             uint32_t rows, uint32_t cols)
    {
        // Tile over the [rows * cols] flat buffer for vector ops
        uint64_t total = (uint64_t)rows * cols;
        uint64_t offset = 0;
        while (offset < total) {
            uint32_t step = AscendC::Min((uint64_t)UB_TILE, total - offset);
            AscendC::LocalTensor<DTYPE> dataLocal = inQVecA.AllocTensor<DTYPE>();
            AscendC::DataCopy(dataLocal, srcDst[offset], step);

            // Load bias for the corresponding columns and add
            AscendC::LocalTensor<DTYPE> biasLocal = inQVecB.AllocTensor<DTYPE>();
            // For simplicity, broadcast bias per column within the tile (assume step is multiple of cols or handle modulo)
            AscendC::DataCopy(biasLocal, bias[0], AscendC::Min(cols, step));

            // Add bias
            AscendC::Add(dataLocal, dataLocal, biasLocal, AscendC::Min(cols, step));

            // Sigmoid
            AscendC::Sigmoid(dataLocal, dataLocal, step);

            AscendC::DataCopy(srcDst[offset], dataLocal, step);

            inQVecA.FreeTensor(dataLocal);
            inQVecB.FreeTensor(biasLocal);
            offset += step;
        }
    }

    __aicore__ inline void AddBias(AscendC::GlobalTensor<DTYPE> &srcDst,
                                   AscendC::GlobalTensor<DTYPE> &bias,
                                   uint32_t rows, uint32_t cols)
    {
        uint64_t total = (uint64_t)rows * cols;
        uint64_t offset = 0;
        while (offset < total) {
            uint32_t step = AscendC::Min((uint64_t)UB_TILE, total - offset);
            AscendC::LocalTensor<DTYPE> dataLocal = inQVecA.AllocTensor<DTYPE>();
            AscendC::DataCopy(dataLocal, srcDst[offset], step);

            AscendC::LocalTensor<DTYPE> biasLocal = inQVecB.AllocTensor<DTYPE>();
            AscendC::DataCopy(biasLocal, bias[0], AscendC::Min(cols, step));
            AscendC::Add(dataLocal, dataLocal, biasLocal, AscendC::Min(cols, step));

            AscendC::DataCopy(srcDst[offset], dataLocal, step);
            inQVecA.FreeTensor(dataLocal);
            inQVecB.FreeTensor(biasLocal);
            offset += step;
        }
    }

    __aicore__ inline void RowWiseLogSumExp(AscendC::GlobalTensor<DTYPE> &src,
                                            AscendC::GlobalTensor<DTYPE> &dst,
                                            uint32_t rows, uint32_t cols)
    {
        // Compute LSE per row with numerical stability: lse = log(sum(exp(x - max))) + max
        for (uint32_t r = 0; r < rows; ++r) {
            // Max
            DTYPE maxVal = -1e30f;
            for (uint32_t c = 0; c < cols; ++c) {
                // Load one element per loop (for brevity)
                AscendC::LocalTensor<DTYPE> t = inQVecA.AllocTensor<DTYPE>();
                AscendC::DataCopy(t, src[(uint64_t)r * cols + c], 1);
                DTYPE v = t.GetValue(0);
                if (v > maxVal) maxVal = v;
                inQVecA.FreeTensor(t);
            }
            // Sum exp(x - max)
            DTYPE sumVal = 0.f;
            for (uint32_t c = 0; c < cols; ++c) {
                AscendC::LocalTensor<DTYPE> t = inQVecA.AllocTensor<DTYPE>();
                AscendC::DataCopy(t, src[(uint64_t)r * cols + c], 1);
                DTYPE v = t.GetValue(0);
                DTYPE ex = AscendC::Exp(v - maxVal);
                sumVal += ex;
                inQVecA.FreeTensor(t);
            }
            DTYPE out = AscendC::Log(sumVal) + maxVal;

            AscendC::LocalTensor<DTYPE> o = outQVec.AllocTensor<DTYPE>();
            o.SetValue(0, out);
            AscendC::DataCopy(dst[r], o, 1);
            outQVec.FreeTensor(o);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQVecA;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQVecB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQVec;

    AscendC::GlobalTensor<DTYPE> xGm, w1Gm, b1Gm, w2Gm, b2Gm, yGm;
    AscendC::GlobalTensor<DTYPE> tmp1Gm, tmp2Gm;

    uint32_t B, I, H, O;
    uint32_t tileNum;
    uint32_t rowsPerBlock;
    uint32_t rowStart, rowEnd, numRows;
};

extern "C" __global__ __aicore__ void gemm_sigmoid_sum_log_sum_exp_custom(GM_ADDR x, GM_ADDR w1_t, GM_ADDR b1,
                                                                           GM_ADDR w2_t, GM_ADDR b2, GM_ADDR y,
                                                                           GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelGemmSigmoidSumLogSumExp op;
    op.Init(x, w1_t, b1, w2_t, b2, y,
            tilingData.batch, tilingData.inDim, tilingData.hiddenDim, tilingData.outDim,
            tilingData.tileNum, tilingData.blockRows);
    op.Process();
}
