host_operator_src="""
#include <cstring>
#include "register/op_def_registry.h"

// 参考文档使用的Matmul Tiling接口
#include "matmul_tiling.h"
#include "platform_ascendc.h"

namespace optiling {

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 获取输入形状
    const auto *shapeA = context->GetInputShape(0);
    const auto *shapeB = context->GetInputShape(1);
    const auto &oriA = shapeA->GetOriginShape();
    const auto &oriB = shapeB->GetOriginShape();

    // 预期 A: (N, M, K), B: (K, L)
    int32_t N = static_cast<int32_t>(oriA.GetDim(0));
    int32_t M = static_cast<int32_t>(oriA.GetDim(1));
    int32_t K = static_cast<int32_t>(oriA.GetDim(2));
    int32_t L = static_cast<int32_t>(oriB.GetDim(1));

    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    matmul_tiling::MultiCoreMatmulTiling tiling(ascendcPlatform);

    // 数据类型均为 float (DT_FLOAT)
    tiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    // 本算子无bias
    tiling.SetBias(false);

    // 单次Matmul的M/N/K
    tiling.SetShape(M, L, K);
    // 原始完整形状 (M, N, Ka, Kb)，此处Ka=Kb=K
    tiling.SetOrgShape(M, L, K);

    // NORMAL布局下设置batch信息：A有N个batch，B按1个batch广播
    tiling.SetBatchInfoForNormal(N, 1, M, L, K);

    // 默认占用全部可用buffer空间
    tiling.SetBufferSpace(-1, -1, -1);

    optiling::TCubeTiling tilingData;
    int ret = tiling.GetTiling(tilingData);
    if (ret == -1) {
        return ge::GRAPH_FAILED;
    }

    // 设置blockDim
    int32_t core_num = tiling.GetCoreNum();
    if (core_num <= 0) { core_num = 1; }
    context->SetBlockDim(static_cast<uint32_t>(core_num));

    // 将TCubeTiling二进制拷贝到框架原始tiling buffer
    auto *raw = context->GetRawTilingData();
    std::memcpy(raw->GetData(), &tilingData, sizeof(optiling::TCubeTiling));
    raw->SetDataSize(sizeof(optiling::TCubeTiling));

    // workspace为0
    size_t *ws_sizes = context->GetWorkspaceSizes(1);
    ws_sizes[0] = 0;

    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* a_shape = context->GetInputShape(0);
    const gert::Shape* b_shape = context->GetInputShape(1);
    gert::Shape* y_shape = context->GetOutputShape(0);
    // 期望输出为 (N, M, L)
    *y_shape = *a_shape;
    if (a_shape->GetDimNum() == 3 && b_shape->GetDimNum() == 2) {
        y_shape->SetDim(2, b_shape->GetDim(1));
    }
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class ThreeDimTensorMatrixMultiplicationCustom : public OpDef {
public:
    explicit ThreeDimTensorMatrixMultiplicationCustom(const char* name) : OpDef(name)
    {
        this->Input("A")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("B")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("C")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(ThreeDimTensorMatrixMultiplicationCustom);
}

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

// 使用TCubeTiling作为host侧下发的tiling结构
extern "C" __global__  __aicore__ void three_dim_tensor_matrix_multiplication_custom(GM_ADDR aGM, GM_ADDR bGM,
                                                                                      GM_ADDR cGM, GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    // 从GM中反序列化TCubeTiling
    TCubeTiling tiling;
    auto tempTilingGM = (__gm__ uint32_t*)tilingGM;
    auto tempTiling = (uint32_t*)&tiling;
    for (int i = 0; i < (int)(sizeof(TCubeTiling) / sizeof(int32_t)); ++i, ++tempTilingGM, ++tempTiling) {
        *tempTiling = *tempTilingGM;
    }

    // GM张量初始化
    using A_T = float;
    using B_T = float;
    using C_T = float;

    AscendC::GlobalTensor<A_T> aGlobal;
    AscendC::GlobalTensor<B_T> bGlobal;
    AscendC::GlobalTensor<C_T> cGlobal;

    int32_t sizeA = tiling.ALayoutInfoB * tiling.singleCoreM * tiling.singleCoreK * sizeof(A_T);
    int32_t sizeB = tiling.BLayoutInfoB * tiling.singleCoreK * tiling.singleCoreN * sizeof(B_T);
    int32_t sizeC = tiling.CLayoutInfoB * tiling.singleCoreM * tiling.singleCoreN * sizeof(C_T);

    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(aGM), sizeA);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(bGM), sizeB);
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(cGM), sizeC);

    // 共享资源配置（参考文档示例）
    tiling.shareMode = 0;
    tiling.shareL1Size = 512 * 1024;
    tiling.shareL0CSize = 128 * 1024;
    tiling.shareUbSize = 0;

    // 构造Matmul类型，NORMAL布局
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::NORMAL> aType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, B_T, false, LayoutMode::NORMAL> bType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::NORMAL> cType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T> biasType;

    constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);
    matmul::Matmul<aType, bType, cType, biasType, MM_CFG> mm;

    TPipe pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm);
    mm.Init(&tiling);

    // 绑定张量
    AscendC::GlobalTensor<A_T> gm_a;
    gm_a.SetGlobalBuffer(const_cast<__gm__ A_T*>(aGlobal[0].GetPhyAddr()), tiling.singleCoreM * tiling.singleCoreK);

    AscendC::GlobalTensor<B_T> gm_b;
    gm_b.SetGlobalBuffer(const_cast<__gm__ B_T*>(bGlobal[0].GetPhyAddr()), tiling.singleCoreK * tiling.singleCoreN);

    AscendC::GlobalTensor<C_T> gm_c;
    gm_c.SetGlobalBuffer(const_cast<__gm__ C_T*>(cGlobal[0].GetPhyAddr()), tiling.singleCoreM * tiling.singleCoreN);

    mm.SetTensorA(gm_a, false);
    mm.SetTensorB(gm_b, false);
    mm.SetWorkspace(workspaceGM, 0);

    // 从tiling读取batch信息
    uint32_t batchA = static_cast<uint32_t>(tiling.ALayoutInfoB);
    uint32_t batchB = static_cast<uint32_t>(tiling.BLayoutInfoB);

    // 执行多batch矩阵乘
    mm.IterateBatch(gm_c, batchA, batchB, false);
}
