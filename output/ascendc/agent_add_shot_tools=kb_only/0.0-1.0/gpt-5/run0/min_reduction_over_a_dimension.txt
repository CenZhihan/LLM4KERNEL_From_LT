host_operator_src="""
#include "min_reduction_over_a_dimension_custom_tiling.h"
#include "register/op_def_registry.h"

namespace {
inline void CanonicalizeDim(int64_t &dim, int64_t rank)
{
    if (dim < 0) dim += rank;
    if (dim < 0) dim = 0;
    if (dim >= rank) dim = rank - 1;
}
}

namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 获取输入形状与属性dim
    const gert::Shape* in_shape = context->GetInputShape(0);
    auto origin = in_shape->GetOriginShape();
    int64_t rank = origin.GetDimNum();

    int64_t dim_attr = 0;
    (void)context->GetAttrInt("dim", dim_attr);
    CanonicalizeDim(dim_attr, rank);

    // 计算 outer, reduceLen, inner
    uint64_t outer = 1;
    uint64_t inner = 1;
    uint64_t reduceLen = 1;

    for (int64_t i = 0; i < dim_attr; ++i) {
        outer *= static_cast<uint64_t>(origin.GetDim(i));
    }
    reduceLen = static_cast<uint64_t>(origin.GetDim(dim_attr));
    for (int64_t i = dim_attr + 1; i < rank; ++i) {
        inner *= static_cast<uint64_t>(origin.GetDim(i));
    }

    // 设置Block与简单tileNum
    const uint32_t MAX_BLOCKS = 8; // 参考示例，固定配置
    uint32_t block_dim = static_cast<uint32_t>((outer < MAX_BLOCKS) ? outer : MAX_BLOCKS);
    if (block_dim == 0) block_dim = 1;
    const uint32_t TILE_NUM = 1;

    MinReductionOverADimensionCustomTilingData tiling;
    tiling.set_outer(static_cast<uint32_t>(outer));
    tiling.set_reduceLen(static_cast<uint32_t>(reduceLen));
    tiling.set_inner(static_cast<uint32_t>(inner));
    tiling.set_dim(static_cast<int32_t>(dim_attr));
    tiling.set_tileNum(TILE_NUM);

    context->SetBlockDim(block_dim);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* in = context->GetInputShape(0);
    auto origin = in->GetOriginShape();
    int64_t rank = origin.GetDimNum();

    int64_t dim_attr = 0;
    (void)context->GetAttrInt("dim", dim_attr);
    if (dim_attr < 0) dim_attr += rank;
    if (dim_attr < 0) dim_attr = 0;
    if (dim_attr >= rank) dim_attr = rank - 1;

    // 输出形状为去掉被约简维度后的形状
    std::vector<int64_t> out_dims;
    out_dims.reserve(static_cast<size_t>(rank > 0 ? rank - 1 : 0));
    for (int64_t i = 0; i < rank; ++i) {
        if (i == dim_attr) continue;
        out_dims.push_back(origin.GetDim(i));
    }
    gert::Shape* out = context->GetOutputShape(0);
    out->SetOriginShape(ge::Shape(out_dims));
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "min_reduction_over_a_dimension_custom_tiling_data.h"

constexpr int32_t IN_Q_NUM = 1;
constexpr int32_t OUT_Q_NUM = 1;

class KernelMinReduce {
public:
    __aicore__ inline KernelMinReduce() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t outer, uint32_t reduceLen, uint32_t inner)
    {
        this->outer = outer;
        this->reduceLen = reduceLen;
        this->inner = inner;

        // 基于outer维度做block划分
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blkIdx = AscendC::GetBlockIdx();

        this->outerPerBlock = (outer + blockNum - 1) / blockNum;
        this->outerStart = blkIdx * this->outerPerBlock;
        if (this->outerStart > outer) this->outerStart = outer;
        uint32_t outerEnd = this->outerStart + this->outerPerBlock;
        if (outerEnd > outer) outerEnd = outer;
        this->outerThisBlock = (outerEnd > this->outerStart) ? (outerEnd - this->outerStart) : 0;

        // 设置GM buffer起始位置到当前block负责的范围
        // 输入大小 = outer * reduceLen * inner
        // 当前block起始偏移
        uint64_t xStartOffset = static_cast<uint64_t>(this->outerStart) * this->reduceLen * this->inner;
        uint64_t yStartOffset = static_cast<uint64_t>(this->outerStart) * this->inner;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_X *>(x) + xStartOffset,
                            static_cast<uint64_t>(this->outerThisBlock) * this->reduceLen * this->inner);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_Y *>(y) + yStartOffset,
                            static_cast<uint64_t>(this->outerThisBlock) * this->inner);

        // 本算子按inner为连续长度进行向量化处理
        pipe.InitBuffer(inQueueX, IN_Q_NUM, this->inner * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, OUT_Q_NUM, this->inner * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t o = 0; o < this->outerThisBlock; ++o) {
            // curMin保存当前o的最小值向量
            AscendC::LocalTensor<DTYPE_Y> curMin = outQueueY.AllocTensor<DTYPE_Y>();
            // 拷入第一帧切片 (k=0)
            uint64_t base0 = static_cast<uint64_t>(o) * this->reduceLen * this->inner;
            AscendC::DataCopy(curMin, xGm[base0], this->inner);

            // 依次与后续帧做按元素最小归约
            for (uint32_t k = 1; k < this->reduceLen; ++k) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                uint64_t basek = static_cast<uint64_t>(o) * this->reduceLen * this->inner
                               + static_cast<uint64_t>(k) * this->inner;
                AscendC::DataCopy(xLocal, xGm[basek], this->inner);
                // 使用向量最小操作
                AscendC::Min(curMin, curMin, xLocal, this->inner);
                inQueueX.FreeTensor(xLocal);
            }

            outQueueY.EnQue<DTYPE_Y>(curMin);

            // 写回输出
            AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
            uint64_t yoff = static_cast<uint64_t>(o) * this->inner;
            AscendC::DataCopy(yGm[yoff], yLocal, this->inner);
            outQueueY.FreeTensor(yLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, IN_Q_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, OUT_Q_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t outer = 0;
    uint32_t reduceLen = 0;
    uint32_t inner = 0;

    uint32_t outerPerBlock = 0;
    uint32_t outerStart = 0;
    uint32_t outerThisBlock = 0;
};

extern "C" __global__ __aicore__ void min_reduction_over_a_dimension_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMinReduce op;
    op.Init(x, y, tiling_data.outer, tiling_data.reduceLen, tiling_data.inner);
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor min_reduction_over_a_dimension_custom_impl_npu(const at::Tensor& self, int64_t dim) {
    int64_t rank = self.dim();
    int64_t d = dim >= 0 ? dim : dim + rank;
    if (d < 0) d = 0;
    if (d >= rank) d = rank - 1;

    std::vector<int64_t> out_sizes = self.sizes().vec();
    out_sizes.erase(out_sizes.begin() + d);

    at::Tensor result = at::empty(out_sizes, self.options());
    EXEC_NPU_CMD(aclnnMinReductionOverADimensionCustom, self, d, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("min_reduction_over_a_dimension_custom", &min_reduction_over_a_dimension_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("min_reduction_over_a_dimension_custom", &min_reduction_over_a_dimension_custom_impl_npu, "min reduce over a dimension");
