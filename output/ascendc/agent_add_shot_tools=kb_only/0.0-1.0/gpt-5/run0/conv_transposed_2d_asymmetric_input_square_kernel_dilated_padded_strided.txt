kernel_src="""
#include \"kernel_operator.h\"
#define __NPU_TILING__
#include \"conv_transposed_2d_asymmetric_input_square_kernel_dilated_padded_strided_custom_tiling_data.h\"

extern \"C\" __global__ __aicore__ void conv_transposed_2d_asymmetric_input_square_kernel_dilated_padded_strided_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    const uint32_t N  = tilingData.n;
    const uint32_t Ci = tilingData.c_in;
    const uint32_t Hi = tilingData.h_in;
    const uint32_t Wi = tilingData.w_in;
    const uint32_t Co = tilingData.c_out;
    const uint32_t K  = tilingData.k;
    const uint32_t S  = tilingData.stride;
    const uint32_t P  = tilingData.padding;
    const uint32_t D  = tilingData.dilation;
    const uint32_t Ho = tilingData.h_out;
    const uint32_t Wo = tilingData.w_out;
    const uint32_t hasBias = tilingData.hasBias;

    __gm__ float* xGm  = reinterpret_cast<__gm__ float*>(x);
    __gm__ float* wGm  = reinterpret_cast<__gm__ float*>(weight);
    __gm__ float* bGm  = reinterpret_cast<__gm__ float*>(bias);
    __gm__ float* yGm  = reinterpret_cast<__gm__ float*>(y);

    // 线性划分输出元素到各个Core
    uint32_t totalOut = tilingData.totalOutLength;
    uint32_t blockNum = AscendC::GetBlockNum();
    uint32_t blockIdx = AscendC::GetBlockIdx();

    uint32_t blockLen = (totalOut + blockNum - 1) / blockNum;
    uint32_t start    = blockIdx * blockLen;
    uint32_t end      = AscendC::Min(totalOut, start + blockLen);

    // 预计算步长
    uint64_t x_stride_n  = (uint64_t)Ci * Hi * Wi;
    uint64_t x_stride_c  = (uint64_t)Hi * Wi;
    uint64_t x_stride_h  = (uint64_t)Wi;

    // weight: [Ci, Co, K, K]
    uint64_t w_stride_ci = (uint64_t)Co * K * K;
    uint64_t w_stride_co = (uint64_t)K * K;
    uint64_t w_stride_kh = (uint64_t)K;

    // y: [N, Co, Ho, Wo]
    uint64_t y_stride_n  = (uint64_t)Co * Ho * Wo;
    uint64_t y_stride_c  = (uint64_t)Ho * Wo;
    uint64_t y_stride_h  = (uint64_t)Wo;

    // 初始化输出为0（保证非覆盖写也正确）
    for (uint32_t idx = start; idx < end; ++idx) {
        uint32_t tmp = idx;
        uint32_t ow = tmp % Wo; tmp /= Wo;
        uint32_t oh = tmp % Ho; tmp /= Ho;
        uint32_t oc = tmp % Co; tmp /= Co;
        uint32_t n  = tmp;

        uint64_t y_off = (uint64_t)n * y_stride_n + (uint64_t)oc * y_stride_c + (uint64_t)oh * y_stride_h + (uint64_t)ow;
        float acc = 0.0f;
        if (hasBias) {
            acc = bGm[oc];
        }

        // 反卷积计算
        // 条件: (oh + P - kh*D) % S == 0 且 (ow + P - kw*D) % S == 0
        for (uint32_t ic = 0; ic < Ci; ++ic) {
            for (uint32_t kh = 0; kh < K; ++kh) {
                int32_t h_in_mod = (int32_t)oh + (int32_t)P - (int32_t)kh * (int32_t)D;
                if (h_in_mod % (int32_t)S != 0) continue;
                int32_t ih = h_in_mod / (int32_t)S;
                if (ih < 0 || ih >= (int32_t)Hi) continue;

                for (uint32_t kw = 0; kw < K; ++kw) {
                    int32_t w_in_mod = (int32_t)ow + (int32_t)P - (int32_t)kw * (int32_t)D;
                    if (w_in_mod % (int32_t)S != 0) continue;
                    int32_t iw = w_in_mod / (int32_t)S;
                    if (iw < 0 || iw >= (int32_t)Wi) continue;

                    uint64_t x_off = (uint64_t)n * x_stride_n + (uint64_t)ic * x_stride_c + (uint64_t)ih * x_stride_h + (uint64_t)iw;
                    uint64_t w_off = (uint64_t)ic * w_stride_ci + (uint64_t)oc * w_stride_co + (uint64_t)kh * w_stride_kh + (uint64_t)kw;

                    acc += xGm[x_off] * wGm[w_off];
                }
            }
        }
        yGm[y_off] = acc;
    }
}
