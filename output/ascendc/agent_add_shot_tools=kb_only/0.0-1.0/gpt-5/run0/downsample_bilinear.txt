static constexpr uint32_t BLOCK_DIM = 8;
static constexpr uint32_t OUT_H = 60;
static constexpr uint32_t OUT_W = 80;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    auto inShape = context->GetInputShape(0);

    // 读取输入维度，假定输入为 NCHW
    uint32_t n = static_cast<uint32_t>(inShape->GetOriginShape().GetDim(0));
    uint32_t c = static_cast<uint32_t>(inShape->GetOriginShape().GetDim(1));
    uint32_t inH = static_cast<uint32_t>(inShape->GetOriginShape().GetDim(2));
    uint32_t inW = static_cast<uint32_t>(inShape->GetOriginShape().GetDim(3));

    DownsampleBilinearCustomTilingData tiling;
    tiling.set_n(n);
    tiling.set_c(c);
    tiling.set_inH(inH);
    tiling.set_inW(inW);
    tiling.set_outH(OUT_H);
    tiling.set_outW(OUT_W);
    tiling.set_totalNC(n * c);

    context->SetBlockDim(BLOCK_DIM);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t* ws = context->GetWorkspaceSizes(1);
    ws[0] = 0;
    return ge::GRAPH_SUCCESS;
}
} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);

    // 输出为 N, C, 60, 80
    auto n = x_shape->GetOriginShape().GetDim(0);
    auto c = x_shape->GetOriginShape().GetDim(1);
    y_shape->SetDimNum(4);
    y_shape->SetDim(0, n);
    y_shape->SetDim(1, c);
    y_shape->SetDim(2, 60);
    y_shape->SetDim(3, 80);
    return GRAPH_SUCCESS;

    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "downsample_bilinear_custom_tiling_data.h"

// 简化实现：按 N*C 维度进行 block 切分；每个 block 处理若干个 (n,c) 平面，
// 针对对齐到 (outH,outW) 的输出，逐点执行 bilinear 插值 (align_corners=false)。

class KernelDownsampleBilinear {
public:
    __aicore__ inline KernelDownsampleBilinear() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t n, uint32_t c,
                                uint32_t inH, uint32_t inW,
                                uint32_t outH, uint32_t outW,
                                uint32_t totalNC)
    {
        this->n = n;
        this->c = c;
        this->inH = inH;
        this->inW = inW;
        this->outH = outH;
        this->outW = outW;
        this->inHW = inH * inW;
        this->outHW = outH * outW;
        this->totalNC = totalNC;

        uint32_t blocks = AscendC::GetBlockNum();
        uint32_t per = (totalNC + blocks - 1) / blocks;
        this->startNc = per * AscendC::GetBlockIdx();
        this->ncCount = (this->startNc >= totalNC) ? 0 : (AscendC::Min(per, totalNC - this->startNc));

        // 直接使用 GM 指针（简化实现）
        xPtr = reinterpret_cast<__gm__ DTYPE_X*>(x) + static_cast<uint64_t>(startNc) * inHW;
        yPtr = reinterpret_cast<__gm__ DTYPE_Z*>(y) + static_cast<uint64_t>(startNc) * outHW;

        // 可选：如需本地缓冲可初始化 pipe/queue，这里核心逻辑为逐点插值，省略队列
    }

    __aicore__ inline void Process()
    {
        if (ncCount == 0) {
            return;
        }

        const float scaleH = static_cast<float>(inH) / static_cast<float>(outH);
        const float scaleW = static_cast<float>(inW) / static_cast<float>(outW);

        for (uint32_t ncIdx = 0; ncIdx < ncCount; ++ncIdx) {
            __gm__ DTYPE_X* src = xPtr + static_cast<uint64_t>(ncIdx) * inHW;
            __gm__ DTYPE_Z* dst = yPtr + static_cast<uint64_t>(ncIdx) * outHW;

            for (uint32_t oh = 0; oh < outH; ++oh) {
                float in_y = (static_cast<float>(oh) + 0.5f) * scaleH - 0.5f;
                int32_t y0 = static_cast<int32_t>(AscendC::Floor(in_y));
                float ly = in_y - static_cast<float>(y0);
                if (y0 < 0) { y0 = 0; ly = 0.f; }
                int32_t y1 = y0 + 1;
                if (y1 >= static_cast<int32_t>(inH)) { y1 = static_cast<int32_t>(inH) - 1; }
                float wy0 = 1.f - ly;
                float wy1 = ly;

                for (uint32_t ow = 0; ow < outW; ++ow) {
                    float in_x = (static_cast<float>(ow) + 0.5f) * scaleW - 0.5f;
                    int32_t x0 = static_cast<int32_t>(AscendC::Floor(in_x));
                    float lx = in_x - static_cast<float>(x0);
                    if (x0 < 0) { x0 = 0; lx = 0.f; }
                    int32_t x1 = x0 + 1;
                    if (x1 >= static_cast<int32_t>(inW)) { x1 = static_cast<int32_t>(inW) - 1; }
                    float wx0 = 1.f - lx;
                    float wx1 = lx;

                    uint64_t off00 = static_cast<uint64_t>(y0) * inW + static_cast<uint64_t>(x0);
                    uint64_t off01 = static_cast<uint64_t>(y0) * inW + static_cast<uint64_t>(x1);
                    uint64_t off10 = static_cast<uint64_t>(y1) * inW + static_cast<uint64_t>(x0);
                    uint64_t off11 = static_cast<uint64_t>(y1) * inW + static_cast<uint64_t>(x1);

                    float v00 = static_cast<float>(src[off00]);
                    float v01 = static_cast<float>(src[off01]);
                    float v10 = static_cast<float>(src[off10]);
                    float v11 = static_cast<float>(src[off11]);

                    float top = v00 * wx0 + v01 * wx1;
                    float bot = v10 * wx0 + v11 * wx1;
                    float v = top * wy0 + bot * wy1;

                    dst[static_cast<uint64_t>(oh) * outW + static_cast<uint64_t>(ow)] = static_cast<DTYPE_Z>(v);
                }
            }
        }
    }

private:
    // 维度与切分信息
    uint32_t n, c, inH, inW, outH, outW;
    uint32_t inHW, outHW;
    uint32_t totalNC;
    uint32_t startNc;
    uint32_t ncCount;

    // GM 基址
    __gm__ DTYPE_X* xPtr = nullptr;
    __gm__ DTYPE_Z* yPtr = nullptr;

    // 可选：pipeline/queue，如需进一步优化可引入
    AscendC::TPipe pipe;
};

extern "C" __global__ __aicore__ void downsample_bilinear_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelDownsampleBilinear op;
    op.Init(x, y,
            tiling_data.n, tiling_data.c,
            tiling_data.inH, tiling_data.inW,
            tiling_data.outH, tiling_data.outW,
            tiling_data.totalNC);
    op.Process();
}
