host_operator_src="""
#include "cumsum_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
// 行粒度分块，避免UB占用过大。默认每次处理1行。
const uint32_t TILE_OUTER = 1;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    CumsumCustomTilingData tiling;

    // 获取输入形状
    const gert::Shape* in_shape = context->GetInputShape(0);
    auto origin = in_shape->GetOriginShape();
    uint32_t rank = origin.GetDimNum();
    if (rank == 0) {
        return ge::GRAPH_FAILED;
    }
    uint32_t inner = static_cast<uint32_t>(origin.GetDim(rank - 1));
    // outter为非最后一维的乘积
    uint64_t totalSize = origin.GetShapeSize();
    uint32_t outter = static_cast<uint32_t>(totalSize / inner);

    // 设置Block维度与tiling参数
    context->SetBlockDim(BLOCK_DIM);
    tiling.set_outter(outter);
    tiling.set_inner(inner);
    tiling.set_tileOuter(TILE_OUTER);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无需workspace
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* in_shape = context->GetInputShape(0);
    gert::Shape* out_shape = context->GetOutputShape(0);
    *out_shape = *in_shape;
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class CumsumCustom : public OpDef {
public:
    explicit CumsumCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);

        this->AICore()
            .SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(CumsumCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "cumsum_custom_tiling_data.h"

// 参考AscendC CumSum高阶API：
// Prototype（不显式传入sharedTmpBuffer时由框架分配临时空间）
// template <typename T, const CumSumConfig &config = defaultCumSumConfig>
// __aicore__ inline void CumSum(LocalTensor<T> &dstTensor, LocalTensor<T> &lastRowTensor,
//                               const LocalTensor<T> &srcTensor, const CumSumInfo &cumSumInfo)
//
// CumSumInfo: { outter, inner }，且 inner*sizeof(T) 需32B对齐。
// 本kernel按最后一维进行逐行前缀和，逐块搬运以控制UB占用。

constexpr int32_t BUFFER_NUM = 2; // queue中的双缓冲
class KernelCumsum {
public:
    __aicore__ inline KernelCumsum() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t outter, uint32_t inner, uint32_t tileOuter)
    {
        height = outter;
        width = inner;
        rowsPerTile = tileOuter;

        // 多核切分：按行均分
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        rowsPerBlock = (height + blockNum - 1) / blockNum;
        uint32_t startRow = blockIdx * rowsPerBlock;
        uint32_t endRow = AscendC::Min(startRow + rowsPerBlock, height);
        rowsPerBlock = (endRow > startRow) ? (endRow - startRow) : 0;
        baseOffset = startRow * width; // 以元素为单位的起始偏移

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + baseOffset, static_cast<uint64_t>(rowsPerBlock) * width);
        yGm.SetGlobalBuffer((__gm__ DTYPE_X *)y + baseOffset, static_cast<uint64_t>(rowsPerBlock) * width);

        // 每个tile搬运 rowsThis*width 元素
        // 预留一个lastRow缓冲（即便不输出，也按API形参占位）
        pipe.InitBuffer(inQueueX, BUFFER_NUM, static_cast<uint64_t>(rowsPerTile) * width * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, static_cast<uint64_t>(rowsPerTile) * width * sizeof(DTYPE_X));
        pipe.InitBuffer(lastRowQueue, BUFFER_NUM, static_cast<uint64_t>(width) * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        if (rowsPerBlock == 0) {
            return;
        }
        uint32_t tiles = (rowsPerBlock + rowsPerTile - 1) / rowsPerTile;
        uint32_t loopCount = tiles * BUFFER_NUM;

        for (uint32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline uint32_t GetRowsThis(uint32_t tileIdx) const
    {
        uint32_t startRowInBlock = tileIdx * rowsPerTile;
        if (startRowInBlock >= rowsPerBlock) return 0;
        uint32_t remain = rowsPerBlock - startRowInBlock;
        return remain > rowsPerTile ? rowsPerTile : remain;
    }

    __aicore__ inline void CopyIn(uint32_t progress)
    {
        uint32_t tileIdx = progress % ((rowsPerBlock + rowsPerTile - 1) / rowsPerTile);
        uint32_t rowsThis = GetRowsThis(tileIdx);
        if (rowsThis == 0) return;

        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        uint64_t elemOffset = static_cast<uint64_t>(tileIdx) * rowsPerTile * width;
        AscendC::DataCopy(xLocal, xGm[elemOffset], static_cast<uint64_t>(rowsThis) * width);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t progress)
    {
        uint32_t tileIdx = progress % ((rowsPerBlock + rowsPerTile - 1) / rowsPerTile);
        uint32_t rowsThis = GetRowsThis(tileIdx);
        if (rowsThis == 0) return;

        AscendC::LocalTensor<DTYPE_X> srcLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> dstLocal = outQueueY.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> lastRowLocal = lastRowQueue.AllocTensor<DTYPE_X>();

        // isLastAxis=true, outputLastRow=false（默认配置），逐行前缀和
        AscendC::CumSumInfo info{rowsThis, width};
        AscendC::CumSum<DTYPE_X>(dstLocal, lastRowLocal, srcLocal, info);

        outQueueY.EnQue<DTYPE_X>(dstLocal);
        inQueueX.FreeTensor(srcLocal);
        lastRowQueue.FreeTensor(lastRowLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        uint32_t tileIdx = progress % ((rowsPerBlock + rowsPerTile - 1) / rowsPerTile);
        uint32_t rowsThis = GetRowsThis(tileIdx);
        if (rowsThis == 0) return;

        AscendC::LocalTensor<DTYPE_X> dstLocal = outQueueY.DeQue<DTYPE_X>();
        uint64_t elemOffset = static_cast<uint64_t>(tileIdx) * rowsPerTile * width;
        AscendC::DataCopy(yGm[elemOffset], dstLocal, static_cast<uint64_t>(rowsThis) * width);
        outQueueY.FreeTensor(dstLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM>  inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM>  lastRowQueue;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_X> yGm;

    uint32_t height = 0;       // 总行数(outter)
    uint32_t width = 0;        // 行宽(inner)
    uint32_t rowsPerTile = 1;  // 每次处理的行数
    uint32_t rowsPerBlock = 0; // 当前核处理的行数
    uint64_t baseOffset = 0;   // 当前核在整体中的基准偏移（元素数）
};

extern "C" __global__ __aicore__ void cumsum_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelCumsum op;
    op.Init(x, y, tiling_data.outter, tiling_data.inner, tiling_data.tileOuter);
    op.Process();
}
"""
