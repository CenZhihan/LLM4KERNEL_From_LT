kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transposed_1d_asymmetric_input_square_kernel_padded_strided_dilated_custom_tiling_data.h"

using DTYPE = float;

class KernelConvT1D {
public:
    __aicore__ inline KernelConvT1D() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
                                uint32_t n, uint32_t cin, uint32_t cout,
                                uint32_t linPad, uint32_t keff, uint32_t lout, uint64_t totalOutElems)
    {
        this->n = n;
        this->cin = cin;
        this->cout = cout;
        this->linPad = linPad;
        this->keff = keff;
        this->lout = lout;
        this->totalOutElems = totalOutElems;

        uint64_t xSize = static_cast<uint64_t>(n) * cin * linPad;
        uint64_t wSize = static_cast<uint64_t>(cout) * cin * keff;
        uint64_t bSize = static_cast<uint64_t>(cout);
        uint64_t ySize = static_cast<uint64_t>(n) * cout * lout;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(x), xSize);
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(w), wSize);
        bGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(b), bSize);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(y), ySize);

        // 准备一个小型计算缓冲区（用于标量/微向量计算）
        pipe.InitBuffer(calcQ, 1, 8 * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        for (uint64_t g = 0; g < totalOutElems; ++g) {
            // 线性下标 -> (n_idx, oc, pos)
            uint64_t tmp = g;
            uint32_t pos = static_cast<uint32_t>(tmp % lout);
            tmp /= lout;
            uint32_t oc = static_cast<uint32_t>(tmp % cout);
            uint32_t n_idx = static_cast<uint32_t>(tmp / cout);

            // acc = bias[oc]
            AscendC::LocalTensor<DTYPE> accLocal = calcQ.AllocTensor<DTYPE>();
            AscendC::DataCopy(accLocal, bGm[oc], 1);

            // 累加: sum_{ic, k} x[n, ic, pos+k] * w[oc, ic, k]
            for (uint32_t ic = 0; ic < cin; ++ic) {
                for (uint32_t k = 0; k < keff; ++k) {
                    uint64_t xIdx = (static_cast<uint64_t>(n_idx) * cin + ic) * linPad + (pos + k);
                    uint64_t wIdx = (static_cast<uint64_t>(oc) * cin + ic) * keff + k;

                    AscendC::LocalTensor<DTYPE> xLocal = calcQ.AllocTensor<DTYPE>();
                    AscendC::LocalTensor<DTYPE> wLocal = calcQ.AllocTensor<DTYPE>();
                    AscendC::LocalTensor<DTYPE> prodLocal = calcQ.AllocTensor<DTYPE>();

                    AscendC::DataCopy(xLocal, xGm[xIdx], 1);
                    AscendC::DataCopy(wLocal, wGm[wIdx], 1);
                    AscendC::Mul(prodLocal, xLocal, wLocal, 1);
                    AscendC::Add(accLocal, accLocal, prodLocal, 1);

                    calcQ.FreeTensor(xLocal);
                    calcQ.FreeTensor(wLocal);
                    calcQ.FreeTensor(prodLocal);
                }
            }

            uint64_t yIdx = (static_cast<uint64_t>(n_idx) * cout + oc) * lout + pos;
            AscendC::DataCopy(yGm[yIdx], accLocal, 1);
            calcQ.FreeTensor(accLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> calcQ;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> wGm;
    AscendC::GlobalTensor<DTYPE> bGm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t n;
    uint32_t cin;
    uint32_t cout;
    uint32_t linPad;
    uint32_t keff;
    uint32_t lout;
    uint64_t totalOutElems;
};

extern "C" __global__ __aicore__ void conv_transposed_1d_asymmetric_input_square_kernel_padded_strided_dilated_custom(
    GM_ADDR x_pad, GM_ADDR w_eff, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    KernelConvT1D op;
    op.Init(x_pad, w_eff, bias, y,
            tiling_data.n, tiling_data.cin, tiling_data.cout,
            tiling_data.linPad, tiling_data.keff, tiling_data.lout, tiling_data.totalOutElems);
    op.Process();
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

namespace {
at::Tensor dilate_kernel_1d(const at::Tensor& w_flip, int64_t dilation) {
    // w_flip: [Cout, Cin, K]
    auto Cout = w_flip.size(0);
    auto Cin = w_flip.size(1);
    auto K = w_flip.size(2);
    int64_t Keff = (K - 1) * dilation + 1;

    auto opts = w_flip.options();
    auto w_eff = at::zeros({Cout, Cin, Keff}, opts);
    // 位置间隔为 dilation 的位置拷贝原核
    w_eff.index_put_({at::indexing::Slice(), at::indexing::Slice(), at::indexing::Slice(0, Keff, dilation)}, w_flip);
    return w_eff;
}

at::Tensor upsample_1d_insert_zeros(const at::Tensor& x, int64_t stride) {
    // x: [N, Cin, L]
    if (stride == 1) return x;
    auto N = x.size(0);
    auto Cin = x.size(1);
    auto L = x.size(2);
    int64_t Lup = (L - 1) * stride + 1;

    auto opts = x.options();
    auto x_up = at::zeros({N, Cin, Lup}, opts);
    x_up.index_put_({at::indexing::Slice(), at::indexing::Slice(), at::indexing::Slice(0, Lup, stride)}, x);
    return x_up;
}
} // namespace

at::Tensor conv_transposed_1d_asymmetric_input_square_kernel_padded_strided_dilated_custom_impl_npu(
    const at::Tensor& x, const at::Tensor& weight, c10::optional<at::Tensor> bias_opt,
    int64_t stride, int64_t padding, int64_t dilation) {

    // 1) 上采样输入: 插零
    auto x_up = upsample_1d_insert_zeros(x, stride);

    // 2) 计算有效padding：P' = dilation*(K-1) - padding
    int64_t K = weight.size(2);
    int64_t pad_eff = dilation * (K - 1) - padding;
    TORCH_CHECK(pad_eff >= 0, "Effective padding must be >= 0");

    // 3) padding 输入
    auto x_pad = at::constant_pad_nd(x_up, {pad_eff, pad_eff}, 0);

    // 4) 权重变换: 先翻转kernel，再交换 in/out 通道，再dilation展开
    // 原始ConvTranspose1d权重形状通常为 [Cin, Cout, K]
    auto w_flip = weight.flip(-1).permute({1, 0, 2}).contiguous(); // [Cout, Cin, K]
    auto w_eff = dilate_kernel_1d(w_flip, dilation).contiguous();  // [Cout, Cin, Keff]

    // 5) Bias
    at::Tensor bias;
    if (bias_opt.has_value() && bias_opt.value().defined()) {
        bias = bias_opt.value().contiguous();
    } else {
        bias = at::zeros({weight.size(1)}, weight.options());
    }

    // 6) 计算输出形状: Lout = LinPad - Keff + 1
    auto N = x_pad.size(0);
    auto Cout = w_eff.size(0);
    auto LinPad = x_pad.size(2);
    auto Keff = w_eff.size(2);
    TORCH_CHECK(LinPad >= Keff, "Invalid shapes: LinPad < Keff");
    int64_t Lout = LinPad - Keff + 1;

    auto result = at::empty({N, Cout, Lout}, x.options());

    // 执行自定义核（输入已预处理为普通Conv1d场景：stride=1, dilation=1, 无padding）
    EXEC_NPU_CMD(aclnnConvTransposed1dAsymmetricInputSquareKernelPaddedStridedDilatedCustom,
                 x_pad, w_eff, bias, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("conv_transposed_1d_asymmetric_input_square_kernel_padded_strided_dilated_custom",
           &conv_transposed_1d_asymmetric_input_square_kernel_padded_strided_dilated_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("conv_transposed_1d_asymmetric_input_square_kernel_padded_strided_dilated_custom",
          &conv_transposed_1d_asymmetric_input_square_kernel_padded_strided_dilated_custom_impl_npu,
          "ConvTranspose1d via custom AscendC (preprocessed to regular Conv1d)");
}

import torch
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义AscendC算子替换ConvTranspose1d，流程：
      - Python侧将反卷积等效为：上采样(插零) + 两侧padding + 翻转/通道交换 + kernel dilation
      - 调用自定义核完成标准Conv1d( stride=1, dilation=1, 无padding ) 的计算
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, stride: int = 1, padding: int = 0, dilation: int = 1, bias: bool = False):
        super(ModelNew, self).__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        self.dilation = dilation
        self.bias_flag = bias

        # 仅用于持有权重和偏置，与原模型保持一致的参数布局: [Cin, Cout, K]
        self.weight = torch.nn.Parameter(torch.empty(in_channels, out_channels, kernel_size))
        torch.nn.init.kaiming_uniform_(self.weight, a=5**0.5)
        if bias:
            self.bias = torch.nn.Parameter(torch.zeros(out_channels))
        else:
            self.register_parameter("bias", None)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return custom_ops_lib.conv_transposed_1d_asymmetric_input_square_kernel_padded_strided_dilated_custom(
            x, self.weight, self.bias, self.stride, self.padding, self.dilation
        )
