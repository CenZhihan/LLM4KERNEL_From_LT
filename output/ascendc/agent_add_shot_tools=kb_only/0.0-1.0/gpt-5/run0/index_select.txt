kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "index_select_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 1; // 简化：每次处理1行

class KernelIndexSelect {
public:
    __aicore__ inline KernelIndexSelect() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR index, GM_ADDR y,
                                uint32_t rows, uint32_t inCols, uint32_t outCols, uint32_t tileRows)
    {
        this->rows = rows;
        this->inCols = inCols;
        this->outCols = outCols;
        this->tileRows = tileRows;

        // 每个Block处理的行数
        this->rowsPerBlock = (this->rows + AscendC::GetBlockNum() - 1) / AscendC::GetBlockNum();
        // 基于block索引计算起始行
        uint32_t startRow = AscendC::GetBlockIdx() * this->rowsPerBlock;
        if (startRow + this->rowsPerBlock > this->rows) {
            this->rowsPerBlock = this->rows - startRow;
        }

        // 绑定GM
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startRow * this->inCols, this->rowsPerBlock * this->inCols);
        indexGm.SetGlobalBuffer((__gm__ int32_t *)index, this->outCols);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Z *)y + startRow * this->outCols, this->rowsPerBlock * this->outCols);

        // 初始化队列缓冲
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileRows * this->inCols * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueIndex, BUFFER_NUM, this->outCols * sizeof(int32_t));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileRows * this->outCols * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        if (this->rowsPerBlock == 0) {
            return;
        }
        uint32_t loopCount = (this->rowsPerBlock + this->tileRows - 1) / this->tileRows;
        for (uint32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<int32_t> indexLocal = inQueueIndex.AllocTensor<int32_t>();

        uint32_t rowsThisTile = this->tileRows;
        uint32_t rowStart = progress * this->tileRows;
        if (rowStart + rowsThisTile > this->rowsPerBlock) {
            rowsThisTile = this->rowsPerBlock - rowStart;
        }

        // 拷贝当前tile的输入行块
        AscendC::DataCopy(xLocal, xGm[rowStart * this->inCols], rowsThisTile * this->inCols);
        // 每次将索引拷贝到本地（索引共享）
        AscendC::DataCopy(indexLocal, indexGm, this->outCols);

        inQueueX.EnQue(xLocal);
        inQueueIndex.EnQue(indexLocal);
    }

    __aicore__ inline void Compute(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<int32_t> indexLocal = inQueueIndex.DeQue<int32_t>();
        AscendC::LocalTensor<DTYPE_Z> yLocal = outQueueY.AllocTensor<DTYPE_Z>();

        uint32_t rowsThisTile = this->tileRows;
        uint32_t rowStart = progress * this->tileRows;
        if (rowStart + rowsThisTile > this->rowsPerBlock) {
            rowsThisTile = this->rowsPerBlock - rowStart;
        }

        // 逐行按索引拣选列
        for (uint32_t r = 0; r < rowsThisTile; ++r) {
            for (uint32_t j = 0; j < this->outCols; ++j) {
                // 读取索引并从该行中取对应列
                int32_t col = indexLocal.GetValue(j);
                // 边界保护（可选）
                col = (col < 0) ? 0 : (col >= static_cast<int32_t>(this->inCols) ? (this->inCols - 1) : col);

                // 从xLocal[r*inCols + col]拷贝1个元素到yLocal[r*outCols + j]
                AscendC::DataCopy(yLocal[r * this->outCols + j], xLocal[r * this->inCols + static_cast<uint32_t>(col)], 1);
            }
        }

        outQueueY.EnQue<DTYPE_Z>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueIndex.FreeTensor(indexLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> yLocal = outQueueY.DeQue<DTYPE_Z>();

        uint32_t rowsThisTile = this->tileRows;
        uint32_t rowStart = progress * this->tileRows;
        if (rowStart + rowsThisTile > this->rowsPerBlock) {
            rowsThisTile = this->rowsPerBlock - rowStart;
        }

        AscendC::DataCopy(yGm[rowStart * this->outCols], yLocal, rowsThisTile * this->outCols);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueIndex;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<int32_t> indexGm;
    AscendC::GlobalTensor<DTYPE_Z> yGm;

    uint32_t rows = 0;
    uint32_t inCols = 0;
    uint32_t outCols = 0;
    uint32_t tileRows = 1;
    uint32_t rowsPerBlock = 0;
};

extern "C" __global__ __aicore__ void index_select_custom(GM_ADDR x, GM_ADDR index, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    if ASCEND_IS_AIC {
        return;
    }
    KernelIndexSelect op;
    op.Init(x, index, y, tiling_data.rows, tiling_data.inCols, tiling_data.outCols, tiling_data.tileRows);
    op.Process();
}
"""

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor index_select_custom_impl_npu(const at::Tensor& x, const at::Tensor& index) {
    // 结果形状: [rows, outCols]
    auto rows = x.size(0);
    auto outCols = index.size(0);
    at::Tensor result = at::empty({rows, outCols}, x.options());

    // 索引使用int32
    at::Tensor index_i32 = index.scalar_type() == at::kInt ? index : index.to(at::kInt);

    EXEC_NPU_CMD(aclnnIndexSelectCustom, x, index_i32, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("index_select_custom", &index_select_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("index_select_custom", &index_select_custom_impl_npu, "index_select over dim=1");
