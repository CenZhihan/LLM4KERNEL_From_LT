inline uint64_t ShapeSize(const gert::Shape& s) {
    uint64_t size = 1;
    for (int i = 0; i < s.GetDimNum(); ++i) {
        size *= static_cast<uint64_t>(s.GetDim(i));
    }
    return size;
}
}

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    const gert::Shape* xShape = context->GetInputShape(0);
    const gert::Shape* yShape = context->GetInputShape(1);

    uint32_t lenX = static_cast<uint32_t>(ShapeSize(xShape->GetOriginShape()));
    uint32_t lenY = static_cast<uint32_t>(ShapeSize(yShape->GetOriginShape()));

    Densenet201CustomTilingData tiling;
    context->SetBlockDim(BLOCK_DIM);
    tiling.set_lenX(lenX);
    tiling.set_lenY(lenY);
    tiling.set_tileNum(TILE_NUM);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(),
                        context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* xShape = context->GetInputShape(0);
    const gert::Shape* yShape = context->GetInputShape(1);
    gert::Shape* zShape = context->GetOutputShape(0);

    // 期望输入都是4D NCHW，通道拼接：C_out = Cx + Cy
    auto xDimNum = xShape->GetDimNum();
    auto yDimNum = yShape->GetDimNum();
    if (xDimNum != yDimNum || (xDimNum != 4 && xDimNum != 3 && xDimNum != 2 && xDimNum != 1)) {
        // 简化处理：若维度不同，回传输入0形状
        *zShape = *xShape;
        return GRAPH_SUCCESS;
    }
    std::vector<int64_t> outDims(xDimNum, 0);
    for (int i = 0; i < xDimNum; ++i) {
        outDims[i] = xShape->GetDim(i);
    }
    if (xDimNum >= 2) {
        // 以第1维为通道维（NCHW/NCH/NC/N）
        outDims[1] = xShape->GetDim(1) + yShape->GetDim(1);
        // 其他维度按输入0
    }
    zShape->SetDimNum(xDimNum);
    for (int i = 0; i < xDimNum; ++i) {
        zShape->SetDim(i, outDims[i]);
    }
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "densenet201_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // 队列中本地tensor个数

class KernelConcatC {
public:
    __aicore__ inline KernelConcatC() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t lenX, uint32_t lenY, uint32_t tileNum)
    {
        this->lenX = lenX;
        this->lenY = lenY;
        this->tileNum = tileNum;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->blockLenX = lenX / blockNum;
        this->blockLenY = lenY / blockNum;
        uint32_t xOffset = this->blockLenX * blockIdx;
        uint32_t yOffset = this->blockLenY * blockIdx;
        uint32_t zOffset = (this->blockLenX + this->blockLenY) * blockIdx;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(x) + xOffset, this->blockLenX);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(y) + yOffset, this->blockLenY);
        zGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(z) + zOffset, this->blockLenX + this->blockLenY);

        this->tileLenX = this->blockLenX / (this->tileNum * BUFFER_NUM);
        this->tileLenY = this->blockLenY / (this->tileNum * BUFFER_NUM);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLenX * sizeof(float));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, this->tileLenY * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // 先拷贝X到Z前半部分
        int32_t loopX = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopX; ++i) {
            CopyInX(i);
            ComputeX(i);
        }
        // 再拷贝Y到Z后半部分
        int32_t loopY = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopY; ++i) {
            CopyInY(i);
            ComputeY(i);
        }
    }

private:
    __aicore__ inline void CopyInX(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLenX], this->tileLenX);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeX(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        // 直接将本地块写到Z的对应位置（前半段）
        AscendC::DataCopy(zGm[progress * this->tileLenX], xLocal, this->tileLenX);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyInY(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = inQueueY.AllocTensor<float>();
        AscendC::DataCopy(yLocal, yGm[progress * this->tileLenY], this->tileLenY);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void ComputeY(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = inQueueY.DeQue<float>();
        // 写到Z的后半段，起始偏移为blockLenX
        AscendC::DataCopy(zGm[this->blockLenX + progress * this->tileLenY], yLocal, this->tileLenY);
        inQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueY;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t lenX;
    uint32_t lenY;
    uint32_t tileNum;
    uint32_t blockLenX;
    uint32_t blockLenY;
    uint32_t tileLenX;
    uint32_t tileLenY;
};

extern "C" __global__ __aicore__ void densenet201_custom(GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConcatC op;
    op.Init(x, y, z, tiling_data.lenX, tiling_data.lenY, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class DenseBlockNew(torch.nn.Module):
    def __init__(self, num_layers: int, num_input_features: int, growth_rate: int):
        super(DenseBlockNew, self).__init__()
        layers = []
        for i in range(num_layers):
            layers.append(self._make_layer(num_input_features + i * growth_rate, growth_rate))
        self.layers = torch.nn.ModuleList(layers)

    def _make_layer(self, in_features: int, growth_rate: int):
        return torch.nn.Sequential(
            torch.nn.BatchNorm2d(in_features),
            torch.nn.ReLU(inplace=True),
            torch.nn.Conv2d(in_features, growth_rate, kernel_size=3, padding=1, bias=False),
            torch.nn.Dropout(0.0)
        )

    def forward(self, x):
        # 使用自定义AscendC算子进行通道拼接，替换torch.cat(features, 1)
        for layer in self.layers:
            new_feature = layer(x)
            x = custom_ops_lib.densenet201_custom(x, new_feature)
        return x

class TransitionLayer(torch.nn.Module):
    def __init__(self, num_input_features: int, num_output_features: int):
        super(TransitionLayer, self).__init__()
        self.transition = torch.nn.Sequential(
            torch.nn.BatchNorm2d(num_input_features),
            torch.nn.ReLU(inplace=True),
            torch.nn.Conv2d(num_input_features, num_output_features, kernel_size=1, bias=False),
            torch.nn.AvgPool2d(kernel_size=2, stride=2)
        )

    def forward(self, x):
        return self.transition(x)

class ModelNew(torch.nn.Module):
    def __init__(self, growth_rate: int = 32, num_classes: int = 1000):
        super(ModelNew, self).__init__()

        self.features = torch.nn.Sequential(
            torch.nn.Conv2d(3, 64, kernel_size=7, stride=2, padding=3, bias=False),
            torch.nn.BatchNorm2d(64),
            torch.nn.ReLU(inplace=True),
            torch.nn.MaxPool2d(kernel_size=3, stride=2, padding=1)
        )

        num_features = 64
        block_layers = [6, 12, 48, 32]

        self.dense_blocks = torch.nn.ModuleList()
        self.transition_layers = torch.nn.ModuleList()

        for i, num_layers in enumerate(block_layers):
            block = DenseBlockNew(num_layers=num_layers, num_input_features=num_features, growth_rate=growth_rate)
            self.dense_blocks.append(block)
            num_features = num_features + num_layers * growth_rate

            if i != len(block_layers) - 1:
                transition = TransitionLayer(num_input_features=num_features, num_output_features=num_features // 2)
                self.transition_layers.append(transition)
                num_features = num_features // 2

        self.final_bn = torch.nn.BatchNorm2d(num_features)
        self.classifier = torch.nn.Linear(num_features, num_classes)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.features(x)

        for i, block in enumerate(self.dense_blocks):
            x = block(x)
            if i != len(self.dense_blocks) - 1:
                x = self.transition_layers[i](x)

        x = self.final_bn(x)
        x = torch.relu(x)
        x = torch.nn.functional.adaptive_avg_pool2d(x, (1, 1)).view(x.size(0), -1)
        x = self.classifier(x)
        return x
'''
