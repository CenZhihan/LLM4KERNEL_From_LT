kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose3d_scale_batch_norm_global_avg_pool_custom_tiling_data.h"

using dataType = float;
constexpr int32_t BUFFER_NUM = 1;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                const optiling::ConvTranspose3dScaleBatchNormGlobalAvgPoolCustomTilingData &t)
    {
        N = t.nLength();
        C = t.cLength();
        D = t.dLength();
        H = t.hLength();
        W = t.wLength();
        DHW = t.dhwLength();

        bnB = t.bnBLength();
        bnS = t.bnSLength();
        bnH = t.bnHLength();

        epsilon = t.epsilon();
        scale = t.scale();

        bshLength = (uint32_t)bnB * bnS * bnH; // C * N * DHW
        ncLength = (uint32_t)N * C;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ dataType *>(x), bshLength);
        gammaGm.SetGlobalBuffer(reinterpret_cast<__gm__ dataType *>(gamma), C);
        betaGm.SetGlobalBuffer(reinterpret_cast<__gm__ dataType *>(beta), C);
        // Output holds N*C elements (pooled to 1x1x1)
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ dataType *>(y), ncLength);

        // Init pipe buffers (allocate full for simplicity as in docs examples)
        pipe.InitBuffer(inQueueX, BUFFER_NUM, sizeof(dataType) * bshLength);
        pipe.InitBuffer(inQueueGamma, BUFFER_NUM, sizeof(dataType) * C);
        pipe.InitBuffer(inQueueBeta, BUFFER_NUM, sizeof(dataType) * C);
        pipe.InitBuffer(outQueueY, BUFFER_NUM, sizeof(dataType) * ncLength);
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        auto xLocal = inQueueX.AllocTensor<dataType>();
        auto gammaLocal = inQueueGamma.AllocTensor<dataType>();
        auto betaLocal = inQueueBeta.AllocTensor<dataType>();

        AscendC::DataCopy(xLocal, xGm, bshLength);
        AscendC::DataCopy(gammaLocal, gammaGm, C);
        AscendC::DataCopy(betaLocal, betaGm, C);

        inQueueX.EnQue(xLocal);
        inQueueGamma.EnQue(gammaLocal);
        inQueueBeta.EnQue(betaLocal);
    }

    __aicore__ inline void Compute()
    {
        auto xLocal = inQueueX.DeQue<dataType>();
        auto gammaLocal = inQueueGamma.DeQue<dataType>();
        auto betaLocal = inQueueBeta.DeQue<dataType>();

        auto pooledLocal = outQueueY.AllocTensor<dataType>();

        // 1) Scale: x = x * scale
        AscendC::Muls(xLocal, xLocal, scale, bshLength);

        // 2) BatchNorm (C,N,DHW) mapped to (B,S,H) = (C,N,DHW)
        // Prepare outputs for BN
        auto tmpOutLocal = tmpOut.AllocTensor<dataType>();
        auto meanLocal = tmpMean.AllocTensor<dataType>();
        auto varLocal = tmpVar.AllocTensor<dataType>();

        AscendC::BatchNorm<dataType, false, false>(tmpOutLocal, meanLocal, varLocal,
                                                   xLocal, gammaLocal, betaLocal,
                                                   (dataType)epsilon, bnTiling);

        // 3) Global Average Pooling over DHW for each (N,C)
        // Reshape logically to [N*C, DHW], compute mean along last axis.
        // Accumulate sum for each row and then multiply by 1.0f/DHW.
        const dataType invDHW = (dataType)(1.0f / (float)DHW);
        // For simplicity, use vector reduction APIs per row (conceptual)
        // Each row length = DHW, row count = N*C
        for (uint32_t row = 0; row < ncLength; ++row) {
            // Offset inside tmpOutLocal: row * DHW
            AscendC::LocalTensor<dataType> rowVec = tmpOutLocal[row * DHW];
            dataType sumScalar = 0;
            AscendC::ReduceSum(sumScalar, rowVec, DHW);
            pooledLocal.SetValue(row, sumScalar * invDHW);
        }

        outQueueY.EnQue<dataType>(pooledLocal);

        // Free temps
        tmpOut.FreeTensor(tmpOutLocal);
        tmpMean.FreeTensor(meanLocal);
        tmpVar.FreeTensor(varLocal);

        inQueueX.FreeTensor(xLocal);
        inQueueGamma.FreeTensor(gammaLocal);
        inQueueBeta.FreeTensor(betaLocal);
    }

    __aicore__ inline void CopyOut()
    {
        auto pooledLocal = outQueueY.DeQue<dataType>();
        AscendC::DataCopy(yGm, pooledLocal, ncLength);
        outQueueY.FreeTensor(pooledLocal);
    }

private:
    // GM tensors
    AscendC::GlobalTensor<dataType> xGm;
    AscendC::GlobalTensor<dataType> gammaGm;
    AscendC::GlobalTensor<dataType> betaGm;
    AscendC::GlobalTensor<dataType> yGm;

    // Queues and temp buffers
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueGamma;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueBeta;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpOut;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpMean;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpVar;

    // Shapes
    uint32_t N, C, D, H, W, DHW;
    uint32_t bnB, bnS, bnH;
    uint32_t bshLength;
    uint32_t ncLength;

    // Params
    dataType epsilon;
    dataType scale;

    // BN tiling
    BatchNormTiling bnTiling;
};

extern "C" __global__ __aicore__ void conv_transpose3d_scale_batch_norm_global_avg_pool_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelFused op;
    op.Init(x, gamma, beta, y, tilingData);
    op.Process();
}

import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized: keep ConvTranspose3d in PyTorch, fuse Scale+BatchNorm+GlobalAvgPool3d to a custom AscendC op.
    """
    def __init__(self, in_channels, out_channels, kernel_size, scale_factor, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size)
        self.scale_factor = float(scale_factor)
        self.batch_norm = torch.nn.BatchNorm3d(out_channels, eps=eps, momentum=momentum)
        # GlobalAvgPool3d is fused into the custom op.

    def forward(self, x):
        x = self.conv_transpose(x)
        # fused: scale -> batchnorm -> global average pool -> [N, C, 1, 1, 1]
        y = custom_ops_lib.conv_transpose3d_scale_batch_norm_global_avg_pool_custom(
            x, self.batch_norm.weight, self.batch_norm.bias, self.scale_factor, float(self.batch_norm.eps))
