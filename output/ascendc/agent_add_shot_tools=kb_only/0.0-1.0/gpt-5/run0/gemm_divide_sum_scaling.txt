    uint32_t inputSize = static_cast<uint32_t>(x_shape->GetOriginShape().GetDim(1));

    uint32_t minValue = 0;
    AscendC::GetSumMaxMinTmpSize(inputSize, dtypesize, false, maxValue, minValue);

    const gert::RuntimeAttrs *attrs = context->GetAttrs();
    float scale = 1.0f;
    if (attrs != nullptr && attrs->GetSize() > 0) {

    tiling.set_inputSize(inputSize);
    tiling.set_tmpSize(minValue);
    tiling.set_scale(foldedScale);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "gemm_divide_sum_scaling_custom_tiling_data.h"

using DTYPE_T = DTYPE_X; // assume both inputs are float

class KernelGemmDivideSumScaling {
public:
    __aicore__ inline KernelGemmDivideSumScaling() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR out, uint32_t batch, uint32_t inputSize,
                                uint32_t tmpSize, float scale)
    {
        this->batch = batch;
        this->inputSize = inputSize;
        this->tmpSize = tmpSize;
        this->scale = scale;

        // split by batch along cores
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (batch + blockNum - 1) / blockNum; // ceil split
        uint32_t startRow = blockIdx * rowsPerBlock;
        if (startRow + rowsPerBlock > batch) {
            rowsPerBlock = (batch > startRow) ? (batch - startRow) : 0;
        }

        xGm.SetGlobalBuffer((__gm__ DTYPE_T*)x + startRow * inputSize, rowsPerBlock * inputSize);
        sGm.SetGlobalBuffer((__gm__ DTYPE_T*)y, inputSize);
        outGm.SetGlobalBuffer((__gm__ DTYPE_T*)out + startRow, rowsPerBlock);

        // queues
        pipe.InitBuffer(inQueueX, 1, inputSize * sizeof(DTYPE_T));       // one row of x
        pipe.InitBuffer(inQueueS, 1, inputSize * sizeof(DTYPE_T));       // s vector
        pipe.InitBuffer(tmpQueue, 1, inputSize * sizeof(DTYPE_T));       // product buffer
        pipe.InitBuffer(workQueue, 1, tmpSize);                          // Sum tmp buffer
        pipe.InitBuffer(outQueue, 1, AscendC::ONE_BLK_SIZE);             // one scalar (aligned)
    }

    __aicore__ inline void Process()
    {
        if (rowsPerBlock == 0) {
            return;
        }
        // preload s vector to local once
        AscendC::LocalTensor<DTYPE_T> sLocal = inQueueS.AllocTensor<DTYPE_T>();
        AscendC::DataCopy(sLocal, sGm, inputSize);
        inQueueS.EnQue(sLocal);

        for (uint32_t r = 0; r < rowsPerBlock; ++r) {
            CopyIn(r);
            Compute();
            CopyOut(r);
        }

        // free s
        sLocal = inQueueS.DeQue<DTYPE_T>();
        inQueueS.FreeTensor(sLocal);
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        AscendC::LocalTensor<DTYPE_T> xLocal = inQueueX.AllocTensor<DTYPE_T>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * inputSize], inputSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_T> xLocal = inQueueX.DeQue<DTYPE_T>();
        AscendC::LocalTensor<DTYPE_T> sLocal = inQueueS.DeQue<DTYPE_T>(); // peek
        AscendC::LocalTensor<DTYPE_T> prodLocal = tmpQueue.AllocTensor<DTYPE_T>();
        AscendC::LocalTensor<DTYPE_T> outLocal = outQueue.AllocTensor<DTYPE_T>();
        AscendC::LocalTensor<uint8_t> workLocal = workQueue.AllocTensor<uint8_t>();

        // elementwise multiply: prod = x * s
        AscendC::Mul(prodLocal, xLocal, sLocal, inputSize);

        // reduce sum over last axis
        AscendC::SumParams params;
        params.outter = 1;
        params.n = inputSize;
        params.inner = (inputSize * sizeof(DTYPE_T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(DTYPE_T);
        AscendC::Duplicate<DTYPE_T>(outLocal, (DTYPE_T)0, AscendC::ONE_BLK_SIZE / sizeof(DTYPE_T));
        AscendC::Sum(outLocal, prodLocal, workLocal, params);

        // scale: multiply by folded scale (scale * 0.5 already folded at tiling)
        AscendC::Muls(outLocal, outLocal, (DTYPE_T)scale, 1);

        // enqueue and free
        outQueue.EnQue(outLocal);
        workQueue.FreeTensor(workLocal);
        tmpQueue.FreeTensor(prodLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueS.EnQue(sLocal); // keep s for next iteration
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        AscendC::LocalTensor<DTYPE_T> outLocal = outQueue.DeQue<DTYPE_T>();
        AscendC::DataCopy(outGm[rowIdx], outLocal, 1);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueS;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpQueue;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> workQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue;

    AscendC::GlobalTensor<DTYPE_T> xGm;
    AscendC::GlobalTensor<DTYPE_T> sGm;
    AscendC::GlobalTensor<DTYPE_T> outGm;

    uint32_t batch = 0;
    uint32_t inputSize = 0;
    uint32_t rowsPerBlock = 0;
    uint32_t tmpSize = 0;
    float scale = 1.0f;
};

extern "C" __global__ __aicore__ void gemm_divide_sum_scaling_custom(GM_ADDR x, GM_ADDR y, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmDivideSumScaling op;
    op.Init(x, y, out, tiling_data.batch, tiling_data.inputSize, tiling_data.tmpSize, tiling_data.scale);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model using custom AscendC kernel:
    Computes out = (scaling_factor/2) * (x @ s), where s = weight.sum(dim=0), then reshaped to [B, 1].
    """
    def __init__(self, input_size, hidden_size, scaling_factor):
        super().__init__()
        self.weight = torch.nn.Parameter(torch.randn(hidden_size, input_size))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        # Precompute s = sum over hidden dimension of weight: shape [I]
        s = self.weight.sum(dim=0)  # small overhead vs fused kernel; large GEMM is fused
        # Call fused kernel: handles dot per row + divide-by-2 + scaling
        return custom_ops_lib.gemm_divide_sum_scaling_custom(x, s, float(self.scaling_factor))
'''
