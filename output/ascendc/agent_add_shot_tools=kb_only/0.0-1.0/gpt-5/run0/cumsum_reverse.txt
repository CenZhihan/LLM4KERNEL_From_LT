#include "cumsum_reverse_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace {
inline uint32_t AlignUpTo32Bytes(uint32_t n, uint32_t typeSize) {
    return ((n * typeSize + 32 - 1) / 32) * 32 / typeSize;
}
}

namespace optiling {
const uint32_t BLOCK_DIM = 1;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    CumsumReverseCustomTilingData tiling;
    // 获取输入形状，要求为二维，接口默认ND
    const gert::Shape* in_shape = context->GetInputShape(0);
    auto origin_shape = in_shape->GetOriginShape();
    // 仅支持二维输入：形如 [B, S]
    uint32_t dim0 = static_cast<uint32_t>(origin_shape.GetDim(0));
    uint32_t dim1 = static_cast<uint32_t>(origin_shape.GetDim(1));

    // 读取运行时属性 dim（沿哪个维度进行reverse cumsum）
    const gert::RuntimeAttrs* attrs = context->GetAttrs();
    int64_t dim = 0;
    if (attrs != nullptr && attrs->GetSize() > 0) {
        dim = *(attrs->GetAttrPointer<int64_t>(0));
    }

    // 归一化 dim：仅支持 0 或 1（等价支持 -1 -> 1）
    if (dim == -1) dim = 1;
    bool isLastAxis = (dim == 1);

    uint32_t outter = dim0;
    uint32_t n = dim1; // 内轴实际长度（未对齐）
    // CumSum API 要求 inner*sizeof(T) 必须32字节对齐
    uint32_t typeSize = 4; // float
    uint32_t inner = AlignUpTo32Bytes(n, typeSize);

    // 选择一次分段拷贝的段长，用于在核内做翻转时的块拷贝
    // 这里给一个经验值，开发者可按UB剩余空间调整
    uint32_t segSize = 1024;
    if (segSize > n) segSize = n;

    context->SetBlockDim(BLOCK_DIM);
    tiling.set_outter(outter);
    tiling.set_n(n);
    tiling.set_inner(inner);
    tiling.set_isLastAxis(isLastAxis ? 1U : 0U);
    tiling.set_segSize(segSize);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "cumsum_reverse_custom_tiling_data.h"

// 参考文档：
// - CumSum API（支持按首轴或末轴累加）
// - 要求 inner*sizeof(T) 32字节对齐
// 这里通过块拷贝在本地反转，再调用 CumSum，最后再次块拷贝反转输出，实现 reverse cumsum。

template <typename T>
class KernelCumsumReverse {
public:
    __aicore__ inline KernelCumsumReverse() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t outter, uint32_t n, uint32_t inner,
                                uint32_t isLastAxis, uint32_t segSize)
    {
        this->outter = outter;
        this->n = n;
        this->inner = inner;
        this->isLastAxis = isLastAxis != 0;
        this->segSize = (segSize == 0) ? n : segSize;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(x), outter * n);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(y), outter * n);

        // 分配本地缓冲：按对齐后的 inner 作为每行跨度
        pipe.InitBuffer(inQueue, 1, outter * inner * sizeof(T));
        pipe.InitBuffer(outQueue, 1, outter * inner * sizeof(T));
        // lastRow 队列即使不输出，也要分配占位
        pipe.InitBuffer(lastRowQueue, 1, inner * sizeof(T));
    }

    __aicore__ inline void Process()
    {
        CopyInAndReverse();
        ComputeCumSum();
        ReverseAndCopyOut();
    }

private:
    __aicore__ inline void CopyInAndReverse()
    {
        // 将GM数据按轴做“翻转”后拷入本地，并按 inner 对齐补0
        srcLocal = inQueue.AllocTensor<T>();
        AscendC::Duplicate<T>(srcLocal, static_cast<T>(0), outter * inner);

        if (isLastAxis) {
            // 沿末轴翻转：逐行处理
            for (uint32_t r = 0; r < outter; ++r) {
                uint32_t gmRowBase = r * n;
                uint32_t lcRowBase = r * inner;
                // 以段为单位从右向左取GM，顺序写入Local
                for (uint32_t done = 0; done < n; ) {
                    uint32_t cpy = AscendC::min(segSize, n - done);
                    uint32_t gmOff = gmRowBase + (n - done - cpy);
                    uint32_t lcOff = lcRowBase + done;
                    AscendC::DataCopy(srcLocal[lcOff], xGm[gmOff], cpy);
                    done += cpy;
                }
                // 行尾 padding 区域已在初始化中置0
            }
        } else {
            // 沿首轴翻转：翻转行顺序
            for (uint32_t r = 0; r < outter; ++r) {
                uint32_t gmRowBase = (outter - 1 - r) * n; // 反向行
                uint32_t lcRowBase = r * inner;
                AscendC::DataCopy(srcLocal[lcRowBase], xGm[gmRowBase], n);
                // 行尾 padding 区域已在初始化中置0
            }
        }
        inQueue.EnQue(srcLocal);
    }

    __aicore__ inline void ComputeCumSum()
    {
        dstLocal = outQueue.AllocTensor<T>();
        lastRowLocal = lastRowQueue.AllocTensor<T>();
        srcLocal = inQueue.DeQue<T>();

        const AscendC::CumSumInfo info{outter, inner};
        if (isLastAxis) {
            static constexpr AscendC::CumSumConfig cfg{true, false, false};
            AscendC::CumSum<T, cfg>(dstLocal, lastRowLocal, srcLocal, info);
        } else {
            static constexpr AscendC::CumSumConfig cfg{false, false, false};
            AscendC::CumSum<T, cfg>(dstLocal, lastRowLocal, srcLocal, info);
        }

        outQueue.EnQue(dstLocal);
        lastRowQueue.EnQue(lastRowLocal);
        inQueue.FreeTensor(srcLocal);
    }

    __aicore__ inline void ReverseAndCopyOut()
    {
        dstLocal = outQueue.DeQue<T>();
        lastRowLocal = lastRowQueue.DeQue<T>();
        // 将CumSum结果再按轴“翻转回去”，并只写回有效的 n 元素
        if (isLastAxis) {
            for (uint32_t r = 0; r < outter; ++r) {
                uint32_t gmRowBase = r * n;
                uint32_t lcRowBase = r * inner;
                for (uint32_t done = 0; done < n; ) {
                    uint32_t cpy = AscendC::min(segSize, n - done);
                    uint32_t lcOff = lcRowBase + done;
                    uint32_t gmOff = gmRowBase + (n - done - cpy);
                    AscendC::DataCopy(yGm[gmOff], dstLocal[lcOff], cpy);
                    done += cpy;
                }
            }
        } else {
            for (uint32_t r = 0; r < outter; ++r) {
                uint32_t gmRowBase = (outter - 1 - r) * n; // 反向行
                uint32_t lcRowBase = r * inner;
                AscendC::DataCopy(yGm[gmRowBase], dstLocal[lcRowBase], n);
            }
        }

        outQueue.FreeTensor(dstLocal);
        lastRowQueue.FreeTensor(lastRowLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> lastRowQueue;

    AscendC::GlobalTensor<T> xGm;
    AscendC::GlobalTensor<T> yGm;

    AscendC::LocalTensor<T> srcLocal;
    AscendC::LocalTensor<T> dstLocal;
    AscendC::LocalTensor<T> lastRowLocal;

    uint32_t outter{1};
    uint32_t n{1};
    uint32_t inner{1};
    bool isLastAxis{true};
    uint32_t segSize{1024};
};

extern "C" __global__ __aicore__ void cumsum_reverse_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelCumsumReverse<DTYPE_X> op;
    op.Init(x, y,
            tiling_data.outter, tiling_data.n, tiling_data.inner,
            tiling_data.isLastAxis, tiling_data.segSize);
    op.Process();
}
