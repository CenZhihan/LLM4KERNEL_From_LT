kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matrix_scalar_multiplication_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelMatrixScalarMul {
public:
    __aicore__ inline KernelMatrixScalarMul() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR s, GM_ADDR c, uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        aGm.SetGlobalBuffer((__gm__ float *)a + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        cGm.SetGlobalBuffer((__gm__ float *)c + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        // scalar buffer is length 1
        sGm.SetGlobalBuffer((__gm__ float *)s, 1);

        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(scalarQueue, 1, sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Load scalar once
        float sValue = LoadScalar();

        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i, sValue);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline float LoadScalar()
    {
        AscendC::LocalTensor<float> sLocal = scalarQueue.AllocTensor<float>();
        AscendC::DataCopy(sLocal, sGm, 1);
        // Assume LocalTensor supports GetValue to fetch scalar
        float val = sLocal.GetValue(0);
        scalarQueue.FreeTensor(sLocal);
        return val;
    }

    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> aLocal = inQueueA.AllocTensor<float>();
        AscendC::DataCopy(aLocal, aGm[progress * this->tileLength], this->tileLength);
        inQueueA.EnQue(aLocal);
    }

    __aicore__ inline void Compute(int32_t progress, const float sValue)
    {
        AscendC::LocalTensor<float> aLocal = inQueueA.DeQue<float>();
        AscendC::LocalTensor<float> cLocal = outQueueC.AllocTensor<float>();
        AscendC::Muls(cLocal, aLocal, sValue, this->tileLength);
        outQueueC.EnQue<float>(cLocal);
        inQueueA.FreeTensor(aLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> cLocal = outQueueC.DeQue<float>();
        AscendC::DataCopy(cGm[progress * this->tileLength], cLocal, this->tileLength);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> scalarQueue;

    AscendC::GlobalTensor<float> aGm;
    AscendC::GlobalTensor<float> cGm;
    AscendC::GlobalTensor<float> sGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void matrix_scalar_multiplication_custom(GM_ADDR a, GM_ADDR s, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatrixScalarMul op;
    op.Init(a, s, c, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}
