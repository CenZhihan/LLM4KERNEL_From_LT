kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "mamba_return_final_state_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 1; // single buffer is enough for pure copy

class KernelMambaReturnFinalState {
public:
    __aicore__ inline KernelMambaReturnFinalState() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t perBatchLength,
                                uint32_t batchSize, uint32_t zLen, uint32_t tileNum)
    {
        this->totalLength = totalLength;         // B * H * P * N
        this->perBatchLen = perBatchLength;      // H * P * N
        this->batchSize   = batchSize;           // B
        this->zLen        = zLen;                // Z
        this->tilesPerB   = tileNum;             // tiles per batch

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, (uint32_t)0);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, (uint32_t)0);

        // Tile length inside a batch slice
        this->tileLen = this->perBatchLen / this->tilesPerB;
        if (this->tileLen == 0) { this->tileLen = this->perBatchLen; this->tilesPerB = 1; }

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLen * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        // total tiles to process = B * tilesPerB
        uint32_t loopCount = this->batchSize * this->tilesPerB;
        for (uint32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline uint32_t GetCalLen(uint32_t tileIdxInBatch) const
    {
        if (tileIdxInBatch < (this->tilesPerB - 1)) {
            return this->tileLen;
        } else {
            uint32_t done = this->tileLen * (this->tilesPerB - 1);
            return this->perBatchLen - done;
        }
    }

    __aicore__ inline void CopyIn(uint32_t progress)
    {
        // map progress -> batch b and tile index t
        uint32_t b = progress / this->tilesPerB;
        uint32_t t = progress % this->tilesPerB;

        uint32_t calLen = GetCalLen(t);
        uint32_t offsetInBatch = t * this->tileLen;

        // x layout: [B, Z, H, P, N], contiguous
        // last z slice offset for batch b
        uint64_t srcBase = (uint64_t)b * (uint64_t)(this->zLen * this->perBatchLen)
                         + (uint64_t)(this->zLen - 1) * (uint64_t)this->perBatchLen
                         + (uint64_t)offsetInBatch;

        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[srcBase], calLen);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        // Pure copy from VECIN to VECOUT
        AscendC::Copy(yLocal, xLocal, xLocal.GetSize());
        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        uint32_t b = progress / this->tilesPerB;
        uint32_t t = progress % this->tilesPerB;

        uint32_t calLen = GetCalLen(t);
        uint32_t offsetInBatch = t * this->tileLen;

        // y layout: [B, H, P, N], contiguous
        uint64_t dstBase = (uint64_t)b * (uint64_t)this->perBatchLen + (uint64_t)offsetInBatch;

        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[dstBase], yLocal, calLen);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t totalLength;
    uint32_t perBatchLen;
    uint32_t batchSize;
    uint32_t zLen;
    uint32_t tilesPerB;
    uint32_t tileLen;
};

extern "C" __global__ __aicore__ void mamba_return_final_state_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMambaReturnFinalState op;
    op.Init(x, y,
            tiling_data.totalLength,
            tiling_data.perBatchLength,
            tiling_data.batchSize,
            tiling_data.zLen,
            tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
from einops import rearrange
import torch.nn.functional as F

class ModelNew(torch.nn.Module):
    def __init__(self, batch_size, seq_length, n_heads, d_head, d_state, block_len=64):
        super(ModelNew, self).__init__()
        assert seq_length % block_len == 0, "Sequence length must be divisible by block length"
        self.batch_size = batch_size
        self.seq_length = seq_length
        self.n_heads = n_heads
        self.d_head = d_head
        self.d_state = d_state
        self.block_len = block_len

        self.A = torch.nn.Parameter(torch.randn(batch_size, seq_length, n_heads))
        self.B = torch.nn.Parameter(torch.randn(batch_size, seq_length, n_heads, d_state))
        self.C = torch.nn.Parameter(torch.randn(batch_size, seq_length, n_heads, d_state))

    def segsum(self, x):
        T = x.size(-1)
        x_cumsum = torch.cumsum(x, dim=-1)
        x_segsum = x_cumsum[..., :, None] - x_cumsum[..., None, :]
        mask = torch.tril(torch.ones(T, T, device=x.device, dtype=torch.bool), diagonal=0)
        x_segsum = x_segsum.masked_fill(~mask, -torch.inf)
        return x_segsum

    def forward(self, X, initial_states=None):
        X_blocks, A_blocks, B_blocks, C_blocks = [
            rearrange(x, "b (c l) ... -> b c l ...", l=self.block_len)
            for x in (X, self.A, self.B, self.C)
        ]

        A_blocks = rearrange(A_blocks, "b c l h -> b h c l")
        A_cumsum = torch.cumsum(A_blocks, dim=-1)

        L = torch.exp(self.segsum(A_blocks))
        Y_diag = torch.einsum("bclhn,bcshn,bhcls,bcshp->bclhp",
                              C_blocks, B_blocks, L, X_blocks)

        decay_states = torch.exp((A_cumsum[:, :, :, -1:] - A_cumsum))
        states = torch.einsum("bclhn,bhcl,bclhp->bchpn",
                              B_blocks, decay_states, X_blocks)

        if initial_states is None:
            initial_states = torch.zeros_like(states[:, :1])
        states = torch.cat([initial_states, states], dim=1)

        decay_chunk = torch.exp(self.segsum(F.pad(A_cumsum[:, :, :, -1], (1, 0))))
        new_states = torch.einsum("bhzc,bchpn->bzhpn", decay_chunk, states)

        # Use custom kernel to take last z slice efficiently.
        return custom_ops_lib.mamba_return_final_state_custom(new_states)
'''
