kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matmul_add_swish_tanh_gelu_hardtanh_custom_tiling_data.h"

using DTYPE = float;

constexpr int32_t BUFFER_NUM = 2; // 双缓冲

class KernelMatmulAddSwishTanhGeluHardtanh {
public:
    __aicore__ inline KernelMatmulAddSwishTanhGeluHardtanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR addv, GM_ADDR y, uint32_t totalLength, uint32_t tileNum, uint32_t vecLen)
    {
        // 将数据按行处理：每个tile处理一整行（长度=vecLen）
        this->vecLen = vecLen;
        this->tileNum = tileNum;
        // 一个block负责的总元素数量
        this->blockLength = totalLength / AscendC::GetBlockNum();
        // 每个tile处理一整行
        this->tileLength = this->vecLen;

        xGm.SetGlobalBuffer((__gm__ DTYPE *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        addvGm.SetGlobalBuffer((__gm__ DTYPE *)addv, this->vecLen);
        yGm.SetGlobalBuffer((__gm__ DTYPE *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(inQueueAddv, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE));

        // 计算当前block内的行数（每行vecLen元素）
        this->rowsPerBlock = this->blockLength / this->vecLen;
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->rowsPerBlock; // 每次一行
        // 双缓冲流水
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute();
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t rowIdx)
    {
        // x一行
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);

        // addv整向量（按行广播）
        AscendC::LocalTensor<DTYPE> addvLocal = inQueueAddv.AllocTensor<DTYPE>();
        AscendC::DataCopy(addvLocal, addvGm[0], this->tileLength);
        inQueueAddv.EnQue(addvLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> addvLocal = inQueueAddv.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.AllocTensor<DTYPE>();

        // y = x + addv（广播）
        AscendC::Add(yLocal, xLocal, addvLocal, this->tileLength);

        // Swish: sigmoid(y) * y
        AscendC::LocalTensor<DTYPE> tmpLocal = pipe.AllocTensor<AscendC::TPosition::VECCALC, DTYPE>(this->tileLength);
        AscendC::Sigmoid(tmpLocal, yLocal, this->tileLength);
        AscendC::Mul(yLocal, yLocal, tmpLocal, this->tileLength);

        // Tanh
        AscendC::Tanh(yLocal, yLocal, this->tileLength);

        // GELU
        AscendC::Gelu(yLocal, yLocal, this->tileLength);

        // Hardtanh: clip到[-1, 1]
        AscendC::Maxs(yLocal, yLocal, (DTYPE)-1.0f, this->tileLength);
        AscendC::Mins(yLocal, yLocal, (DTYPE)1.0f, this->tileLength);

        outQueueY.EnQue<DTYPE>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueAddv.FreeTensor(addvLocal);
        pipe.FreeTensor(tmpLocal);
    }

    __aicore__ inline void CopyOut(int32_t rowIdx)
    {
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm[rowIdx * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueAddv;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> addvGm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t vecLen;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void matmul_add_swish_tanh_gelu_hardtanh_custom(GM_ADDR x, GM_ADDR addv, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulAddSwishTanhGeluHardtanh op;
    op.Init(x, addv, y, tiling_data.totalLength, tiling_data.tileNum, tiling_data.vecLen);
    op.Process();
}
