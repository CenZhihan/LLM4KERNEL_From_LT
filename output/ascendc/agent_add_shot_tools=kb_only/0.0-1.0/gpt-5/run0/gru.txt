kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "gru_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGru {
public:
    __aicore__ inline KernelGru() {}
    __aicore__ inline void Init(GM_ADDR xr, GM_ADDR xz, GM_ADDR xn,
                                GM_ADDR hr, GM_ADDR hz, GM_ADDR hn,
                                GM_ADDR hprev, GM_ADDR hout,
                                uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xrGm.SetGlobalBuffer((__gm__ float *)xr + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        xzGm.SetGlobalBuffer((__gm__ float *)xz + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        xnGm.SetGlobalBuffer((__gm__ float *)xn + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        hrGm.SetGlobalBuffer((__gm__ float *)hr + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        hzGm.SetGlobalBuffer((__gm__ float *)hz + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        hnGm.SetGlobalBuffer((__gm__ float *)hn + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        hprevGm.SetGlobalBuffer((__gm__ float *)hprev + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        houtGm.SetGlobalBuffer((__gm__ float *)hout + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueXR, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueXZ, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueXN, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueHR, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueHZ, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueHN, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueHPrev, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueHOut, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xrLocal = inQueueXR.AllocTensor<float>();
        AscendC::LocalTensor<float> xzLocal = inQueueXZ.AllocTensor<float>();
        AscendC::LocalTensor<float> xnLocal = inQueueXN.AllocTensor<float>();
        AscendC::LocalTensor<float> hrLocal = inQueueHR.AllocTensor<float>();
        AscendC::LocalTensor<float> hzLocal = inQueueHZ.AllocTensor<float>();
        AscendC::LocalTensor<float> hnLocal = inQueueHN.AllocTensor<float>();
        AscendC::LocalTensor<float> hprevLocal = inQueueHPrev.AllocTensor<float>();

        AscendC::DataCopy(xrLocal, xrGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(xzLocal, xzGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(xnLocal, xnGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(hrLocal, hrGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(hzLocal, hzGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(hnLocal, hnGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(hprevLocal, hprevGm[progress * this->tileLength], this->tileLength);

        inQueueXR.EnQue(xrLocal);
        inQueueXZ.EnQue(xzLocal);
        inQueueXN.EnQue(xnLocal);
        inQueueHR.EnQue(hrLocal);
        inQueueHZ.EnQue(hzLocal);
        inQueueHN.EnQue(hnLocal);
        inQueueHPrev.EnQue(hprevLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<float> xrLocal = inQueueXR.DeQue<float>();
        AscendC::LocalTensor<float> xzLocal = inQueueXZ.DeQue<float>();
        AscendC::LocalTensor<float> xnLocal = inQueueXN.DeQue<float>();
        AscendC::LocalTensor<float> hrLocal = inQueueHR.DeQue<float>();
        AscendC::LocalTensor<float> hzLocal = inQueueHZ.DeQue<float>();
        AscendC::LocalTensor<float> hnLocal = inQueueHN.DeQue<float>();
        AscendC::LocalTensor<float> hprevLocal = inQueueHPrev.DeQue<float>();
        AscendC::LocalTensor<float> houtLocal = outQueueHOut.AllocTensor<float>();

        // r = sigmoid(xr + hr) -> reuse xrLocal
        AscendC::Add(xrLocal, xrLocal, hrLocal, this->tileLength);
        AscendC::Sigmoid(xrLocal, xrLocal, this->tileLength);

        // z = sigmoid(xz + hz) -> reuse xzLocal
        AscendC::Add(xzLocal, xzLocal, hzLocal, this->tileLength);
        AscendC::Sigmoid(xzLocal, xzLocal, this->tileLength);

        // n = tanh(xn + r * hn) -> reuse xnLocal; compute hnLocal = r * hn
        AscendC::Mul(hnLocal, xrLocal, hnLocal, this->tileLength); // hnLocal = r * hn
        AscendC::Add(xnLocal, xnLocal, hnLocal, this->tileLength);
        AscendC::Tanh(xnLocal, xnLocal, this->tileLength); // xnLocal now is n

        // hout = n + z * (hprev - n)
        AscendC::Sub(hprevLocal, hprevLocal, xnLocal, this->tileLength); // hprevLocal = hprev - n
        AscendC::Mul(hprevLocal, xzLocal, hprevLocal, this->tileLength); // hprevLocal = z * (hprev - n)
        AscendC::Add(houtLocal, xnLocal, hprevLocal, this->tileLength);  // hout = n + z*(hprev - n)

        outQueueHOut.EnQue(houtLocal);

        inQueueXR.FreeTensor(xrLocal);
        inQueueXZ.FreeTensor(xzLocal);
        inQueueXN.FreeTensor(xnLocal);
        inQueueHR.FreeTensor(hrLocal);
        inQueueHZ.FreeTensor(hzLocal);
        inQueueHN.FreeTensor(hnLocal);
        inQueueHPrev.FreeTensor(hprevLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> houtLocal = outQueueHOut.DeQue<float>();
        AscendC::DataCopy(houtGm[progress * this->tileLength], houtLocal, this->tileLength);
        outQueueHOut.FreeTensor(houtLocal);
    }

private:
    AscendC::TPipe pipe;

    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueXR;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueXZ;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueXN;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueHR;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueHZ;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueHN;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueHPrev;

    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueHOut;

    AscendC::GlobalTensor<float> xrGm, xzGm, xnGm;
    AscendC::GlobalTensor<float> hrGm, hzGm, hnGm;
    AscendC::GlobalTensor<float> hprevGm;
    AscendC::GlobalTensor<float> houtGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void gru_custom(GM_ADDR xr, GM_ADDR xz, GM_ADDR xn,
                                                 GM_ADDR hr, GM_ADDR hz, GM_ADDR hn,
                                                 GM_ADDR hprev, GM_ADDR hout,
                                                 GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGru op;
    op.Init(xr, xz, xn, hr, hz, hn, hprev, hout, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, input_size, hidden_size, num_layers=3, bias=True, batch_first=False):
        super().__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.bias = bias
        self.batch_first = batch_first

        self.weight_ih = nn.ParameterList()
        self.weight_hh = nn.ParameterList()
        self.bias_ih = nn.ParameterList()
        self.bias_hh = nn.ParameterList()

        for _ in range(num_layers):
            w_ih = nn.Parameter(torch.empty(3 * hidden_size, input_size))
            w_hh = nn.Parameter(torch.empty(3 * hidden_size, hidden_size))
            nn.init.xavier_uniform_(w_ih)
            nn.init.orthogonal_(w_hh)
            self.weight_ih.append(w_ih)
            self.weight_hh.append(w_hh)
            if bias:
                b_ih = nn.Parameter(torch.zeros(3 * hidden_size))
                b_hh = nn.Parameter(torch.zeros(3 * hidden_size))
            else:
                b_ih = nn.Parameter(torch.zeros(3 * hidden_size), requires_grad=False)
                b_hh = nn.Parameter(torch.zeros(3 * hidden_size), requires_grad=False)
            self.bias_ih.append(b_ih)
            self.bias_hh.append(b_hh)
            input_size = hidden_size  # next layer input size

    def forward(self, x, h0):
        if self.batch_first:
            # (B, T, C) -> (T, B, C)
            x = x.transpose(0, 1)

        seq_len, batch_size, _ = x.shape
        h_prev_all = list(torch.unbind(h0, dim=0))
        layer_input = x
        for layer in range(self.num_layers):
            w_ih = self.weight_ih[layer]
            w_hh = self.weight_hh[layer]
            b_ih = self.bias_ih[layer]
            b_hh = self.bias_hh[layer]

            h_prev = h_prev_all[layer]
            outputs = []
            for t in range(seq_len):
                x_t = layer_input[t]  # (B, input_size or hidden_size)

                gate_x = F.linear(x_t, w_ih, b_ih)  # (B, 3H)
                gate_h = F.linear(h_prev, w_hh, b_hh)  # (B, 3H)

                xr, xz, xn = gate_x.chunk(3, dim=1)
                hr, hz, hn = gate_h.chunk(3, dim=1)

                # Custom fused gate update kernel
                h_t = custom_ops_lib.gru_custom(xr, xz, xn, hr, hz, hn, h_prev)

                outputs.append(h_t)
                h_prev = h_t

            layer_out = torch.stack(outputs, dim=0)  # (T, B, H)
            layer_input = layer_out  # next layer input
            h_prev_all[layer] = h_prev

        output = layer_input  # final layer output
        if self.batch_first:
            output = output.transpose(0, 1)
        return output
'''
