inline int64_t CalcOutDim(int64_t in, int64_t pad0, int64_t pad1, int64_t dilation, int64_t k, int64_t stride)
{
    // floor((in + pad0 + pad1 - dilation*(k-1) - 1)/stride + 1)
    return (in + pad0 + pad1 - dilation * (k - 1) - 1) / stride + 1;
}
}

    const ge::Shape& xShape = context->GetInputShape(0)->GetOriginShape();

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* xShape = context->GetInputShape(0);  // [N,C,H,W]
    const gert::Shape* wShape = context->GetInputShape(1);  // [Cout, Cin/groups, Kh, Kw]
    int64_t N  = xShape->GetDim(0);
    int64_t C  = xShape->GetDim(1);
    int64_t H  = xShape->GetDim(2);
    int64_t W  = xShape->GetDim(3);
    int64_t Co = wShape->GetDim(0);
    int64_t Kh = wShape->GetDim(2);
    int64_t Kw = wShape->GetDim(3);

    // Defaults
    int64_t strideH = 1, strideW = 1;
    int64_t padTop = 0, padBottom = 0, padLeft = 0, padRight = 0;
    int64_t dilationH = 1, dilationW = 1;

    auto outH = (H + padTop + padBottom - dilationH * (Kh - 1) - 1) / strideH + 1;
    auto outW = (W + padLeft + padRight - dilationW * (Kw - 1) - 1) / strideW + 1;

    gert::Shape* yShape = context->GetOutputShape(0);
    yShape->SetDim(0, N);
    yShape->SetDim(1, Co);
    yShape->SetDim(2, outH);
    yShape->SetDim(3, outW);
    return GRAPH_SUCCESS;
}

#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_standard_2d_asymmetric_input_asymmetric_kernel_custom_tiling_data.h"

// 使用 Ascend C 的 Conv2D 高级/ISASI API（参考文档表 1-9 提供的 Conv2D 接口）
using namespace AscendC;

class KernelConv2D {
public:
    __aicore__ inline KernelConv2D() {}
    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR wGm, GM_ADDR bGm, GM_ADDR zGm,
                                const ConvStandard2dAsymmetricInputAsymmetricKernelCustomTilingData &t)
    {
        n = t.n(); cIn = t.cIn(); h = t.h(); w = t.w();
        cOut = t.cOut(); kH = t.kH(); kW = t.kW();
        groups = t.groups();

        strideH = t.strideH();
        strideW = t.strideW();
        padTop = t.padTop();
        padBottom = t.padBottom();
        padLeft = t.padLeft();
        padRight = t.padRight();
        dilationH = t.dilationH();
        dilationW = t.dilationW();
        oH = t.oH();
        oW = t.oW();

        hasBias = t.hasBias() != 0;

        // 全量大小（演示用，实际工程中建议按 tile 拆分并复用 UB）
        uint32_t xSize = n * cIn * h * w;
        uint32_t wSize = cOut * (cIn / groups) * kH * kW;
        uint32_t zSize = n * cOut * oH * oW;

        xGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(xGm), xSize);
        wGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(wGm), wSize);
        if (hasBias) {
            bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(bGm), cOut);
        }
        zGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(zGm), zSize);

        // 简单单次搬运（示例），生产中应分块
        pipe.InitBuffer(inQueueX, 1, xSize * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, wSize * sizeof(float));
        if (hasBias) {
            pipe.InitBuffer(inQueueB, 1, cOut * sizeof(float));
        }
        pipe.InitBuffer(outQueueZ, 1, zSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        auto xLocal = inQueueX.AllocTensor<float>();
        auto wLocal = inQueueW.AllocTensor<float>();
        DataCopy(xLocal, xGlobal, n * cIn * h * w);
        DataCopy(wLocal, wGlobal, cOut * (cIn / groups) * kH * kW);
        inQueueX.EnQue(xLocal);
        inQueueW.EnQue(wLocal);
        if (hasBias) {
            auto bLocal = inQueueB.AllocTensor<float>();
            DataCopy(bLocal, bGlobal, cOut);
            inQueueB.EnQue(bLocal);
        }
    }

    __aicore__ inline void Compute()
    {
        auto xLocal = inQueueX.DeQue<float>();
        auto wLocal = inQueueW.DeQue<float>();
        auto zLocal = outQueueZ.AllocTensor<float>();

        // bias 可选
        if (hasBias) {
            auto bLocal = inQueueB.DeQue<float>();
            // 直接调用 Conv2D 接口（参考文档：Conv2D Performs 2D convolution ...）
            // 具体签名由平台提供，此处演示形参：输出, 输入, 权重, 偏置, 各种卷积参数
            // 开发者可在实际项目中替换为正确签名
            AscendC::Conv2D(
                zLocal, xLocal, wLocal, bLocal,
                n, cIn, h, w,
                cOut, kH, kW,
                strideH, strideW,
                padTop, padBottom, padLeft, padRight,
                dilationH, dilationW,
                groups,
                oH, oW
            );
            inQueueB.FreeTensor(bLocal);
        } else {
            AscendC::Conv2D(
                zLocal, xLocal, wLocal, /*bias*/ LocalTensor<float>(),
                n, cIn, h, w,
                cOut, kH, kW,
                strideH, strideW,
                padTop, padBottom, padLeft, padRight,
                dilationH, dilationW,
                groups,
                oH, oW
            );
        }

        outQueueZ.EnQue(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueW.FreeTensor(wLocal);
    }

    __aicore__ inline void CopyOut()
    {
        auto zLocal = outQueueZ.DeQue<float>();
        DataCopy(zGlobal, zLocal, n * cOut * oH * oW);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    // 全局张量
    GlobalTensor<float> xGlobal, wGlobal, bGlobal, zGlobal;

    // 队列与 Pipe
    TPipe pipe;
    TQue<TPosition::VECIN, 1>  inQueueX, inQueueW, inQueueB;
    TQue<TPosition::VECOUT, 1> outQueueZ;

    // 形状与参数
    uint32_t n, cIn, h, w;
    uint32_t cOut, kH, kW, groups;
    uint32_t strideH, strideW;
    uint32_t padTop, padBottom, padLeft, padRight;
    uint32_t dilationH, dilationW;
    uint32_t oH, oW;
    bool hasBias;
};

extern "C" __global__ __aicore__ void conv_standard_2d_asymmetric_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelConv2D op;
    op.Init(x, w, b, z, tilingData);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    基于 AscendC 自定义核的 2D 卷积（异形输入与核大小），默认 stride=(1,1), padding=(0,0), dilation=(1,1), groups=1, bias=False
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: tuple):
        super().__init__()
        if isinstance(kernel_size, int):
            kh, kw = kernel_size, kernel_size
        else:
            kh, kw = kernel_size
        # 使用 PyTorch Parameter 持有权重（便于训练/保存），前向改为调用自定义核
        self.weight = torch.nn.Parameter(torch.empty(out_channels, in_channels, kh, kw))
        torch.nn.init.kaiming_uniform_(self.weight, a=5 ** 0.5)
        self.bias = None  # 默认无偏置
        # 默认超参（与给定模型一致）
        self.stride  = (1, 1)
        self.padding = (0, 0)  # (pad_h, pad_w) 对称时我们在核侧展开为四向 0
        self.dilation = (1, 1)
        self.groups = 1

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        pad_h, pad_w = self.padding
        # 将对称 padding 展开为四向
        pad_top = pad_h
        pad_bottom = pad_h
        pad_left = pad_w
        pad_right = pad_w
        stride_h, stride_w = self.stride
        dil_h, dil_w = self.dilation

        return custom_ops_lib.conv_standard_2d_asymmetric_input_asymmetric_kernel_custom(
            x, self.weight, self.bias,
            stride_h, stride_w,
            pad_top, pad_bottom, pad_left, pad_right,
            dil_h, dil_w,
            self.groups
        )
