kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matmul_batch_norm_bias_add_divide_swish_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 1; // 每次处理一行（一个特征向量）

class KernelBnBiasDivSwish {
public:
    __aicore__ inline KernelBnBiasDivSwish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR runMean, GM_ADDR runVar,
                                GM_ADDR bias, GM_ADDR divide, GM_ADDR bn_eps, GM_ADDR y,
                                uint32_t totalLength, uint32_t featureSize, uint32_t tileNum)
    {
        this->featureSize = featureSize;                 // N
        this->blockLength = totalLength / AscendC::GetBlockNum(); // M * N / blocks
        this->tileNum = tileNum;                         // M / blocks
        this->tileLength = this->featureSize;            // 每行长度 N

        xGm.SetGlobalBuffer((__gm__ float*)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ float*)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, this->featureSize);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, this->featureSize);
        meanGm.SetGlobalBuffer((__gm__ float*)runMean, this->featureSize);
        varGm.SetGlobalBuffer((__gm__ float*)runVar, this->featureSize);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, 1);     // 标量或长度为1
        divideGm.SetGlobalBuffer((__gm__ float*)divide, 1); // 标量或长度为1
        epsGm.SetGlobalBuffer((__gm__ float*)bn_eps, 1);    // 标量

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));

        // 常量向量缓存（通道参数）
        pipe.InitBuffer(paramQueGamma, 1, this->featureSize * sizeof(float));
        pipe.InitBuffer(paramQueBeta, 1, this->featureSize * sizeof(float));
        pipe.InitBuffer(paramQueMean, 1, this->featureSize * sizeof(float));
        pipe.InitBuffer(paramQueVar, 1, this->featureSize * sizeof(float));

        // 复制一次通道向量
        localGamma = paramQueGamma.AllocTensor<float>();
        localBeta  = paramQueBeta.AllocTensor<float>();
        localMean  = paramQueMean.AllocTensor<float>();
        localVar   = paramQueVar.AllocTensor<float>();
        AscendC::DataCopy(localGamma, gammaGm[0], this->featureSize);
        AscendC::DataCopy(localBeta,  betaGm[0],  this->featureSize);
        AscendC::DataCopy(localMean,  meanGm[0],  this->featureSize);
        AscendC::DataCopy(localVar,   varGm[0],   this->featureSize);
        paramQueGamma.EnQue(localGamma);
        paramQueBeta.EnQue(localBeta);
        paramQueMean.EnQue(localMean);
        paramQueVar.EnQue(localVar);

        // 标量读入
        AscendC::LocalTensor<float> tmpScalar = scalarBuf.AllocTensor<float>();
        pipe.InitBuffer(scalarBuf, 1, 3 * sizeof(float)); // bias/div/eps 共 3 个标量
        // bias
        AscendC::DataCopy(tmpScalar, biasGm[0], 1);
        biasScalar = tmpScalar.GetPhyAddr();
        // divide
        AscendC::DataCopy(tmpScalar, divideGm[0], 1);
        divideScalar = tmpScalar.GetPhyAddr();
        // eps
        AscendC::DataCopy(tmpScalar, epsGm[0], 1);
        epsScalar = tmpScalar.GetPhyAddr();
    }

    __aicore__ inline void Process()
    {
        // tileNum 为当前 block 需要处理的行数
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // 取出通道向量
        AscendC::LocalTensor<float> gammaLocal = paramQueGamma.DeQue<float>();
        AscendC::LocalTensor<float> betaLocal  = paramQueBeta.DeQue<float>();
        AscendC::LocalTensor<float> meanLocal  = paramQueMean.DeQue<float>();
        AscendC::LocalTensor<float> varLocal   = paramQueVar.DeQue<float>();

        // y = ((x - mean) / sqrt(var + eps)) * gamma + beta
        AscendC::Sub(yLocal, xLocal, meanLocal, this->tileLength);      // y = x - mean
        // tmp = var + eps
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.AllocTensor<float>();
        AscendC::Add(tmpLocal, varLocal, *epsScalar, this->tileLength); // 按标量 eps 相加
        AscendC::Sqrt(tmpLocal, tmpLocal, this->tileLength);            // sqrt(var + eps)
        AscendC::Div(yLocal, yLocal, tmpLocal, this->tileLength);       // (x-mean)/sqrt(var+eps)
        AscendC::Mul(yLocal, yLocal, gammaLocal, this->tileLength);     // * gamma
        AscendC::Add(yLocal, yLocal, betaLocal, this->tileLength);      // + beta

        // + bias（标量）
        AscendC::Add(yLocal, yLocal, *biasScalar, this->tileLength);
        // / divide（标量）
        AscendC::Div(yLocal, yLocal, *divideScalar, this->tileLength);

        // Swish: y = y * sigmoid(y) = y / (1 + exp(-y))
        AscendC::Mul(tmpLocal, yLocal, -1.0f, this->tileLength);        // -y
        AscendC::Exp(tmpLocal, tmpLocal, this->tileLength);             // exp(-y)
        AscendC::Add(tmpLocal, tmpLocal, 1.0f, this->tileLength);       // 1 + exp(-y)
        AscendC::Div(yLocal, yLocal, tmpLocal, this->tileLength);       // y / (1 + exp(-y))

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);

        // 归还通道常量
        paramQueGamma.EnQue(gammaLocal);
        paramQueBeta.EnQue(betaLocal);
        paramQueMean.EnQue(meanLocal);
        paramQueVar.EnQue(varLocal);
        tmpBuf.FreeTensor(tmpLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;

    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM>  inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::TQue<AscendC::TPosition::VECIN, 1>  paramQueGamma;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  paramQueBeta;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  paramQueMean;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  paramQueVar;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  scalarBuf;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  tmpBuf;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> meanGm;
    AscendC::GlobalTensor<float> varGm;

    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> divideGm;
    AscendC::GlobalTensor<float> epsGm;

    AscendC::LocalTensor<float> localGamma;
    AscendC::LocalTensor<float> localBeta;
    AscendC::LocalTensor<float> localMean;
    AscendC::LocalTensor<float> localVar;

    float* biasScalar = nullptr;
    float* divideScalar = nullptr;
    float* epsScalar = nullptr;

    uint32_t featureSize;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void matmul_batch_norm_bias_add_divide_swish_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR running_mean, GM_ADDR running_var,
    GM_ADDR bias, GM_ADDR divide, GM_ADDR bn_eps,
    GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelBnBiasDivSwish op;
    op.Init(x, gamma, beta, running_mean, running_var, bias, divide, bn_eps, y,
            tilingData.totalLength, tilingData.featureSize, tilingData.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Fused kernel replaces: BatchNorm + bias-add + divide + Swish.
    Matmul (Linear) is kept as-is to produce the input of fused kernel.
    """
    def __init__(self, in_features, out_features, bn_eps=1e-5, bn_momentum=0.1, bias_shape=(1,), divide_value=1.0):
        super(ModelNew, self).__init__()
        self.matmul = torch.nn.Linear(in_features, out_features)  # same as original
        self.bn = torch.nn.BatchNorm1d(out_features, eps=bn_eps, momentum=bn_momentum)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.divide_value = divide_value

    def forward(self, x):
        # 1) Matmul (Linear)
        x = self.matmul(x)
        # 2) Fused: BatchNorm(inference stats) + bias-add + divide + Swish
        gamma = self.bn.weight
        beta = self.bn.bias
        running_mean = self.bn.running_mean
        running_var = self.bn.running_var
        bias_scalar = self.bias
        divide = torch.as_tensor([self.divide_value], device=x.device, dtype=x.dtype)
        bn_eps = torch.as_tensor([self.bn.eps], device=x.device, dtype=x.dtype)
        x = custom_ops_lib.matmul_batch_norm_bias_add_divide_swish_custom(
            x, gamma, beta, running_mean, running_var, bias_scalar, divide, bn_eps
        )
        return x
