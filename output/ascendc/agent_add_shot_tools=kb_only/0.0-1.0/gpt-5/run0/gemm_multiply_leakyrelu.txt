kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "gemm_multiply_leakyrelu_custom_tiling_data.h"

// 数据类型设定
using DTYPE = float;

constexpr int32_t BUFFER_NUM = 2; // 每个队列的张量数量

class KernelMulLeakyRelu {
public:
    __aicore__ inline KernelMulLeakyRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum,
                                float multiplier, float negativeSlope)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->multiplier = multiplier;
        this->negativeSlope = negativeSlope;

        xGm.SetGlobalBuffer((__gm__ DTYPE *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(tmpQueue, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> tmpLocal = tmpQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.AllocTensor<DTYPE>();

        // 先进行标量乘法: tmp = x * multiplier
        AscendC::Muls(tmpLocal, xLocal, (DTYPE)multiplier, this->tileLength);

        // 再进行LeakyReLU: y = LeakyRelu(tmp, negativeSlope)
        AscendC::LeakyRelu(yLocal, tmpLocal, (DTYPE)negativeSlope, this->tileLength);

        outQueueY.EnQue<DTYPE>(yLocal);
        tmpQueue.FreeTensor(tmpLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    float multiplier;
    float negativeSlope;
};

extern "C" __global__ __aicore__ void gemm_multiply_leakyrelu_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMulLeakyRelu op;
    op.Init(x, y,
            tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.multiplier, tiling_data.negativeSlope);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Replace PyTorch multiply + LeakyReLU with custom AscendC fused kernel.
    GEMM remains as nn.Linear to produce input for the fused kernel.
    """
    def __init__(self, in_features, out_features, multiplier, negative_slope):
        super(ModelNew, self).__init__()
        self.gemm = torch.nn.Linear(in_features, out_features)
        self.multiplier = float(multiplier)
        self.negative_slope = float(negative_slope)

    def forward(self, x):
        x = self.gemm(x)  # GEMM
        # Fused: x * multiplier + LeakyReLU
        x = custom_ops_lib.gemm_multiply_leakyrelu_custom(x, self.multiplier, self.negative_slope)
        return x
'''
