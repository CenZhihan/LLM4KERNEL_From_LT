host_tiling_src="""
#include <cstring>
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace optiling {

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // Read input shapes (assume square matrices N x N)
    const gert::Shape* aShape = context->GetInputShape(0);
    const gert::Shape* bShape = context->GetInputShape(1);
    if (aShape->GetOriginShape().GetDimNum() != 2 || bShape->GetOriginShape().GetDimNum() != 2) {
        return ge::GRAPH_FAILED;
    }
    int64_t M = aShape->GetOriginShape().GetDim(0);
    int64_t K = aShape->GetOriginShape().GetDim(1);
    int64_t K2 = bShape->GetOriginShape().GetDim(0);
    int64_t N = bShape->GetOriginShape().GetDim(1);
    if (K != K2) {
        return ge::GRAPH_FAILED;
    }

    // Build tiling using Matmul tiling API
    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    matmul_tiling::MultiCoreMatmulTiling tiling(ascendcPlatform);
    tiling.SetDim(1); // default, will be updated by tiling result
    tiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetBiasType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    // Shapes
    tiling.SetShape((uint32_t)M, (uint32_t)N, (uint32_t)K);
    tiling.SetSingleShape((uint32_t)M, (uint32_t)N, (uint32_t)K);
    tiling.SetOrgShape((uint32_t)M, (uint32_t)N, (uint32_t)K);
    tiling.SetBias(false);
    tiling.SetBufferSpace(-1, -1, -1);

    optiling::TCubeTiling tilingData;
    int ret = tiling.GetTiling(tilingData);
    if (ret == -1) {
        return ge::GRAPH_FAILED;
    }

    // Set block dim suggested by tiling
    context->SetBlockDim(tiling.GetCoreNum());

    // Serialize TCubeTiling directly to raw buffer
    auto raw = context->GetRawTilingData();
    if (raw->GetCapacity() < sizeof(optiling::TCubeTiling)) {
        return ge::GRAPH_FAILED;
    }
    std::memcpy(raw->GetData(), &tilingData, sizeof(optiling::TCubeTiling));
    raw->SetDataSize(sizeof(optiling::TCubeTiling));

    // No extra workspace
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    return ge::GRAPH_SUCCESS;
}

} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* aShape = context->GetInputShape(0);
    const gert::Shape* bShape = context->GetInputShape(1);
    if (aShape->GetOriginShape().GetDimNum() != 2 || bShape->GetOriginShape().GetDimNum() != 2) {
        return GRAPH_FAILED;
    }
    // Matmul [M,K] x [K,N] -> [M,N]
    int64_t M = aShape->GetOriginShape().GetDim(0);
    int64_t N = bShape->GetOriginShape().GetDim(1);
    gert::Shape* outShape = context->GetOutputShape(0);
    outShape->GetOriginShape().SetDimNum(2);
    outShape->GetOriginShape().SetDim(0, M);
    outShape->GetOriginShape().SetDim(1, N);
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return GRAPH_SUCCESS;
}
} // namespace ge

namespace ops {
class MatmulForUpperTriangularMatricesCustom : public OpDef {
public:
    explicit MatmulForUpperTriangularMatricesCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("z")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);

        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};
OP_ADD(MatmulForUpperTriangularMatricesCustom);
} // namespace ops

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

extern "C" __global__ __aicore__ void matmul_for_upper_triangular_matrices_custom(GM_ADDR aGM, GM_ADDR bGM, GM_ADDR cGM, GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    // Load TCubeTiling data from GM
    optiling::TCubeTiling tiling;
    auto src = reinterpret_cast<__gm__ uint32_t*>(tilingGM);
    auto dst = reinterpret_cast<uint32_t*>(&tiling);
    for (uint32_t i = 0; i < sizeof(optiling::TCubeTiling) / sizeof(uint32_t); ++i) {
        dst[i] = src[i];
    }

    // Define Matmul types: A/B/C are float ND in GM
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NONE> aType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NONE> bType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NONE> cType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float> biasType;

    // Init GM tensors (size params are not strictly used by the MM runtime here)
    AscendC::GlobalTensor<float> aGlobal;
    AscendC::GlobalTensor<float> bGlobal;
    AscendC::GlobalTensor<float> cGlobal;
    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(aGM), 0);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(bGM), 0);
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(cGM), 0);

    // Create and register Matmul instance
    matmul::Matmul<aType, bType, cType, biasType> mm;
    AscendC::TPipe pipe;
    g_cubeTPipePtr = &pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm);
    mm.Init(&tiling);

    // Bind inputs; no transpose, no bias
    mm.SetTensorA(aGlobal[0], false);
    mm.SetTensorB(bGlobal[0], false);
    mm.SetWorkspace(workspaceGM, 0);

    // Single compute (non-batch)
    // Use IterateNBatch with single loop for compatibility
    const uint32_t batchLoop = 1;
    const uint32_t batchA = 1;
    const uint32_t batchB = 1;
    const bool enSequentialWrite = false;
    mm.IterateNBatch(batchLoop, batchA, batchB, enSequentialWrite);
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"

at::Tensor matmul_for_upper_triangular_matrices_custom_impl_npu(const at::Tensor& a, const at::Tensor& b) {
    TORCH_CHECK(a.dim() == 2 && b.dim() == 2, "Inputs must be 2D matrices.");
    TORCH_CHECK(a.size(1) == b.size(0), "Incompatible matmul shapes.");
    at::Tensor result = at::empty({a.size(0), b.size(1)}, a.options());
    EXEC_NPU_CMD(aclnnMatmulForUpperTriangularMatricesCustom, a, b, result);
    return result;
}
