host_operator_src="""
#include "max_broadcast_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace {
static inline int64_t GetDimSafe(const gert::Shape& s, int idx) {
    if (s.GetDimNum() <= idx) return 1;
    return s.GetDim(idx);
}
static inline int64_t BroadcastDim(int64_t a, int64_t b) {
    return a == b ? a : (a == 1 ? b : (b == 1 ? a : -1));
}
}

namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 读取输入形状
    const gert::Shape& aShape = context->GetInputShape(0)->GetOriginShape();
    const gert::Shape& bShape = context->GetInputShape(1)->GetOriginShape();

    int64_t aN = GetDimSafe(aShape, 0);
    int64_t aC = GetDimSafe(aShape, 1);
    int64_t aK = GetDimSafe(aShape, 2);

    int64_t bN = GetDimSafe(bShape, 0);
    int64_t bC = GetDimSafe(bShape, 1);
    int64_t bK = GetDimSafe(bShape, 2);

    int64_t N = BroadcastDim(aN, bN);
    int64_t C = BroadcastDim(aC, bC);
    int64_t K = BroadcastDim(aK, bK);
    if (N <= 0 || C <= 0 || K <= 0) {
        return ge::GRAPH_FAILED; // 非法广播
    }

    // 简单设置block维度
    const uint32_t BLOCK_DIM = 8;
    context->SetBlockDim(BLOCK_DIM);

    // 简单沿最后一维K做分块
    uint32_t tileK = 256;
    if (K < tileK) tileK = static_cast<uint32_t>(K);

    MaxBroadcastCustomTilingData tiling;
    tiling.set_n(static_cast<uint32_t>(N));
    tiling.set_c(static_cast<uint32_t>(C));
    tiling.set_k(static_cast<uint32_t>(K));
    tiling.set_tileK(tileK);

    // 保存tiling数据与workspace
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    return ge::GRAPH_SUCCESS;
}
} // namespace optiling


namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape& a = context->GetInputShape(0)->GetOriginShape();
    const gert::Shape& b = context->GetInputShape(1)->GetOriginShape();

    int64_t aN = (a.GetDimNum() > 0) ? a.GetDim(0) : 1;
    int64_t aC = (a.GetDimNum() > 1) ? a.GetDim(1) : 1;
    int64_t aK = (a.GetDimNum() > 2) ? a.GetDim(2) : 1;

    int64_t bN = (b.GetDimNum() > 0) ? b.GetDim(0) : 1;
    int64_t bC = (b.GetDimNum() > 1) ? b.GetDim(1) : 1;
    int64_t bK = (b.GetDimNum() > 2) ? b.GetDim(2) : 1;

    auto f = [](int64_t u, int64_t v)->int64_t {
        if (u == v) return u;
        if (u == 1) return v;
        if (v == 1) return u;
        return -1;
    };

    int64_t N = f(aN, bN);
    int64_t C = f(aC, bC);
    int64_t K = f(aK, bK);
    if (N <= 0 || C <= 0 || K <= 0) {
        return GRAPH_FAILED;
    }

    gert::Shape* yShape = context->GetOutputShape(0);
    std::vector<int64_t> out = {N, C, K};
    yShape->SetDimNum(out.size());
    for (size_t i = 0; i < out.size(); ++i) {
        yShape->SetDim(i, out[i]);
    }
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "max_broadcast_custom_tiling_data.h"

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 1;

class KernelMaxBroadcast {
public:
    __aicore__ inline KernelMaxBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t n, uint32_t c, uint32_t k, uint32_t tileK)
    {
        this->n = n;
        this->c = c;
        this->k = k;
        this->tileK = tileK;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(x), n * k);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(y), c);
        zGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(z), n * c * k);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileK * sizeof(DTYPE));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileK * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        // 将N维度在不同block间拆分
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t nPerBlk = n / blockNum;
        uint32_t nBeg = blockIdx * nPerBlk;
        uint32_t nEnd = nBeg + nPerBlk;

        for (uint32_t i = nBeg; i < nEnd; ++i) {
            for (uint32_t j = 0; j < c; ++j) {
                for (uint32_t off = 0; off < k; off += tileK) {
                    uint32_t curr = AscendC::Min(tileK, k - off);

                    // Copy X tile: shape [k], slice of length curr
                    AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
                    AscendC::DataCopy(xLocal, xGm[i * k + off], curr);

                    // Load scalar y_j (broadcast on K)
                    DTYPE yScalar = yGm.GetValue(j);

                    // Compute max(xTile, yScalar)
                    AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.AllocTensor<DTYPE>();
                    // 使用标量版本算子（类似LayerNorm示例中的 Adds/Muls），对每个元素执行 max(x, yScalar)
                    AscendC::Maxs(zLocal, xLocal, yScalar, curr);

                    // Enqueue & write back
                    outQueueZ.EnQue<DTYPE>(zLocal);
                    inQueueX.FreeTensor(xLocal);

                    AscendC::LocalTensor<DTYPE> zLocalOut = outQueueZ.DeQue<DTYPE>();
                    AscendC::DataCopy(zGm[(i * c + j) * k + off], zLocalOut, curr);
                    outQueueZ.FreeTensor(zLocalOut);
                }
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> yGm;
    AscendC::GlobalTensor<DTYPE> zGm;
    uint32_t n;
    uint32_t c;
    uint32_t k;
    uint32_t tileK;
};

extern "C" __global__ __aicore__ void max_broadcast_custom(GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelMaxBroadcast op;
    op.Init(x, y, z, tilingData.n, tilingData.c, tilingData.k, tilingData.tileK);
    op.Process();
}
