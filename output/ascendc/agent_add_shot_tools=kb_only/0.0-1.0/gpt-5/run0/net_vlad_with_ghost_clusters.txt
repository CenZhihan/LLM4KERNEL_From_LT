kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "net_vlad_with_ghost_clusters_custom_tiling_data.h"

// 本内核实现 NetVLAD 的核心计算：
// 输入：
//   assignment: [B, N, KTotal]，已softmax，KTotal = K + G，内核仅取前K列
//   x         : [B, N, D]
//   clusters2 : [1, D, K]
// 输出：
//   vlad      : [B, D, K]，对应 vlad = (assignment^T @ x)^T - a，a = sum_N(assignment) * clusters2
// 说明：
// - 为了简洁与稳定，我们采用逐行/逐列分块+向量操作组合实现：
//   vlad_accum[K, D] = sum_{n=0..N-1} outer(assignment[n, :K], x[n, :D])
//   a_sum[K] = sum_{n=0..N-1} assignment[n, :K]
//   a[D, K]  = clusters2[D, K] .* a_sum[None, K]
//   最终输出转置为 [D, K] 并减去a。
// - 参考文档：
//   * 高阶API示例（Sum/More Examples）展示了队列、UB缓冲与LocalTensor用法。
//   * 本实现主要用到DataCopy、Add、Muls、Duplicate等常用向量API（见LayerNorm/More Examples中的使用风格）。

constexpr int32_t IN_Q_DEPTH  = 1;
constexpr int32_t OUT_Q_DEPTH = 1;
constexpr int32_t CALC_Q_DEPTH = 1;

class KernelNetVladWithGhostClusters {
public:
    __aicore__ inline KernelNetVladWithGhostClusters() {}

    __aicore__ inline void Init(GM_ADDR assignment, GM_ADDR x, GM_ADDR clusters2, GM_ADDR vlad,
                                uint32_t B, uint32_t N, uint32_t D, uint32_t K, uint32_t KTotal)
    {
        this->B = B;
        this->N = N;
        this->D = D;
        this->K = K;
        this->KTotal = KTotal;

        assignmentGm.SetGlobalBuffer((__gm__ DTYPE_X *)assignment, static_cast<uint64_t>(B) * N * KTotal);
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, static_cast<uint64_t>(B) * N * D);
        clusters2Gm.SetGlobalBuffer((__gm__ DTYPE_X *)clusters2, static_cast<uint64_t>(D) * K);
        vladGm.SetGlobalBuffer((__gm__ DTYPE_X *)vlad, static_cast<uint64_t>(B) * D * K);

        // 初始化队列与UB缓存
        pipe.InitBuffer(inAssRowQ, IN_Q_DEPTH, K * sizeof(DTYPE_X));
        pipe.InitBuffer(inXRowQ, IN_Q_DEPTH, D * sizeof(DTYPE_X));
        pipe.InitBuffer(inC2Q, IN_Q_DEPTH, D * K * sizeof(DTYPE_X));
        pipe.InitBuffer(outZRowQ, OUT_Q_DEPTH, K * sizeof(DTYPE_X));
        pipe.InitBuffer(calcQ, CALC_Q_DEPTH,
                        (K * D + K + D) * sizeof(DTYPE_X)); // vladAcc(K*D) + aSum(K) + temp(D)
    }

    __aicore__ inline void Process()
    {
        // 预取 clusters2 到UB
        AscendC::LocalTensor<DTYPE_X> clusters2Local = inC2Q.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(clusters2Local, clusters2Gm, static_cast<uint64_t>(D) * K);
        inC2Q.EnQue(clusters2Local);

        // 分批处理每个batch
        for (uint32_t b = 0; b < B; ++b) {
            ComputeOneBatch(b);
        }

        // 释放
        clusters2Local = inC2Q.DeQue<DTYPE_X>();
        inC2Q.FreeTensor(clusters2Local);
    }

private:
    __aicore__ inline void ComputeOneBatch(uint32_t b)
    {
        // 分配计算缓冲
        AscendC::LocalTensor<DTYPE_X> vladAcc = calcQ.AllocTensor<DTYPE_X>();        // 大小 K*D
        AscendC::LocalTensor<DTYPE_X> aSum    = calcQ.Get<DTYPE_X>(K * D);           // 大小 K
        AscendC::LocalTensor<DTYPE_X> tempD   = calcQ.Get<DTYPE_X>(K * D + K);       // 大小 D

        // 初始化 vladAcc = 0, aSum = 0
        DTYPE_X zero(0);
        AscendC::Duplicate<DTYPE_X>(vladAcc, zero, static_cast<uint64_t>(K) * D);
        AscendC::Duplicate<DTYPE_X>(aSum, zero, K);

        // 外层循环：沿N累加
        for (uint32_t n = 0; n < N; ++n) {
            // 读取 assignment[b, n, :K] 一行（前K列）
            AscendC::LocalTensor<DTYPE_X> aRow = inAssRowQ.AllocTensor<DTYPE_X>();
            {
                // GM起始下标
                uint64_t assOff = static_cast<uint64_t>(b) * N * KTotal + static_cast<uint64_t>(n) * KTotal;
                AscendC::DataCopy(aRow, assignmentGm[assOff], K);
            }
            inAssRowQ.EnQue(aRow);

            // 读取 x[b, n, :D] 一行
            AscendC::LocalTensor<DTYPE_X> xRow = inXRowQ.AllocTensor<DTYPE_X>();
            {
                uint64_t xOff = static_cast<uint64_t>(b) * N * D + static_cast<uint64_t>(n) * D;
                AscendC::DataCopy(xRow, xGm[xOff], D);
            }
            inXRowQ.EnQue(xRow);

            // a_sum 累积： aSum += aRow
            aRow = inAssRowQ.DeQue<DTYPE_X>();
            AscendC::Add(aSum, aSum, aRow, K);  // 向量加
            inAssRowQ.FreeTensor(aRow);

            // vlad_accum 累积：对于每个k，vladAcc[k, :] += aRow[k] * xRow[:]
            xRow = inXRowQ.DeQue<DTYPE_X>();
            for (uint32_t k = 0; k < K; ++k) {
                // tempD[:] = xRow[:] * aRow[k]
                // 为了获得 aRow[k]，复用一个小的读取：再次拉取该行（或在上面缓存）。此处简化起见再次读取。
                AscendC::LocalTensor<DTYPE_X> aRow2 = inAssRowQ.AllocTensor<DTYPE_X>();
                {
                    uint64_t assOff = static_cast<uint64_t>(b) * N * KTotal + static_cast<uint64_t>(n) * KTotal;
                    AscendC::DataCopy(aRow2, assignmentGm[assOff], K);
                }
                // 标量
                DTYPE_X scalar = aRow2.GetValue(k);
                inAssRowQ.FreeTensor(aRow2);

                AscendC::Muls<DTYPE_X>(tempD, xRow, scalar, D);

                // vladAcc[k, :] += tempD[:]
                AscendC::Add(vladAcc[k * D], vladAcc[k * D], tempD, D);
            }
            inXRowQ.FreeTensor(xRow);
        }

        // clusters2 本地缓存
        AscendC::LocalTensor<DTYPE_X> clusters2Local = inC2Q.DeQue<DTYPE_X>();
        inC2Q.EnQue(clusters2Local); // 复用

        // 输出：按d行写出 [D, K]，并做减法：vlad[d, k] = vladAcc[k, d] - clusters2[d, k] * aSum[k]
        for (uint32_t d = 0; d < D; ++d) {
            AscendC::LocalTensor<DTYPE_X> outRow = outZRowQ.AllocTensor<DTYPE_X>();
            for (uint32_t k = 0; k < K; ++k) {
                DTYPE_X valV = vladAcc[k * D + d];
                DTYPE_X c2   = clusters2Local[d + k * D];
                DTYPE_X as   = aSum[k];
                outRow.SetValue(k, valV - c2 * as);
            }
            outZRowQ.EnQue(outRow);
            outRow = outZRowQ.DeQue<DTYPE_X>();
            uint64_t zOff = static_cast<uint64_t>(b) * D * K + static_cast<uint64_t>(d) * K;
            AscendC::DataCopy(vladGm[zOff], outRow, K);
            outZRowQ.FreeTensor(outRow);
        }

        // 释放计算缓冲
        calcQ.FreeTensor(vladAcc);
    }

private:
    AscendC::TPipe pipe;

    AscendC::TQue<AscendC::TPosition::VECIN,  IN_Q_DEPTH>  inAssRowQ;
    AscendC::TQue<AscendC::TPosition::VECIN,  IN_Q_DEPTH>  inXRowQ;
    AscendC::TQue<AscendC::TPosition::VECIN,  IN_Q_DEPTH>  inC2Q;
    AscendC::TQue<AscendC::TPosition::VECOUT, OUT_Q_DEPTH> outZRowQ;
    AscendC::TQue<AscendC::TPosition::VECCALC, CALC_Q_DEPTH> calcQ;

    AscendC::GlobalTensor<DTYPE_X> assignmentGm;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_X> clusters2Gm;
    AscendC::GlobalTensor<DTYPE_X> vladGm;

    uint32_t B, N, D, K, KTotal;
};

extern "C" __global__ __aicore__ void net_vlad_with_ghost_clusters_custom(GM_ADDR assignment,
                                                                          GM_ADDR x,
                                                                          GM_ADDR clusters2,
                                                                          GM_ADDR vlad,
                                                                          GM_ADDR workspace,
                                                                          GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelNetVladWithGhostClusters op;
    op.Init(assignment, x, clusters2, vlad,
            tilingData.B(), tilingData.N(), tilingData.D(), tilingData.K(), tilingData.KTotal());
    op.Process();
}

model_src='''
import torch
import torch.nn.functional as F
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, cluster_size, feature_size, ghost_clusters):
        super().__init__()
        self.feature_size = feature_size
        self.cluster_size = cluster_size
        self.ghost_clusters = ghost_clusters

        import math
        init_sc = (1 / math.sqrt(feature_size))
        clusters = cluster_size + ghost_clusters

        # 与原始模型保持一致
        self.clusters = torch.nn.Parameter(init_sc * torch.randn(feature_size, clusters))
        self.batch_norm = torch.nn.BatchNorm1d(clusters)
        self.clusters2 = torch.nn.Parameter(init_sc * torch.randn(1, feature_size, cluster_size))
        self.out_dim = self.cluster_size * feature_size

    def forward(self, x, mask=None):
        # x: [B, N, D]
        B, N, D = x.shape
        # 1) 计算 assignment（含BN与Softmax）
        assignment = torch.matmul(x.view(-1, self.feature_size), self.clusters)         # [B*N, K+G]
        assignment = self.batch_norm(assignment)                                        # BN1d通道数=K+G
        assignment = F.softmax(assignment, dim=1)                                       # [B*N, K+G]
        assignment = assignment.view(B, N, -1)                                          # [B, N, K+G]

        # 2) 使用自定义AscendC内核计算NetVLAD核心：去ghost、累积、减去a
        #    输出 vlad: [B, D, K]
        vlad = custom_ops_lib.net_vlad_with_ghost_clusters_custom(assignment, x, self.clusters2)

        # 3) L2内归一化（沿D轴）
        vlad = F.normalize(vlad, dim=1)

        # 4) 拉平 + L2归一化
        vlad = vlad.reshape(B, self.cluster_size * self.feature_size)  # [B, D*K]
        vlad = F.normalize(vlad, dim=1)
        return vlad
'''
