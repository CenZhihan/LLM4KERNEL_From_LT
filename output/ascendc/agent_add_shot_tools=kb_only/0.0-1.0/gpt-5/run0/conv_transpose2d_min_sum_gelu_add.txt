kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose2d_min_sum_gelu_add_custom_tiling_data.h"

// 本算子输入: x (N,C,H,W), bias (1,1,1,W或可广播到W), 输出: z (N,1,1,W)
// 计算: 对x在通道维做min(keepdim=True)->(N,1,H,W)，再对H维求和->(N,1,1,W)，GELU，再与bias相加。

class KernelConvTranspose2dMinSumGeluAdd {
public:
    __aicore__ inline KernelConvTranspose2dMinSumGeluAdd() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                uint32_t N, uint32_t C, uint32_t H, uint32_t W, uint32_t tileNum)
    {
        this->N = N;
        this->C = C;
        this->H = H;
        this->W = W;
        this->tileNum = tileNum;

        uint32_t coreNum = AscendC::GetBlockNum();
        uint32_t coreId = AscendC::GetBlockIdx();

        // 以batch维按core均分，最后一核可能少一些
        uint32_t nPerCore = (N + coreNum - 1) / coreNum;
        startN = coreId * nPerCore;
        endN = (startN + nPerCore > N) ? N : (startN + nPerCore);

        xGm.SetGlobalBuffer((__gm__ float*)x + startN * (uint64_t)C * H * W, (endN - startN) * (uint64_t)C * H * W);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, W); // 期望bias为长度W的向量
        zGm.SetGlobalBuffer((__gm__ float*)z + startN * (uint64_t)W, (endN - startN) * (uint64_t)W);

        // 为每个向量计算分配本地缓冲
        pipe.InitBuffer(bufSum, 1, W * sizeof(float));
        pipe.InitBuffer(bufCur, 1, W * sizeof(float));
        pipe.InitBuffer(bufTmp, 1, W * sizeof(float));
        pipe.InitBuffer(bufBias, 1, W * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (startN >= endN) {
            return;
        }
        // 预取bias宽度向量
        AscendC::LocalTensor<float> biasLocal = bufBias.Get<float>();
        AscendC::DataCopy(biasLocal, biasGm, W);

        for (uint32_t n = 0; n < (endN - startN); ++n) {
            // sum向量: 累加所有H上min_c(x[n,:,h,:])
            AscendC::LocalTensor<float> sumLocal = bufSum.Get<float>();

            // 先用h=0初始化sumLocal
            MinOnChannelsTo(sumLocal, n, 0);

            for (uint32_t h = 1; h < H; ++h) {
                AscendC::LocalTensor<float> curLocal = bufCur.Get<float>();
                MinOnChannelsTo(curLocal, n, h);
                AscendC::Add(sumLocal, sumLocal, curLocal, W);
            }

            // GELU
            AscendC::Gelu(sumLocal, sumLocal, W);

            // + bias (宽度向量)
            AscendC::Add(sumLocal, sumLocal, biasLocal, W);

            // 写回 (N,1,1,W) => 线性(N*W)
            AscendC::DataCopy(zGm[n * (uint64_t)W], sumLocal, W);
        }
    }

private:
    // 计算: cur(h,:) = min_c x[n,c,h,:] 向量写入dstLocal
    __aicore__ inline void MinOnChannelsTo(AscendC::LocalTensor<float>& dstLocal, uint32_t nLocal, uint32_t h)
    {
        // 先拷入c=0
        uint64_t baseOffset = ((uint64_t)nLocal * C * H + (uint64_t)0 * H + h) * W;
        AscendC::DataCopy(dstLocal, xGm[baseOffset], W);

        // 依次与其它c做elementwise min
        for (uint32_t c = 1; c < C; ++c) {
            AscendC::LocalTensor<float> tmpLocal = bufTmp.Get<float>();
            uint64_t off = ((uint64_t)nLocal * C * H + (uint64_t)c * H + h) * W;
            AscendC::DataCopy(tmpLocal, xGm[off], W);
            AscendC::Min(dstLocal, dstLocal, tmpLocal, W);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufSum;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufCur;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufTmp;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufBias;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t N;
    uint32_t C;
    uint32_t H;
    uint32_t W;
    uint32_t tileNum;

    uint32_t startN;
    uint32_t endN;
};

extern "C" __global__ __aicore__ void conv_transpose2d_min_sum_gelu_add_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose2dMinSumGeluAdd op;
    op.Init(x, bias, z, tiling_data.N, tiling_data.C, tiling_data.H, tiling_data.W, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model: keep ConvTranspose2d in PyTorch, fuse Min(S=channel, keepdim=True) + Sum(H, keepdim=True) + GELU + Add(bias) into custom AscendC op.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, bias_shape):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride, padding, output_padding)
        # bias as parameter, will be expanded to match width before calling custom op
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        y = self.conv_transpose(x)  # (N, C, H, W)
        # custom op expects: x=y (N,C,H,W), bias shaped to width (1,1,1,W)
        bias_vec = self.bias.view(1, 1, 1, 1).expand(1, 1, 1, y.size(3)).contiguous()
        return custom_ops_lib.conv_transpose2d_min_sum_gelu_add_custom(y, bias_vec)
