namespace optiling {
// 依据文档：通过 MultiCoreMatmulTiling 获取 TCubeTiling 并原样下发
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 输入0: x [M, K], 输入1: weight [N, K] (PyTorch Linear 权重为 [N, K])
    auto xShape = context->GetInputShape(0)->GetOriginShape();
    auto wShape = context->GetInputShape(1)->GetOriginShape();
    if (xShape.GetDimNum() != 2 || wShape.GetDimNum() != 2) {
        return ge::GRAPH_PARAM_INVALID;
    }
    uint32_t M = static_cast<uint32_t>(xShape.GetDim(0));
    uint32_t K = static_cast<uint32_t>(xShape.GetDim(1));
    uint32_t N = static_cast<uint32_t>(wShape.GetDim(0)); // weight: [N, K]

    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    matmul_tiling::MultiCoreMatmulTiling tiling(ascendcPlatform);
    tiling.SetDim(1);
    tiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetBiasType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    // 依据文档 SetShape/SetSingleShape/SetOrgShape
    tiling.SetShape(M, N, K);
    tiling.SetSingleShape(M, N, K);
    tiling.SetOrgShape(M, N, K);
    tiling.SetBias(true);
    tiling.SetBufferSpace(-1, -1, -1);

    optiling::TCubeTiling cubeTiling;
    int ret = tiling.GetTiling(cubeTiling);
    if (ret == -1) {
        return ge::GRAPH_FAILED;
    }

    // 设置 blockDim
    int32_t blockDim = tiling.GetCoreNum();
    if (blockDim <= 0) { blockDim = 1; }
    context->SetBlockDim(static_cast<uint32_t>(blockDim));

    // 将 TCubeTiling 的原始字节下发
    auto raw = context->GetRawTilingData();
    if (raw->GetCapacity() < sizeof(optiling::TCubeTiling)) {
        return ge::GRAPH_FAILED;
    }
    std::memcpy(raw->GetData(), &cubeTiling, sizeof(optiling::TCubeTiling));
    raw->SetDataSize(sizeof(optiling::TCubeTiling));

    // 无需工作空间
    size_t *workspace = context->GetWorkspaceSizes(1);
    workspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    // 输出形状为 [M, N]
    const gert::Shape* x_shape = context->GetInputShape(0);   // [M, K]
    const gert::Shape* w_shape = context->GetInputShape(1);   // [N, K]
    if (x_shape->GetDimNum() != 2 || w_shape->GetDimNum() != 2) {
        return GRAPH_PARAM_INVALID;
    }
    gert::Shape* y_shape = context->GetOutputShape(0);
    std::vector<int64_t> dims(2);
    dims[0] = x_shape->GetDim(0);
    dims[1] = w_shape->GetDim(0);
    y_shape->SetDim(dims);
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto dt = context->GetInputDataType(0);
    context->SetOutputDataType(0, dt);
    return ge::GRAPH_SUCCESS;
}
} // namespace ge

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using A_T = float;
using B_T = float;
using C_T = float;
using BiasT = float;

// 依据文档：使用 LayoutMode::NORMAL 的 Matmul 高级 API
// B 设为转置以适配 PyTorch Linear 的权重布局 [N, K]
typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::NORMAL> aType;
typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, B_T, true,  LayoutMode::NORMAL> bType;
typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::NORMAL> cType;
typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, BiasT> biasType;

extern "C" __global__ __aicore__ void matmul_gelu_softmax_custom(GM_ADDR xGM, GM_ADDR wGM, GM_ADDR bGM, GM_ADDR yGM, GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    // 从 GM 读取 TCubeTiling（参考文档示例）
    TCubeTiling tiling;
    auto tempTilingGM = (__gm__ uint32_t*)tilingGM;
    auto tempTiling = (uint32_t*)&tiling;
    for (int i = 0; i < (int)(sizeof(TCubeTiling) / sizeof(int32_t)); ++i, ++tempTilingGM, ++tempTiling) {
        *tempTiling = *tempTilingGM;
    }

    // 初始化 GM Tensor（参考 NORMAL 示例）
    AscendC::GlobalTensor<A_T> aGlobal;
    AscendC::GlobalTensor<B_T> bGlobal;
    AscendC::GlobalTensor<C_T> cGlobal;
    AscendC::GlobalTensor<BiasT> biasGlobal;

    int32_t sizeA = tiling.ALayoutInfoB * tiling.singleCoreM * tiling.singleCoreK * sizeof(A_T);
    int32_t sizeB = tiling.BLayoutInfoB * tiling.singleCoreK * tiling.singleCoreN * sizeof(B_T);
    int32_t sizeC = tiling.CLayoutInfoB * tiling.singleCoreM * tiling.singleCoreN * sizeof(C_T);
    int32_t sizeBias = tiling.CLayoutInfoB * tiling.singleCoreN * sizeof(BiasT);

    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(xGM), sizeA);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(wGM), sizeB);
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(yGM), sizeC);
    biasGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ BiasT*>(bGM), sizeBias);

    // L1/L0C/UB 共享配置（与文档示例保持一致）
    tiling.shareMode = 0;
    tiling.shareL1Size = 512 * 1024;
    tiling.shareL0CSize = 128 * 1024;
    tiling.shareUbSize = 0;

    int offset_a = 0, offset_b = 0, offset_c = 0, offset_bias = 0;

    AscendC::GlobalTensor<A_T> gm_a;
    gm_a.SetGlobalBuffer(const_cast<__gm__ A_T*>(aGlobal[offset_a].GetPhyAddr()),
                         tiling.singleCoreM * tiling.singleCoreK);

    AscendC::GlobalTensor<B_T> gm_b;
    gm_b.SetGlobalBuffer(const_cast<__gm__ B_T*>(bGlobal[offset_b].GetPhyAddr()),
                         tiling.singleCoreK * tiling.singleCoreN);

    AscendC::GlobalTensor<C_T> gm_c;
    gm_c.SetGlobalBuffer(const_cast<__gm__ C_T*>(cGlobal[offset_c].GetPhyAddr()),
                         tiling.singleCoreM * tiling.singleCoreN);

    AscendC::GlobalTensor<BiasT> gm_bias;
    gm_bias.SetGlobalBuffer(const_cast<__gm__ BiasT*>(biasGlobal[offset_bias].GetPhyAddr()),
                            tiling.singleCoreN);

    // 创建 Matmul 实例并初始化（参考文档：REGIST_MATMUL_OBJ, Init, IterateBatch）
    constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);
    matmul::Matmul<aType, bType, cType, biasType, MM_CFG> mm1;
    AscendC::TPipe pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm1);
    mm1.Init(&tiling);

    // A 不转置，B 转置（PyTorch Linear 权重为 [N, K]）
    mm1.SetTensorA(gm_a, false);
    mm1.SetTensorB(gm_b, true);
    mm1.SetWorkspace(workspaceGM, 0);
    if (tiling.isBias) {
        mm1.SetBias(gm_bias);
    }

    // 单批次计算
    mm1.IterateBatch(gm_c, 1, 1, false);

    // 说明：本内核完成 MatMul(+Bias)。后续 GELU + Softmax 在上层 PyTorch 继续处理。
}

#include <torch/library.h>

// NPU 实现：调用自定义 AscendC 算子完成 MatMul(+Bias)
at::Tensor matmul_gelu_softmax_custom_impl_npu(const at::Tensor& x,
                                               const at::Tensor& weight,
                                               const at::Tensor& bias) {
    // 输出形状 [M, N] = [x.size(0), weight.size(0)]
    auto opts = x.options();
    at::Tensor result = at::empty({x.size(0), weight.size(0)}, opts);
    EXEC_NPU_CMD(aclnnMatmulGeluSoftmaxCustom, x, weight, bias, result);
    return result;
}

    m.impl("matmul_gelu_softmax_custom", &matmul_gelu_softmax_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("matmul_gelu_softmax_custom", &matmul_gelu_softmax_custom_impl_npu, "Matmul + Bias (GELU+Softmax on PyTorch)");

model_src='''
import torch
import torch_npu
import torch.nn.functional as F
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features):
        super(ModelNew, self).__init__()
        self.linear = torch.nn.Linear(in_features, out_features)

    def forward(self, x):
        # 使用自定义 AscendC 算子替换 Linear 的 MatMul(+Bias)
        y = custom_ops_lib.matmul_gelu_softmax_custom(x, self.linear.weight, self.linear.bias)
        # 其后在 PyTorch 中继续完成 GELU + Softmax
        y = F.gelu(y)
        y = F.softmax(y, dim=1)
        return y
'''
