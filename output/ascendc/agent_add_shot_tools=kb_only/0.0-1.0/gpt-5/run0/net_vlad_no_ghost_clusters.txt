    tiling.set_batch(B);
    tiling.set_numFeatures(N);
    tiling.set_featDim(D);
    tiling.set_numClusters(K);

    context->SetBlockDim(blockDim);
    tiling.set_blockDim(blockDim);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "net_vlad_no_ghost_clusters_custom_tiling_data.h"

// 说明：本核实现 vlad = assignment @ x - a_bias
// 输入：assignment [B, K, N], x [B, N, D], a_bias [B, K, D]
// 输出：vlad_bkd   [B, K, D]
// 计算：对每个 batch b、cluster k，计算 D 维向量：sum_n(assignment[b,k,n] * x[b,n,:]) - a_bias[b,k,:]
// 参考：两操作数模板及数据拷贝/加法 API（AscendC 基础 API 示例）

using DTYPE = float;

constexpr int32_t Q_DEPTH = 1;

class KernelNetVladNoGhostClusters {
public:
    __aicore__ inline KernelNetVladNoGhostClusters() {}

    __aicore__ inline void Init(GM_ADDR assignment, GM_ADDR x, GM_ADDR a_bias, GM_ADDR out,
                                uint32_t B, uint32_t N, uint32_t D, uint32_t K, uint32_t blockDim)
    {
        this->B_ = B;
        this->N_ = N;
        this->D_ = D;
        this->K_ = K;

        assignmentGm_.SetGlobalBuffer((__gm__ DTYPE*)assignment, (uint32_t)(B_ * K_ * N_));
        xGm_.SetGlobalBuffer((__gm__ DTYPE*)x, (uint32_t)(B_ * N_ * D_));
        aGm_.SetGlobalBuffer((__gm__ DTYPE*)a_bias, (uint32_t)(B_ * K_ * D_));
        outGm_.SetGlobalBuffer((__gm__ DTYPE*)out, (uint32_t)(B_ * K_ * D_));

        // 队列与缓冲区：行向量大小为 D
        pipe_.InitBuffer(inQXRow_, Q_DEPTH, D_ * sizeof(DTYPE));
        pipe_.InitBuffer(inQArow_, Q_DEPTH, D_ * sizeof(DTYPE));
        pipe_.InitBuffer(inQScalar_, Q_DEPTH, sizeof(DTYPE));
        pipe_.InitBuffer(outQRow_, Q_DEPTH, D_ * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        // 将 batch 维按 block 维切分
        for (uint32_t b = blockIdx; b < B_; b += blockNum) {
            // 遍历 K
            for (uint32_t k = 0; k < K_; ++k) {
                // 累加器向量 acc 存放在 outQRow_ 对应的 LocalTensor
                AscendC::LocalTensor<DTYPE> accLocal = outQRow_.AllocTensor<DTYPE>();
                bool firstAcc = true;

                // 遍历 N 做加权求和
                for (uint32_t n = 0; n < N_; ++n) {
                    // 载入 x[b, n, :]
                    AscendC::LocalTensor<DTYPE> xLocal = inQXRow_.AllocTensor<DTYPE>();
                    uint32_t xOff = b * (N_ * D_) + n * D_;
                    AscendC::DataCopy(xLocal, xGm_[xOff], D_);
                    inQXRow_.EnQue(xLocal);

                    // 载入 assignment[b, k, n] 标量
                    AscendC::LocalTensor<DTYPE> sLocal = inQScalar_.AllocTensor<DTYPE>();
                    uint32_t aOffScalar = b * (K_ * N_) + k * N_ + n;
                    AscendC::DataCopy(sLocal, assignmentGm_[aOffScalar], 1);
                    inQScalar_.EnQue(sLocal);

                    // 出队计算
                    AscendC::LocalTensor<DTYPE> xRow = inQXRow_.DeQue<DTYPE>();
                    AscendC::LocalTensor<DTYPE> sSca = inQScalar_.DeQue<DTYPE>();

                    // acc = sum_n ( assignment * xRow )
                    if (firstAcc) {
                        // acc = xRow * s
                        AscendC::Muls(accLocal, xRow, sSca.GetValue(0), D_);
                        firstAcc = false;
                    } else {
                        // tmp = xRow * s; acc += tmp
                        AscendC::LocalTensor<DTYPE> tmp = inQXRow_.AllocTensor<DTYPE>();
                        AscendC::DataCopy(tmp, xRow, D_);
                        AscendC::Muls(tmp, tmp, sSca.GetValue(0), D_);
                        AscendC::Add(accLocal, accLocal, tmp, D_);
                        inQXRow_.FreeTensor(tmp);
                    }

                    inQXRow_.FreeTensor(xRow);
                    inQScalar_.FreeTensor(sSca);
                }

                // 减去 a_bias[b, k, :]
                AscendC::LocalTensor<DTYPE> aRow = inQArow_.AllocTensor<DTYPE>();
                uint32_t aOff = b * (K_ * D_) + k * D_;
                AscendC::DataCopy(aRow, aGm_[aOff], D_);
                AscendC::Sub(accLocal, accLocal, aRow, D_);
                inQArow_.FreeTensor(aRow);

                // 写回 out[b, k, :]
                uint32_t outOff = b * (K_ * D_) + k * D_;
                AscendC::DataCopy(outGm_[outOff], accLocal, D_);

                outQRow_.FreeTensor(accLocal);
            }
        }
    }

private:
    AscendC::TPipe pipe_;
    AscendC::TQue<AscendC::QuePosition::VECIN, Q_DEPTH> inQXRow_;
    AscendC::TQue<AscendC::QuePosition::VECIN, Q_DEPTH> inQArow_;
    AscendC::TQue<AscendC::QuePosition::VECIN, Q_DEPTH> inQScalar_;
    AscendC::TQue<AscendC::QuePosition::VECOUT, Q_DEPTH> outQRow_;
    AscendC::GlobalTensor<DTYPE> assignmentGm_;
    AscendC::GlobalTensor<DTYPE> xGm_;
    AscendC::GlobalTensor<DTYPE> aGm_;
    AscendC::GlobalTensor<DTYPE> outGm_;

    uint32_t B_{0};
    uint32_t N_{0};
    uint32_t D_{0};
    uint32_t K_{0};
};

extern "C" __global__ __aicore__ void net_vlad_no_ghost_clusters_custom(GM_ADDR assignment, GM_ADDR x, GM_ADDR a_bias,
                                                                        GM_ADDR vlad_bkd, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelNetVladNoGhostClusters op;
    op.Init(assignment, x, a_bias, vlad_bkd,
            tilingData.batch, tilingData.numFeatures, tilingData.featDim, tilingData.numClusters, tilingData.blockDim);
    op.Process();
}

model_src='''
import math
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, cluster_size, feature_size, ghost_clusters):
        super(ModelNew, self).__init__()
        self.feature_size = feature_size
        self.cluster_size = cluster_size
        self.ghost_clusters = ghost_clusters

        init_sc = (1 / math.sqrt(feature_size))
        clusters = cluster_size + ghost_clusters

        self.clusters = nn.Parameter(init_sc * torch.randn(feature_size, clusters))
        self.batch_norm = nn.BatchNorm1d(clusters)
        self.clusters2 = nn.Parameter(init_sc * torch.randn(1, feature_size, cluster_size))
        self.out_dim = self.cluster_size * feature_size

    def forward(self, x, mask=None):
        # x: [B, N, D]
        B, N, D = x.shape
        x_flat = x.reshape(-1, self.feature_size)  # BN x D

        if x_flat.device != self.clusters.device:
            raise ValueError(f"x.device {x_flat.device} != cluster.device {self.clusters.device}")

        # logits: BN x (K+G)
        assignment = torch.matmul(x_flat, self.clusters)
        assignment = self.batch_norm(assignment)
        assignment = F.softmax(assignment, dim=1)  # BN x (K+G)

        # remove ghost clusters
        assignment = assignment[:, :self.cluster_size]
        assignment = assignment.view(B, N, self.cluster_size)  # B x N x K

        # a_sum: B x 1 x K, a: B x D x K
        a_sum = torch.sum(assignment, dim=1, keepdim=True)  # B x 1 x K
        a = a_sum * self.clusters2  # (B x 1 x K) * (1 x D x K) -> B x D x K

        # Use custom AscendC fused kernel to compute:
        # vlad_bkd = assignment^T @ x - a^T
        # Prepare inputs for custom op:
        assignment_bkn = assignment.transpose(1, 2).contiguous()  # B x K x N
        x_bnd = x.contiguous()                                    # B x N x D
        a_bkd = a.transpose(1, 2).contiguous()                    # B x K x D

        vlad_bkd = custom_ops_lib.net_vlad_no_ghost_clusters_custom(assignment_bkn, x_bnd, a_bkd)  # B x K x D
        vlad = vlad_bkd.transpose(1, 2).contiguous()  # -> B x D x K

        # L2 intra norm
        vlad = F.normalize(vlad, dim=1)

        # flatten + L2 norm
        vlad = vlad.reshape(B, self.cluster_size * self.feature_size)  # B x (D*K)
        vlad = F.normalize(vlad, dim=1)
        return vlad

def get_inputs():
  batch_size = 2048
  num_features = 100
  feature_size = 512
  return [torch.rand(batch_size, num_features, feature_size)]

def get_init_inputs():
  num_clusters = 32
  feature_size = 512
  ghost_clusters = 0
  return [num_clusters, feature_size, ghost_clusters]
'''
