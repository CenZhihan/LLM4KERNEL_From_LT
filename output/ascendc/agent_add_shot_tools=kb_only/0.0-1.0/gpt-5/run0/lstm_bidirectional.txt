kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "lstm_bidirectional_custom_tiling_data.h"

// 本算子功能：从输入张量 [B, S, D] 中抽取最后一个时间步 (S-1) 的特征，得到 [B, D]

constexpr int32_t BUFFER_NUM = 1;

class KernelLstmBidirectionalCustom {
public:
    __aicore__ inline KernelLstmBidirectionalCustom() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batch, uint32_t seqLen, uint32_t hidden2, uint32_t tileNum)
    {
        this->batch = batch;
        this->seqLen = seqLen;
        this->hidden2 = hidden2;
        this->tileNum = tileNum;

        // 设定全局内存句柄
        xGm.SetGlobalBuffer((__gm__ float*)x, (uint64_t)batch * seqLen * hidden2);
        yGm.SetGlobalBuffer((__gm__ float*)y, (uint64_t)batch * hidden2);

        // 每次处理一个 batch 的最后一步，局部缓存长度为 hidden2
        pipe.InitBuffer(inQueue, BUFFER_NUM, hidden2 * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, hidden2 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // 多个 block 按 batch 维度进行交错处理
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        for (uint32_t b = blockIdx; b < batch; b += blockNum) {
            CopyIn(b);
            Compute(b);
            CopyOut(b);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t b)
    {
        // 源偏移：b * S * D + (S - 1) * D
        uint64_t srcOffset = (uint64_t)b * seqLen * hidden2 + (uint64_t)(seqLen - 1) * hidden2;

        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[srcOffset], hidden2);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t /*b*/)
    {
        // 该算子仅为抽取操作，不进行计算，直接透传
        AscendC::LocalTensor<float> xLocal = inQueue.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueue.AllocTensor<float>();
        AscendC::DataCopy(yLocal, xLocal, hidden2);
        outQueue.EnQue(yLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t b)
    {
        // 目标偏移：b * D
        uint64_t dstOffset = (uint64_t)b * hidden2;

        AscendC::LocalTensor<float> yLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(yGm[dstOffset], yLocal, hidden2);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t batch;
    uint32_t seqLen;
    uint32_t hidden2;
    uint32_t tileNum;
};

extern "C" __global__ __aicore__ void lstm_bidirectional_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelLstmBidirectionalCustom op;
    op.Init(x, y,
            tiling_data.batch,
            tiling_data.seqLen,
            tiling_data.hidden2,
            tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_size, num_layers, output_size, dropout=0.0):
        super().__init__()
        self.lstm = torch.nn.LSTM(input_size, hidden_size, num_layers,
                                  batch_first=True, dropout=dropout, bidirectional=True)
        self.fc = torch.nn.Linear(hidden_size * 2, output_size)

    def forward(self, x, h0, c0):
        # 先使用原生 LSTM 获得输出序列
        out, _ = self.lstm(x, (h0, c0))  # out: [B, S, 2H]
        # 使用自定义 AscendC 算子高效抽取最后一个时间步的特征: [B, 2H]
        last = custom_ops_lib.lstm_bidirectional_custom(out)
        # 全连接映射到最终输出
        logits = self.fc(last)  # [B, O]
        return logits
