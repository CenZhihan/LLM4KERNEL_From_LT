#include <cmath>
#include "adam_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    AdamCustomTilingData tiling;

    // tensor length
    uint32_t totalLength = context->GetInputShape(0)->GetOriginShape().GetShapeSize();

    // read attributes
    const gert::RuntimeAttrs *attrs = context->GetAttrs();
    float beta1 = *(attrs->GetAttrPointer<float>(0));
    float beta2 = *(attrs->GetAttrPointer<float>(1));
    float lr    = *(attrs->GetAttrPointer<float>(2));
    float eps   = *(attrs->GetAttrPointer<float>(3));
    int64_t step = *(attrs->GetAttrPointer<int64_t>(4));

    // precompute constants for kernel
    float bc1 = 1.0f - std::pow(beta1, static_cast<float>(step));
    float bc2 = 1.0f - std::pow(beta2, static_cast<float>(step));
    float invBc1 = 1.0f / bc1;
    float invBc2 = 1.0f / bc2;
    float oneMinusB1 = 1.0f - beta1;
    float oneMinusB2 = 1.0f - beta2;

    context->SetBlockDim(BLOCK_DIM);

    tiling.set_totalLength(totalLength);
    tiling.set_tileNum(TILE_NUM);
    tiling.set_beta1(beta1);
    tiling.set_beta2(beta2);
    tiling.set_oneMinusBeta1(oneMinusB1);
    tiling.set_oneMinusBeta2(oneMinusB2);
    tiling.set_lr(lr);
    tiling.set_eps(eps);
    tiling.set_invBiasCorr1(invBc1);
    tiling.set_invBiasCorr2(invBc2);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(),
                        context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* in_shape = context->GetInputShape(0);
    gert::Shape* out_shape = context->GetOutputShape(0);
    *out_shape = *in_shape;
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "adam_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelAdam {
public:
    __aicore__ inline KernelAdam() {}
    __aicore__ inline void Init(GM_ADDR param, GM_ADDR grad, GM_ADDR m, GM_ADDR v, GM_ADDR out,
                                uint32_t totalLength, uint32_t tileNum,
                                float beta1, float beta2, float oneMinusB1, float oneMinusB2,
                                float lr, float eps, float invBc1, float invBc2)
    {
        ASSERT(AscendC::GetBlockNum() != 0 && "block dim can not be zero!");
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        ASSERT(tileNum != 0 && "tile num can not be zero!");
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        paramGm.SetGlobalBuffer((__gm__ DTYPE_PARAM *)param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        gradGm.SetGlobalBuffer((__gm__ DTYPE_GRAD  *)grad  + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        mGm.SetGlobalBuffer   ((__gm__ DTYPE_M     *)m     + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        vGm.SetGlobalBuffer   ((__gm__ DTYPE_V     *)v     + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outGm.SetGlobalBuffer ((__gm__ DTYPE_OUT   *)out   + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueParam, BUFFER_NUM, this->tileLength * sizeof(DTYPE_PARAM));
        pipe.InitBuffer(inQueueGrad,  BUFFER_NUM, this->tileLength * sizeof(DTYPE_GRAD));
        pipe.InitBuffer(inQueueM,     BUFFER_NUM, this->tileLength * sizeof(DTYPE_M));
        pipe.InitBuffer(inQueueV,     BUFFER_NUM, this->tileLength * sizeof(DTYPE_V));
        pipe.InitBuffer(outQueueOut,  BUFFER_NUM, this->tileLength * sizeof(DTYPE_OUT));
        pipe.InitBuffer(tmpQueue,     BUFFER_NUM, this->tileLength * sizeof(DTYPE_PARAM));

        this->beta1 = beta1;
        this->beta2 = beta2;
        this->oneMinusB1 = oneMinusB1;
        this->oneMinusB2 = oneMinusB2;
        this->lr = lr;
        this->eps = eps;
        this->invBc1 = invBc1;
        this->invBc2 = invBc2;
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_PARAM> paramLocal = inQueueParam.AllocTensor<DTYPE_PARAM>();
        AscendC::LocalTensor<DTYPE_GRAD>  gradLocal  = inQueueGrad.AllocTensor<DTYPE_GRAD>();
        AscendC::LocalTensor<DTYPE_M>     mLocal     = inQueueM.AllocTensor<DTYPE_M>();
        AscendC::LocalTensor<DTYPE_V>     vLocal     = inQueueV.AllocTensor<DTYPE_V>();

        AscendC::DataCopy(paramLocal, paramGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(gradLocal,  gradGm [progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(mLocal,     mGm    [progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(vLocal,     vGm    [progress * this->tileLength], this->tileLength);

        inQueueParam.EnQue(paramLocal);
        inQueueGrad.EnQue(gradLocal);
        inQueueM.EnQue(mLocal);
        inQueueV.EnQue(vLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_PARAM> paramLocal = inQueueParam.DeQue<DTYPE_PARAM>();
        AscendC::LocalTensor<DTYPE_GRAD>  gradLocal  = inQueueGrad.DeQue<DTYPE_GRAD>();
        AscendC::LocalTensor<DTYPE_M>     mLocal     = inQueueM.DeQue<DTYPE_M>();
        AscendC::LocalTensor<DTYPE_V>     vLocal     = inQueueV.DeQue<DTYPE_V>();

        AscendC::LocalTensor<DTYPE_OUT>   outLocal   = outQueueOut.AllocTensor<DTYPE_OUT>();
        AscendC::LocalTensor<DTYPE_PARAM> tmpLocal   = tmpQueue.AllocTensor<DTYPE_PARAM>();

        // m = beta1 * m + (1 - beta1) * grad
        AscendC::Muls(tmpLocal, mLocal, (DTYPE_PARAM)beta1, this->tileLength);                    // tmp = beta1*m
        AscendC::Muls(outLocal, gradLocal, (DTYPE_PARAM)oneMinusB1, this->tileLength);            // out = (1-b1)*grad
        AscendC::Add(mLocal, tmpLocal, outLocal, this->tileLength);                               // m = tmp + out

        // v = beta2 * v + (1 - beta2) * grad^2
        AscendC::Mul(outLocal, gradLocal, gradLocal, this->tileLength);                           // out = grad^2
        AscendC::Muls(outLocal, outLocal, (DTYPE_PARAM)oneMinusB2, this->tileLength);             // out = (1-b2)*grad^2
        AscendC::Muls(tmpLocal, vLocal, (DTYPE_PARAM)beta2, this->tileLength);                    // tmp = beta2*v
        AscendC::Add(vLocal, tmpLocal, outLocal, this->tileLength);                               // v = tmp + out

        // m_hat = m * invBc1
        AscendC::Muls(tmpLocal, mLocal, (DTYPE_PARAM)invBc1, this->tileLength);                   // tmp = m_hat

        // v_hat = v * invBc2
        AscendC::Muls(outLocal, vLocal, (DTYPE_PARAM)invBc2, this->tileLength);                   // out = v_hat
        AscendC::Sqrt(outLocal, outLocal, this->tileLength);                                      // out = sqrt(v_hat)
        AscendC::Adds(outLocal, outLocal, (DTYPE_PARAM)eps, this->tileLength);                    // out = sqrt(v_hat)+eps

        // step_vec = lr * (m_hat / (sqrt(v_hat)+eps))
        AscendC::Div(tmpLocal, tmpLocal, outLocal, this->tileLength);                             // tmp = m_hat / denom
        AscendC::Muls(tmpLocal, tmpLocal, (DTYPE_PARAM)lr, this->tileLength);                     // tmp = lr * ( ... )

        // param_new = param - step_vec
        AscendC::Sub(outLocal, paramLocal, tmpLocal, this->tileLength);

        outQueueOut.EnQue(outLocal);
        inQueueParam.FreeTensor(paramLocal);
        inQueueGrad.FreeTensor(gradLocal);
        inQueueM.FreeTensor(mLocal);
        inQueueV.FreeTensor(vLocal);
        tmpQueue.FreeTensor(tmpLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_OUT> outLocal = outQueueOut.DeQue<DTYPE_OUT>();
        AscendC::DataCopy(outGm[progress * this->tileLength], outLocal, this->tileLength);
        outQueueOut.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueParam, inQueueGrad, inQueueM, inQueueV;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueOut;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpQueue;

    AscendC::GlobalTensor<DTYPE_PARAM> paramGm;
    AscendC::GlobalTensor<DTYPE_GRAD>  gradGm;
    AscendC::GlobalTensor<DTYPE_M>     mGm;
    AscendC::GlobalTensor<DTYPE_V>     vGm;
    AscendC::GlobalTensor<DTYPE_OUT>   outGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;

    float beta1, beta2, oneMinusB1, oneMinusB2, lr, eps, invBc1, invBc2;
};

extern "C" __global__ __aicore__ void adam_custom(GM_ADDR param, GM_ADDR grad, GM_ADDR m, GM_ADDR v, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelAdam op;
    op.Init(param, grad, m, v, out,
            tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.beta1, tiling_data.beta2, tiling_data.oneMinusBeta1, tiling_data.oneMinusBeta2,
            tiling_data.lr, tiling_data.eps, tiling_data.invBiasCorr1, tiling_data.invBiasCorr2);
    op.Process();
}
