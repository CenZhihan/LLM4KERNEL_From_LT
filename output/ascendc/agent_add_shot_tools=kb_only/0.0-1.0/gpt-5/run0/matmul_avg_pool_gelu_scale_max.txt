kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matmul_avg_pool_gelu_scale_max_custom_tiling_data.h"
#include <cmath>

// 该内核实现模式：Y = ReduceMax_dim1( GELU( AvgPool1D( X @ W^T + b, ksize, stride=ksize ) ) * scale )
// 参考文档：
// - GET_TILING_DATA 用法（示例 5.4.1 LayerNorm）
// - REGISTER_TILING_DATA_CLASS（6.2.2）
// 注：为演示清晰性，以下实现采用简化的分块与 DataCopy 流程，侧重结构完整性。

class KernelMatmulAvgPoolGeluScaleMax {
public:
    __aicore__ inline KernelMatmulAvgPoolGeluScaleMax() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR scale, GM_ADDR ksize, GM_ADDR y,
                                uint32_t bLength, uint32_t inLength, uint32_t outLength, uint32_t /*tileNum*/)
    {
        this->bLength = bLength;
        this->inLength = inLength;
        this->outLength = outLength;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(x), static_cast<uint32_t>(bLength * inLength));
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(w), static_cast<uint32_t>(outLength * inLength));
        bGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(b), static_cast<uint32_t>(outLength));
        scaleGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(scale), 1);
        ksizeGm.SetGlobalBuffer(reinterpret_cast<__gm__ int32_t*>(ksize), 1);
        yGmPtr = reinterpret_cast<__gm__ float*>(y);

        // 队列与本地缓冲
        pipe.InitBuffer(inQueueX, 1, sizeof(float) * inLength);
        pipe.InitBuffer(inQueueW, 1, sizeof(float) * inLength);
        pipe.InitBuffer(tmpQueue, 1, sizeof(float) * 1); // 用于临时标量搬运
    }

    __aicore__ inline void Process()
    {
        // 读取 scale 与 ksize（标量）
        float scaleVal = 1.0f;
        {
            AscendC::LocalTensor<float> scaleLocal = tmpQueue.AllocTensor<float>();
            AscendC::DataCopy(scaleLocal, scaleGm, 1);
            tmpQueue.EnQue(scaleLocal);
            AscendC::LocalTensor<float> scaleLocalGet = tmpQueue.DeQue<float>();
            scaleVal = *(scaleLocalGet.GetData());
            tmpQueue.FreeTensor(scaleLocalGet);
        }

        int32_t ksizeVal = 1;
        {
            AscendC::LocalTensor<int32_t> kLocal = tmpQueue.AllocTensor<int32_t>();
            AscendC::DataCopy(kLocal, ksizeGm, 1);
            tmpQueue.EnQue(kLocal);
            AscendC::LocalTensor<int32_t> kLocalGet = tmpQueue.DeQue<int32_t>();
            ksizeVal = *(kLocalGet.GetData());
            tmpQueue.FreeTensor(kLocalGet);
        }
        if (ksizeVal <= 0) {
            ksizeVal = 1;
        }

        // 逐 batch 处理
        for (uint32_t bIdx = 0; bIdx < bLength; ++bIdx) {
            // 读入一行 x[b]
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[bIdx * inLength], inLength);
            inQueueX.EnQue(xLocal);
            AscendC::LocalTensor<float> xLocalGet = inQueueX.DeQue<float>();

            float maxVal = -3.4e38f; // 初始最小值
            const uint32_t windows = static_cast<uint32_t>(outLength / static_cast<uint32_t>(ksizeVal));
            for (uint32_t p = 0; p < windows; ++p) {
                float sumWindow = 0.f;
                // 累加 ksize 个输出位置的 matmul+bias
                for (int32_t t = 0; t < ksizeVal; ++t) {
                    uint32_t j = static_cast<uint32_t>(p * ksizeVal + t);
                    // 读取 W[j] 行（长度 inLength）
                    AscendC::LocalTensor<float> wRowLocal = inQueueW.AllocTensor<float>();
                    AscendC::DataCopy(wRowLocal, wGm[j * inLength], inLength);
                    inQueueW.EnQue(wRowLocal);
                    AscendC::LocalTensor<float> wRow = inQueueW.DeQue<float>();

                    // 读取 bias[j]
                    float bj = 0.f;
                    {
                        AscendC::LocalTensor<float> bLocal = tmpQueue.AllocTensor<float>();
                        AscendC::DataCopy(bLocal, bGm[j], 1);
                        tmpQueue.EnQue(bLocal);
                        AscendC::LocalTensor<float> bLocalGet = tmpQueue.DeQue<float>();
                        bj = *(bLocalGet.GetData());
                        tmpQueue.FreeTensor(bLocalGet);
                    }

                    // dot(xLocal, wRow) + b
                    float acc = bj;
                    const float* xPtr = xLocalGet.GetData();
                    const float* wPtr = wRow.GetData();
                    for (uint32_t k = 0; k < inLength; ++k) {
                        acc += xPtr[k] * wPtr[k];
                    }

                    sumWindow += acc;
                    inQueueW.FreeTensor(wRow);
                }

                // AvgPool: 求平均
                float avgVal = sumWindow / static_cast<float>(ksizeVal);

                // GELU 近似：0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
                float v = avgVal;
                float u = 0.7978845608028654f * (v + 0.044715f * v * v * v);
                float geluOut = 0.5f * v * (1.0f + tanhf(u));

                // scale
                float val = geluOut * scaleVal;

                // reduce max
                if (val > maxVal) {
                    maxVal = val;
                }
            }

            // 写回 y[bIdx]
            yGmPtr[bIdx] = maxVal;

            inQueueX.FreeTensor(xLocalGet);
        }
    }

private:
    // 管道与队列
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpQueue;

    // 全局张量
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> scaleGm;
    AscendC::GlobalTensor<int32_t> ksizeGm;
    __gm__ float* yGmPtr;

    // 维度信息
    uint32_t bLength;
    uint32_t inLength;
    uint32_t outLength;
};

extern "C" __global__ __aicore__ void matmul_avg_pool_gelu_scale_max_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR scale, GM_ADDR ksize,
    GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    // 从 tiling 结构中取出维度信息（参考 GET_TILING_DATA 用法）
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulAvgPoolGeluScaleMax op;
    op.Init(x, weight, bias, scale, ksize, y,
            tiling_data.bLength, tiling_data.inLength, tiling_data.outLength, tiling_data.tileNum);
    op.Process();
}

import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Fused: Matmul -> AvgPool1D -> GELU -> Scale -> ReduceMax(dim=1)
    """

        super().__init__()

        w = self.matmul.weight   # [out_features, in_features]
        b = self.matmul.bias     # [out_features]
        scale = torch.tensor(self.scale_factor, dtype=x.dtype, device=x.device)
        ksize = torch.tensor(self.pool_kernel_size, dtype=torch.int32, device=x.device)
