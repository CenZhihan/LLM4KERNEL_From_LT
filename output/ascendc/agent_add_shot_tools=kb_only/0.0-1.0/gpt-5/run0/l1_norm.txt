host_operator_src="""
#include "l1_norm_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    L1NormCustomTilingData tiling;

    const auto* inShape = context->GetInputShape(0);
    auto origin = inShape->GetOriginShape();
    uint32_t dims = origin.GetDimNum();
    if (dims < 1) {
        return ge::GRAPH_PARAM_INVALID;
    }
    // 兼容一维以上输入，按最后一维做L1归一化，前面维度视作batch
    uint32_t bLength = 1;
    for (uint32_t i = 0; i + 1 < dims; ++i) {
        bLength *= static_cast<uint32_t>(origin.GetDim(i));
    }
    uint32_t hLength = static_cast<uint32_t>(origin.GetDim(dims - 1));
    uint32_t totalLength = bLength * hLength;

    context->SetBlockDim(BLOCK_DIM);
    tiling.set_totalLength(totalLength);
    tiling.set_tileNum(TILE_NUM);
    tiling.set_bLength(bLength);
    tiling.set_hLength(hLength);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class L1NormCustom : public OpDef {
public:
    explicit L1NormCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};
OP_ADD(L1NormCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "l1_norm_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // 每个队列的张量数量

class KernelL1Norm {
public:
    __aicore__ inline KernelL1Norm() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum,
                                uint32_t bLength, uint32_t hLength)
    {
        this->bLength = bLength;
        this->hLength = hLength;
        this->tileNum = tileNum;

        // 每个core处理的行数
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        rowsPerBlock = (bLength + blockNum - 1) / blockNum;
        rowStart = blockIdx * rowsPerBlock;
        rowEnd = rowStart + rowsPerBlock;
        if (rowEnd > bLength) rowEnd = bLength;

        // 沿最后一维的分块长度
        this->tileCols = (hLength / tileNum);
        if (this->tileCols == 0) this->tileCols = hLength;
        // 初始化全局缓冲，偏移到本core负责的首行
        uint32_t offsetElems = rowStart * hLength;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x + offsetElems, (rowEnd - rowStart) * hLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y + offsetElems, (rowEnd - rowStart) * hLength);

        // 队列与UB缓冲
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileCols * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileCols * sizeof(DTYPE_Y));
        // 临时缓冲用于中间计算
        pipe.InitBuffer(tmpQueue, 1, this->tileCols * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t r = 0; r < (rowEnd - rowStart); ++r) {
            // Pass-1: 计算该行 mean(|x|)
            float sumVal = 0.0f;
            uint32_t colProcessed = 0;
            while (colProcessed < hLength) {
                uint32_t cur = AscendC::Min(this->tileCols, hLength - colProcessed);
                // Load tile
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal, xGm[r * hLength + colProcessed], cur);
                inQueueX.EnQue(xLocal);

                // abs and reduce
                AscendC::LocalTensor<DTYPE_X> xLocalDeq = inQueueX.DeQue<DTYPE_X>();
                AscendC::LocalTensor<DTYPE_X> tmpLocal = tmpQueue.AllocTensor<DTYPE_X>();
                AscendC::Abs(tmpLocal, xLocalDeq, cur);

                // 使用ReduceSum累加到累加寄存器，再读取结果
                AscendC::ReduceSum(tmpLocal, cur);
                float part = AscendC::GetAccVal<float>();
                sumVal += part;

                // free
                inQueueX.FreeTensor(xLocalDeq);
                tmpQueue.FreeTensor(tmpLocal);

                colProcessed += cur;
            }
            float meanVal = sumVal / static_cast<float>(hLength);
            // 防止除0
            float invMean = 1.0f / (meanVal + 1e-12f);

            // Pass-2: 输出 y = x * invMean
            colProcessed = 0;
            while (colProcessed < hLength) {
                uint32_t cur = AscendC::Min(this->tileCols, hLength - colProcessed);
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
                AscendC::DataCopy(xLocal, xGm[r * hLength + colProcessed], cur);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<DTYPE_X> xLocalDeq = inQueueX.DeQue<DTYPE_X>();
                // 逐元素乘以标量 invMean
                AscendC::Muls(yLocal, xLocalDeq, invMean, cur);

                outQueueY.EnQue<DTYPE_Y>(yLocal);
                inQueueX.FreeTensor(xLocalDeq);

                AscendC::LocalTensor<DTYPE_Y> yLocalDeq = outQueueY.DeQue<DTYPE_Y>();
                AscendC::DataCopy(yGm[r * hLength + colProcessed], yLocalDeq, cur);
                outQueueY.FreeTensor(yLocalDeq);

                colProcessed += cur;
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpQueue;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t bLength;
    uint32_t hLength;
    uint32_t tileNum;
    uint32_t tileCols;

    uint32_t rowsPerBlock;
    uint32_t rowStart;
    uint32_t rowEnd;
};

extern "C" __global__ __aicore__ void l1_norm_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelL1Norm op;
    op.Init(x, y,
            tiling_data.totalLength,
            tiling_data.tileNum,
            tiling_data.bLength,
            tiling_data.hLength);
    op.Process();
}
