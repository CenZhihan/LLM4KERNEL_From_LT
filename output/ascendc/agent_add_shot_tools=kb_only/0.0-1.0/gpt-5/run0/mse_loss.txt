static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    MseLossCustomTilingData tiling;

    // 将整个输入当作一维向量进行处理：outter=1, n=总元素数
    uint32_t outter = 1;
    uint32_t n = static_cast<uint32_t>(context->GetInputShape(0)->GetOriginShape().GetShapeSize());

    // dtype 大小
    auto dt = context->GetInputTensor(0)->GetDataType();
    uint32_t dtypesize = (dt == ge::DT_FLOAT16) ? 2 : 4;

    // 计算 inner 的 32B 对齐
    uint32_t inner = ((n * dtypesize + 32 - 1) / 32) * 32 / dtypesize;

    // 获取 Sum 所需的最小临时空间（与最大值一致）
    uint32_t maxValue = 0;
    uint32_t minValue = 0;
    AscendC::GetSumMaxMinTmpSize(n, dtypesize, false, maxValue, minValue);

    tiling.set_size(minValue);
    tiling.set_outter(outter);
    tiling.set_inner(inner);
    tiling.set_n(n);

    // 单核执行，便于一次性完成全量规约
    context->SetBlockDim(1);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "mse_loss_custom_tiling_data.h"

using namespace AscendC;

class KernelMseLoss {
public:
    __aicore__ inline KernelMseLoss() {}
    __aicore__ inline void Init(GM_ADDR predGm, GM_ADDR tgtGm, GM_ADDR lossGm, uint32_t tmpSize, uint32_t outter, uint32_t inner, uint32_t n)
    {
        this->height = outter; // 1
        this->width = inner;   // 32B 对齐后长度
        this->nNum = n;        // 实际元素数

        predGlobal.SetGlobalBuffer((__gm__ DTYPE_X *)predGm);
        tgtGlobal.SetGlobalBuffer((__gm__ DTYPE_Y *)tgtGm);
        lossGlobal.SetGlobalBuffer((__gm__ DTYPE_Z *)lossGm);

        // 输入缓冲：按 inner 对齐大小分配
        pipe.InitBuffer(inPredQ, 1, this->width * sizeof(DTYPE_X));
        pipe.InitBuffer(inTgtQ, 1, this->width * sizeof(DTYPE_Y));

        // 输出本地缓冲：Sum 输出为 outter 个元素，这里 outter=1；UB 内做 32B 对齐
        uint32_t outAlignedElems = (this->height * sizeof(DTYPE_Z) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(DTYPE_Z);
        pipe.InitBuffer(outLossQ, 1, outAlignedElems * sizeof(DTYPE_Z));

        // 临时工作空间
        pipe.InitBuffer(workQ, 1, tmpSize);
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        LocalTensor<DTYPE_X> predLocal = inPredQ.AllocTensor<DTYPE_X>();
        LocalTensor<DTYPE_Y> tgtLocal  = inTgtQ.AllocTensor<DTYPE_Y>();
        // 仅拷贝实际 n 个元素，剩余对齐部分留空
        DataCopy(predLocal, predGlobal, this->nNum);
        DataCopy(tgtLocal,  tgtGlobal,  this->nNum);
        inPredQ.EnQue(predLocal);
        inTgtQ.EnQue(tgtLocal);
    }

    __aicore__ inline void Compute()
    {
        LocalTensor<DTYPE_X> predLocal = inPredQ.DeQue<DTYPE_X>();
        LocalTensor<DTYPE_Y> tgtLocal  = inTgtQ.DeQue<DTYPE_Y>();
        LocalTensor<uint8_t> workLocal = workQ.AllocTensor<uint8_t>();
        LocalTensor<DTYPE_Z> lossLocal = outLossQ.AllocTensor<DTYPE_Z>();

        // 将输出对齐区清零（UB 内），便于 API 写入
        uint32_t outAlignedElems = (this->height * sizeof(DTYPE_Z) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(DTYPE_Z);
        DTYPE_Z zero(0);
        Duplicate<DTYPE_Z>(lossLocal, zero, outAlignedElems);

        // 计算 (pred - tgt)^2，复用 predLocal 作为中间缓冲
        Sub(predLocal, predLocal, tgtLocal, this->nNum);          // diff
        Mul(predLocal, predLocal, predLocal, this->nNum);         // diff^2

        // 对一维向量规约求和
        AscendC::SumParams params{this->height, this->width, this->nNum}; // outter=1, inner=width, n=nNum
        Sum(lossLocal, predLocal, workLocal, params);

        // 计算 mean：sum / n
        float scale = 1.0f / static_cast<float>(this->nNum);
        Muls(lossLocal, lossLocal, scale, 1); // 仅对第一个元素做缩放

        outLossQ.EnQue<DTYPE_Z>(lossLocal);
        workQ.FreeTensor(workLocal);
        inPredQ.FreeTensor(predLocal);
        inTgtQ.FreeTensor(tgtLocal);
    }

    __aicore__ inline void CopyOut()
    {
        LocalTensor<DTYPE_Z> lossLocal = outLossQ.DeQue<DTYPE_Z>();
        // 仅拷贝 1 个标量到 GM
        DataCopy(lossGlobal, lossLocal, 1);
        outLossQ.FreeTensor(lossLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  inPredQ;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  inTgtQ;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outLossQ;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  workQ;

    AscendC::GlobalTensor<DTYPE_X> predGlobal;
    AscendC::GlobalTensor<DTYPE_Y> tgtGlobal;
    AscendC::GlobalTensor<DTYPE_Z> lossGlobal;

    uint32_t width = 0;   // inner
    uint32_t height = 0;  // outter (1)
    uint32_t nNum = 0;    // 实际元素数
};

extern "C" __global__ __aicore__ void mse_loss_custom(GM_ADDR predictions, GM_ADDR targets, GM_ADDR loss, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMseLoss op;
    op.Init(predictions, targets, loss, tiling_data.size, tiling_data.outter, tiling_data.inner, tiling_data.n);
    op.Process();
}
