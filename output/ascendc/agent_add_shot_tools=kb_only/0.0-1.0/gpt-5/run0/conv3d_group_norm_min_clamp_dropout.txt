kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv3d_group_norm_min_clamp_dropout_custom_tiling_data.h"

// 基于参考资料：
// - 1) Ascend C 基础向量API：Mins/Maxs/Min/Max/Mul/Cast/Reciprocal/DataCopy/SetQueue等（Ascend C API Reference 表1-2、表1-3等）。
// - 2) DropOut可用字节掩码模式（高阶API说明，支持uint8掩码；也可用Cast+Mul模拟过滤再按keepProb缩放，参考DropOut章节描述）。
// 这里采用：字节掩码mask(uint8)->Cast为float，逐元素乘法，再乘以1/keepProb，实现与DropOut一致的功能。
// min/clamp使用Mins/Maxs/Min/Max组合实现。
// 注意：本核不实现Conv3D与GroupNorm（由PyTorch完成），本核完成min(x, min_val) + clamp + dropout三者融合。

using DTYPE = float;
using MASKTYPE = uint8_t;
constexpr int32_t BUFFER_NUM = 2;

class KernelFusedOp {
public:
    __aicore__ inline KernelFusedOp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mask, GM_ADDR minVal, GM_ADDR maxVal, GM_ADDR keepProb,
                                GM_ADDR z, uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE*)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        maskGm.SetGlobalBuffer((__gm__ MASKTYPE*)mask + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        // 标量全局地址（单元素）
        minGm.SetGlobalBuffer((__gm__ DTYPE*)minVal, 1);
        maxGm.SetGlobalBuffer((__gm__ DTYPE*)maxVal, 1);
        keepGm.SetGlobalBuffer((__gm__ DTYPE*)keepProb, 1);

        zGm.SetGlobalBuffer((__gm__ DTYPE*)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(inQueueMask, BUFFER_NUM, this->tileLength * sizeof(MASKTYPE));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE));

        // 临时向量缓冲：用于min/max广播结果、mask转浮点、keepProb广播及其倒数
        pipe.InitBuffer(tmpQueue, 3, this->tileLength * sizeof(DTYPE));
        // 标量缓冲：持有单元素min/max/keepProb
        pipe.InitBuffer(scalarQueue, 3, 1 * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        // 读取一次标量
        minScalar = scalarQueue.AllocTensor<DTYPE>();
        maxScalar = scalarQueue.AllocTensor<DTYPE>();
        keepScalar = scalarQueue.AllocTensor<DTYPE>();
        AscendC::DataCopy(minScalar, minGm, 1);
        AscendC::DataCopy(maxScalar, maxGm, 1);
        AscendC::DataCopy(keepScalar, keepGm, 1);

        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }

        scalarQueue.FreeTensor(minScalar);
        scalarQueue.FreeTensor(maxScalar);
        scalarQueue.FreeTensor(keepScalar);
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::LocalTensor<MASKTYPE> maskLocal = inQueueMask.AllocTensor<MASKTYPE>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(maskLocal, maskGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueMask.EnQue(maskLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<MASKTYPE> maskLocal = inQueueMask.DeQue<MASKTYPE>();
        AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.AllocTensor<DTYPE>();

        // 构造广播向量
        AscendC::LocalTensor<DTYPE> minVec = tmpQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> maxVec = tmpQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> keepVec = tmpQueue.AllocTensor<DTYPE>();

        AscendC::BroadCast(minVec, minScalar, this->tileLength);
        AscendC::BroadCast(maxVec, maxScalar, this->tileLength);
        AscendC::BroadCast(keepVec, keepScalar, this->tileLength);

        // 1) 先做 x <- min(x, min_val)
        AscendC::Min(zLocal, xLocal, minVec, this->tileLength);
        // 2) 再 clamp 到[min_val, max_val]
        AscendC::Max(zLocal, zLocal, minVec, this->tileLength);
        AscendC::Min(zLocal, zLocal, maxVec, this->tileLength);

        // 3) Dropout: mask(uint8)->cast(float) ，与z逐元素乘，再乘以(1/keepProb)
        AscendC::LocalTensor<DTYPE> maskF = tmpQueue.AllocTensor<DTYPE>();
        AscendC::Cast(maskF, maskLocal, this->tileLength);
        AscendC::Mul(zLocal, zLocal, maskF, this->tileLength);

        // 计算1/keepProb向量
        AscendC::LocalTensor<DTYPE> invKeep = tmpQueue.AllocTensor<DTYPE>();
        AscendC::Reciprocal(invKeep, keepVec, this->tileLength);
        AscendC::Mul(zLocal, zLocal, invKeep, this->tileLength);

        // 释放tmp与输入
        tmpQueue.FreeTensor(minVec);
        tmpQueue.FreeTensor(maxVec);
        tmpQueue.FreeTensor(keepVec);
        tmpQueue.FreeTensor(maskF);
        tmpQueue.FreeTensor(invKeep);

        outQueueZ.EnQue<DTYPE>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueMask.FreeTensor(maskLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.DeQue<DTYPE>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;

    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueMask;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::TQue<AscendC::TPosition::VECCALC, 3> tmpQueue;
    AscendC::TQue<AscendC::TPosition::VECCALC, 3> scalarQueue;

    AscendC::GlobalTensor<DTYPE> xGm, zGm;
    AscendC::GlobalTensor<MASKTYPE> maskGm;
    AscendC::GlobalTensor<DTYPE> minGm, maxGm, keepGm;

    AscendC::LocalTensor<DTYPE> minScalar;
    AscendC::LocalTensor<DTYPE> maxScalar;
    AscendC::LocalTensor<DTYPE> keepScalar;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv3d_group_norm_min_clamp_dropout_custom(
    GM_ADDR x, GM_ADDR mask, GM_ADDR minVal, GM_ADDR maxVal, GM_ADDR keepProb,
    GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedOp op;
    op.Init(x, mask, minVal, maxVal, keepProb, z, tiling_data.totalLength(), tiling_data.tileNum());
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    优化版：保留PyTorch Conv3d与GroupNorm；将后续min/clamp/dropout融合为自定义AscendC核
    conv3d_group_norm_min_clamp_dropout_custom。
    """
    def __init__(self, in_channels, out_channels, kernel_size, groups, min_value, max_value, dropout_p):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.norm = torch.nn.GroupNorm(groups, out_channels)
        self.min_value = float(min_value)
        self.max_value = float(max_value)
        self.dropout_p = float(dropout_p)

    def forward(self, x):
        # 先由PyTorch完成Conv3d与GroupNorm
        x = self.conv(x)
        x = self.norm(x)

        # 生成Dropout字节掩码（Byte mode）；参考AscendC DropOut说明：mask为uint8(0/1)，与输入同形状
        keep_prob = 1.0 - self.dropout_p
        if self.training:
            mask = (torch.rand_like(x) < keep_prob).to(torch.uint8)
        else:
            # 推理阶段通常不做随机丢弃，保持全1；与常规Dropout一致
            mask = torch.ones_like(x, dtype=torch.uint8)

        # 将min/max/keep_prob作为单元素张量输入（ND标量）
        min_val_t = torch.tensor(self.min_value, dtype=x.dtype, device=x.device)
        max_val_t = torch.tensor(self.max_value, dtype=x.dtype, device=x.device)
        keep_prob_t = torch.tensor(keep_prob, dtype=x.dtype, device=x.device)

        # 调用融合核：min(x, min_val) + clamp([min_val, max_val]) + dropout(byte mask)并按1/keep_prob缩放
        out = custom_ops_lib.conv3d_group_norm_min_clamp_dropout_custom(x, mask, min_val_t, max_val_t, keep_prob_t)
        return out
'''
