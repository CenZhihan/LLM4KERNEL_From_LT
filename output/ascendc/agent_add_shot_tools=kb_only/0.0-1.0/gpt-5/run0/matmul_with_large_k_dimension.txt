host_operator_src="""
#include "matmul_with_large_k_dimension_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
static const uint32_t BLOCK_DIM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    auto *shapeA = context->GetInputShape(0);
    auto *shapeB = context->GetInputShape(1);
    if (shapeA == nullptr || shapeB == nullptr) {
        return ge::GRAPH_FAILED;
    }

    // Expect A: [M, K], B: [K, N]
    uint32_t M = static_cast<uint32_t>(shapeA->GetDim(0));
    uint32_t K = static_cast<uint32_t>(shapeA->GetDim(1));
    uint32_t N = static_cast<uint32_t>(shapeB->GetDim(1));

    MatmulWithLargeKDimensionCustomTilingData tiling;
    tiling.set_M(M);
    tiling.set_N(N);
    tiling.set_K(K);

    context->SetBlockDim(BLOCK_DIM);

    // Save tiling to buffer
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // Workspace not required
    size_t *workspaceSizes = context->GetWorkspaceSizes(1);
    workspaceSizes[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* a_shape = context->GetInputShape(0); // [M, K]
    const gert::Shape* b_shape = context->GetInputShape(1); // [K, N]
    if (a_shape == nullptr || b_shape == nullptr) {
        return GRAPH_FAILED;
    }
    gert::Shape* c_shape = context->GetOutputShape(0);
    // Output: [M, N]
    c_shape->Resize(2);
    c_shape->SetDim(0, a_shape->GetDim(0));
    c_shape->SetDim(1, b_shape->GetDim(1));
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return GRAPH_SUCCESS;
}
}

namespace ops {
class MatmulWithLargeKDimensionCustom : public OpDef {
public:
    explicit MatmulWithLargeKDimensionCustom(const char* name) : OpDef(name)
    {
        this->Input("A")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("B")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("C")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(MatmulWithLargeKDimensionCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matmul_with_large_k_dimension_custom_tiling_data.h"
#include "lib/matmul_intf.h"

// This kernel uses the Matmul NORMAL layout with IterateBatch as shown in the CANN Ascend C API reference.

using A_T = float;
using B_T = float;
using C_T = float;

extern "C" __global__ __aicore__ void matmul_with_large_k_dimension_custom(GM_ADDR aGM, GM_ADDR bGM,
                                                                           GM_ADDR cGM, GM_ADDR workspaceGM,
                                                                           GM_ADDR tilingGM)
{
    // Load tiling data (M, N, K)
    GET_TILING_DATA(tilingData, tilingGM);
    const uint32_t M = tilingData.M;
    const uint32_t N = tilingData.N;
    const uint32_t K = tilingData.K;

    // Define Matmul types for NORMAL layout
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::NORMAL> aType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, B_T, true,  LayoutMode::NORMAL> bType; // not transposed compute expects B as KN
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::NORMAL> cType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T> biasType;

    // Initialize GM tensors
    AscendC::GlobalTensor<A_T> aGlobal;
    AscendC::GlobalTensor<B_T> bGlobal;
    AscendC::GlobalTensor<C_T> cGlobal;

    const int32_t sizeA = static_cast<int32_t>(M) * static_cast<int32_t>(K);
    const int32_t sizeB = static_cast<int32_t>(K) * static_cast<int32_t>(N);
    const int32_t sizeC = static_cast<int32_t>(M) * static_cast<int32_t>(N);

    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(aGM), sizeA);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(bGM), sizeB);
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(cGM), sizeC);

    // Prepare NORMAL layout tiling
    TCubeTiling tiling;
    // Zero-initialize the structure
    __builtin_memset(&tiling, 0, sizeof(TCubeTiling));
    tiling.singleCoreM = M;
    tiling.singleCoreK = K;
    tiling.singleCoreN = N;
    tiling.ALayoutInfoB = 1;
    tiling.BLayoutInfoB = 1;
    tiling.CLayoutInfoB = 1;
    tiling.isBias = 0;
    // Shared buffer sizes (example values)
    tiling.shareMode = 0;
    tiling.shareL1Size = 512 * 1024;
    tiling.shareL0CSize = 128 * 1024;
    tiling.shareUbSize = 0;

    // Wrap GM buffers for Matmul
    AscendC::GlobalTensor<A_T> gm_a;
    AscendC::GlobalTensor<B_T> gm_b;
    AscendC::GlobalTensor<C_T> gm_c;

    gm_a.SetGlobalBuffer(const_cast<__gm__ A_T*>(aGlobal[0].GetPhyAddr()), M * K);
    gm_b.SetGlobalBuffer(const_cast<__gm__ B_T*>(bGlobal[0].GetPhyAddr()), K * N);
    gm_c.SetGlobalBuffer(const_cast<__gm__ C_T*>(cGlobal[0].GetPhyAddr()), M * N);

    // Create and register Matmul instance
    constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);
    matmul::Matmul<aType, bType, cType, biasType, MM_CFG> mm1;

    TPipe pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm1);

    // Initialize and run Matmul in NORMAL layout
    mm1.Init(&tiling);
    // A: [M, K], B: [K, N]
    const uint32_t isTransposeAIn = 0U;
    const uint32_t isTransposeBIn = 0U;
    mm1.SetTensorA(gm_a, isTransposeAIn);
    mm1.SetTensorB(gm_b, isTransposeBIn);
    mm1.SetWorkspace(workspaceGM, 0);

    // No bias
    // Execute Matmul for a single batch
    const int32_t batchA = 1;
    const int32_t batchB = 1;
    mm1.IterateBatch(gm_c, batchA, batchB, false);
}

at::Tensor matmul_with_large_k_dimension_custom_impl_npu(const at::Tensor& A, const at::Tensor& B) {
    auto M = A.size(0);
    auto K = A.size(1);
    TORCH_CHECK(B.size(0) == K, \\"B's first dim must equal A's second dim\\");
    auto N = B.size(1);
    at::Tensor C = at::empty({M, N}, A.options());
    EXEC_NPU_CMD(aclnnMatmulWithLargeKDimensionCustom, A, B, C);
    return C;
}

    m.impl("matmul_with_large_k_dimension_custom", &matmul_with_large_k_dimension_custom_impl_npu);
