    LeakyReluCustomTiling t;
    t.total_elems = static_cast<uint32_t>(total_elems);
    t.dtype_flag = is_fp32 ? 1u : 0u;
    t.negative_slope = negative_slope;
    return t;

// 伪代码：获取元素总数
static inline uint64_t GetNumel(const int64_t* shape, int64_t dim) {
    uint64_t numel = 1;
    for (int64_t i = 0; i < dim; ++i) {
        numel *= static_cast<uint64_t>(shape[i]);
    }
    return numel;
}

int LeakyReluCustomRun(void* x_gm, void* y_gm,
                       const int64_t* shape, int64_t dim,

    // 生成tiling
    uint64_t total_elems_u64 = GetNumel(shape, dim);

kernel_src = r"""
// leaky_relu_custom_kernel.cpp
// 说明：AscendC内核，使用基础API LeakyRelu 完成按元素计算。
// 参考：
// - 4.1.3.7 LeakyRelu（Basic APIs）：
//   template <typename T, bool isSetMask = true>
//   __aicore__ inline void LeakyRelu(const LocalTensor<T>& dstLocal, const LocalTensor<T>& srcLocal,
//                                    const T& scalarValue, const int32_t& calCount)
// - 数据搬运、队列、TPipe 参考：Silu/GeGLU 示例（5.3.5.1、5.3.7.1 示例）

#include "kernel_operator.h"

using namespace AscendC;

template <typename srcType>
class KernelLeakyReluCustom {
public:
    __aicore__ inline KernelLeakyReluCustom() {}

    __aicore__ inline void Init(GM_ADDR x_gm, GM_ADDR y_gm, uint32_t inputSize) {
        dataSize_ = inputSize;
        xGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ srcType*>(x_gm), dataSize_);
        yGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ srcType*>(y_gm), dataSize_);
        pipe_.InitBuffer(inQueue_, 1, dataSize_ * sizeof(srcType));
        pipe_.InitBuffer(outQueue_, 1, dataSize_ * sizeof(srcType));
    }

    __aicore__ inline void Process(float negative_slope) {
        CopyIn();
        Compute(negative_slope);
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn() {
        LocalTensor<srcType> xLocal = inQueue_.AllocTensor<srcType>();
        DataCopy(xLocal, xGlobal_, dataSize_);
        inQueue_.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float negative_slope) {
        LocalTensor<srcType> yLocal = outQueue_.AllocTensor<srcType>();
        LocalTensor<srcType> xLocal = inQueue_.DeQue<srcType>();

        // 按API要求，scalarValue的数据类型需与Tensor元素类型一致
        srcType slopeVal = static_cast<srcType>(negative_slope);

        // calCount使用int32范围，保证不超过输入size
        int32_t calCount = static_cast<int32_t>(dataSize_);
        LeakyRelu<srcType>(yLocal, xLocal, slopeVal, calCount);

        outQueue_.EnQue(yLocal);
        inQueue_.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut() {
        LocalTensor<srcType> yLocal = outQueue_.DeQue<srcType>();
        DataCopy(yGlobal_, yLocal, dataSize_);
        outQueue_.FreeTensor(yLocal);
    }

private:
    GlobalTensor<srcType> xGlobal_;
    GlobalTensor<srcType> yGlobal_;
    TPipe pipe_;
    TQue<QuePosition::VECIN, 1> inQueue_;
    TQue<QuePosition::VECOUT, 1> outQueue_;
    uint32_t dataSize_{0};
};

// AscendC Kernel Entry（模板实例）
// 注意：导出两个实例以匹配Host侧按dtype选择
extern "C" {

__aicore__ void leaky_relu_custom_fp16(GM_ADDR x_gm, GM_ADDR y_gm, uint32_t total_elems, float negative_slope) {
    KernelLeakyReluCustom<half> op;
    op.Init(x_gm, y_gm, total_elems);
    op.Process(negative_slope);
}

__aicore__ void leaky_relu_custom_fp32(GM_ADDR x_gm, GM_ADDR y_gm, uint32_t total_elems, float negative_slope) {
    KernelLeakyReluCustom<float> op;
    op.Init(x_gm, y_gm, total_elems);
    op.Process(negative_slope);
}

}

python_bind_src = r"""
# python_bindings.py
# 说明：
# - 定义Python侧绑定与Fallback。
# - 注册Torch库schema以便在未加载NPU自定义算子so时也可正常fallback到PyTorch实现。
# - 实际部署时，需通过torch.ops.load_library加载由AscendC工程编译出的so，以启用NPU实现。

import torch
from torch.library import Library

# 定义算子schema
_lib = Library("leaky_relu_custom", "DEF")
_lib.define("leaky_relu_custom(Tensor x, float negative_slope) -> Tensor")

# CPU/Meta后端fallback实现（用于开发调试或无自定义算子so时）
def _cpu_fallback(x: torch.Tensor, negative_slope: float) -> torch.Tensor:
    return torch.nn.functional.leaky_relu(x, negative_slope=negative_slope)

_lib.impl("leaky_relu_custom", _cpu_fallback, "CPU")
_lib.impl("leaky_relu_custom", _cpu_fallback, "Meta")

def leaky_relu_custom(x: torch.Tensor, negative_slope: float):
    # 如已通过torch.ops.load_library加载自定义Ascend算子so，以下调用将路由到NPU实现
    try:
        return torch.ops.leaky_relu_custom.leaky_relu_custom(x, float(negative_slope))
    except Exception:
        # 回退到CPU/Meta实现（或其它可用设备）
        return _cpu_fallback(x, float(negative_slope))
"""
