// gelu_custom_op.cpp
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"
#include "gelu_custom_tiling.h"

using namespace ge;
using namespace gert;

namespace {
static graphStatus InferShapeAndType(Operator& op, const std::vector<TensorDesc*>& inputs,
                                     const std::vector<TensorDesc*>& outputs) {
    // 形状与类型推导：输出与输入一致
    if (inputs.empty() || outputs.empty()) {
        return GRAPH_FAILED;
    }
    auto* in_desc = inputs[0];
    auto* out_desc = outputs[0];
    out_desc->SetDataType(in_desc->GetDataType());
    out_desc->SetFormat(in_desc->GetFormat());
    out_desc->SetShape(in_desc->GetShape());
    return GRAPH_SUCCESS;
}

static graphStatus TilingFunc(TilingContext* context) {
    optiling::GeluCustomTilingData tiling;

    const auto* in_shape = context->GetInputShape(0);
    if (in_shape == nullptr) {
        return GRAPH_FAILED;
    }
    const auto storage_shape = in_shape->GetStorageShape();

    // 计算总元素数 dataSize
    uint64_t total = 1;
    for (uint32_t i = 0; i < storage_shape.GetDimNum(); ++i) {
        total *= static_cast<uint64_t>(storage_shape.GetDim(i));
    }
    // AscendC API 使用 uint32_t 作为 dataSize，这里进行裁剪（大多数场景足够）
    tiling.set_dataSize(static_cast<uint32_t>(total));

    // 设置 BlockDim
    context->SetBlockDim(1);

    // 保存 tiling 数据
    auto* raw = context->GetRawTilingData();
    tiling.SaveToBuffer(raw->GetData(), raw->GetCapacity());
    raw->SetDataSize(tiling.GetDataSize());

    // 无需 workspace
    size_t* workspaces = context->GetWorkspaceSizes(1);
    workspaces[0] = 0;

    return GRAPH_SUCCESS;
}
} // namespace

extern "C" {
REGISTER_OP_DEF(GeluCustom)
    .FrameworkType("AscendC")
    .Input("x")
    .Output("y")
    .SetInferShapeAndType(InferShapeAndType)
    .SetTiling(TilingFunc);
}

kernel_src = r"""
// gelu_custom.cpp
#include "kernel_operator.h"

template <typename T>
class KernelGelu {
public:
    __aicore__ inline KernelGelu() {}

    __aicore__ inline void Init(GM_ADDR x_gm, GM_ADDR y_gm, uint32_t inputSize) {
        dataSize_ = inputSize;
        xGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(x_gm), dataSize_);
        yGlobal_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(y_gm), dataSize_);
        pipe_.InitBuffer(inQueue_, 1, dataSize_ * sizeof(T));
        pipe_.InitBuffer(outQueue_, 1, dataSize_ * sizeof(T));
    }

    __aicore__ inline void Process() {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn() {
        AscendC::LocalTensor<T> xLocal = inQueue_.AllocTensor<T>();
        AscendC::DataCopy(xLocal, xGlobal_, dataSize_);
        inQueue_.EnQue(xLocal);
    }

    __aicore__ inline void Compute() {
        AscendC::LocalTensor<T> yLocal = outQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> xLocal = inQueue_.DeQue<T>();
        // 使用 AscendC 高级 API: Gelu（参考 CANN Ascend C API Reference 5.3.3.1 Gelu）
        AscendC::Gelu(yLocal, xLocal, dataSize_);
        outQueue_.EnQue<T>(yLocal);
        inQueue_.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut() {
        AscendC::LocalTensor<T> yLocal = outQueue_.DeQue<T>();
        AscendC::DataCopy(yGlobal_, yLocal, dataSize_);
        outQueue_.FreeTensor(yLocal);
    }

private:
    AscendC::GlobalTensor<T> xGlobal_;
    AscendC::GlobalTensor<T> yGlobal_;
    AscendC::TPipe pipe_;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueue_;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue_;
    uint32_t dataSize_ = 0;
};

// 内核入口函数名必须与算子名一致：gelu_custom
extern "C" __global__ __aicore__ void gelu_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    if ASCEND_IS_AIC {
        return;
    }
    // 获取 tiling 数据（参考文档示例）
    GET_TILING_DATA(tiling_data, tiling);
    // 数据类型通过 TILING_KEY 传递的更通用方式在参考示例中演示，
    // 这里直接根据上层注册的数据类型分发模板，在 Gelu 中无需共享临时缓冲区。
    auto dt = GetInputDataType(0);
    if (dt == ge::DT_FLOAT16) {
        KernelGelu<half> op;
        op.Init(x, y, tiling_data.dataSize);
        op.Process();
    } else {
        KernelGelu<float> op;
        op.Init(x, y, tiling_data.dataSize);
        op.Process();
    }
}
"""

python_bind_src = r"""
# Python 绑定：优先尝试加载自定义 AscendC 算子库，不可用时回退至 PyTorch 自带 GELU
import os
import torch
import torch.nn.functional as F

_loaded = False

def _try_load_lib():
    global _loaded
    if _loaded:
        return True
    candidates = [
        os.environ.get("GELU_CUSTOM_LIB", ""),
        "./build/libgelu_custom.so",
        "libgelu_custom.so"
    ]
    for path in candidates:
        if not path:
            continue
        try:
            torch.ops.load_library(path)
            _loaded = True
            break
        except Exception:
            continue
    return _loaded

def gelu_custom(x: torch.Tensor) -> torch.Tensor:
    # 若已加载并注册，则直接调用自定义 OP
    if _try_load_lib():
        try:
            return torch.ops.GeluCustom.gelu_custom(x)
        except Exception:
            pass
    # 回退路径：使用 PyTorch 的 GELU
    return F.gelu(x)
"""
