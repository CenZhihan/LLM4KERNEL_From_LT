kernel_src = r"""
// 说明：AscendC Kernel 源码（L0），实现 SELU 激活
// 要点：
// - 使用 OP_TYPE_REGISTER 注册 L0 算子（知识库 6.4.2.1.18）
// - 在 Kernel 中注册默认 TilingData 以及按 TilingKey 的匹配（知识库 4.7.6）
#include <stdint.h>

namespace l0op {
OP_TYPE_REGISTER(SeluCustom); // 必须在 L0 API 开头注册算子类型
}

extern "C" {

typedef struct {
    uint32_t blkDim;
    uint32_t totalSize;
    uint32_t splitTile;
} SeluCustomTilingPlain;

// AscendC 设备侧 Kernel（示例：简化单核串行演示）
__global__ __aicore__ void selu_custom(__gm__ const float* x, __gm__ float* y, __gm__ uint8_t* tiling) {
    // 基于知识库示例注册默认 TilingData 和条件匹配（若可用）
    // REGISTER_TILING_DEFAULT(optiling::SeluCustomTilingData);
    // REGISTER_TILING_FOR_TILINGKEY("TILING_KEY_VAR == 1", optiling::SeluCustomTilingData);
    // 为避免依赖框架宏展开，这里直接解析平铺结构（与 host_tiling.h 对齐）
    const SeluCustomTilingPlain* td = (const SeluCustomTilingPlain*)tiling;
    uint32_t n = td->totalSize;

    const float alpha = 1.6732632423543772f;
    const float lambda_ = 1.0507009873554805f;

    // 简化演示：单线程串行处理
    for (uint32_t i = 0; i < n; ++i) {
        float v = x[i];
        float out = (v > 0.0f) ? v : (alpha * (expf(v) - 1.0f));
        y[i] = lambda_ * out;
    }
}

} // extern "C"
"""
