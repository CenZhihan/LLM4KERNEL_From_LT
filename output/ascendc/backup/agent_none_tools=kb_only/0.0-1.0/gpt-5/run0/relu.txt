host_tiling_src = r"""
#ifndef RELU_CUSTOM_TILING_H_
#define RELU_CUSTOM_TILING_H_

// 本文件仅提供极简的“平铺参数”定义，便于 Host 侧计算需要传递给 Kernel 的元素个数。
// 参考文档（知识库检索）：
// - 基础 API ReLU：4.1.1.8 Relu，接口形如 Relu(dstLocal, srcLocal, calCount)
// - 示例模式（模板）：4.1.1.9 More Samples

#include <cstdint>

struct ReluCustomTiling {
    // 实际需要计算的元素个数，对应 AscendC::Relu 的 calCount
    uint32_t dataSize = 0;
};

// 简单的“生成 tiling”函数：直接将总元素数写入 dataSize
inline ReluCustomTiling GenReluCustomTiling(uint64_t numel) {
    ReluCustomTiling t;
    // AscendC Relu 的 calCount 为 int32_t/uint32_t，确保不溢出（大张量可在 Host 侧分批次调度）
    if (numel > 0xFFFFFFFFull) {
        // 极端情况下可在上层切分，这里做最简保护
        t.dataSize = 0xFFFFFFFFu;
    } else {
        t.dataSize = static_cast<uint32_t>(numel);
    }
    return t;
}

#endif // RELU_CUSTOM_TILING_H_
"""

host_operator_src = r"""
// Host 侧算子封装：负责形状与 dtype 处理、生成 Tiling、以及发起 AscendC Kernel 启动。
// 参考文档（知识库检索）：
// - 基础 API ReLU：4.1.1.8 Relu（参数 calCount 为参与计算的元素个数）
// - 示例：5.3.5.1 Silu 与 4.1.1.9 模板示例（DataCopy/队列/流程）

#include <torch/extension.h>
#include <ATen/ATen.h>
#include <vector>
#include <stdexcept>
#include "relu_custom_tiling.h"

// Kernel 符号由 kernel 源文件提供
extern "C" void kernel_relu_custom(uint8_t* xGm, uint8_t* yGm, uint32_t dataSize);

// 简单的 Launch 包装（演示用）：实际工程中请使用 Ascend 运行时接口配置 stream/grid/block 并 Launch
static inline void LaunchKernelReluCustom(void* x, void* y, uint32_t nElem) {
    // 在 AscendC 工程中，这里应由 runtime 完成 __global__ kernel 的调度。
    // 为了示例自洽，我们直接声明并调用符号（假设工具链提供符号解析与调度）。
    kernel_relu_custom(reinterpret_cast<uint8_t*>(x), reinterpret_cast<uint8_t*>(y), nElem);
}

// 对外导出：PyTorch 张量 -> 自定义 AscendC ReLU
at::Tensor ReluCustom(at::Tensor x) {
    // 要求输入为 contiguous
    x = x.contiguous();

    // AscendC 基础 ReLU 文档（4.1.1.8）在 Atlas 训练系列仅明确 half 支持；
    // 为兼容实际训练，我们在 Host 侧将 float32 自动转换为 float16，再交给 Kernel。
    at::Tensor x_half;
    bool cast_back_to_fp32 = false;

    if (x.scalar_type() == at::kFloat) {
        x_half = x.to(at::kHalf);
        cast_back_to_fp32 = true;
    } else if (x.scalar_type() == at::kHalf) {
        x_half = x;
    } else {
        throw std::runtime_error("ReluCustom only supports float16/float32.");
    }

    auto y_half = at::empty_like(x_half);

    // 计算元素总数并生成 tiling（calCount）
    const uint64_t numel = static_cast<uint64_t>(x_half.numel());
    ReluCustomTiling tiling = GenReluCustomTiling(numel);
    if (tiling.dataSize == 0) {
        // 空张量直接返回
        if (cast_back_to_fp32) {
            return x.to(at::kFloat);
        }
        return x_half;
    }

    // 取设备指针并发起 Kernel（此处演示直接调用，实际需使用 Ascend 运行时）
    void* x_dev = x_half.data_ptr();
    void* y_dev = y_half.data_ptr();
    LaunchKernelReluCustom(x_dev, y_dev, tiling.dataSize);

    // 如输入为 float32，则将结果转回 float32
    if (cast_back_to_fp32) {
        return y_half.to(at::kFloat);
    }
    return y_half;
}
"""

kernel_src = r"""
// AscendC Kernel 源码：基于基础 API Relu 实现逐元素 ReLU
// 参考文档（知识库检索）：4.1.1.8 Relu 与 4.1.1.9 More Samples
// - 原型：template <typename T> __aicore__ inline void Relu(const LocalTensor<T>& dstLocal,
//   const LocalTensor<T>& srcLocal, const int32_t& calCount)
// - 约束：源/目的 LocalTensor 类型一致；calCount 为参与计算的元素数；ND 格式；地址对齐等。

#include "kernel_operator.h"

template <typename srcType>
class KernelRelu {
public:
    __aicore__ inline KernelRelu() {}
    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR yGm, uint32_t inputSize)
    {
        dataSize = inputSize;
        xGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ srcType *>(xGm), dataSize);
        yGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ srcType *>(yGm), dataSize);
        pipe.InitBuffer(inQueueX, 1, dataSize * sizeof(srcType));
        pipe.InitBuffer(outQueueY, 1, dataSize * sizeof(srcType));
    }
    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }
private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<srcType> xLocal = inQueueX.AllocTensor<srcType>();
        AscendC::DataCopy(xLocal, xGlobal, dataSize);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<srcType> yLocal = outQueueY.AllocTensor<srcType>();
        AscendC::LocalTensor<srcType> xLocal = inQueueX.DeQue<srcType>();
        // 使用基础 API Relu 计算；calCount = dataSize
        AscendC::Relu(yLocal, xLocal, static_cast<int32_t>(dataSize));
        outQueueY.EnQue<srcType>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<srcType> yLocal = outQueueY.DeQue<srcType>();
        AscendC::DataCopy(yGlobal, yLocal, dataSize);
        outQueueY.FreeTensor(yLocal);
    }
private:
    AscendC::GlobalTensor<srcType> xGlobal;
    AscendC::GlobalTensor<srcType> yGlobal;
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueY;
    uint32_t dataSize = 0;
};

// 单一导出 Kernel：按文档（Atlas 训练系列）采用 half 进行计算
extern "C" __global__ __aicore__ void kernel_relu_custom(GM_ADDR xGm, GM_ADDR yGm, uint32_t dataSize)
{
    KernelRelu<half> op;
    op.Init(xGm, yGm, dataSize);
    op.Process();
}
"""
