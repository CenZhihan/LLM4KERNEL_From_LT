kernel_src = r"""
// AscendC 内核：以基本矢量 API 组合实现 ELU：y = max(0, x) + min(0, alpha * (exp(x) - 1))
// 利用表 1-2 矢量/标量指令：Exp, Adds, Muls, Mins, Maxs, Add
// 参考：5.1.31（GET_TILING_DATA/队列/TPipe），5.3.x（Exp 示例使用）
#include "kernel_operator.h"

template <typename srcType>
class KernelEluCustom {
public:
    __aicore__ inline KernelEluCustom() {}

    __aicore__ inline void Init(GM_ADDR srcGm, GM_ADDR dstGm, uint32_t inputSize, float alphaIn)
    {
        dataSize = inputSize;
        alpha = static_cast<srcType>(alphaIn);

        srcGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ srcType *>(srcGm), dataSize);
        dstGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ srcType *>(dstGm), dataSize);

        pipe.InitBuffer(inQueue, 1, dataSize * sizeof(srcType));
        pipe.InitBuffer(outQueue, 1, dataSize * sizeof(srcType));
        // 两个 VECCALC 临时缓冲：tmp0/tmp1
        pipe.InitBuffer(tmpQueue0, 1, dataSize * sizeof(srcType));
        pipe.InitBuffer(tmpQueue1, 1, dataSize * sizeof(srcType));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<srcType> srcLocal = inQueue.AllocTensor<srcType>();
        AscendC::DataCopy(srcLocal, srcGlobal, dataSize);
        inQueue.EnQue(srcLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<srcType> dstLocal = outQueue.AllocTensor<srcType>();
        AscendC::LocalTensor<srcType> srcLocal = inQueue.DeQue<srcType>();

        AscendC::LocalTensor<srcType> tmp0 = tmpQueue0.AllocTensor<srcType>(); // 保存中间结果
        AscendC::LocalTensor<srcType> tmp1 = tmpQueue1.AllocTensor<srcType>(); // 保存 ReLU(x)

        // tmp1 = max(0, x)
        AscendC::Maxs(tmp1, srcLocal, static_cast<srcType>(0), dataSize);

        // tmp0 = exp(x)
        // 参考文档 Example 3 对 Exp 的调用形式（AscendC::Exp<T, PAR, isReuseSource>(dst, src, count)）
        AscendC::Exp<srcType, 15, false>(tmp0, srcLocal, dataSize);

        // tmp0 = exp(x) - 1
        AscendC::Adds(tmp0, tmp0, static_cast<srcType>(-1), dataSize);

        // tmp0 = alpha * (exp(x) - 1)
        AscendC::Muls(tmp0, tmp0, alpha, dataSize);

        // tmp0 = min(0, alpha * (exp(x) - 1))
        AscendC::Mins(tmp0, tmp0, static_cast<srcType>(0), dataSize);

        // dst = max(0, x) + min(0, alpha*(exp(x)-1))
        AscendC::Add(dstLocal, tmp1, tmp0, dataSize);

        outQueue.EnQue<srcType>(dstLocal);

        // 释放本地张量
        inQueue.FreeTensor(srcLocal);
        tmpQueue0.FreeTensor(tmp0);
        tmpQueue1.FreeTensor(tmp1);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<srcType> dstLocal = outQueue.DeQue<srcType>();
        AscendC::DataCopy(dstGlobal, dstLocal, dataSize);
        outQueue.FreeTensor(dstLocal);
    }

private:
    AscendC::GlobalTensor<srcType> srcGlobal;
    AscendC::GlobalTensor<srcType> dstGlobal;
    AscendC::TPipe pipe;

    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpQueue0;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpQueue1;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue;

    uint32_t dataSize = 0;
    srcType alpha = static_cast<srcType>(1.0f);
};

// 内核入口：读取 tiling，初始化并执行
extern "C" __global__ __aicore__ void elu_custom(GM_ADDR srcGm, GM_ADDR dstGm, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    KernelEluCustom<float> op; // 示例以 float 实例化，实际可按需支持 half
    op.Init(srcGm, dstGm, tilingData.dataSize, tilingData.alpha);

    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}
"""
