host_operator_src="""
#include "softmax_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
static constexpr uint32_t DEFAULT_BLOCK_DIM = 48;

static ge::graphStatus TilingFunc(gert::TilingContext *context)
{
    SoftmaxCustomTilingData tiling;

    const gert::Shape *in_shape = context->GetInputShape(0);
    // 仅支持ND输入，按最后一个维度做softmax
    uint32_t dims = in_shape->GetDimNum();
    if (dims == 0) {
        return ge::GRAPH_FAILED;
    }

    // 计算行数(m)与列数(n)，按最后一维为n
    uint64_t m = 1;
    for (uint32_t i = 0; i + 1 < dims; ++i) {
        m *= static_cast<uint64_t>(in_shape->GetDim(i));
    }
    uint64_t n = static_cast<uint64_t>(in_shape->GetDim(dims - 1));

    // 设置block并行在m维（行）
    uint32_t blockDim = static_cast<uint32_t>(std::min<uint64_t>(DEFAULT_BLOCK_DIM, m == 0 ? 1 : m));
    if (blockDim == 0) blockDim = 1;

    uint32_t rowsPerBlock = static_cast<uint32_t>(m / blockDim);
    if (rowsPerBlock == 0) {
        // 当m < blockDim时，确保前(m-1)个block处理1行，最后一个处理剩余
        rowsPerBlock = 1;
        blockDim = static_cast<uint32_t>(m);
    }
    uint32_t lastBlockRows = static_cast<uint32_t>(m - static_cast<uint64_t>(rowsPerBlock) * (blockDim - 1));

    const gert::RuntimeAttrs *attrs = context->GetAttrs();
    int64_t dim = 1;
    if (attrs != nullptr && attrs->GetAttrNum() > 0) {
        const int64_t *pdim = attrs->GetAttrPointer<int64_t>(0);
        if (pdim != nullptr) dim = *pdim;
    }

    context->SetBlockDim(blockDim);

    tiling.set_rowsPerBlock(rowsPerBlock);
    tiling.set_lastBlockRows(lastBlockRows);
    tiling.set_rowLength(static_cast<uint32_t>(n));
    tiling.set_dim(dim);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *workspace = context->GetWorkspaceSizes(1);
    workspace[0] = 0; // 不使用workspace
    return ge::GRAPH_SUCCESS;
}
} // namespace optiling

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext *context)
{
    const gert::Shape *x_shape = context->GetInputShape(0);
    gert::Shape *y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const ge::DataType x_dtype = context->GetInputDataType(0);
    context->SetOutputDataType(0, x_dtype);
    return GRAPH_SUCCESS;
}
} // namespace ge

namespace ops {
class SoftmaxCustom : public OpDef {
public:
    explicit SoftmaxCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Attr("dim").AttrType(OPTIONAL).Int(1);
        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};
OP_ADD(SoftmaxCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "softmax_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // 每个队列的tensor个数

class KernelSoftmax {
public:
    __aicore__ inline KernelSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t rowsPerBlock, uint32_t lastBlockRows,
                                uint32_t rowLength)
    {
        this->rowLength = rowLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        if (blockIdx == blockNum - 1) {
            this->rowsThisBlock = lastBlockRows;
        } else {
            this->rowsThisBlock = rowsPerBlock;
        }
        if (this->rowsThisBlock == 0 || this->rowLength == 0) return;

        uint64_t startRow = static_cast<uint64_t>(rowsPerBlock) * blockIdx;
        uint64_t totalElemsThisBlock = static_cast<uint64_t>(this->rowsThisBlock) * this->rowLength;

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * this->rowLength, totalElemsThisBlock);
        yGm.SetGlobalBuffer((__gm__ float*)y + startRow * this->rowLength, totalElemsThisBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->rowLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->rowLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->rowsThisBlock == 0 || this->rowLength == 0) return;

        for (uint32_t r = 0; r < this->rowsThisBlock; ++r) {
            CopyIn(r);
            Compute();
            CopyOut(r);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->rowLength], this->rowLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // 使用高阶SoftMax API，按最后一维(rowLength)进行softmax
        SoftMaxTiling tilingInfo; // 由API内部根据shape校验并确定
        SoftMaxShapeInfo shapeInfo = {1, this->rowLength}; // [m=1, k=rowLength]
        AscendC::SoftMax(yLocal, xLocal, tilingInfo, shapeInfo);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[rowIdx * this->rowLength], yLocal, this->rowLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm, yGm;

    uint32_t rowLength = 0;
    uint32_t rowsThisBlock = 0;
};

extern "C" __global__ __aicore__ void softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelSoftmax op;
    op.Init(x, y, tiling_data.rowsPerBlock, tiling_data.lastBlockRows, tiling_data.rowLength);
    op.Process();
}
"""
