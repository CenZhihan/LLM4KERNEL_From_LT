kernel_src="""
#include \"kernel_operator.h\"
#define __NPU_TILING__
#include \"gemm_group_norm_min_bias_add_custom_tiling_data.h\"
#include \"groupnorm_base.h\"  // 假设包含 KernelGroupNorm 定义（与参考资料示例一致）

using DTYPE = float;

extern \"C\" __global__ __aicore__ void gemm_group_norm_min_bias_add_custom(
    GM_ADDR input_x_gm, GM_ADDR gamma_gm, GM_ADDR beta_gm, GM_ADDR bias_gm,
    GM_ADDR y_gm, GM_ADDR output_mean_gm, GM_ADDR output_variance_gm, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    // 读取必要形状信息
    const uint32_t N = tilingData.n;
    const uint32_t C = tilingData.c;
    const uint32_t H = tilingData.h;
    const uint32_t W = tilingData.w;
    (void)H; (void)W; // 本例固定为 1x1

    // 第一步：调用 GroupNorm 高级 API 完成归一化
    KernelGroupNorm<DTYPE, false> gnOp;
    gnOp.Init(input_x_gm, gamma_gm, beta_gm, y_gm, output_mean_gm, output_variance_gm, tilingData.groupNormTilingData);
    gnOp.Process();

    // 第二步：对归一化结果在通道维(C)做 reduce min(keepdim=True, dim=1)，并与 bias 相加
    // 目标：y[n, c, 1, 1] = min_{k in C}(norm[n, k, 1, 1]) + bias[c, 1, 1]
    // 简化实现：逐 batch 处理，每次拉取一整条通道向量到本地缓冲，先算通道最小值，再写回加上 bias 的结果。

    AscendC::GlobalTensor<DTYPE> normOut;
    normOut.SetGlobalBuffer((__gm__ DTYPE*)y_gm, (uint32_t)(N * C));

    AscendC::GlobalTensor<DTYPE> biasG;
    biasG.SetGlobalBuffer((__gm__ DTYPE*)bias_gm, C);

    // 使用 UB 小缓冲进行分块
    constexpr int32_t UB_TILE = 1024; // 简化示例块大小
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> qIn;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> qBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> qOut;
    pipe.InitBuffer(qIn, 1, UB_TILE * sizeof(DTYPE));
    pipe.InitBuffer(qBias, 1, UB_TILE * sizeof(DTYPE));
    pipe.InitBuffer(qOut, 1, UB_TILE * sizeof(DTYPE));

    // 预处理：bias 按通道维 C 进行块式访问
    for (uint32_t n = 0; n < N; ++n) {
        // 计算每个 batch 的通道最小值
        DTYPE min_val = static_cast<DTYPE>(3.402823466e+38F); // float 最大值
        uint32_t cRemain = C;
        uint32_t cOffset = 0;
        while (cRemain > 0) {
            uint32_t cur = cRemain > (uint32_t)UB_TILE ? (uint32_t)UB_TILE : cRemain;
            // 拉取归一化输出的一段通道数据
            auto inLocal = qIn.AllocTensor<DTYPE>();
            AscendC::DataCopy(inLocal, normOut[n * C + cOffset], cur);

            // 在本地计算该段的最小值
            // 简化：直接在标量循环中比较
            for (uint32_t i = 0; i < cur; ++i) {
                DTYPE v = inLocal.GetValue(i);
                if (v < min_val) min_val = v;
            }

            qIn.FreeTensor(inLocal);
            cRemain -= cur;
            cOffset += cur;
        }

        // 根据 min_val 与 bias 形成最终输出 y[n, c] = min_val + bias[c]
        cRemain = C;
        cOffset = 0;
        while (cRemain > 0) {
            uint32_t cur = cRemain > (uint32_t)UB_TILE ? (uint32_t)UB_TILE : cRemain;

            // 读 bias
            auto biasLocal = qBias.AllocTensor<DTYPE>();
            AscendC::DataCopy(biasLocal, biasG[cOffset], cur);

            // 生成输出块
            auto outLocal = qOut.AllocTensor<DTYPE>();
            for (uint32_t i = 0; i < cur; ++i) {
                DTYPE b = biasLocal.GetValue(i);
                outLocal.SetValue(i, min_val + b);
            }

            // 回写到 y（覆盖原归一化输出）
            AscendC::DataCopy(normOut[n * C + cOffset], outLocal, cur);

            qBias.FreeTensor(biasLocal);
            qOut.FreeTensor(outLocal);

            cRemain -= cur;
            cOffset += cur;
        }
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super(ModelNew, self).__init__()
        self.gemm = torch.nn.Linear(in_features, out_features)
        # 仅复用权重/偏置参数定义，计算由自定义算子完成
        self.group_norm = torch.nn.GroupNorm(num_groups, out_features)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        # GEMM：保持与原始模型一致
        x = self.gemm(x)                     # [N, C]
        x = x.view(x.size(0), x.size(1), 1, 1).contiguous()  # [N, C, 1, 1]
        # 使用自定义 AscendC 算子完成：GroupNorm -> reduce_min(dim=1, keepdim=True) -> + bias
        # 其中 GroupNorm 的 gamma/beta 来自 self.group_norm
        return custom_ops_lib.gemm_group_norm_min_bias_add_custom(
            x, self.group_norm.weight.view(1, -1, 1, 1), self.group_norm.bias.view(1, -1, 1, 1), self.bias
        )
