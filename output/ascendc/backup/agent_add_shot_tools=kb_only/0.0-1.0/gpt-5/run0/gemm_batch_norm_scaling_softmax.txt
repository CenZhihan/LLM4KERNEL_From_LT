kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "gemm_batch_norm_scaling_softmax_custom_tiling_data.h"
#include <cmath>
#include <cfloat>

using DTYPE = float;

extern "C" __global__ __aicore__ void gemm_batch_norm_scaling_softmax_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
    GM_ADDR bn_weight, GM_ADDR bn_bias,
    GM_ADDR running_mean, GM_ADDR running_var,
    GM_ADDR scale, GM_ADDR eps,
    GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    uint32_t M = tiling_data.M();
    uint32_t N = tiling_data.N();
    uint32_t K = tiling_data.K();

    __gm__ DTYPE* xGm = reinterpret_cast<__gm__ DTYPE*>(x);
    __gm__ DTYPE* wGm = reinterpret_cast<__gm__ DTYPE*>(weight); // [N, K] row-major
    __gm__ DTYPE* bGm = reinterpret_cast<__gm__ DTYPE*>(bias);   // [N]

    __gm__ DTYPE* bnW = reinterpret_cast<__gm__ DTYPE*>(bn_weight); // gamma [N]
    __gm__ DTYPE* bnB = reinterpret_cast<__gm__ DTYPE*>(bn_bias);   // beta  [N]
    __gm__ DTYPE* runMean = reinterpret_cast<__gm__ DTYPE*>(running_mean); // [N]
    __gm__ DTYPE* runVar  = reinterpret_cast<__gm__ DTYPE*>(running_var);  // [N]
    __gm__ DTYPE* scaleGm = reinterpret_cast<__gm__ DTYPE*>(scale);        // broadcastable
    __gm__ DTYPE* epsGm   = reinterpret_cast<__gm__ DTYPE*>(eps);          // scalar tensor [1]
    __gm__ DTYPE* yGm     = reinterpret_cast<__gm__ DTYPE*>(y);            // [M, N]

    DTYPE epsVal = epsGm[0];
    DTYPE scaleVal = scaleGm[0];

    int32_t blockNum = AscendC::GetBlockNum();
    int32_t blockIdx = AscendC::GetBlockIdx();
    uint32_t rowsPerBlock = (M + (uint32_t)blockNum - 1U) / (uint32_t)blockNum;
    uint32_t rowStart = (uint32_t)blockIdx * rowsPerBlock;
    uint32_t rowEnd = rowStart + rowsPerBlock;
    if (rowEnd > M) rowEnd = M;

    for (uint32_t r = rowStart; r < rowEnd; ++r) {
        // 1) GEMM: y_tmp[n] = sum_k x[r,k] * W[n,k] + bias[n]
        for (uint32_t n = 0; n < N; ++n) {
            DTYPE acc = 0.0f;
            __gm__ DTYPE* wRow = wGm + n * K;
            __gm__ DTYPE* xRow = xGm + r * K;
            for (uint32_t k = 0; k < K; ++k) {
                acc += xRow[k] * wRow[k];
            }
            acc += bGm[n];

            // 2) BatchNorm1d (inference): ((acc - mean) / sqrt(var + eps)) * gamma + beta
            DTYPE invStd = 1.0f / sqrtf(runVar[n] + epsVal);
            DTYPE bnOut = (acc - runMean[n]) * invStd * bnW[n] + bnB[n];

            // 3) scaling: multiply
            DTYPE scaled = bnOut * scaleVal;

            // write pre-softmax value
            yGm[r * N + n] = scaled;
        }

        // 4) Softmax along last dim (N)
        DTYPE maxv = -FLT_MAX;
        for (uint32_t n = 0; n < N; ++n) {
            DTYPE v = yGm[r * N + n];
            maxv = v > maxv ? v : maxv;
        }
        DTYPE sumExp = 0.0f;
        for (uint32_t n = 0; n < N; ++n) {
            DTYPE v = yGm[r * N + n] - maxv;
            DTYPE ev = expf(v);
            yGm[r * N + n] = ev; // store temporary exp
            sumExp += ev;
        }
        DTYPE invSum = 1.0f / sumExp;
        for (uint32_t n = 0; n < N; ++n) {
            yGm[r * N + n] = yGm[r * N + n] * invSum;
        }
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, bn_eps=1e-5, bn_momentum=0.1, scale_shape=(1,)):
        super().__init__()
        self.gemm = torch.nn.Linear(in_features, out_features)
        self.bn = torch.nn.BatchNorm1d(out_features, eps=bn_eps, momentum=bn_momentum)
        self.scale = torch.nn.Parameter(torch.ones(scale_shape))

    def forward(self, x):
        # fused custom op expects:
        # x, weight, bias, bn_weight, bn_bias, running_mean, running_var, scale, eps
        eps_tensor = torch.tensor(self.bn.eps, dtype=x.dtype, device=x.device)
        return custom_ops_lib.gemm_batch_norm_scaling_softmax_custom(
            x,
            self.gemm.weight,
            self.gemm.bias,
            self.bn.weight,
            self.bn.bias,
            self.bn.running_mean,
            self.bn.running_var,
            self.scale,
            eps_tensor
        )
