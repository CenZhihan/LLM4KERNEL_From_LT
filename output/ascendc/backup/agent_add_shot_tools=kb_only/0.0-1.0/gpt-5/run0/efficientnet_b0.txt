kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "efficientnet_b0_custom_tiling_data.h"

template <typename T, typename accType>
class KernelEfficientnetB0 {
public:
    __aicore__ inline KernelEfficientnetB0() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t outter, uint32_t inner, uint32_t n, uint32_t tmpSize)
    {
        meanParams.outter = outter;  // 行数：N*C
        meanParams.inner = inner;    // 每行长度：H*W
        meanParams.n = n;            // 参与平均的有效元素数：H*W
        this->tmpSize = tmpSize;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(x));
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(y));
        // 输入全部拷入，本示例直接一次完成
        pipe.InitBuffer(inQueueX, 1, meanParams.outter * meanParams.inner * sizeof(T));
        // 输出按 32B 对齐长度分配空间
        uint32_t outAlignedElt = (meanParams.outter * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(T);
        pipe.InitBuffer(outQueueY, 1, outAlignedElt * sizeof(T));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<T> xLocal = inQueueX.AllocTensor<T>();
        AscendC::DataCopy(xLocal, xGm, meanParams.outter * meanParams.inner);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<T> xLocal = inQueueX.DeQue<T>();
        AscendC::LocalTensor<T> yLocal = outQueueY.AllocTensor<T>();
        if (tmpSize != 0) {
            pipe.InitBuffer(tmpBuf, tmpSize);
            AscendC::LocalTensor<uint8_t> tmpTensor = tmpBuf.Get<uint8_t>();
            AscendC::Mean<T, accType>(yLocal, xLocal, tmpTensor, meanParams);
        } else {
            AscendC::Mean<T, accType>(yLocal, xLocal, meanParams);
        }
        outQueueY.EnQue<T>(yLocal);
    }
    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<T> yLocal = outQueueY.DeQue<T>();
        uint32_t outAlignedElt = (meanParams.outter * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(T);
        AscendC::DataCopy(yGm, yLocal, outAlignedElt);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::GlobalTensor<T> xGm;
    AscendC::GlobalTensor<T> yGm;
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::MeanParams meanParams;
    uint32_t tmpSize = 0;
};

extern "C" __global__ __aicore__ void efficientnet_b0_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    // 仅示例 float 精度
    KernelEfficientnetB0<float, float> op;
    op.Init(x, y, tiling_data.outter, tiling_data.inner, tiling_data.n, tiling_data.tmpSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class MBConv(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, expand_ratio):
        super(MBConv, self).__init__()
        self.use_residual = (stride == 1 and in_channels == out_channels)
        hidden_dim = in_channels * expand_ratio
        if expand_ratio != 1:
            self.expand_conv = torch.nn.Sequential(
                torch.nn.Conv2d(in_channels, hidden_dim, kernel_size=1, stride=1, padding=0, bias=False),
                torch.nn.BatchNorm2d(hidden_dim),
                torch.nn.ReLU6(inplace=True)
            )
        self.depthwise_conv = torch.nn.Sequential(
            torch.nn.Conv2d(hidden_dim, hidden_dim, kernel_size=kernel_size, stride=stride, padding=(kernel_size-1)//2, groups=hidden_dim, bias=False),
            torch.nn.BatchNorm2d(hidden_dim),
            torch.nn.ReLU6(inplace=True)
        )
        self.project_conv = torch.nn.Sequential(
            torch.nn.Conv2d(hidden_dim, out_channels, kernel_size=1, stride=1, padding=0, bias=False),
            torch.nn.BatchNorm2d(out_channels)
        )

    def forward(self, x):
        identity = x
        if hasattr(self, "expand_conv"):
            x = self.expand_conv(x)
        x = self.depthwise_conv(x)
        x = self.project_conv(x)
        if self.use_residual:
            x = x + identity
        return x

class ModelNew(torch.nn.Module):
    def __init__(self, num_classes=1000):
        super(ModelNew, self).__init__()
        self.conv1 = torch.nn.Conv2d(3, 32, kernel_size=3, stride=2, padding=1, bias=False)
        self.bn1 = torch.nn.BatchNorm2d(32)
        self.blocks = torch.nn.Sequential(
            MBConv(32, 16, kernel_size=3, stride=1, expand_ratio=1),
            MBConv(16, 24, kernel_size=3, stride=2, expand_ratio=6),
            MBConv(24, 24, kernel_size=3, stride=1, expand_ratio=6),
            MBConv(24, 40, kernel_size=5, stride=2, expand_ratio=6),
            MBConv(40, 40, kernel_size=5, stride=1, expand_ratio=6),
            MBConv(40, 80, kernel_size=3, stride=2, expand_ratio=6),
            MBConv(80, 80, kernel_size=3, stride=1, expand_ratio=6),
            MBConv(80, 112, kernel_size=5, stride=1, expand_ratio=6),
            MBConv(112, 112, kernel_size=5, stride=1, expand_ratio=6),
            MBConv(112, 192, kernel_size=5, stride=2, expand_ratio=6),
            MBConv(192, 192, kernel_size=5, stride=1, expand_ratio=6),
            MBConv(192, 192, kernel_size=5, stride=1, expand_ratio=6),
            MBConv(192, 320, kernel_size=3, stride=1, expand_ratio=6),
        )
        self.conv2 = torch.nn.Conv2d(320, 1280, kernel_size=1, stride=1, padding=0, bias=False)
        self.bn2 = torch.nn.BatchNorm2d(1280)
        self.fc = torch.nn.Linear(1280, num_classes)

    def forward(self, x):
        x = torch.nn.functional.relu(self.bn1(self.conv1(x)))
        x = self.blocks(x)
        x = torch.nn.functional.relu(self.bn2(self.conv2(x)))
        # 用自定义 AscendC GAP 内核替换自适应平均池化 + 展平
        x = custom_ops_lib.efficientnet_b0_custom(x)  # [N, C]
        x = self.fc(x)
        return x
'''
