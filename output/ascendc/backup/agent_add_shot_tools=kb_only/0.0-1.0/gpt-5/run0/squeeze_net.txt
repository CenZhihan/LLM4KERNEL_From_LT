kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "squeeze_net_custom_tiling_data.h"

// 参考：4.3 基础 API (TPipe/TQue) 与 5.x 高阶 API 示例的流水线写法
// 功能：按通道维(1维)拼接两个 NCHW Tensor。由于内存为 N*C*H*W 排列，
//       cat(dim=1) 等价于将 x 的整块数据拷贝到 z 起始，再将 y 紧接其后拷贝。

constexpr int32_t BUFFER_NUM = 2; // 每个队列的 tensor 数

class KernelConcatC {
public:
    __aicore__ inline KernelConcatC() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t lenX, uint32_t lenY, uint32_t tileNum)
    {
        // 多核切分：按元素线性空间做 1D 切分
        this->lenX = lenX;
        this->lenY = lenY;
        this->tileNum = tileNum;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->blockLenX = lenX / blockNum;
        this->blockLenY = lenY / blockNum;

        // 每块再按 tile+双缓冲切分
        this->tileLenX = this->blockLenX / (this->tileNum * BUFFER_NUM);
        this->tileLenY = this->blockLenY / (this->tileNum * BUFFER_NUM);

        // 绑定本 block 的全局内存窗口
        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x + blockIdx * this->blockLenX, this->blockLenX);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y + blockIdx * this->blockLenY, this->blockLenY);
        // 输出 block 长度是两者之和
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z*)z + blockIdx * (this->blockLenX + this->blockLenY),
                            this->blockLenX + this->blockLenY);

        // 初始化队列与缓冲
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLenX * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, this->tileLenY * sizeof(DTYPE_Y));
        // 分别为 x/y 路径准备输出队列，避免大小不一致
        pipe.InitBuffer(outQueueZX, BUFFER_NUM, this->tileLenX * sizeof(DTYPE_Z));
        pipe.InitBuffer(outQueueZY, BUFFER_NUM, this->tileLenY * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        // 先拷贝 x -> z 的前半部分
        int32_t loopX = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopX; ++i) {
            CopyInX(i);
            ComputeX(i);
            CopyOutX(i);
        }
        // 再拷贝 y -> z 的后半部分
        int32_t loopY = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopY; ++i) {
            CopyInY(i);
            ComputeY(i);
            CopyOutY(i);
        }
    }

private:
    __aicore__ inline void CopyInX(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLenX], this->tileLenX);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeX(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZX.AllocTensor<DTYPE_Z>();
        // 直接搬运（数据类型一致）
        AscendC::DataCopy(zLocal, xLocal, this->tileLenX);
        outQueueZX.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOutX(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZX.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->tileLenX], zLocal, this->tileLenX);
        outQueueZX.FreeTensor(zLocal);
    }

    __aicore__ inline void CopyInY(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(yLocal, yGm[progress * this->tileLenY], this->tileLenY);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void ComputeY(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZY.AllocTensor<DTYPE_Z>();
        AscendC::DataCopy(zLocal, yLocal, this->tileLenY);
        outQueueZY.EnQue<DTYPE_Z>(zLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOutY(int32_t progress)
    {
        // y 的写入偏移需要从 block 的后半部分开始
        uint32_t yBase = this->blockLenX; // 先写完 x，再接着写 y
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZY.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[yBase + progress * this->tileLenY], zLocal, this->tileLenY);
        outQueueZY.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;

    uint32_t lenX;
    uint32_t lenY;
    uint32_t blockLenX;
    uint32_t blockLenY;
    uint32_t tileNum;
    uint32_t tileLenX;
    uint32_t tileLenY;
};

extern "C" __global__ __aicore__ void squeeze_net_custom(GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConcatC op;
    op.Init(x, y, z, tiling_data.lenX, tiling_data.lenY, tiling_data.tileNum);
    op.Process();
}
"""

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

# 使用 AscendC 自定义算子替换 FireModule 中的通道拼接（cat dim=1）

class FireModuleNew(nn.Module):
    def __init__(self, in_channels, squeeze_channels, expand1x1_channels, expand3x3_channels):
        super(FireModuleNew, self).__init__()
        self.squeeze = nn.Conv2d(in_channels, squeeze_channels, kernel_size=1)
        self.squeeze_activation = nn.ReLU(inplace=True)
        self.expand1x1 = nn.Conv2d(squeeze_channels, expand1x1_channels, kernel_size=1)
        self.expand1x1_activation = nn.ReLU(inplace=True)
        self.expand3x3 = nn.Conv2d(squeeze_channels, expand3x3_channels, kernel_size=3, padding=1)
        self.expand3x3_activation = nn.ReLU(inplace=True)

    def forward(self, x):
        x = self.squeeze_activation(self.squeeze(x))
        y1 = self.expand1x1_activation(self.expand1x1(x))
        y3 = self.expand3x3_activation(self.expand3x3(x))
        # 使用自定义 AscendC 算子进行拼接
        return custom_ops_lib.squeeze_net_custom(y1, y3)

class ModelNew(nn.Module):
    def __init__(self, num_classes=1000):
        super(ModelNew, self).__init__()
        self.features = nn.Sequential(
            nn.Conv2d(3, 96, kernel_size=7, stride=2),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2, ceil_mode=True),
            FireModuleNew(96, 16, 64, 64),
            FireModuleNew(128, 16, 64, 64),
            FireModuleNew(128, 32, 128, 128),
            nn.MaxPool2d(kernel_size=3, stride=2, ceil_mode=True),
            FireModuleNew(256, 32, 128, 128),
            FireModuleNew(256, 48, 192, 192),
            FireModuleNew(384, 48, 192, 192),
            FireModuleNew(384, 64, 256, 256),
            nn.MaxPool2d(kernel_size=3, stride=2, ceil_mode=True),
            FireModuleNew(512, 64, 256, 256),
        )
        self.classifier = nn.Sequential(
            nn.Dropout(p=0.0),
            nn.Conv2d(512, num_classes, kernel_size=1),
            nn.ReLU(inplace=True),
            nn.AdaptiveAvgPool2d((1, 1))
        )

    def forward(self, x):
        x = self.features(x)
        x = self.classifier(x)
        return torch.flatten(x, 1)

# 输入与初始化辅助函数（与原模型一致）
batch_size = 64
input_channels = 3
height = 512
width = 512
num_classes = 1000

def get_inputs():
    return [torch.rand(batch_size, input_channels, height, width)]

def get_init_inputs():
    return [num_classes]
