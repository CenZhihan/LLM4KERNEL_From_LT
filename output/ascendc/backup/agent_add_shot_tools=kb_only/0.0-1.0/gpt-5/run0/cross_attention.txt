host_operator_src="""
#include "register/op_def_registry.h"

// 参考资料：
// - 5.2.2.1.1 Instructions for Use（Matmul tiling 获取流程）
// - 5.2.1.22 IterateBatch（NORMAL 布局批处理）
// - 5.2.1.24 IterateNBatch / SetBatchInfoForNormal（设置 NORMAL 布局 M/N/K 轴与 batch 数）
// 这里在 TilingFunc 中为 NORMAL（BMNK）布局的批量 Matmul 设置 tiling，计算 scores = Q @ KV^T

namespace optiling {

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 获取输入形状：q:[B, Lq, D], kv:[B, Lkv, D]
    auto q_shape = context->GetInputShape(0)->GetOriginShape();
    auto kv_shape = context->GetInputShape(1)->GetOriginShape();

    int64_t B   = q_shape.GetDim(0);
    int64_t Lq  = q_shape.GetDim(1);
    int64_t D   = q_shape.GetDim(2);
    int64_t Lkv = kv_shape.GetDim(1);

    // 平台与 tiling 对象
    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    matmul_tiling::MultiCoreMatmulTiling tiling(ascendcPlatform);

    // 设置并行核数（示例设置1，由 tiling 自行决定也可）
    tiling.SetDim(1);

    // 设置 A/B/C 数据类型与位置、格式（ND，DT_FLOAT）
    tiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling.SetBiasType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);

    // 单次 Matmul 形状（NORMAL: M, N, K）
    // A: [M=Lq, K=D], B: [K=D, N=Lkv]（在 kernel 中对 B 设置转置）
    tiling.SetShape(static_cast<int32_t>(Lq), static_cast<int32_t>(Lkv), static_cast<int32_t>(D));
    tiling.SetOrgShape(static_cast<int32_t>(Lq), static_cast<int32_t>(Lkv), static_cast<int32_t>(D));

    // NORMAL 布局设置 batch 信息：两侧 batch 都为 B
    // 参考 5.2.1.24 中 SetBatchInfoForNormal
    tiling.SetBatchInfoForNormal(static_cast<int32_t>(B), static_cast<int32_t>(B),
                                 static_cast<int32_t>(Lq), static_cast<int32_t>(Lkv), static_cast<int32_t>(D));

    // 可用空间（默认全可用）
    tiling.SetBufferSpace(-1, -1, -1);

    // 输出 tiling 数据到 raw buffer（TCubeTiling）
    optiling::TCubeTiling tilingData;
    int ret = tiling.GetTiling(tilingData);
    if (ret == -1) {
        return ge::GRAPH_FAILED;
    }

    context->SetBlockDim(tiling.GetCoreNum());
    tilingData.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tilingData.GetDataSize());

    size_t* workspaces = context->GetWorkspaceSizes(1);
    workspaces[0] = 0; // 无额外 workspace

    return ge::GRAPH_SUCCESS;
}

} // namespace optiling


namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    // 输出 scores 形状为 [B, Lq, Lkv]
    const gert::Shape* q_shape = context->GetInputShape(0);
    const gert::Shape* kv_shape = context->GetInputShape(1);

    gert::Shape* out_shape = context->GetOutputShape(0);

    std::vector<int64_t> o = {
        q_shape->GetDim(0), // B
        q_shape->GetDim(1), // Lq
        kv_shape->GetDim(1) // Lkv
    };
    out_shape->SetDim(o);

    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto dt = context->GetInputDataType(0);
    context->SetOutputDataType(0, dt);
    return ge::GRAPH_SUCCESS;
}
}


namespace ops {
class CrossAttentionCustom : public OpDef {
public:
    explicit CrossAttentionCustom(const char* name) : OpDef(name)
    {
        this->Input("q")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("kv")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("scores")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);

        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(CrossAttentionCustom);
}

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

// 参考资料：
// - 5.2.1.1 Matmul 使用说明（C = A x B [+ Bias]）
// - 5.2.1.22 IterateBatch（NORMAL 布局批处理执行）
// - 5.2.2.1.1 获取 TCubeTiling 并在 kernel 中使用

extern "C" __global__ __aicore__ void cross_attention_custom(GM_ADDR qGM, GM_ADDR kvGM, GM_ADDR scoresGM, GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    // 从 GM 读取 TCubeTiling
    TCubeTiling tiling;
    {
        auto tempTilingGM = (__gm__ uint32_t*)tilingGM;
        auto tempTiling = (uint32_t*)&tiling;
        for (int i = 0; i < (int)(sizeof(TCubeTiling) / sizeof(uint32_t)); ++i, ++tempTilingGM, ++tempTiling) {
            *tempTiling = *tempTilingGM;
        }
    }

    // 定义 Matmul 类型：NORMAL 布局，float
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> aType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, float, true,  LayoutMode::NORMAL> bType; // B 设置可转置
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> cType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, float> biasType;

    // 初始化 GM Tensor 尺寸（NORMAL 示例）
    AscendC::GlobalTensor<float> aGlobal;
    AscendC::GlobalTensor<float> bGlobal;
    AscendC::GlobalTensor<float> cGlobal;

    int32_t sizeA = tiling.ALayoutInfoB * tiling.singleCoreM * tiling.singleCoreK * sizeof(float);
    int32_t sizeB = tiling.BLayoutInfoB * tiling.singleCoreK * tiling.singleCoreN * sizeof(float);
    int32_t sizeC = tiling.CLayoutInfoB * tiling.singleCoreM * tiling.singleCoreN * sizeof(float);

    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(qGM), sizeA);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(kvGM), sizeB);
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(scoresGM), sizeC);

    tiling.shareMode = 0;
    tiling.shareL1Size = 512 * 1024;
    tiling.shareL0CSize = 128 * 1024;
    tiling.shareUbSize = 0;

    int offset_a = 0, offset_b = 0, offset_c = 0;

    AscendC::GlobalTensor<float> gm_a;
    gm_a.SetGlobalBuffer(const_cast<__gm__ float*>(aGlobal[offset_a].GetPhyAddr()),
                         tiling.singleCoreM * tiling.singleCoreK);

    AscendC::GlobalTensor<float> gm_b;
    gm_b.SetGlobalBuffer(const_cast<__gm__ float*>(bGlobal[offset_b].GetPhyAddr()),
                         tiling.singleCoreK * tiling.singleCoreN);

    AscendC::GlobalTensor<float> gm_c;
    gm_c.SetGlobalBuffer(const_cast<__gm__ float*>(cGlobal[offset_c].GetPhyAddr()),
                         tiling.singleCoreM * tiling.singleCoreN);

    // 创建 Matmul 实例
    constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);
    matmul::Matmul<aType, bType, cType, biasType, MM_CFG> mm1;
    AscendC::TPipe pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm1);

    mm1.Init(&tiling);

    // 设置 A/B，B 侧按需转置（这里为 KV^T）
    uint32_t isTransposeAIn = 0;
    uint32_t isTransposeBIn = 1;
    mm1.SetTensorA(gm_a, isTransposeAIn);
    mm1.SetTensorB(gm_b, isTransposeBIn);
    mm1.SetWorkspace(workspaceGM, 0);

    // 执行批量 Matmul：scores = Q @ KV^T
    uint32_t batchA = tiling.ALayoutInfoB; // B
    uint32_t batchB = tiling.BLayoutInfoB; // B
    mm1.IterateBatch(gm_c, batchA, batchB, false);
}

at::Tensor cross_attention_custom_impl_npu(const at::Tensor& q, const at::Tensor& kv) {
    // q:[B,Lq,D], kv:[B,Lkv,D] -> scores:[B,Lq,Lkv]
    auto B = q.sizes()[0];
    auto Lq = q.sizes()[1];
    auto Lkv = kv.sizes()[1];

    at::Tensor scores = at::empty({B, Lq, Lkv}, q.options());
    EXEC_NPU_CMD(aclnnCrossAttentionCustom, q, kv, scores);
    return scores;

model_src='''
import math
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义 AscendC 内核加速 Cross-Attention 的主 GEMM（scores = Q @ K^T）。
    余下的 softmax 与与 V 的乘法仍由 PyTorch 端完成，确保数值稳定性与易用性。
    """
    def __init__(self, d_model=512, num_heads=8):
        super().__init__()
        self.d_model = d_model
        self.num_heads = num_heads

    def forward(self, q, kv):
        # 自定义核：scores = q @ kv^T
        scores = custom_ops_lib.cross_attention_custom(q, kv)   # [B, Lq, Lkv]
        scores = scores / math.sqrt(self.d_model)
        attn = torch.softmax(scores, dim=-1)                    # [B, Lq, Lkv]
        out = torch.matmul(attn, kv)                            # [B, Lq, D]
        return out
'''
