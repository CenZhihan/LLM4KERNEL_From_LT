kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_pointwise_2d_custom_tiling_data.h"

// 本算子实现1x1点卷积：在Python侧将输入转为NHWC并展平成[M, Cin]，权重为[Cout, Cin]（按原始Conv2d存储），
// 在核内做简化的分块计算。参照Ascend C API的TPipe/TQue及一元算子样例进行队列/拷贝组织。

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 2;

class KernelPWConv1x1 {
public:
    __aicore__ inline KernelPWConv1x1() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR z,
                                uint32_t n, uint32_t h, uint32_t wimg,
                                uint32_t cin, uint32_t cout,
                                uint32_t hasBias, uint32_t tileNum)
    {
        // 输入在Python端已变换为NHWC并重排为二维: X2D shape = [M, Cin], M = N*H*W
        this->N = n;
        this->H = h;
        this->W = wimg;
        this->Cin = cin;
        this->Cout = cout;
        this->hasBias = hasBias;
        this->tileNum = tileNum;

        this->M = N * H * W;

        // 均分到各block
        this->blockM = this->M / AscendC::GetBlockNum();
        this->tileM = this->blockM / this->tileNum / BUFFER_NUM;

        // 设定GM缓冲（按二维扁平内存设置长度）
        xGm.SetGlobalBuffer((__gm__ DTYPE*)x + (size_t)AscendC::GetBlockIdx() * blockM * Cin, blockM * Cin);
        wGm.SetGlobalBuffer((__gm__ DTYPE*)w, (size_t)Cout * Cin);
        zGm.SetGlobalBuffer((__gm__ DTYPE*)z + (size_t)AscendC::GetBlockIdx() * blockM * Cout, blockM * Cout);
        if (hasBias) {
            bGm.SetGlobalBuffer((__gm__ DTYPE*)b, Cout);
        }

        // 初始化队列
        // X分块: tileM * Cin
        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileM * Cin * sizeof(DTYPE));
        // Z分块: tileM * Cout
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, tileM * Cout * sizeof(DTYPE));
        // 权重和bias常驻UB一次性拷贝
        pipe.InitBuffer(weightQueue, 1, (size_t)Cout * Cin * sizeof(DTYPE));
        if (hasBias) {
            pipe.InitBuffer(biasQueue, 1, Cout * sizeof(DTYPE));
        }
    }

    __aicore__ inline void Process()
    {
        // 将权重、偏置拷贝至本地
        AscendC::LocalTensor<DTYPE> wLocal = weightQueue.AllocTensor<DTYPE>();
        AscendC::DataCopy(wLocal, wGm, (uint32_t)((size_t)Cout * Cin));

        AscendC::LocalTensor<DTYPE> bLocal;
        if (hasBias) {
            bLocal = biasQueue.AllocTensor<DTYPE>();
            AscendC::DataCopy(bLocal, bGm, Cout);
        }

        // 按tile遍历
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i, wLocal, bLocal);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        // 拷入X分块
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        uint32_t offset = progress * tileM * Cin;
        AscendC::DataCopy(xLocal, xGm[offset], tileM * Cin);
        inQueueX.EnQue(xLocal);

        // 预分配Z分块
        AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.AllocTensor<DTYPE>();
        outQueueZ.EnQue<DTYPE>(zLocal); // 先入队占位，Compute中DeQue出来写入
    }

    __aicore__ inline void Compute(int32_t /*progress*/,
                                   AscendC::LocalTensor<DTYPE> &wLocal,
                                   AscendC::LocalTensor<DTYPE> &bLocal)
    {
        // 取出X/Z分块本地tensor
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.DeQue<DTYPE>();

        // 朴素GEMM: [tileM, Cin] * [Cout, Cin]^T -> [tileM, Cout]
        // 内存布局:
        //   xLocal: 行主序, 行长 Cin
        //   wLocal: 连续存储为 [Cout, Cin]，权重按 oc*C_in + k 访问
        //   zLocal: 行主序, 行长 Cout
        for (uint32_t m = 0; m < tileM; ++m) {
            for (uint32_t oc = 0; oc < Cout; ++oc) {
                DTYPE acc = (hasBias ? bLocal[oc] : (DTYPE)0);
                for (uint32_t k = 0; k < Cin; ++k) {
                    DTYPE xv = xLocal[m * Cin + k];
                    DTYPE wv = wLocal[oc * Cin + k];
                    acc = acc + xv * wv;
                }
                zLocal[m * Cout + oc] = acc;
            }
        }

        // 回填到队列
        outQueueZ.EnQue<DTYPE>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.DeQue<DTYPE>();
        uint32_t offset = progress * tileM * Cout;
        AscendC::DataCopy(zGm[offset], zLocal, tileM * Cout);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    // 维度/参数
    uint32_t N, H, W;
    uint32_t Cin, Cout;
    uint32_t M;          // N*H*W
    uint32_t blockM;
    uint32_t tileM;
    uint32_t tileNum;
    uint32_t hasBias;

    // 管道与队列
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> weightQueue;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> biasQueue;

    // GM张量
    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> wGm;
    AscendC::GlobalTensor<DTYPE> bGm;
    AscendC::GlobalTensor<DTYPE> zGm;
};

extern "C" __global__ __aicore__ void conv_pointwise_2d_custom(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelPWConv1x1 op;
    op.Init(x, w, b, z,
            tilingData.n(), tilingData.h(), tilingData.w(),
            tilingData.cin(), tilingData.cout(),
            tilingData.hasBias(), tilingData.tileNum());
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

static inline at::Tensor to_channels_last_2d(const at::Tensor& t) {
    // NHWC contiguous
    if (!t.is_contiguous(at::MemoryFormat::ChannelsLast)) {
        return t.contiguous(at::MemoryFormat::ChannelsLast);
    }
    return t;
}

at::Tensor conv_pointwise_2d_custom_impl_npu(const at::Tensor& x, const at::Tensor& w, c10::optional<at::Tensor> b_opt) {
    // 预处理：
    // 1) 将输入x转为channels_last并展平成[M, Cin]
    // 2) 将权重w视为[Cout, Cin, 1, 1]，在核内按[Cout, Cin]访问
    TORCH_CHECK(x.dim() == 4, "x must be 4D NCHW");
    TORCH_CHECK(w.dim() >= 2, "w must be at least 2D [Cout, Cin, 1, 1]");

    auto x_cl = to_channels_last_2d(x);
    int64_t N = x_cl.size(0);
    int64_t Cin = x_cl.size(1); // 原NCHW的C
    int64_t H = x_cl.size(2);
    int64_t W = x_cl.size(3);
    int64_t Cout = w.size(0);

    // 将NHWC视图重塑为二维[M, Cin]
    auto x2d = x_cl.view({N * H * W, Cin}).contiguous();

    // 权重按[Cout, Cin, 1, 1]展平为二维[Cout, Cin]
    at::Tensor w2d = w.view({Cout, Cin}).contiguous();

    // 输出二维[M, Cout]
    at::Tensor z2d = at::empty({N * H * W, Cout}, x.options());

    // 偏置
    at::Tensor b = b_opt.has_value() ? b_opt.value() : at::Tensor();
    if (b.defined()) {
        TORCH_CHECK(b.numel() == Cout, "bias length must equal Cout");
    }

    if (b.defined()) {
        EXEC_NPU_CMD(aclnnConvPointwise2dCustom, x2d, w2d, b, z2d);
    } else {
        // 传入空Tensor作为可选参数
        EXEC_NPU_CMD(aclnnConvPointwise2dCustom, x2d, w2d, at::Tensor(), z2d);
    }

    // 还原为NHWC再转回NCHW
    auto z_nhwc = z2d.view({N, H, W, Cout}).contiguous();
    auto z_nchw = z_nhwc.permute({0, 3, 1, 2}).contiguous();
    return z_nchw;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("conv_pointwise_2d_custom", &conv_pointwise_2d_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("conv_pointwise_2d_custom", &conv_pointwise_2d_custom_impl_npu, "pointwise 1x1 conv2d");
}
