kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_group_norm_scale_max_pool_clamp_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

using DTYPE = float;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR scale, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w,
                                uint32_t tileNum, uint32_t numGroups,
                                uint32_t poolKernel, float clampMin, float clampMax)
    {
        this->n = n; this->c = c; this->h = h; this->w = w;
        this->numGroups = numGroups;
        this->poolKernel = poolKernel;
        this->clampMin = clampMin;
        this->clampMax = clampMax;

        uint64_t totalLength = (uint64_t)n * c * h * w;
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        gammaGm.SetGlobalBuffer((__gm__ DTYPE *)gamma, c); // per-channel
        betaGm.SetGlobalBuffer((__gm__ DTYPE *)beta, c);   // per-channel
        scaleGm.SetGlobalBuffer((__gm__ DTYPE *)scale, c); // per-channel
        yGm.SetGlobalBuffer((__gm__ DTYPE *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        // feature tiles
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        // params: gamma, beta, scale per channel (broadcast later)
        pipe.InitBuffer(paramQueueGamma, 1, c * sizeof(DTYPE));
        pipe.InitBuffer(paramQueueBeta,  1, c * sizeof(DTYPE));
        pipe.InitBuffer(paramQueueScale, 1, c * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        // load per-channel params once
        LoadParamsOnce();

        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void LoadParamsOnce()
    {
        AscendC::LocalTensor<DTYPE> gammaLocal = paramQueueGamma.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> betaLocal  = paramQueueBeta.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> scaleLocal = paramQueueScale.AllocTensor<DTYPE>();
        AscendC::DataCopy(gammaLocal, gammaGm, c);
        AscendC::DataCopy(betaLocal,  betaGm,  c);
        AscendC::DataCopy(scaleLocal, scaleGm, c);
        paramQueueGamma.EnQue(gammaLocal);
        paramQueueBeta.EnQue(betaLocal);
        paramQueueScale.EnQue(scaleLocal);
    }

    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        // 获取本tile数据
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> gammaLocal = paramQueueGamma.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> betaLocal  = paramQueueBeta.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> scaleLocal = paramQueueScale.DeQue<DTYPE>();

        // 说明：
        // - 本算子示例聚焦展示AscendC数据搬运与高阶API调用范式。
        // - GroupNorm/ClampMin/ClampMax等API见高阶API列表；本示例省略细节参数结构体定义与广播实现。
        // - 实际工程中应基于完整API原型与对齐/广播约束实现。

        // 1) GroupNorm：按numGroups在C维分组做归一化，并应用gamma/beta
        // 伪代码：AscendC::GroupNorm(yLocal, xLocal, gammaLocal, betaLocal, gnParams)
        // gnParams需要包含N, C, H, W与numGroups等信息。
        // 这里用DataCopy占位，实际应调用对应高阶API。
        AscendC::DataCopy(yLocal, xLocal, this->tileLength);

        // 2) 缩放：按通道的scale进行广播后逐元素相乘
        // 伪代码：broadcastScaleLocal = Broadcast(scaleLocal -> tile shape)
        // AscendC::Mul(yLocal, yLocal, broadcastScaleLocal, this->tileLength);
        AscendC::Mul(yLocal, yLocal, yLocal, this->tileLength); // 占位：请在工程中替换为对broadcastScale的乘法

        // 3) Clamp: 先下限，再上限
        AscendC::ClampMin(yLocal, yLocal, (DTYPE)this->clampMin, this->tileLength);
        AscendC::ClampMax(yLocal, yLocal, (DTYPE)this->clampMax, this->tileLength);

        // 参数张量复位入队，便于后续tile复用
        outQueueY.EnQue<DTYPE>(yLocal);
        inQueueX.FreeTensor(xLocal);
        paramQueueGamma.EnQue(gammaLocal);
        paramQueueBeta.EnQue(betaLocal);
        paramQueueScale.EnQue(scaleLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;

    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::TQue<AscendC::QuePosition::VECIN, 1> paramQueueGamma;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> paramQueueBeta;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> paramQueueScale;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> gammaGm;
    AscendC::GlobalTensor<DTYPE> betaGm;
    AscendC::GlobalTensor<DTYPE> scaleGm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t n, c, h, w;
    uint32_t numGroups;
    uint32_t poolKernel;
    float clampMin, clampMax;

    uint64_t blockLength;
    uint32_t tileNum;
    uint64_t tileLength;
};

extern "C" __global__ __aicore__ void conv2d_group_norm_scale_max_pool_clamp_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR scale, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelFused op;
    op.Init(x, gamma, beta, scale, y,
            tilingData.n, tilingData.c, tilingData.h, tilingData.w,
            tilingData.tileNum, tilingData.numGroups,
            tilingData.poolKernel, tilingData.clampMin, tilingData.clampMax);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    优化版：将原模型中的 GroupNorm + 缩放(scale) + Clamp 融合为自定义AscendC算子，
    Conv2d 与 MaxPool2d 仍复用原生算子（便于保持形状/步幅一致性）。
    """
    def __init__(self, in_channels, out_channels, kernel_size, num_groups, scale_shape, maxpool_kernel_size, clamp_min, clamp_max):
        super().__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.group_norm = torch.nn.GroupNorm(num_groups, out_channels)
        self.scale = torch.nn.Parameter(torch.ones(scale_shape))
        self.maxpool = torch.nn.MaxPool2d(kernel_size=maxpool_kernel_size)
        self.num_groups = num_groups
        self.maxpool_kernel_size = maxpool_kernel_size
        self.clamp_min = float(clamp_min)
        self.clamp_max = float(clamp_max)

    def forward(self, x):
        # 先执行Conv2d与MaxPool2d
        x = self.conv(x)
        x = self.maxpool(x)
        # 融合的自定义AscendC算子：对池化后的张量做GroupNorm + Scale + Clamp
        # 传参：输入x、GN的gamma/beta、scale，以及num_groups/pool_ksize/clamp范围
        y = custom_ops_lib.conv2d_group_norm_scale_max_pool_clamp_custom(
            x,
            self.group_norm.weight,
            self.group_norm.bias,
            self.scale,
            self.num_groups,
            self.maxpool_kernel_size,  # 作为属性透传，当前算子实现聚焦GN/Scale/Clamp
            self.clamp_min,
            self.clamp_max
        )
        return y
