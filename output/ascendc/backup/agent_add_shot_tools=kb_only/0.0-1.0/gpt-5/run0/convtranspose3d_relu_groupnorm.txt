static ge::graphStatus TilingFunc(gert::TilingContext* context)

    Convtranspose3dReluGroupnormCustomTilingData tiling;

    const auto* in0_shape_ptr = context->GetInputShape(0);
    const auto* gamma_shape_ptr = context->GetInputShape(1);
    if (in0_shape_ptr == nullptr || gamma_shape_ptr == nullptr) {
        return ge::GRAPH_FAILED;
    }

    const auto in0_origin = in0_shape_ptr->GetOriginShape();
    const auto gamma_origin = gamma_shape_ptr->GetOriginShape();

    uint64_t totalElems = GetShapeSize(in0_origin);
    uint32_t hLength = static_cast<uint32_t>(GetShapeSize(gamma_origin)); // channels
    if (hLength == 0) {
        return ge::GRAPH_FAILED;
    }
    uint32_t bs = static_cast<uint32_t>(totalElems / hLength);

    // Flatten to [B, S, H] = [1, bs, hLength]
    tiling.set_bLength(1);
    tiling.set_sLength(bs);
    tiling.set_hLength(hLength);
    tiling.set_alpha(0.0f);        // DeepNorm: LayerNorm(0*X + SubLayer(X)) = LayerNorm(SubLayer(X))
    tiling.set_epsilon(1e-5f);

    context->SetBlockDim(BLOCK_DIM);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *workspace = context->GetWorkspaceSizes(1);
    workspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "convtranspose3d_relu_groupnorm_custom_tiling_data.h"

// 说明：本自定义核使用 AscendC::DeepNorm API 实现通道维 LayerNorm（作为 GroupNorm 的高效替代）。
// 输入张量视为 [B,S,H] 排布，其中 H=通道数，B*S=批量*空间体素数。
// 为实现 LayerNorm(y) = DeepNorm(alpha*X + SubLayer(X))，取 alpha=0 且 X=SubLayer(X)=输入 y。

class KernelChannelLayerNorm {
public:
    __aicore__ inline KernelChannelLayerNorm() {}

    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR gammaGm, GM_ADDR betaGm,
                                GM_ADDR zGm,
                                uint32_t bLength, uint32_t sLength, uint32_t hLength,
                                float alpha, float epsilon)
    {
        this->bLength = bLength;
        this->sLength = sLength;
        this->hLength = hLength;
        this->alpha = static_cast<float>(alpha);
        this->epsilon = static_cast<float>(epsilon);

        bshLength = bLength * sLength * hLength;
        bsLength = bLength * sLength;

        inputGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(xGm), bshLength);
        gammaGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(gammaGm), hLength);
        betaGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(betaGm), hLength);
        outputGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(zGm), bshLength);

        pipe.InitBuffer(inQueueX, 1, bshLength * sizeof(float));
        pipe.InitBuffer(inQueueGamma, 1, hLength * sizeof(float));
        pipe.InitBuffer(inQueueBeta, 1, hLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, bshLength * sizeof(float));
        // 为 DeepNorm 输出均值/方差分配队列（本算子不回传到 GM）
        pipe.InitBuffer(tmpQueueMean, 1, bsLength * sizeof(float));
        pipe.InitBuffer(tmpQueueRstd, 1, bsLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.AllocTensor<float>();
        AscendC::LocalTensor<float> betaLocal = inQueueBeta.AllocTensor<float>();

        AscendC::DataCopy(xLocal, inputGlobal, bshLength);
        AscendC::DataCopy(gammaLocal, gammaGlobal, hLength);
        AscendC::DataCopy(betaLocal, betaGlobal, hLength);

        inQueueX.EnQue(xLocal);
        inQueueGamma.EnQue(gammaLocal);
        inQueueBeta.EnQue(betaLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.DeQue<float>();
        AscendC::LocalTensor<float> betaLocal = inQueueBeta.DeQue<float>();

        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> meanLocal = tmpQueueMean.AllocTensor<float>();
        AscendC::LocalTensor<float> rstdLocal = tmpQueueRstd.AllocTensor<float>();

        // 构造 DeepNorm tiling
        DeepNormTiling tiling;
        tiling.bLength = bLength;
        tiling.sLength = sLength;
        tiling.hLength = hLength;

        // 使用 DeepNorm 完成 LayerNorm：alpha=0，srcLocal 与 gxLocal 复用输入
        AscendC::DeepNorm<float, false, false>(
            zLocal, meanLocal, rstdLocal,
            xLocal,            // srcLocal
            xLocal,            // gxLocal (SubLayer(X))
            betaLocal, gammaLocal,
            static_cast<float>(alpha),
            static_cast<float>(epsilon),
            tiling
        );

        // 释放输入本地张量
        inQueueX.FreeTensor(xLocal);
        inQueueGamma.FreeTensor(gammaLocal);
        inQueueBeta.FreeTensor(betaLocal);

        // 推送输出与临时张量
        outQueueZ.EnQue(zLocal);
        tmpQueueMean.EnQue(meanLocal);
        tmpQueueRstd.EnQue(rstdLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(outputGlobal, zLocal, bshLength);
        outQueueZ.FreeTensor(zLocal);

        // 丢弃均值/方差本地张量（不写回 GM）
        AscendC::LocalTensor<float> meanLocal = tmpQueueMean.DeQue<float>();
        AscendC::LocalTensor<float> rstdLocal = tmpQueueRstd.DeQue<float>();
        tmpQueueMean.FreeTensor(meanLocal);
        tmpQueueRstd.FreeTensor(rstdLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1>  inQueueX, inQueueGamma, inQueueBeta;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueZ;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> tmpQueueMean, tmpQueueRstd;

    AscendC::GlobalTensor<float> inputGlobal;
    AscendC::GlobalTensor<float> gammaGlobal;
    AscendC::GlobalTensor<float> betaGlobal;
    AscendC::GlobalTensor<float> outputGlobal;

    uint32_t bLength;
    uint32_t sLength;
    uint32_t hLength;
    uint32_t bshLength;
    uint32_t bsLength;
    float alpha;
    float epsilon;
};

extern "C" __global__ __aicore__ void convtranspose3d_relu_groupnorm_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelChannelLayerNorm op;
    op.Init(x, gamma, beta, z,
            tiling_data.bLength, tiling_data.sLength, tiling_data.hLength,
            tiling_data.alpha, tiling_data.epsilon);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义 AscendC 核将 ReLU+GroupNorm 阶段替换为 DeepNorm(=LayerNorm) 的高效实现。
    ConvTranspose3d 仍使用原生算子。
    """
    def __init__(self, in_channels, out_channels, kernel_size, groups, bias=False):
        super().__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size, bias=bias)
        self.relu = torch.nn.ReLU()
        # 仅复用 GroupNorm 的缩放/偏置参数作为 gamma/beta
        self.group_norm = torch.nn.GroupNorm(num_groups=groups, num_channels=out_channels)

    def forward(self, x):
        y = self.conv_transpose(x)
        y = self.relu(y)
        # 调用自定义核完成归一化（DeepNorm 通道层归一化）
        return custom_ops_lib.convtranspose3d_relu_groupnorm_custom(
            y, self.group_norm.weight, self.group_norm.bias
        )
