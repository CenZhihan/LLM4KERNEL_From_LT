kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_batch_norm_scaling_custom_tiling_data.h"

// 参考高阶/基础API样例，采用块内分片 + 环形队列写回
constexpr int32_t BUFFER_NUM = 2;

using DTYPE = float;

class KernelConvBnScale {
public:
    __aicore__ inline KernelConvBnScale() {}

    __aicore__ inline void Init(GM_ADDR x,
                                GM_ADDR weight,
                                GM_ADDR bias,
                                GM_ADDR bn_gamma,
                                GM_ADDR bn_beta,
                                GM_ADDR bn_mean,
                                GM_ADDR bn_var,
                                GM_ADDR scaling,
                                GM_ADDR y,
                                const optiling::Conv2dBatchNormScalingCustomTilingData &t) {
        // 读取tiling信息
        N   = t.get_N();
        Cin = t.get_Cin();
        Hin = t.get_Hin();
        Win = t.get_Win();
        Cout= t.get_Cout();
        Kh  = t.get_Kh();
        Kw  = t.get_Kw();
        Hout= t.get_Hout();
        Wout= t.get_Wout();
        eps = t.get_eps();

        totalLength = t.get_totalLength();
        tileNum     = t.get_tileNum();

        blockLength = totalLength / AscendC::GetBlockNum();
        tileLength  = blockLength / tileNum / BUFFER_NUM;
        blockStart  = blockLength * AscendC::GetBlockIdx();

        // 全局指针
        xGm      = reinterpret_cast<__gm__ DTYPE*>(x);
        wGm      = reinterpret_cast<__gm__ DTYPE*>(weight);
        bGm      = reinterpret_cast<__gm__ DTYPE*>(bias);
        gammaGm  = reinterpret_cast<__gm__ DTYPE*>(bn_gamma);
        betaGm   = reinterpret_cast<__gm__ DTYPE*>(bn_beta);
        meanGm   = reinterpret_cast<__gm__ DTYPE*>(bn_mean);
        varGm    = reinterpret_cast<__gm__ DTYPE*>(bn_var);
        scaleGm  = reinterpret_cast<__gm__ DTYPE*>(scaling);

        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(y) + blockStart, blockLength);

        // 仅对输出建立环形队列（示例做法）
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(DTYPE));
    }

    __aicore__ inline void Process() {
        int32_t loopCount = tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void Compute(int32_t progress) {
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.AllocTensor<DTYPE>();

        uint32_t base = progress * tileLength;
        for (uint32_t j = 0; j < tileLength; ++j) {
            uint32_t outIndex = base + j;
            if (outIndex >= blockLength) {
                // 越界保护
                // 简化处理：填0
                // yLocal[j] = 0; // 假设可下标访问
                continue;
            }

            // 线性index映射到(N, Cout, Hout, Wout)
            uint32_t globalIndex = blockStart + outIndex;
            uint32_t coSpatial = Hout * Wout;
            uint32_t nStride   = Cout * coSpatial;

            uint32_t n  = globalIndex / nStride;
            uint32_t r1 = globalIndex - n * nStride;
            uint32_t co = r1 / coSpatial;
            uint32_t r2 = r1 - co * coSpatial;
            uint32_t ho = r2 / Wout;
            uint32_t wo = r2 - ho * Wout;

            // 卷积累加
            DTYPE acc = bGm[co];
            for (uint32_t c = 0; c < Cin; ++c) {
                for (uint32_t kh = 0; kh < Kh; ++kh) {
                    uint32_t ih = ho + kh;
                    for (uint32_t kw = 0; kw < Kw; ++kw) {
                        uint32_t iw = wo + kw;
                        // x索引: ((n*Cin + c)*Hin + ih)*Win + iw
                        uint64_t xIdx = (((uint64_t)n * Cin + c) * Hin + ih) * Win + iw;
                        // w索引: (((co*Cin + c)*Kh + kh)*Kw + kw)
                        uint64_t wIdx = (((uint64_t)co * Cin + c) * Kh + kh) * Kw + kw;
                        acc += xGm[xIdx] * wGm[wIdx];
                    }
                }
            }

            // BatchNorm: y = (acc - mean)/sqrt(var+eps) * gamma + beta
            DTYPE mean = meanGm[co];
            DTYPE var  = varGm[co];
            DTYPE gamma= gammaGm[co];
            DTYPE beta = betaGm[co];
            DTYPE norm = (acc - mean) * AscendC::Rsqrt(var + eps);
            DTYPE bnOut = norm * gamma + beta;

            // scaling
            DTYPE scale = scaleGm[0];
            DTYPE finalVal = bnOut * scale;

            // 写入本地输出
            // 这里假设LocalTensor支持下标访问
            yLocal[j] = finalVal;
        }

        outQueueY.EnQue<DTYPE>(yLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress) {
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm[progress * tileLength], yLocal, tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    // 维度
    uint32_t N, Cin, Hin, Win, Cout, Kh, Kw, Hout, Wout;
    float eps;

    // 分块
    uint32_t totalLength;
    uint32_t tileNum;
    uint32_t blockLength;
    uint32_t tileLength;
    uint32_t blockStart;

    // 全局内存指针
    __gm__ DTYPE* xGm;
    __gm__ DTYPE* wGm;
    __gm__ DTYPE* bGm;
    __gm__ DTYPE* gammaGm;
    __gm__ DTYPE* betaGm;
    __gm__ DTYPE* meanGm;
    __gm__ DTYPE* varGm;
    __gm__ DTYPE* scaleGm;

    // 输出
    AscendC::GlobalTensor<DTYPE> yGm;

    // 管道与队列
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
};

extern "C" __global__ __aicore__ void conv2d_batch_norm_scaling_custom(
    GM_ADDR x,
    GM_ADDR weight,
    GM_ADDR bias,
    GM_ADDR bn_gamma,
    GM_ADDR bn_beta,
    GM_ADDR bn_mean,
    GM_ADDR bn_var,
    GM_ADDR scaling,
    GM_ADDR y,
    GM_ADDR workspace,
    GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelConvBnScale op;
    op.Init(x, weight, bias, bn_gamma, bn_beta, bn_mean, bn_var, scaling, y, tilingData);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义AscendC算子，将 Conv2d + BatchNorm + 标量放大 融合为一个核函数调用。
    约束：stride=1, padding=0, dilation=1, groups=1。
    """
    def __init__(self, in_channels, out_channels, kernel_size, scaling_factor):
        super().__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size, bias=True)
        self.bn = torch.nn.BatchNorm2d(out_channels, affine=True, track_running_stats=True)
        self.scaling_factor = scaling_factor

    def forward(self, x):
        # 将权重与BN参数传入自定义算子
        scale = torch.tensor([self.scaling_factor], device=x.device, dtype=x.dtype)
        return custom_ops_lib.conv2d_batch_norm_scaling_custom(
            x,
            self.conv.weight,
            self.conv.bias if self.conv.bias is not None else torch.zeros(self.conv.weight.size(0), device=x.device, dtype=x.dtype),
            self.bn.weight,
            self.bn.bias,
            self.bn.running_mean,
            self.bn.running_var,
            scale
        )
