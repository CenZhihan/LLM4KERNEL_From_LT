kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"
#define __NPU_TILING__
#include "shallow_wide_mlp_custom_tiling_data.h"

using A_T = float;
using B_T = float;
using C_T = float;
using BiasT = float;

typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::NORMAL> aType;
typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, B_T, false, LayoutMode::NORMAL> bType;
typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::NORMAL> cType;
typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, BiasT> biasType;

__aicore__ inline void relu_inplace(__gm__ float* gmPtr, uint32_t elemNum)
{
    constexpr int32_t BUF_NUM = 2;
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUF_NUM> inQ;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUF_NUM> outQ;

    const uint32_t tileLen = 4096; // 基础tile
    pipe.InitBuffer(inQ, BUF_NUM, tileLen * sizeof(float));
    pipe.InitBuffer(outQ, BUF_NUM, tileLen * sizeof(float));

    uint32_t tiles = (elemNum + tileLen - 1) / tileLen;
    for (uint32_t t = 0; t < tiles; ++t) {
        uint32_t cur = (t == tiles - 1) ? (elemNum - t * tileLen) : tileLen;

        // copy in
        AscendC::LocalTensor<float> inL = inQ.AllocTensor<float>();
        AscendC::DataCopy(inL, gmPtr + t * tileLen, cur);
        inQ.EnQue(inL);

        // compute
        AscendC::LocalTensor<float> xL = inQ.DeQue<float>();
        AscendC::LocalTensor<float> yL = outQ.AllocTensor<float>();
        AscendC::Relu(yL, xL, cur); // 直接使用Relu算子
        outQ.EnQue(yL);
        inQ.FreeTensor(xL);

        // copy out
        AscendC::LocalTensor<float> outL = outQ.DeQue<float>();
        AscendC::DataCopy(gmPtr + t * tileLen, outL, cur);
        outQ.FreeTensor(outL);
    }
}

extern "C" __global__ __aicore__ void shallow_wide_mlp_custom(
    GM_ADDR xGM, GM_ADDR w1GM, GM_ADDR b1GM,
    GM_ADDR w2GM, GM_ADDR b2GM,
    GM_ADDR w3GM, GM_ADDR b3GM,
    GM_ADDR yGM,
    GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    GET_TILING_DATA(td, tilingGM);

    const uint32_t B = td.batch;
    const uint32_t IN = td.inSize;
    const uint32_t H1 = td.h1Size;
    const uint32_t H2 = td.h2Size;
    const uint32_t OUT = td.outSize;

    // 设置GM全局tensor
    AscendC::GlobalTensor<A_T> aX, aH1, aH2;
    AscendC::GlobalTensor<B_T> w1, w2, w3;
    AscendC::GlobalTensor<C_T> y;
    AscendC::GlobalTensor<BiasT> b1, b2, b3;

    aX.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(xGM), B * IN);

    w1.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(w1GM), IN * H1);
    b1.SetGlobalBuffer(reinterpret_cast<__gm__ BiasT*>(b1GM), H1);

    w2.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(w2GM), H1 * H2);
    b2.SetGlobalBuffer(reinterpret_cast<__gm__ BiasT*>(b2GM), H2);

    w3.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(w3GM), H2 * OUT);
    b3.SetGlobalBuffer(reinterpret_cast<__gm__ BiasT*>(b3GM), OUT);

    y.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(yGM), B * OUT);

    // workspace: 前半 B*H1, 中间 B*H2, 其余可留给库内部使用(示例)
    __gm__ float* wsFloat = reinterpret_cast<__gm__ float*>(workspaceGM);
    __gm__ float* h1GMPtr = wsFloat;
    __gm__ float* h2GMPtr = wsFloat + (static_cast<uint64_t>(B) * H1);
    aH1.SetGlobalBuffer(h1GMPtr, B * H1);
    aH2.SetGlobalBuffer(h2GMPtr, B * H2);

    // 注册Matmul对象
    AscendC::TPipe pipe;
    matmul::Matmul<aType, bType, cType, biasType> mm1;
    matmul::Matmul<aType, bType, cType, biasType> mm2;
    matmul::Matmul<aType, bType, cType, biasType> mm3;

    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm1, &td.t1, mm2, &td.t2, mm3, &td.t3);

    // Layer 1: [B, IN] * [IN, H1] + b1 -> ReLU
    {
        AscendC::GlobalTensor<A_T> gm_a;
        gm_a.SetGlobalBuffer(const_cast<__gm__ A_T*>(aX.GetPhyAddr()), B * IN);
        AscendC::GlobalTensor<B_T> gm_b;
        gm_b.SetGlobalBuffer(const_cast<__gm__ B_T*>(w1.GetPhyAddr()), IN * H1);
        AscendC::GlobalTensor<C_T> gm_c;
        gm_c.SetGlobalBuffer(const_cast<__gm__ C_T*>(aH1.GetPhyAddr()), B * H1);
        AscendC::GlobalTensor<BiasT> gm_bias;
        gm_bias.SetGlobalBuffer(const_cast<__gm__ BiasT*>(b1.GetPhyAddr()), H1);

        mm1.SetTensorA(gm_a, false);
        mm1.SetTensorB(gm_b, false);
        mm1.SetWorkspace(workspaceGM, 0);
        mm1.SetBias(gm_bias);
        mm1.IterateAll();

        relu_inplace(h1GMPtr, B * H1);
    }

    // Layer 2: [B, H1] * [H1, H2] + b2 -> ReLU
    {
        AscendC::GlobalTensor<A_T> gm_a;
        gm_a.SetGlobalBuffer(const_cast<__gm__ A_T*>(aH1.GetPhyAddr()), B * H1);
        AscendC::GlobalTensor<B_T> gm_b;
        gm_b.SetGlobalBuffer(const_cast<__gm__ B_T*>(w2.GetPhyAddr()), H1 * H2);
        AscendC::GlobalTensor<C_T> gm_c;
        gm_c.SetGlobalBuffer(const_cast<__gm__ C_T*>(aH2.GetPhyAddr()), B * H2);
        AscendC::GlobalTensor<BiasT> gm_bias;
        gm_bias.SetGlobalBuffer(const_cast<__gm__ BiasT*>(b2.GetPhyAddr()), H2);

        mm2.SetTensorA(gm_a, false);
        mm2.SetTensorB(gm_b, false);
        mm2.SetWorkspace(workspaceGM, 0);
        mm2.SetBias(gm_bias);
        mm2.IterateAll();

        relu_inplace(h2GMPtr, B * H2);
    }

    // Layer 3: [B, H2] * [H2, OUT] + b3
    {
        AscendC::GlobalTensor<A_T> gm_a;
        gm_a.SetGlobalBuffer(const_cast<__gm__ A_T*>(aH2.GetPhyAddr()), B * H2);
        AscendC::GlobalTensor<B_T> gm_b;
        gm_b.SetGlobalBuffer(const_cast<__gm__ B_T*>(w3.GetPhyAddr()), H2 * OUT);
        AscendC::GlobalTensor<C_T> gm_c;
        gm_c.SetGlobalBuffer(const_cast<__gm__ C_T*>(y.GetPhyAddr()), B * OUT);
        AscendC::GlobalTensor<BiasT> gm_bias;
        gm_bias.SetGlobalBuffer(const_cast<__gm__ BiasT*>(b3.GetPhyAddr()), OUT);

        mm3.SetTensorA(gm_a, false);
        mm3.SetTensorB(gm_b, false);
        mm3.SetWorkspace(workspaceGM, 0);
        mm3.SetBias(gm_bias);
        mm3.IterateAll();
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_layer_sizes, output_size):
        super().__init__()
        layers = []
        current_input_size = input_size
        for hidden_size in hidden_layer_sizes:
            layers.append(torch.nn.Linear(current_input_size, hidden_size))
            layers.append(torch.nn.ReLU())
            current_input_size = hidden_size
        layers.append(torch.nn.Linear(current_input_size, output_size))
        self.network = torch.nn.Sequential(*layers)

    def forward(self, x):
        # 提取三层权重与偏置，并在前向中一次调用自定义核
        # 约定: 将权重按[K,N]=[in,out]传入kernel，因此需转置Linear权重
        w1 = self.network[0].weight.t().contiguous()
        b1 = self.network[0].bias
        w2 = self.network[2].weight.t().contiguous()
        b2 = self.network[2].bias
        w3 = self.network[4].weight.t().contiguous()
        b3 = self.network[4].bias
        return custom_ops_lib.shallow_wide_mlp_custom(x, w1, b1, w2, b2, w3, b3)
'''
