    uint32_t c4 = static_cast<uint32_t>(s3.GetDim(1));
    uint32_t h  = static_cast<uint32_t>(s0.GetDim(2));

    tiling.set_hw(hw);

    tiling.set_c2(c2);

    tiling.set_c4(c4);

    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    ws[0] = 0;
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "googlenet_inception_module_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGooglenetInceptionModule {
public:
    __aicore__ inline KernelGooglenetInceptionModule() {}
    __aicore__ inline void Init(GM_ADDR x1, GM_ADDR x2, GM_ADDR x3, GM_ADDR x4, GM_ADDR y,
                                uint32_t n, uint32_t hw,
                                uint32_t c1, uint32_t c2, uint32_t c3, uint32_t c4,
                                uint32_t tileNum)
    {
        this->n = n;
        this->hw = hw;
        this->c1 = c1;
        this->c2 = c2;
        this->c3 = c3;
        this->c4 = c4;
        this->cOut = c1 + c2 + c3 + c4;
        this->tileNum = tileNum;

        uint32_t total1 = n * c1 * hw;
        uint32_t total2 = n * c2 * hw;
        uint32_t total3 = n * c3 * hw;
        uint32_t total4 = n * c4 * hw;
        uint32_t totalY = n * cOut * hw;

        x1Gm.SetGlobalBuffer((__gm__ float *)x1, total1);
        x2Gm.SetGlobalBuffer((__gm__ float *)x2, total2);
        x3Gm.SetGlobalBuffer((__gm__ float *)x3, total3);
        x4Gm.SetGlobalBuffer((__gm__ float *)x4, total4);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalY);

        uint32_t blocks = AscendC::GetBlockNum();
        uint32_t idx = AscendC::GetBlockIdx();
        this->batchPerBlock = (n + blocks - 1) / blocks;
        this->batchStart = idx * this->batchPerBlock;
        uint32_t tentativeEnd = this->batchStart + this->batchPerBlock;
        this->batchEnd = tentativeEnd > n ? n : tentativeEnd;

        uint32_t maxBranchElems = hw;
        if (c1 > 0) maxBranchElems = c1 * hw;
        if (c2 * hw > maxBranchElems) maxBranchElems = c2 * hw;
        if (c3 * hw > maxBranchElems) maxBranchElems = c3 * hw;
        if (c4 * hw > maxBranchElems) maxBranchElems = c4 * hw;

        this->tileLength = (maxBranchElems + (tileNum * BUFFER_NUM) - 1) / (tileNum * BUFFER_NUM);
        if (this->tileLength == 0) this->tileLength = 1;

        pipe.InitBuffer(tmpQue, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t b = batchStart; b < batchEnd; ++b) {
            uint32_t outBase = b * cOut * hw;
            uint32_t offset = 0;
            CopyBranch(x1Gm, b, c1, outBase + offset);
            offset += c1 * hw;
            CopyBranch(x2Gm, b, c2, outBase + offset);
            offset += c2 * hw;
            CopyBranch(x3Gm, b, c3, outBase + offset);
            offset += c3 * hw;
            CopyBranch(x4Gm, b, c4, outBase + offset);
        }
    }

private:
    __aicore__ inline void CopyBranch(AscendC::GlobalTensor<float>& xGm, uint32_t b, uint32_t c, uint32_t yOffset)
    {
        if (c == 0) return;
        uint32_t inBase = b * c * hw;
        uint32_t total = c * hw;
        uint32_t loops = (total + tileLength - 1) / tileLength;
        for (uint32_t i = 0; i < loops; ++i) {
            uint32_t start = i * tileLength;
            uint32_t len = (start + tileLength <= total) ? tileLength : (total - start);
            AscendC::LocalTensor<float> buf = tmpQue.AllocTensor<float>();
            AscendC::DataCopy(buf, xGm[inBase + start], len);
            AscendC::DataCopy(yGm[yOffset + start], buf, len);
            tmpQue.FreeTensor(buf);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpQue;
    AscendC::GlobalTensor<float> x1Gm, x2Gm, x3Gm, x4Gm, yGm;

    uint32_t n, hw, c1, c2, c3, c4, cOut;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t batchPerBlock, batchStart, batchEnd;
};

extern "C" __global__ __aicore__ void googlenet_inception_module_custom(GM_ADDR x1, GM_ADDR x2, GM_ADDR x3, GM_ADDR x4, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGooglenetInceptionModule op;
    op.Init(x1, x2, x3, x4, y,
            tiling_data.n, tiling_data.hw,
            tiling_data.c1, tiling_data.c2, tiling_data.c3, tiling_data.c4,
            tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_1x1, reduce_3x3, out_3x3, reduce_5x5, out_5x5, pool_proj):
        super().__init__()
        self.branch1x1 = torch.nn.Conv2d(in_channels, out_1x1, kernel_size=1)
        self.branch3x3 = torch.nn.Sequential(
            torch.nn.Conv2d(in_channels, reduce_3x3, kernel_size=1),
            torch.nn.Conv2d(reduce_3x3, out_3x3, kernel_size=3, padding=1)
        )
        self.branch5x5 = torch.nn.Sequential(
            torch.nn.Conv2d(in_channels, reduce_5x5, kernel_size=1),
            torch.nn.Conv2d(reduce_5x5, out_5x5, kernel_size=5, padding=2)
        )
        self.branch_pool = torch.nn.Sequential(
            torch.nn.MaxPool2d(kernel_size=3, stride=1, padding=1),
            torch.nn.Conv2d(in_channels, pool_proj, kernel_size=1)
        )

    def forward(self, x):
        b1 = self.branch1x1(x)
        b3 = self.branch3x3(x)
        b5 = self.branch5x5(x)
        bp = self.branch_pool(x)
        return custom_ops_lib.googlenet_inception_module_custom(b1, b3, b5, bp)
'''
