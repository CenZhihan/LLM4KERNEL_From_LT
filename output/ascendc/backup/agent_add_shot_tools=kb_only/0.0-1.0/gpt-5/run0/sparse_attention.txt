host_operator_src="""
#include "sparse_attention_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace optiling {
extern uint32_t GetSoftMaxMinTmpSize(const ge::Shape& srcShape, const uint32_t dataTypeSize, const bool isReuseSource);
extern void SoftMaxTilingFunc(const ge::Shape& srcShape, const uint32_t dataTypeSize, const uint32_t localWorkSpaceSize, optiling::SoftMaxTiling& softmaxTiling);

static uint32_t GetLastDim(const gert::Shape* shape)
{
    auto origin = shape->GetOriginShape();
    std::vector<int64_t> dims = origin.GetDims();
    if (dims.size() == 0) { return 1; }
    return static_cast<uint32_t>(dims.back());
}

static uint64_t GetTotalSize(const gert::Shape* shape)
{
    return static_cast<uint64_t>(shape->GetOriginShape().GetShapeSize());
}

const uint32_t BLOCK_DIM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    SparseAttentionCustomTilingData tiling;

    const gert::Shape* inShape = context->GetInputShape(0);
    uint32_t K = GetLastDim(inShape);
    uint64_t totalLen = GetTotalSize(inShape);
    if (K == 0) { return ge::GRAPH_FAILED; }

    // SoftMax tiling expects a 2D layout [M, K]
    uint64_t M = totalLen / K;
    std::vector<int64_t> shapeVec = { static_cast<int64_t>(M), static_cast<int64_t>(K) };
    ge::Shape srcShape(shapeVec);

    // Minimum tmp size is enough as per doc
    const uint32_t localWorkSpaceSize = AscendC::GetSoftMaxMinTmpSize(srcShape, sizeof(float), false);
    AscendC::SoftMaxTilingFunc(srcShape, sizeof(float), localWorkSpaceSize, tiling.softmaxTilingData);

    // Set block dim and basic tiling fields
    context->SetBlockDim(BLOCK_DIM);
    tiling.set_totalLength(static_cast<uint32_t>(totalLen));
    tiling.set_seqK(K);

    // Make each tile exactly one row of length K
    // tileNum * BUFFER_NUM * K == blockLength; let runtime split by core
    // Here set tileNum as total rows per core; runtime kernel will compute per-core length
    tiling.set_tileNum(static_cast<uint32_t>(M / AscendC::GetBlockNum() == 0 ? 1 : M / AscendC::GetBlockNum()));

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class SparseAttentionCustom : public OpDef {
public:
    explicit SparseAttentionCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)

            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

            .ParamType(REQUIRED)

            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);

OP_ADD(SparseAttentionCustom);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "sparse_attention_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSparseAttention {
public:
    __aicore__ inline KernelSparseAttention() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum, uint32_t seqK,
                                const optiling::SoftMaxTiling &softmaxTilingData)
    {
        this->totalLength = totalLength;
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum == 0 ? 1 : tileNum;
        this->seqK = seqK;
        this->tileLength = this->seqK; // process one row (softmax over last dim)

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));

        this->softmaxTiling = softmaxTilingData;
    }

    __aicore__ inline void Process()
    {
        // number of rows per core
        uint32_t rowsPerCore = (this->blockLength / this->seqK);
        int32_t loopCount = rowsPerCore * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        // Use SoftMax high-level API with precomputed tiling
        // The xLocal is one row, softmax over K dimension
        AscendC::SoftMax(yLocal, xLocal, this->softmaxTiling);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t seqK;

    optiling::SoftMaxTiling softmaxTiling;
};

extern "C" __global__ __aicore__ void sparse_attention_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelSparseAttention op;
    op.Init(x, y, tilingData.totalLength, tilingData.tileNum, tilingData.seqK, tilingData.softmaxTilingData);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

model_src='''
import math
import torch
import torch_npu
import custom_ops_lib
import torch.nn as nn

class ModelNew(nn.Module):
    """
    Replace attention softmax with custom AscendC kernel (SparseAttentionCustom).
    Masking is applied on PyTorch side, then softmax is computed by the custom kernel.
    """
    def __init__(self, d_model=256, num_heads=4, window_size=32, max_seq_len=4096):
        super().__init__()
        assert d_model % num_heads == 0
        self.d_model = d_model
        self.num_heads = num_heads
        self.head_dim = d_model // num_heads
        self.window_size = window_size
        self.max_seq_len = max_seq_len

        self.q_proj = nn.Linear(d_model, d_model, bias=True)
        self.k_proj = nn.Linear(d_model, d_model, bias=True)
        self.v_proj = nn.Linear(d_model, d_model, bias=True)
        self.out_proj = nn.Linear(d_model, d_model, bias=True)

        # Precompute sparse mask: True means masked
        mask = torch.ones(max_seq_len, max_seq_len)
        for i in range(max_seq_len):
            start = max(0, i - window_size)
            end = min(max_seq_len, i + window_size)
            mask[i, start:end] = 0
        self.register_buffer("attn_mask", mask.bool(), persistent=False)

    def forward(self, x):
        """
        x: (B, S, D)
        """
        B, S, D = x.shape
        H = self.num_heads
        Hd = self.head_dim
        scale = 1.0 / math.sqrt(Hd)

        q = self.q_proj(x).view(B, S, H, Hd).transpose(1, 2)  # (B, H, S, Hd)
        k = self.k_proj(x).view(B, S, H, Hd).transpose(1, 2)  # (B, H, S, Hd)
        v = self.v_proj(x).view(B, S, H, Hd).transpose(1, 2)  # (B, H, S, Hd)

        # Attention logits
        scores = torch.matmul(q, k.transpose(-2, -1)) * scale  # (B, H, S, S)

        # Apply sparse band mask on CPU/PyTorch side: masked positions -> -inf
        attn_mask = self.attn_mask[:S, :S]
        scores = scores.masked_fill(attn_mask.view(1, 1, S, S), float("-inf"))

        # Custom AscendC kernel to compute softmax along the last dim
        # Output probs shape: (B, H, S, S)
        probs = custom_ops_lib.sparse_attention_custom(scores)

        # Weighted sum
        context = torch.matmul(probs, v)  # (B, H, S, Hd)
        context = context.transpose(1, 2).contiguous().view(B, S, D)
        out = self.out_proj(context)
        return out
'''
