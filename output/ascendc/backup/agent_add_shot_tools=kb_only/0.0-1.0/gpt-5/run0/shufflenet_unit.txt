kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "shufflenet_unit_custom_tiling_data.h"

// 本算子实现 ShuffleNet 中的 ChannelShuffle 操作：
// 将输入 [N, C, H, W] 视为 [N, groups, C/groups, H, W]，交换维度(1, 2)后再展平为 [N, C, H, W]。
constexpr int32_t BUFFER_NUM = 1;

class KernelChannelShuffle {
public:
    __aicore__ inline KernelChannelShuffle() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w, uint32_t groups)
    {
        this->n = n;
        this->c = c;
        this->h = h;
        this->w = w;
        this->groups = groups;
        this->hw = h * w;
        this->elems = static_cast<uint64_t>(n) * c * this->hw;

        srcGlobal.SetGlobalBuffer((__gm__ float*)x, elems);
        dstGlobal.SetGlobalBuffer((__gm__ float*)y, elems);

        // 每次搬运一个通道的 H*W 连续数据块
        pipe.InitBuffer(inQueueSrc,  BUFFER_NUM, this->hw * sizeof(float));
        pipe.InitBuffer(outQueueDst, BUFFER_NUM, this->hw * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t cpg = c / groups; // channels per group
        for (uint32_t ni = 0; ni < n; ++ni) {
            for (uint32_t co = 0; co < c; ++co) {
                // co -> (q, r)
                uint32_t q = co / cpg;     // which group in output
                uint32_t r = co % cpg;     // which channel inside each group
                // input channel index after inverse mapping: ci = r * groups + q
                uint32_t ci = r * groups + q;

                uint64_t srcOffset = ((uint64_t)ni * c + ci) * hw;
                uint64_t dstOffset = ((uint64_t)ni * c + co) * hw;

                CopyIn(srcOffset);
                Compute();
                CopyOut(dstOffset);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint64_t srcOffset)
    {
        AscendC::LocalTensor<float> srcLocal = inQueueSrc.AllocTensor<float>();
        AscendC::DataCopy(srcLocal, srcGlobal[srcOffset], hw);
        inQueueSrc.EnQue(srcLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> srcLocal = inQueueSrc.DeQue<float>();
        AscendC::LocalTensor<float> dstLocal = outQueueDst.AllocTensor<float>();
        // 本操作仅为搬运，直接复制
        AscendC::DataCopy(dstLocal, srcLocal, hw);
        outQueueDst.EnQue<float>(dstLocal);
        inQueueSrc.FreeTensor(srcLocal);
    }

    __aicore__ inline void CopyOut(uint64_t dstOffset)
    {
        AscendC::LocalTensor<float> dstLocal = outQueueDst.DeQue<float>();
        AscendC::DataCopy(dstGlobal[dstOffset], dstLocal, hw);
        outQueueDst.FreeTensor(dstLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueSrc;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueDst;

    AscendC::GlobalTensor<float> srcGlobal;
    AscendC::GlobalTensor<float> dstGlobal;

    uint32_t n;
    uint32_t c;
    uint32_t h;
    uint32_t w;
    uint32_t groups;

    uint32_t hw;
    uint64_t elems;
};

extern "C" __global__ __aicore__ void shufflenet_unit_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelChannelShuffle op;
    op.Init(x, y,
            tiling_data.n,
            tiling_data.c,
            tiling_data.h,
            tiling_data.w,
            tiling_data.groups);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class ChannelShuffleCustom(nn.Module):
    def __init__(self, groups: int):
        super().__init__()
        self.groups = groups
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return custom_ops_lib.shufflenet_unit_custom(x, self.groups)

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, groups=3):
        super(ModelNew, self).__init__()
        assert out_channels % 4 == 0
        mid_channels = out_channels // 4

        self.conv1 = nn.Conv2d(in_channels, mid_channels, kernel_size=1, stride=1, padding=0, groups=groups, bias=False)
        self.bn1 = nn.BatchNorm2d(mid_channels)

        self.conv2 = nn.Conv2d(mid_channels, mid_channels, kernel_size=3, stride=1, padding=1, groups=mid_channels, bias=False)
        self.bn2 = nn.BatchNorm2d(mid_channels)

        self.conv3 = nn.Conv2d(mid_channels, out_channels, kernel_size=1, stride=1, padding=0, groups=groups, bias=False)
        self.bn3 = nn.BatchNorm2d(out_channels)

        # 使用自定义 AscendC kernel 实现 ChannelShuffle
        self.shuffle = ChannelShuffleCustom(groups)

        if in_channels == out_channels:
            self.shortcut = nn.Sequential()
        else:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=1, padding=0, bias=False),
                nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out = self.shuffle(out)
        out = F.relu(self.bn3(self.conv3(out)))
        out = out + self.shortcut(x)
        return out
'''
