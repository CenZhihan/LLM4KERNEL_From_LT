kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose3d_clamp_min_divide_custom_tiling_data.h"

// 本Kernel对输入张量执行：y = clamp_min(x, min_value); y = y * (1/divisor)
// 使用Maxs实现clamp_min，使用Muls实现乘以标量

constexpr int32_t BUFFER_NUM = 2; // pipeline队列的tensor个数

class KernelClampMinDivide {
public:
    __aicore__ inline KernelClampMinDivide() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum,
                                float minValue, float invDiv)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->minValue = minValue;
        this->invDiv = invDiv;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(calcBuf, 1, this->tileLength * sizeof(float)); // 临时计算缓冲
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> tmpLocal = calcBuf.Get<float>(); // 用作clamp后的中间结果
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // clamp_min: tmp = max(x, minValue)
        AscendC::Maxs(tmpLocal, xLocal, this->minValue, this->tileLength);

        // divide: y = tmp * invDiv
        AscendC::Muls(yLocal, tmpLocal, this->invDiv, this->tileLength);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> calcBuf;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;

    float minValue;
    float invDiv;
};

extern "C" __global__ __aicore__ void conv_transpose3d_clamp_min_divide_custom(GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelClampMinDivide op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.tileNum, tiling_data.minValue, tiling_data.invDiv);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义AscendC算子将 clamp_min 和 divide 融合为单算子，加速 ConvTranspose3d 后处理
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, min_value, divisor):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.min_value = float(min_value)
        self.divisor = float(divisor)

    def forward(self, x):
        x = self.conv_transpose(x)  # 先进行转置三维卷积
        # 将后处理融合算子在NPU侧执行：y = clamp_min(x, min_value); y = y / divisor
        x = custom_ops_lib.conv_transpose3d_clamp_min_divide_custom(x, self.min_value, self.divisor)
        return x
