kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "average_pooling_2d_custom_tiling_data.h"

inline __aicore__ uint32_t AlignToBlock(const uint32_t inputValue, const uint32_t typeSize)
{
    constexpr uint32_t ONE_BLK_SIZE = 32;
    uint32_t alignUnit = ONE_BLK_SIZE / typeSize;
    return (inputValue + alignUnit - 1) / alignUnit * alignUnit;
}

class KernelAveragePooling2D {
public:
    __aicore__ inline KernelAveragePooling2D() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w,
                                uint32_t out_h, uint32_t out_w,
                                uint32_t ksize, uint32_t stride, uint32_t padding,
                                uint32_t totalLength, uint32_t tmpSize)
    {
        this->n = n; this->c = c; this->h = h; this->w = w;
        this->out_h = out_h; this->out_w = out_w;
        this->ksize = ksize; this->stride = stride; this->padding = padding;
        this->totalLength = totalLength;
        this->blockLength = (AscendC::GetBlockIdx() == AscendC::GetBlockNum() - 1)
                            ? (totalLength - (totalLength / AscendC::GetBlockNum()) * (AscendC::GetBlockNum() - 1))
                            : (totalLength / AscendC::GetBlockNum());
        this->blockStart = (totalLength / AscendC::GetBlockNum()) * AscendC::GetBlockIdx();

        // set global buffers for the whole tensor to allow arbitrary indexing
        uint32_t inTotal = n * c * h * w;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x, inTotal);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y, totalLength);

        // buffers
        this->rowInnerAligned = AlignToBlock(ksize, sizeof(DTYPE_X));
        this->scalarAligned = AlignToBlock(1, sizeof(DTYPE_X));

        // tmp buffer for Sum API
        this->tmpSize = tmpSize;
        if (this->tmpSize != 0) {
            pipe.InitBuffer(workQueue, 1, this->tmpSize);
        }
        pipe.InitBuffer(rowQueue, 1, rowInnerAligned * sizeof(DTYPE_X));
        pipe.InitBuffer(accQueue, 1, scalarAligned * sizeof(DTYPE_X));
        pipe.InitBuffer(rowSumQueue, 1, scalarAligned * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->blockLength; ++i) {
            ComputeOne(i);
        }
    }

private:
    __aicore__ inline void ComputeOne(uint32_t progress)
    {
        uint32_t outIdx = blockStart + progress;

        // unravel index: n, c, oh, ow
        uint32_t ow = outIdx % out_w;
        uint32_t tmp1 = outIdx / out_w;
        uint32_t oh = tmp1 % out_h;
        uint32_t tmp2 = tmp1 / out_h;
        uint32_t cc = tmp2 % c;
        uint32_t nn = tmp2 / c;

        int32_t ihStart = static_cast<int32_t>(oh) * static_cast<int32_t>(stride) - static_cast<int32_t>(padding);
        int32_t iwStart = static_cast<int32_t>(ow) * static_cast<int32_t>(stride) - static_cast<int32_t>(padding);

        AscendC::LocalTensor<DTYPE_Y> accLocal = accQueue.AllocTensor<DTYPE_Y>();
        // zero initialize accumulator
        DTYPE_Y zeroVal = (DTYPE_Y)0;
        AscendC::Duplicate<DTYPE_Y>(accLocal, zeroVal, scalarAligned);

        for (uint32_t kh = 0; kh < ksize; ++kh) {
            int32_t ih = ihStart + static_cast<int32_t>(kh);
            if (ih < 0 || ih >= static_cast<int32_t>(h)) {
                continue;
            }
            int32_t validStart = iwStart < 0 ? 0 : iwStart;
            int32_t validEnd = iwStart + static_cast<int32_t>(ksize);
            if (validEnd > static_cast<int32_t>(w)) validEnd = static_cast<int32_t>(w);
            int32_t cnt = validEnd - validStart;
            if (cnt <= 0) {
                continue;
            }

            AscendC::LocalTensor<DTYPE_X> rowLocal = rowQueue.AllocTensor<DTYPE_X>();
            // zero pad the row buffer up to rowInnerAligned
            AscendC::Duplicate<DTYPE_X>(rowLocal, (DTYPE_X)0, rowInnerAligned);

            // copy contiguous valid row segment
            uint32_t gmRowOffset = ((nn * c + cc) * h + static_cast<uint32_t>(ih)) * w + static_cast<uint32_t>(validStart);
            AscendC::DataCopy(rowLocal, xGm[gmRowOffset], static_cast<uint32_t>(cnt));

            // reduce sum over this row segment
            AscendC::LocalTensor<DTYPE_Y> rowSumLocal = rowSumQueue.AllocTensor<DTYPE_Y>();
            AscendC::Duplicate<DTYPE_Y>(rowSumLocal, (DTYPE_Y)0, scalarAligned);
            AscendC::SumParams params{1, rowInnerAligned, static_cast<uint32_t>(cnt)};

            if (this->tmpSize != 0) {
                AscendC::LocalTensor<uint8_t> workLocal = workQueue.AllocTensor<uint8_t>();
                AscendC::Sum<DTYPE_Y>(rowSumLocal, rowLocal, params);
                workQueue.FreeTensor(workLocal);
            } else {
                AscendC::Sum<DTYPE_Y>(rowSumLocal, rowLocal, params);
            }

            // accumulate: accLocal = accLocal + rowSumLocal (count = 1)
            AscendC::Add(accLocal, accLocal, rowSumLocal, 1);

            rowSumQueue.FreeTensor(rowSumLocal);
            rowQueue.FreeTensor(rowLocal);
        }

        // average: multiply by 1/(ksize*ksize)
        DTYPE_Y invArea = (DTYPE_Y)(1.0f / ((float)ksize * (float)ksize));
        AscendC::Muls(accLocal, accLocal, invArea, 1);

        // write back single value
        AscendC::DataCopy(yGm[outIdx], accLocal, 1);
        accQueue.FreeTensor(accLocal);
    }

private:
    AscendC::TPipe pipe;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> workQueue;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> rowQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> accQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> rowSumQueue;

    uint32_t n, c, h, w;
    uint32_t out_h, out_w;
    uint32_t ksize, stride, padding;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t blockStart;
    uint32_t rowInnerAligned;
    uint32_t scalarAligned;
    uint32_t tmpSize;
};

extern "C" __global__ __aicore__ void average_pooling_2d_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelAveragePooling2D op;
    op.Init(x, y,
            tilingData.n(), tilingData.c(), tilingData.h(), tilingData.w(),
            tilingData.out_h(), tilingData.out_w(),
            tilingData.ksize(), tilingData.stride(), tilingData.padding(),
            tilingData.totalLength(), tilingData.tmpSize());
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

static inline int64_t floor_div(int64_t a, int64_t b) { return a / b; }

at::Tensor average_pooling_2d_custom_impl_npu(const at::Tensor& x, int64_t kernel_size, c10::optional<int64_t> stride_opt, int64_t padding) {
    TORCH_CHECK(x.dim() == 4, "Input must be 4D NCHW");
    int64_t stride = stride_opt.has_value() ? stride_opt.value() : kernel_size;

    auto sizes = x.sizes();
    int64_t n = sizes[0], c = sizes[1], h = sizes[2], w = sizes[3];

    TORCH_CHECK(kernel_size > 0 && stride > 0, "kernel_size and stride must be > 0");

    int64_t out_h = 0;
    int64_t out_w = 0;
    if (h + 2 * padding >= kernel_size) {
        out_h = floor_div(h + 2 * padding - kernel_size, stride) + 1;
    }
    if (w + 2 * padding >= kernel_size) {
        out_w = floor_div(w + 2 * padding - kernel_size, stride) + 1;
    }
    at::Tensor result = at::empty({n, c, out_h, out_w}, x.options());

    // Pass attrs: kernel_size, stride, padding
    EXEC_NPU_CMD(aclnnAveragePooling2dCustom, x, result, (int)kernel_size, (int)stride, (int)padding);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("average_pooling_2d_custom", &average_pooling_2d_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
