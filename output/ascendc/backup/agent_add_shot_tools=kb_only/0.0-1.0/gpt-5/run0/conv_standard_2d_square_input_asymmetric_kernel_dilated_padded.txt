inline uint32_t calc_out(uint32_t in, uint32_t pad, uint32_t dilation, uint32_t k, uint32_t stride) {
    // floor((in + 2*pad - dilation*(k - 1) - 1)/stride + 1)
    int64_t num = static_cast<int64_t>(in) + 2LL * pad - static_cast<int64_t>(dilation) * (static_cast<int64_t>(k) - 1) - 1;
    if (num < 0) num = 0;
    return static_cast<uint32_t>(num / stride + 1);
}

#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_standard_2d_square_input_asymmetric_kernel_dilated_padded_custom_tiling_data.h"

class KernelConv2d {
public:
    __aicore__ inline KernelConv2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y,
                                uint32_t N, uint32_t Cin, uint32_t Cout,
                                uint32_t H, uint32_t W,
                                uint32_t Kh, uint32_t Kw,
                                uint32_t StrideH, uint32_t StrideW,
                                uint32_t PadTop, uint32_t PadLeft,
                                uint32_t DilationH, uint32_t DilationW,
                                uint32_t HOut, uint32_t WOut) {
        this->N = N; this->Cin = Cin; this->Cout = Cout;
        this->H = H; this->W = W; this->Kh = Kh; this->Kw = Kw;
        this->StrideH = StrideH; this->StrideW = StrideW;
        this->PadTop = PadTop; this->PadLeft = PadLeft;
        this->DilationH = DilationH; this->DilationW = DilationW;
        this->HOut = HOut; this->WOut = WOut;

        xGm = reinterpret_cast<__gm__ float *>(x);
        wGm = reinterpret_cast<__gm__ float *>(w);
        yGm = reinterpret_cast<__gm__ float *>(y);
    }

    __aicore__ inline void Process() {
        Compute();
    }

private:
    __aicore__ inline void Compute() {
        // Naive reference convolution on GM memory for clarity
        // Shapes:
        // x: [N, Cin, H, W]
        // w: [Cout, Cin, Kh, Kw]
        // y: [N, Cout, HOut, WOut]
        for (uint32_t n = 0; n < N; ++n) {
            for (uint32_t oc = 0; oc < Cout; ++oc) {
                for (uint32_t oh = 0; oh < HOut; ++oh) {
                    for (uint32_t ow = 0; ow < WOut; ++ow) {
                        float acc = 0.0f;
                        for (uint32_t ic = 0; ic < Cin; ++ic) {
                            for (uint32_t kh = 0; kh < Kh; ++kh) {
                                int32_t ih = static_cast<int32_t>(oh) * static_cast<int32_t>(StrideH)
                                             - static_cast<int32_t>(PadTop)
                                             + static_cast<int32_t>(kh) * static_cast<int32_t>(DilationH);
                                if (ih < 0 || ih >= static_cast<int32_t>(H)) continue;
                                for (uint32_t kw = 0; kw < Kw; ++kw) {
                                    int32_t iw = static_cast<int32_t>(ow) * static_cast<int32_t>(StrideW)
                                                 - static_cast<int32_t>(PadLeft)
                                                 + static_cast<int32_t>(kw) * static_cast<int32_t>(DilationW);
                                    if (iw < 0 || iw >= static_cast<int32_t>(W)) continue;

                                    uint64_t x_idx = (((uint64_t)n * Cin + ic) * H + (uint32_t)ih) * W + (uint32_t)iw;
                                    uint64_t w_idx = ((((uint64_t)oc) * Cin + ic) * Kh + kh) * Kw + kw;
                                    acc += xGm[x_idx] * wGm[w_idx];
                                }
                            }
                        }
                        uint64_t y_idx = (((uint64_t)n * Cout + oc) * HOut + oh) * WOut + ow;
                        yGm[y_idx] = acc;
                    }
                }
            }
        }
    }

private:
    __gm__ float* xGm;
    __gm__ float* wGm;
    __gm__ float* yGm;

    uint32_t N, Cin, Cout;
    uint32_t H, W, Kh, Kw;
    uint32_t StrideH, StrideW;
    uint32_t PadTop, PadLeft;
    uint32_t DilationH, DilationW;
    uint32_t HOut, WOut;
};

extern "C" __global__ __aicore__ void conv_standard_2d_square_input_asymmetric_kernel_dilated_padded_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2d op;
    op.Init(x, w, y,
            tiling_data.N, tiling_data.Cin, tiling_data.Cout,
            tiling_data.H, tiling_data.W,
            tiling_data.Kh, tiling_data.Kw,
            tiling_data.StrideH, tiling_data.StrideW,
            tiling_data.PadTop, tiling_data.PadLeft,
            tiling_data.DilationH, tiling_data.DilationW,
            tiling_data.HOut, tiling_data.WOut);
    op.Process();
}
