kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose3d_layer_norm_gelu_scaling_custom_tiling_data.h"

// 本自定义算子对输入张量执行 GELU 激活并按标量进行缩放：y = scale * GELU(x)
// 参考文档：High-level API Gelu（AscendC::Gelu），向量标量乘法（Muls）
// 输入：x (ND, float)，scale (1-element float 或可广播标量)
// 输出：y (同形状，同类型)

constexpr int32_t BUFFER_NUM = 2; // 每个队列的tensor个数

class KernelConvTranspose3dLayerNormGeluScaling {
public:
    __aicore__ inline KernelConvTranspose3dLayerNormGeluScaling() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        // 标量scale（要求输入一个元素）
        scaleGm.SetGlobalBuffer((__gm__ DTYPE_SCALE *)scale, 1);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));

        // 预取标量scale到本地（一次即可复用）
        scaleLocal = pipe.AllocScalar<DTYPE_SCALE>();
        AscendC::DataCopy(scaleLocal, scaleGm, 1);
        pipe.EnQueScalar(scaleLocal);
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        // 1) GELU 激活（AscendC::Gelu 支持 highPrecision / highPerformance 模式）
        AscendC::Gelu(yLocal, xLocal, this->tileLength);

        // 2) 取出标量并对结果进行缩放（向量乘标量）
        DTYPE_SCALE s = pipe.DeQueScalar<DTYPE_SCALE>(); // 从队列取出已缓存的标量
        AscendC::Muls(yLocal, yLocal, s, this->tileLength);

        // 标量可复用，重新入队用于下一tile
        pipe.EnQueScalar(s);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    AscendC::GlobalTensor<DTYPE_SCALE> scaleGm;
    AscendC::LocalScalar<DTYPE_SCALE> scaleLocal;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv_transpose3d_layer_norm_gelu_scaling_custom(
    GM_ADDR x, GM_ADDR scale, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dLayerNormGeluScaling op;
    op.Init(x, scale, y, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    结构：ConvTranspose3d -> LayerNorm -> (自定义AscendC算子: GELU + 缩放)
    将原模型中的 GELU 与缩放融合到 AscendC 自定义算子中。
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, bias=True, eps=1e-5, scaling_factor=1.0):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, bias=bias)
        self.layer_norm = torch.nn.LayerNorm(out_channels, eps=eps)
        self.scaling_factor = float(scaling_factor)

    def forward(self, x):
        x = self.conv_transpose(x)                      # (N, C, D, H, W)
        x = self.layer_norm(x)                          # 仍使用 PyTorch 的 LayerNorm
        scale = torch.tensor(self.scaling_factor, device=x.device, dtype=x.dtype)
        # 使用自定义AscendC算子完成 GELU + 缩放
        x = custom_ops_lib.conv_transpose3d_layer_norm_gelu_scaling_custom(x, scale)
        return x
