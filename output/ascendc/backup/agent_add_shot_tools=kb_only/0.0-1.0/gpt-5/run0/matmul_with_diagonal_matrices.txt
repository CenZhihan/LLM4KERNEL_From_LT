kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matmul_with_diagonal_matrices_custom_tiling_data.h"

// 使用行缩放：c[i, :] = a[i] * b[i, :]
constexpr int32_t INBUF_NUM = 1;

class KernelMatmulWithDiagonalMatrices {
public:
    __aicore__ inline KernelMatmulWithDiagonalMatrices() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR c, uint32_t nRows, uint32_t nCols)
    {
        this->nRows = nRows;
        this->nCols = nCols;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t baseRows = nRows / blockNum;
        uint32_t rem = nRows % blockNum;
        if (blockIdx < rem) {
            this->rowsThisBlock = baseRows + 1;
            this->rowStart = blockIdx * (baseRows + 1);
        } else {
            this->rowsThisBlock = baseRows;
            this->rowStart = rem * (baseRows + 1) + (blockIdx - rem) * baseRows;
        }

        aGm.SetGlobalBuffer((__gm__ DTYPE_X *)a + this->rowStart, this->rowsThisBlock);
        bGm.SetGlobalBuffer((__gm__ DTYPE_Y *)b + static_cast<uint64_t>(this->rowStart) * this->nCols,
                            static_cast<uint64_t>(this->rowsThisBlock) * this->nCols);
        cGm.SetGlobalBuffer((__gm__ DTYPE_Z *)c + static_cast<uint64_t>(this->rowStart) * this->nCols,
                            static_cast<uint64_t>(this->rowsThisBlock) * this->nCols);

        // 队列与缓冲区
        pipe.InitBuffer(inQueueA, INBUF_NUM, sizeof(DTYPE_X));                 // 标量a[i]
        pipe.InitBuffer(inQueueB, INBUF_NUM, this->nCols * sizeof(DTYPE_Y));   // 一整行b[i, :]
        pipe.InitBuffer(outQueueC, INBUF_NUM, this->nCols * sizeof(DTYPE_Z));  // 一整行c[i, :]
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsThisBlock; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        // 拷贝 a[rowIdx] 标量
        AscendC::LocalTensor<DTYPE_X> aLocal = inQueueA.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(aLocal, aGm[rowIdx], 1);
        inQueueA.EnQue(aLocal);

        // 拷贝 b[rowIdx, :]
        AscendC::LocalTensor<DTYPE_Y> bLocal = inQueueB.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(bLocal, bGm[static_cast<uint64_t>(rowIdx) * this->nCols], this->nCols);
        inQueueB.EnQue(bLocal);
    }

    __aicore__ inline void Compute(uint32_t /*rowIdx*/)
    {
        AscendC::LocalTensor<DTYPE_X> aLocal = inQueueA.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> bLocal = inQueueB.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> cLocal = outQueueC.AllocTensor<DTYPE_Z>();

        // 将标量加载到寄存器（示例用法）
        DTYPE_X alpha = aLocal.GetValue(0);

        // 使用Muls进行向量与标量乘法
        AscendC::Muls<DTYPE_Z>(cLocal, bLocal, static_cast<DTYPE_Z>(alpha), static_cast<int32_t>(this->nCols));

        inQueueA.FreeTensor(aLocal);
        inQueueB.FreeTensor(bLocal);
        outQueueC.EnQue<DTYPE_Z>(cLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        AscendC::LocalTensor<DTYPE_Z> cLocal = outQueueC.DeQue<DTYPE_Z>();
        AscendC::DataCopy(cGm[static_cast<uint64_t>(rowIdx) * this->nCols], cLocal, this->nCols);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, INBUF_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, INBUF_NUM> outQueueC;

    AscendC::GlobalTensor<DTYPE_X> aGm;
    AscendC::GlobalTensor<DTYPE_Y> bGm;
    AscendC::GlobalTensor<DTYPE_Z> cGm;

    uint32_t nRows;
    uint32_t nCols;
    uint32_t rowStart;
    uint32_t rowsThisBlock;
};

extern "C" __global__ __aicore__ void matmul_with_diagonal_matrices_custom(GM_ADDR a, GM_ADDR b, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulWithDiagonalMatrices op;
    op.Init(a, b, c, tiling_data.nRows, tiling_data.nCols);
    op.Process();
}
