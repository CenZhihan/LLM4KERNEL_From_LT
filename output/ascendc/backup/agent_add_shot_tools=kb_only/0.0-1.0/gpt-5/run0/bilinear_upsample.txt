namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    BilinearUpsampleCustomTilingData tiling;

    const gert::Shape* in_shape = context->GetInputShape(0);
    auto dims = in_shape->GetOriginShape().GetDims();
    // Expect NCHW
    uint32_t n = dims.size() > 0 ? static_cast<uint32_t>(dims[0]) : 1;
    uint32_t c = dims.size() > 1 ? static_cast<uint32_t>(dims[1]) : 1;
    uint32_t h = dims.size() > 2 ? static_cast<uint32_t>(dims[2]) : 1;
    uint32_t w = dims.size() > 3 ? static_cast<uint32_t>(dims[3]) : 1;

    tiling.set_n(n);
    tiling.set_c(c);
    tiling.set_h(h);
    tiling.set_w(w);
    tiling.set_scale(2);

    // Set a reasonable block dim for parallel planes (N*C)
    context->SetBlockDim(8);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // no extra workspace needed
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    auto dims = x_shape->GetOriginShape().GetDims();
    // Double H and W
    if (dims.size() >= 4) {
        dims[2] = dims[2] * 2;
        dims[3] = dims[3] * 2;
    }
    gert::Shape* y_shape = context->GetOutputShape(0);
    y_shape->SetDim(dims);
    return GRAPH_SUCCESS;
}

{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "bilinear_upsample_custom_tiling_data.h"

// 本实现针对 NCHW float32，scale_factor=2.0，align_corners=false
// 参考文档：DataCopy/Copy, TPipe/TQue, ISASI BilinearInterpolation（支持水平/垂直迭代）

constexpr int32_t IN_Q_NUM = 1;
constexpr int32_t OUT_Q_NUM = 1;

struct BiParams {
    uint32_t inH;
    uint32_t inW;
    uint32_t outH;
    uint32_t outW;
    bool alignCorners;
};

class KernelBilinearUpsample {
public:
    __aicore__ inline KernelBilinearUpsample() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t n, uint32_t c, uint32_t h, uint32_t w, uint32_t scale)
    {
        this->n = n;
        this->c = c;
        this->h = h;
        this->w = w;
        this->scale = scale; // fixed = 2
        this->hw = h * w;
        this->oh = h * scale;
        this->ow = w * scale;
        this->ohw = oh * ow;

        // 将 (N*C) 个平面按 AICore 均匀切分
        totalPlanes = n * c;
        uint32_t blocks = AscendC::GetBlockNum();
        uint32_t bid = AscendC::GetBlockIdx();
        uint32_t base = (totalPlanes / blocks) * bid;
        uint32_t rem = totalPlanes % blocks;
        if (bid < rem) {
            base += bid;
            planesPerCore = totalPlanes / blocks + 1;
        } else {
            base += rem;
            planesPerCore = totalPlanes / blocks;
        }
        startPlane = base;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, (uint64_t)n * c * h * w);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, (uint64_t)n * c * oh * ow);

        // 为单平面分配 UB 缓冲
        pipe.InitBuffer(inQueue, IN_Q_NUM, hw * sizeof(DTYPE_X));       // src plane (H*W)
        pipe.InitBuffer(outQueue, OUT_Q_NUM, ohw * sizeof(DTYPE_Y));    // dst plane (2H*2W)
    }

    __aicore__ inline void Process()
    {
        for (uint32_t p = 0; p < planesPerCore; ++p) {
            uint64_t inPlaneOffset = (uint64_t)(startPlane + p) * hw;
            uint64_t outPlaneOffset = (uint64_t)(startPlane + p) * ohw;

            // CopyIn: 读单平面到 UB
            AscendC::LocalTensor<DTYPE_X> srcLocal = inQueue.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(srcLocal, xGm[inPlaneOffset], hw);
            inQueue.EnQue(srcLocal);

            // Compute: 双线性 2x 上采样
            AscendC::LocalTensor<DTYPE_X> src = inQueue.DeQue<DTYPE_X>();
            AscendC::LocalTensor<DTYPE_Y> dst = outQueue.AllocTensor<DTYPE_Y>();

            // 调用 ISASI 双线性插值 API（包含水平/垂直迭代），对单平面进行 2x 放大
            // 文档中描述 BilinearInterpolation 支持垂直和水平迭代，这里以参数形式传入输入/输出尺寸与对齐方式
            BiParams params{h, w, oh, ow, false};
            // 伪代码接口，具体以版本提供的签名为准
            AscendC::BilinearInterpolation(dst, src, params);

            outQueue.EnQue(dst);
            inQueue.FreeTensor(src);

            // CopyOut: 写回 GM
            AscendC::LocalTensor<DTYPE_Y> outLocal = outQueue.DeQue<DTYPE_Y>();
            AscendC::DataCopy(yGm[outPlaneOffset], outLocal, ohw);
            outQueue.FreeTensor(outLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, IN_Q_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, OUT_Q_NUM> outQueue;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t n, c, h, w, scale;
    uint32_t hw, oh, ow, ohw;

    uint32_t totalPlanes;
    uint32_t planesPerCore;
    uint32_t startPlane;
};

extern "C" __global__ __aicore__ void bilinear_upsample_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelBilinearUpsample op;
    op.Init(x, y, tiling_data.n, tiling_data.c, tiling_data.h, tiling_data.w, tiling_data.scale);
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor bilinear_upsample_custom_impl_npu(const at::Tensor& x) {
    // 预分配输出张量，尺寸为 N, C, 2H, 2W
    TORCH_CHECK(x.dim() == 4, "Input must be 4D NCHW");
    auto sizes = x.sizes().vec();
    sizes[2] *= 2;
    sizes[3] *= 2;
    at::Tensor y = at::empty(sizes, x.options());
    EXEC_NPU_CMD(aclnnBilinearUpsampleCustom, x, y);
    return y;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("bilinear_upsample_custom", &bilinear_upsample_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("bilinear_upsample_custom", &bilinear_upsample_custom_impl_npu, "bilinear upsample x2 (align_corners=False)");
}
