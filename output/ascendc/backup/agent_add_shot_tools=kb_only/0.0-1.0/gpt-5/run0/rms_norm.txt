host_operator_src="""
#include "rms_norm_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/tiling_api.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    RmsnormCustomTilingData tiling;
    // 输出总元素数
    uint32_t totalLength = context->GetInputShape(0)->GetOriginShape().GetShapeSize();
    context->SetBlockDim(BLOCK_DIM);
    tiling.set_totalLength(totalLength);
    tiling.set_tileNum(TILE_NUM);

    // 获取输入原始形状，约定输入传入为 [B, S, H]（H为归一化轴，已在Python侧做了reshape）
    const gert::Shape* in_shape = context->GetInputShape(0);
    auto originShape = in_shape->GetOriginShape();
    std::vector<int64_t> shapeVec;
    for (size_t i = 0; i < originShape.GetDimNum(); ++i) {
        shapeVec.push_back(originShape.GetDim(i));
    }
    // 若维度不是3，则在Python侧应已reshape，这里直接按origin传入
    ge::Shape oriSrcShape(shapeVec);
    ge::Shape srcShape(shapeVec); // 简化处理：直接使用原始形状作为对齐后形状

    // 查询RmsNorm所需临时空间范围，并选择最小值以保证功能正确
    uint32_t minValue = 0;
    uint32_t maxValue = 0;
    // 输入类型为float32
    AscendC::GetRmsNormMaxMinTmpSize(srcShape, sizeof(float), maxValue, minValue, false);
    tiling.set_tmpBufSize(minValue);

    // 根据剩余可用UB空间(此处用minValue示例)获取RmsNorm Tiling信息
    AscendC::GetRmsNormTilingInfo(srcShape, oriSrcShape, minValue, sizeof(float), tiling.rmsnormTilingData, false);

    // 保存tiling数据
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无额外workspace
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    // 设定tiling key
    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}
}


namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    // 输出形状与输入x保持一致
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}


namespace ops {
class RmsNormCustom : public OpDef {
public:
    explicit RmsNormCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("gamma")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);

        this->AICore()
            .SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(RmsNormCustom);
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义AscendC算子(RmsNormCustom)对输入做RMS归一化。
    算子在最后一维执行RMSNorm，因此在前后需要做维度变换。
    """
    def __init__(self, num_features: int, eps: float = 1e-5):
        super().__init__()
        self.num_features = num_features
        self.eps = eps
        # gamma设为常量1，等价于无缩放；保持与原始模型一致
        self.register_buffer("gamma", torch.ones(num_features, dtype=torch.float32))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # 输入形状: (B, C, D1, D2, ...)
        # 自定义RMSNorm算子约定对最后一维H做归一化，故将通道维移动到最后
        x_perm = x.movedim(1, -1)  # (B, D1, D2, ..., C)
        B = x_perm.shape[0]
        H = self.num_features  # 归一化轴长度
        # 其余维度合并为S
        S = int(x_perm.numel() // (B * H))
        x_reshaped = x_perm.reshape(B, S, H).contiguous()

        # 调用自定义AscendC算子，输入为 [B, S, H] 与 [H]
        # 该算子内部基于RmsNorm高阶API与Tiling接口(GetRmsNormMaxMinTmpSize/GetRmsNormTilingInfo)实现
        y_reshaped = custom_ops_lib.rms_norm_custom(x_reshaped, self.gamma)

        # 还原形状与维度
        y_perm = y_reshaped.reshape_as(x_perm)
        y = y_perm.movedim(-1, 1)

        return y
'''
