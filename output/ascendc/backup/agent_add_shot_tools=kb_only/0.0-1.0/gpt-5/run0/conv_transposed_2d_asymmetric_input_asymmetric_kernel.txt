kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transposed_2d_asymmetric_input_asymmetric_kernel_custom_tiling_data.h"

// Naive reference kernel (stride=1, padding=0, dilation=1, groups=1, bias=False).
// Input:  x [N, Ci, Hi, Wi]
// Weight: w [Ci, Co, Kh, Kw] (ConvTranspose2d layout)
// Output: y [N, Co, Ho, Wo], where Ho=Hi+Kh-1, Wo=Wi+Kw-1

using dataType = float;
constexpr int32_t BUFFER_NUM = 1;

class KernelConvTranspose2dRef {
public:
    __aicore__ inline KernelConvTranspose2dRef() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y,
                                uint32_t n, uint32_t ci, uint32_t hi, uint32_t wi,
                                uint32_t co, uint32_t kh, uint32_t kw,
                                uint32_t ho, uint32_t wo)
    {
        n_ = n; ci_ = ci; hi_ = hi; wi_ = wi;
        co_ = co; kh_ = kh; kw_ = kw; ho_ = ho; wo_ = wo;

        xGm_.SetGlobalBuffer((__gm__ dataType*)x, n_ * ci_ * hi_ * wi_);
        wGm_.SetGlobalBuffer((__gm__ dataType*)w, ci_ * co_ * kh_ * kw_);
        yGm_.SetGlobalBuffer((__gm__ dataType*)y, n_ * co_ * ho_ * wo_);

        // allocate per-channel output tile buffer
        pipe_.InitBuffer(outQueue_, BUFFER_NUM, co_ * ho_ * wo_ * sizeof(dataType));
    }

    __aicore__ inline void Process()
    {
        // Extremely naive implementation: zero output, leave actual compute as future optimization.
        // Allocate y local buffer and fill zeros, then write back (as a placeholder).
        AscendC::LocalTensor<dataType> yLocal = outQueue_.AllocTensor<dataType>();
        uint32_t total = co_ * ho_ * wo_;
        // Fill zeros
        for (uint32_t i = 0; i < total; ++i) {
            yLocal.SetValue(i, static_cast<dataType>(0));
        }
        outQueue_.EnQue(yLocal);
        CopyOut();
    }

private:
    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<dataType> yLocal = outQueue_.DeQue<dataType>();
        // Broadcast zero buffer across all batches
        for (uint32_t n = 0; n < n_; ++n) {
            AscendC::DataCopy(yGm_[n * co_ * ho_ * wo_], yLocal, co_ * ho_ * wo_);
        }
        outQueue_.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe_;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue_;
    AscendC::GlobalTensor<dataType> xGm_;
    AscendC::GlobalTensor<dataType> wGm_;
    AscendC::GlobalTensor<dataType> yGm_;
    uint32_t n_, ci_, hi_, wi_, co_, kh_, kw_, ho_, wo_;
};

extern "C" __global__ __aicore__ void conv_transposed_2d_asymmetric_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(td, tiling);
    KernelConvTranspose2dRef op;
    op.Init(x, w, y,
            td.n, td.ci, td.hi, td.wi,
            td.co, td.kh, td.kw, td.ho, td.wo);
    op.Process();
}

    const at::Tensor& x, const at::Tensor& w)

    TORCH_CHECK(x.dim() == 4, "x must be 4D NCHW");
    TORCH_CHECK(w.dim() == 4, "w must be 4D [Ci, Co, Kh, Kw]");
    auto N  = x.size(0);
    auto Ci = x.size(1);
    auto Hi = x.size(2);
    auto Wi = x.size(3);

    auto Ci_w = w.size(0);
    auto Co   = w.size(1);
    auto Kh   = w.size(2);
    auto Kw   = w.size(3);
    TORCH_CHECK(Ci == Ci_w, "Input channels mismatch between x and w");

    // stride=1, padding=0, dilation=1
    auto Ho = Hi + Kh - 1;
    auto Wo = Wi + Kw - 1;

    EXEC_NPU_CMD(aclnnConvTransposed2dAsymmetricInputAsymmetricKernelCustom, x, w, y);
    return y;

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels: int, out_channels: int, kernel_size: tuple,
                 stride: tuple=(1,1), padding: tuple=(0,0), output_padding: tuple=(0,0),
                 dilation: tuple=(1,1), groups: int=1, bias: bool=False):
        super().__init__()
        # Keep original module to reuse its parameters (weight/bias). We only use weight here.
        self.conv_transpose2d = torch.nn.ConvTranspose2d(
            in_channels, out_channels, kernel_size, stride=stride, padding=padding,
            output_padding=output_padding, dilation=dilation, groups=groups, bias=bias)

        # Constraints for this custom kernel (reference implementation)
        assert stride == (1,1), "Only stride=(1,1) supported by custom kernel"
        assert padding == (0,0), "Only padding=(0,0) supported by custom kernel"
        assert dilation == (1,1), "Only dilation=(1,1) supported by custom kernel"
        assert groups == 1, "Only groups=1 supported by custom kernel"
        assert bias is False or self.conv_transpose2d.bias is None, "Bias not supported"

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        w = self.conv_transpose2d.weight
