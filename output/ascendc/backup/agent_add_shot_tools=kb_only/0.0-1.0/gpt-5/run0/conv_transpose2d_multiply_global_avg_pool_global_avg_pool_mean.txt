kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose2d_multiply_global_avg_pool_global_avg_pool_mean_custom_tiling_data.h"

class KernelCT2dMulGAPMean {
public:
    __aicore__ inline KernelCT2dMulGAPMean() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t outter, uint32_t inner, uint32_t n,
                                uint32_t tmpSize, float multiplier)
    {
        this->outter = outter;
        this->inner = inner;
        this->n = n;
        this->tmpSize = tmpSize;
        this->multiplier = multiplier;

        // GM buffers
        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(x), outter * inner);
        zGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(z), outter);

        // UB queues
        // Input UB holds all elements (simplified sample, actual tiling recommended in practice)
        pipe.InitBuffer(inQueueX, 1, outter * inner * sizeof(float));
        // Output aligned to ONE_BLK_SIZE
        uint32_t outAlignBytes = (outter * sizeof(float) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE;
        pipe.InitBuffer(outQueueZ, 1, outAlignBytes);
        // Scale vector buffer
        pipe.InitBuffer(tmpScaleQueue, 1, outter * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm, outter * inner);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        // Mean along the last axis (treat xLocal as [outter, inner], reduce 'n' = H*W)
        AscendC::MeanParams meanParams;
        meanParams.outter = outter;
        meanParams.inner  = inner;
        meanParams.n      = n;

        if (tmpSize != 0) {
            pipe.InitBuffer(tmpBuf, tmpSize);
            AscendC::LocalTensor<uint8_t> tmpTensor = tmpBuf.Get<uint8_t>();
            AscendC::Mean<float, float>(zLocal, xLocal, tmpTensor, meanParams);
        } else {
            AscendC::Mean<float, float>(zLocal, xLocal, meanParams);
        }

        // Multiply by scalar multiplier after mean: zLocal = zLocal * multiplier
        AscendC::LocalTensor<float> scaleLocal = tmpScaleQueue.AllocTensor<float>();
        AscendC::Duplicate(scaleLocal, multiplier, outter);
        AscendC::Mul(zLocal, zLocal, scaleLocal, outter);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        tmpScaleQueue.FreeTensor(scaleLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        uint32_t outAlignElems = (outter * sizeof(float) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(float);
        AscendC::DataCopy(zGm, zLocal, outAlignElems);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1>  inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueZ;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpScaleQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t outter;
    uint32_t inner;
    uint32_t n;
    uint32_t tmpSize;
    float multiplier;
};

extern "C" __global__ __aicore__ void conv_transpose2d_multiply_global_avg_pool_global_avg_pool_mean_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelCT2dMulGAPMean op;
    op.Init(x, z,
            tilingData.outter, tilingData.inner, tilingData.n,
            tilingData.tmpSize, tilingData.multiplier);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Fuses: multiply by scalar + global average pooling (H,W) twice into a custom AscendC kernel.
    ConvTranspose2d remains as native layer.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, multiplier):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size,
                                                       stride=stride, padding=padding, output_padding=output_padding)
        self.multiplier = float(multiplier)

    def forward(self, x):
        x = self.conv_transpose(x)
        # Fused kernel performs: x * multiplier -> mean over H,W (keepdim=True) -> mean over H,W (keepdim=True)
        x = custom_ops_lib.conv_transpose2d_multiply_global_avg_pool_global_avg_pool_mean_custom(x, self.multiplier)
        return x
