kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_standard_3d_asymmetric_input_asymmetric_kernel_custom_tiling_data.h"

// 本样例参考 AscendC 文档中的队列/拷贝/计算三段式模板（参见两输入指令模板与Mean/ConfusionTranspose示例）。
// 为了便于理解，使用 float 数据类型与简单整形下发参数，采用单核整张处理的写法示例化实现流程。
// 真实性能优化可在此基础上做分块/分片与流水优化。

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 1;

class KernelConv3D {
public:
    __aicore__ inline KernelConv3D() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y,
                                const uint32_t N, const uint32_t Cin,
                                const uint32_t D, const uint32_t H, const uint32_t W,
                                const uint32_t Cout,
                                const uint32_t Kd, const uint32_t Kh, const uint32_t Kw,
                                const uint32_t stride_d, const uint32_t stride_h, const uint32_t stride_w,
                                const uint32_t pad_d, const uint32_t pad_h, const uint32_t pad_w,
                                const uint32_t dilation_d, const uint32_t dilation_h, const uint32_t dilation_w,
                                const uint32_t groups,
                                const uint32_t Do, const uint32_t Ho, const uint32_t Wo)
    {
        // 保存参数
        this->N = N; this->Cin = Cin; this->D = D; this->H = H; this->W = W;
        this->Cout = Cout; this->Kd = Kd; this->Kh = Kh; this->Kw = Kw;
        this->stride_d = stride_d; this->stride_h = stride_h; this->stride_w = stride_w;
        this->pad_d = pad_d; this->pad_h = pad_h; this->pad_w = pad_w;
        this->dilation_d = dilation_d; this->dilation_h = dilation_h; this->dilation_w = dilation_w;
        this->groups = groups;
        this->Do = Do; this->Ho = Ho; this->Wo = Wo;

        inSize  = N * Cin * D * H * W;
        wtSize  = Cout * (Cin / groups) * Kd * Kh * Kw;
        outSize = N * Cout * Do * Ho * Wo;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(x), inSize);
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(w), wtSize);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(y), outSize);

        // 简单起见整张搬入/搬出
        pipe.InitBuffer(inQueueX, 1, inSize * sizeof(DTYPE));
        pipe.InitBuffer(inQueueW, 1, wtSize * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, 1, outSize * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> wLocal = inQueueW.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm, inSize);
        AscendC::DataCopy(wLocal, wGm, wtSize);
        inQueueX.EnQue(xLocal);
        inQueueW.EnQue(wLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> wLocal = inQueueW.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.AllocTensor<DTYPE>();

        // 朴素卷积：默认 stride=1 pad=0 dilation=1 groups=1
        // 为演示目的使用标量循环；实际工程中应采用更高效的向量API与分块/流水。
        // 获取本地指针（与 AscendC::TBuf Get 类似用法，具体接口以实际版本为准）
        DTYPE* xPtr = reinterpret_cast<DTYPE*>(xLocal.GetPhyAddr());
        DTYPE* wPtr = reinterpret_cast<DTYPE*>(wLocal.GetPhyAddr());
        DTYPE* yPtr = reinterpret_cast<DTYPE*>(yLocal.GetPhyAddr());

        // 清零输出
        for (uint32_t i = 0; i < outSize; ++i) {
            yPtr[i] = static_cast<DTYPE>(0);
        }

        const uint32_t CinPerG = Cin / groups;

        // N, Cout, Do, Ho, Wo
        for (uint32_t n = 0; n < N; ++n) {
          for (uint32_t g = 0; g < groups; ++g) {
            for (uint32_t oc = 0; oc < Cout / groups; ++oc) {
              uint32_t ocg = g * (Cout / groups) + oc;
              for (uint32_t od = 0; od < Do; ++od) {
                int32_t in_d_start = static_cast<int32_t>(od * stride_d) - static_cast<int32_t>(pad_d);
                for (uint32_t oh = 0; oh < Ho; ++oh) {
                  int32_t in_h_start = static_cast<int32_t>(oh * stride_h) - static_cast<int32_t>(pad_h);
                  for (uint32_t ow = 0; ow < Wo; ++ow) {
                    int32_t in_w_start = static_cast<int32_t>(ow * stride_w) - static_cast<int32_t>(pad_w);
                    DTYPE acc = 0;

                    for (uint32_t ic = 0; ic < CinPerG; ++ic) {
                      uint32_t icg = g * CinPerG + ic;
                      for (uint32_t kd = 0; kd < Kd; ++kd) {
                        int32_t id = in_d_start + kd * dilation_d;
                        if (id < 0 || id >= static_cast<int32_t>(D)) continue;
                        for (uint32_t kh = 0; kh < Kh; ++kh) {
                          int32_t ih = in_h_start + kh * dilation_h;
                          if (ih < 0 || ih >= static_cast<int32_t>(H)) continue;
                          for (uint32_t kw = 0; kw < Kw; ++kw) {
                            int32_t iw = in_w_start + kw * dilation_w;
                            if (iw < 0 || iw >= static_cast<int32_t>(W)) continue;

                            // x index: (((n*Cin + icg)*D + id)*H + ih)*W + iw
                            uint64_t xIdx = (((static_cast<uint64_t>(n) * Cin + icg) * D + id) * H + ih) * W + iw;

                            // w index: (((ocg)*CinPerG + ic)*Kd + kd)*Kh + kh)*Kw + kw
                            uint64_t wIdx = (((static_cast<uint64_t>(ocg) * CinPerG + ic) * Kd + kd) * Kh + kh) * Kw + kw;

                            acc += xPtr[xIdx] * wPtr[wIdx];
                          }
                        }
                      }
                    }

                    // y index: (((n*Cout + ocg)*Do + od)*Ho + oh)*Wo + ow
                    uint64_t yIdx = (((static_cast<uint64_t>(n) * Cout + ocg) * Do + od) * Ho + oh) * Wo + ow;
                    yPtr[yIdx] = acc;
                  }
                }
              }
            }
          }
        }

        outQueueY.EnQue<DTYPE>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueW.FreeTensor(wLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm, yLocal, outSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1>  inQueueX, inQueueW;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueY;

    AscendC::GlobalTensor<DTYPE> xGm, wGm, yGm;

    uint32_t N, Cin, D, H, W;
    uint32_t Cout, Kd, Kh, Kw;
    uint32_t stride_d, stride_h, stride_w;
    uint32_t pad_d, pad_h, pad_w;
    uint32_t dilation_d, dilation_h, dilation_w;
    uint32_t groups;
    uint32_t Do, Ho, Wo;

    uint32_t inSize, wtSize, outSize;
};

extern "C" __global__ __aicore__ void conv_standard_3d_asymmetric_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(t, tiling);
    KernelConv3D op;
    op.Init(x, w, y,
            t.N(), t.Cin(), t.D(), t.H(), t.W(),
            t.Cout(), t.Kd(), t.Kh(), t.Kw(),
            t.stride_d(), t.stride_h(), t.stride_w(),
            t.pad_d(), t.pad_h(), t.pad_w(),
            t.dilation_d(), t.dilation_h(), t.dilation_w(),
            t.groups(),
            t.Do(), t.Ho(), t.Wo());
    op.Process();
}
