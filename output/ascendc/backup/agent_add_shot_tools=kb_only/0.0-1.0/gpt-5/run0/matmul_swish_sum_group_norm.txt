kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matmul_swish_sum_group_norm_custom_tiling_data.h"

// 参考：
// - Ascend C 高阶API Matmul/GroupNorm/Sigmoid/Swish/Add/Mul/DataCopy 等
//   (Ascend C API Reference: 高阶API Matmul, GroupNorm, Sigmoid, Swish, Add, Mul, DataCopy, TPipe/TQue/GET_TILING_DATA/TILING_KEY_IS)
// - 本内核完成：Y = GroupNorm( Swish( X @ W^T + bias_linear ) + bias_add )

using DTYPE = float;

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulSwishSumGroupNorm {
public:
    __aicore__ inline KernelMatmulSwishSumGroupNorm() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b0, GM_ADDR b1,
                                GM_ADDR gamma, GM_ADDR beta,
                                GM_ADDR y,
                                uint32_t m, uint32_t n, uint32_t k,
                                uint32_t groups, uint32_t tileNum)
    {
        this->M = m;
        this->N = n;
        this->K = k;
        this->numGroups = groups;
        this->tileNum = tileNum;

        xGm.SetGlobalBuffer((__gm__ DTYPE*)x, (uint64_t)m * k);
        wGm.SetGlobalBuffer((__gm__ DTYPE*)w, (uint64_t)n * k);
        if (b0 != nullptr) {
            b0Gm.SetGlobalBuffer((__gm__ DTYPE*)b0, n);
            hasB0 = true;
        } else {
            hasB0 = false;
        }
        b1Gm.SetGlobalBuffer((__gm__ DTYPE*)b1, n);
        gammaGm.SetGlobalBuffer((__gm__ DTYPE*)gamma, n);
        betaGm.SetGlobalBuffer((__gm__ DTYPE*)beta, n);
        yGm.SetGlobalBuffer((__gm__ DTYPE*)y, (uint64_t)m * n);

        // 基于输出总长度进行简单切分
        totalLength = (uint64_t)m * n;
        blockLength = totalLength / AscendC::GetBlockNum();
        tileLen = blockLength / tileNum / BUFFER_NUM;

        // 为向量阶段初始化队列
        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLen * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLen * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        // 1) 矩阵乘法 Y = X @ W^T (+ bias_linear)
        // 参考 Ascend C 高阶API: Matmul/Gemm
        // 简化示例：直接调用高阶Gemm接口（伪代码示意，具体接口以版本为准）
        // AscendC::Gemm(yGm, xGm, wGm, M, N, K, false, true);
        // 这里以逐行块化为例，实际请使用 Matmul 高阶API以获得最佳性能
        MatmulCompute();

        // 2) Swish: y = sigmoid(y) * y
        // 3) y = y + bias_add
        // 以上向量操作分块执行
        int32_t loopCount = tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyInY(i);
            SwishAndBiasAdd(i);
            CopyOutY(i);
        }

        // 4) GroupNorm 按输出通道N维分组
        // 参考 Ascend C 高阶API: GroupNorm
        GroupNormCompute();
    }

private:
    __aicore__ inline void MatmulCompute()
    {
        // 这里给出一个逻辑示意（推荐使用 Matmul 高阶API）
        // 为简单起见，按行处理（不作为性能最优实现）
        // y[m, n] = sum_k x[m, k] * w[n, k] (+ b0[n])

        const uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t blockStart = blockIdx * blockLength;
        uint64_t rowStart = blockStart / N;
        uint64_t rowsToProcess = blockLength / N;

        for (uint64_t r = 0; r < rowsToProcess; ++r) {
            uint64_t mIdx = rowStart + r;
            for (uint32_t nIdx = 0; nIdx < N; ++nIdx) {
                float acc = 0.f;
                for (uint32_t kIdx = 0; kIdx < K; ++kIdx) {
                    acc += xGm.GetValue(mIdx * K + kIdx) * wGm.GetValue(nIdx * K + kIdx);
                }
                if (hasB0) {
                    acc += b0Gm.GetValue(nIdx);
                }
                yGm.SetValue(mIdx * N + nIdx, acc);
            }
        }
    }

    __aicore__ inline void CopyInY(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> yLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::DataCopy(yLocal, yGm[progress * tileLen], tileLen);
        inQueueX.EnQue(yLocal);
    }

    __aicore__ inline void SwishAndBiasAdd(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> yLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> zLocal = outQueueY.AllocTensor<DTYPE>();

        // zLocal = sigmoid(yLocal) * yLocal
        AscendC::Sigmoid(zLocal, yLocal, tileLen);
        AscendC::Mul(zLocal, zLocal, yLocal, tileLen);

        // 加 bias_add（按通道N广播）
        // 计算该tile的全局起始位置，定位到对应列通道索引
        uint64_t globalStart = AscendC::GetBlockIdx() * blockLength + progress * tileLen;
        for (uint32_t i = 0; i < tileLen; ++i) {
            uint64_t linearIndex = globalStart + i;
            uint32_t c = linearIndex % N;
            float val = zLocal.GetValue(i) + b1Gm.GetValue(c);
            zLocal.SetValue(i, val);
        }

        outQueueY.EnQue<DTYPE>(zLocal);
        inQueueX.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOutY(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> zLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm[progress * tileLen], zLocal, tileLen);
        outQueueY.FreeTensor(zLocal);
    }

    __aicore__ inline void GroupNormCompute()
    {
        // 对每个样本行做 GroupNorm，C=N 通道，分组数为 numGroups
        // 参考 Ascend C 高阶API: GroupNorm(B, C 分组)
        // 简化为行循环，组内做均值方差，再仿照 GroupNorm 计算：
        // y = (y - mean)/sqrt(var + eps) * gamma + beta

        const float eps = 1e-5f;
        const uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t blockStart = blockIdx * blockLength;
        uint64_t rowStart = blockStart / N;
        uint64_t rowsToProcess = blockLength / N;

        uint32_t groupSize = N / numGroups;

        for (uint64_t r = 0; r < rowsToProcess; ++r) {
            uint64_t mIdx = rowStart + r;

            for (uint32_t g = 0; g < numGroups; ++g) {
                uint32_t cStart = g * groupSize;
                // 均值
                float mean = 0.f;
                for (uint32_t ci = 0; ci < groupSize; ++ci) {
                    mean += yGm.GetValue(mIdx * N + cStart + ci);
                }
                mean /= (float)groupSize;

                // 方差
                float var = 0.f;
                for (uint32_t ci = 0; ci < groupSize; ++ci) {
                    float v = yGm.GetValue(mIdx * N + cStart + ci) - mean;
                    var += v * v;
                }
                var /= (float)groupSize;

                float rstd = 1.0f / AscendC::Sqrt(var + eps);

                // 归一化 + 仿射
                for (uint32_t ci = 0; ci < groupSize; ++ci) {
                    uint32_t c = cStart + ci;
                    float v = yGm.GetValue(mIdx * N + c);
                    float yhat = (v - mean) * rstd;
                    float outv = yhat * gammaGm.GetValue(c) + betaGm.GetValue(c);
                    yGm.SetValue(mIdx * N + c, outv);
                }
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> wGm;
    AscendC::GlobalTensor<DTYPE> b0Gm;
    AscendC::GlobalTensor<DTYPE> b1Gm;
    AscendC::GlobalTensor<DTYPE> gammaGm;
    AscendC::GlobalTensor<DTYPE> betaGm;
    AscendC::GlobalTensor<DTYPE> yGm;

    bool hasB0 = false;

    uint32_t M;
    uint32_t N;
    uint32_t K;
    uint32_t numGroups;

    uint64_t totalLength;
    uint64_t blockLength;
    uint64_t tileLen;
    uint32_t tileNum;
};

extern "C" __global__ __aicore__ void matmul_swish_sum_group_norm_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias_linear, GM_ADDR bias_add,
    GM_ADDR gn_gamma, GM_ADDR gn_beta, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSwishSumGroupNorm op;
    op.Init(
        x, weight, bias_linear, bias_add,
        gn_gamma, gn_beta, y,
        tiling_data.m, tiling_data.n, tiling_data.k,
        tiling_data.numGroups, tiling_data.tileNum
    );
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义 AscendC 内核将 Linear(含bias) + Swish + 加法偏置 + GroupNorm 融合为单算子
    """
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super(ModelNew, self).__init__()
        self.matmul = torch.nn.Linear(in_features, out_features)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.group_norm = torch.nn.GroupNorm(num_groups, out_features)

    def forward(self, x):
        # 权重: [out_features, in_features]，输入: [batch, in_features]
        # 自定义算子：y = GroupNorm( Swish( x @ W^T + bias_linear ) + bias_add )

            x,
            self.matmul.weight,
            self.matmul.bias,
            self.bias,
            self.group_norm.weight,
            self.group_norm.bias,
            self.group_norm.num_groups
