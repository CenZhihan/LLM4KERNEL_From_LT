host_operator_src="""
#include "group_norm_custom_tiling.h"
#include "register/op_def_registry.h"
#include "register/op_impl_registry.h"
#include "graph/tensor.h"
#include <vector>

namespace optiling {
using namespace ge;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    GroupNormCustomTilingData tiling;

    // 读取输入形状 [N, C, H, W]
    const gert::Shape* in_shape = context->GetInputShape(0);
    int64_t n = 1, c = 1, h = 1, w = 1;
    if (in_shape->GetStorageShape().GetDimNum() >= 4) {
        n = in_shape->GetStorageShape().GetDim(0);
        c = in_shape->GetStorageShape().GetDim(1);
        h = in_shape->GetStorageShape().GetDim(2);
        w = in_shape->GetStorageShape().GetDim(3);
    }

    // 获取 groups 属性
    int64_t groups = 1;
    (void)context->GetAttr("groups", groups);

    // 构造 ge::Shape，用于调用 GroupNorm Tiling 高阶 API
    std::vector<int64_t> shapeVec = {n, c, h, w};
    ge::Shape srcShape(shapeVec);

    // 依据参考文档 5.4.16：先查询临时空间范围，再计算 Tiling
    uint32_t minSize = 0;
    uint32_t maxSize = 0;
    AscendC::GetGroupNormMaxMinTmpSize(srcShape, sizeof(float), false, static_cast<uint32_t>(groups), maxSize, minSize);
    AscendC::GetGroupNormNDTilingInfo(srcShape, maxSize, sizeof(float), false, static_cast<uint32_t>(groups), tiling.groupNormTilingData);

    tiling.set_n(static_cast<uint32_t>(n));
    tiling.set_c(static_cast<uint32_t>(c));
    tiling.set_h(static_cast<uint32_t>(h));
    tiling.set_w(static_cast<uint32_t>(w));
    tiling.set_group(static_cast<uint32_t>(groups));

    // 保存 Tiling 数据
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());
    // 无需额外 workspace
    size_t* wk = context->GetWorkspaceSizes(1);
    wk[0] = 0;
    // 设置 TILING KEY
    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}

} // namespace optiling

namespace ge {

static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    // y 与 x 同形状
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;

    // mean 与 rstd 形状为 [N, groups]
    int64_t n = 1;
    int64_t groups = 1;
    const auto* in_shape = context->GetInputShape(0);
    if (in_shape->GetStorageShape().GetDimNum() >= 1) {
        n = in_shape->GetStorageShape().GetDim(0);
    }
    (void)context->GetAttr("groups", groups);
    gert::Shape* mean_shape = context->GetOutputShape(1);
    gert::Shape* rstd_shape = context->GetOutputShape(2);
    std::vector<int64_t> mv = {n, groups};
    mean_shape->SetDim(mv);
    rstd_shape->SetDim(mv);
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext* context)
{
    const auto in_dtype = context->GetInputDataType(0);
    context->SetOutputDataType(0, in_dtype);
    context->SetOutputDataType(1, in_dtype);
    context->SetOutputDataType(2, in_dtype);
    return GRAPH_SUCCESS;
}

} // namespace ge

namespace ops {
class GroupNormCustom : public OpDef {
public:
    explicit GroupNormCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("gamma")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("beta")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->Output("y")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("mean")
            .ParamType(OPTIONAL)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("rstd")
            .ParamType(OPTIONAL)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        this->Attr("groups").Int(1);

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);

        this->AICore()
            .SetTiling(optiling::TilingFunc);
        // 请根据实际芯片配置，这里示例性添加 910b
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(GroupNormCustom);
} // namespace ops

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "group_norm_custom_tiling_data.h"

using dataType = float;

extern "C" __global__ __aicore__ void group_norm_custom(GM_ADDR inputX_gm,
                                                        GM_ADDR gamma_gm,
                                                        GM_ADDR beta_gm,
                                                        GM_ADDR output_gm,
                                                        GM_ADDR outputMean_gm,
                                                        GM_ADDR outputRstd_gm,
                                                        GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    // 直接参考文档 5.4.16 的用法：将 Tiling 结构传入高阶 API 封装的 Kernel
    KernelGroupNorm<dataType, false> op;
    op.Init(inputX_gm, gamma_gm, beta_gm, output_gm, outputMean_gm, outputRstd_gm,
            tilingData.groupNormTilingData);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor group_norm_custom_impl_npu(const at::Tensor& x,
                                      const at::Tensor& gamma,
                                      const at::Tensor& beta,
                                      int64_t groups) {
    TORCH_CHECK(x.dim() >= 2, \"Input must have at least 2 dims\");
    TORCH_CHECK(gamma.dim() == 1 && beta.dim() == 1, \"gamma/beta must be 1-D\");
    TORCH_CHECK(gamma.size(0) == beta.size(0) && gamma.size(0) == x.size(1),
                \"gamma/beta size must match C dimension\");

    at::Tensor y = at::empty_like(x);
    // mean/rstd shape: [N, groups]
    auto opts = x.options();
    at::Tensor mean = at::empty({x.size(0), groups}, opts);
    at::Tensor rstd = at::empty({x.size(0), groups}, opts);

    // 将 groups 作为属性参数传入，自定义算子内部使用
    EXEC_NPU_CMD(aclnnGroupNormCustom, x, gamma, beta, groups, y, mean, rstd);
    return y;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl(\"group_norm_custom\", &group_norm_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def(\"group_norm_custom\", &group_norm_custom_impl_npu, \"GroupNorm forward\");
}
"""

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Model using custom AscendC GroupNorm kernel.
    """
    def __init__(self, num_features: int, num_groups: int):
        super().__init__()
        self.num_features = num_features
        self.num_groups = num_groups
        # 与 PyTorch GroupNorm 的可学习参数一致
        self.weight = torch.nn.Parameter(torch.ones(num_features))
        self.bias = torch.nn.Parameter(torch.zeros(num_features))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return custom_ops_lib.group_norm_custom(x, self.weight, self.bias, self.num_groups)
