namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 期望输入为4D: (N, C, H, W)，此处H、W为maxpool+hardtanh后的尺寸
    const gert::Shape* inShape = context->GetInputShape(0);

    // 获取形状各维，接口在不同版本略有差异，下述调用依赖于典型CANN接口
    // 若GetOriginShape().GetDims()不可用，可替换为等效API
    auto originShape = inShape->GetOriginShape();
    std::vector<int64_t> dims = originShape.GetDims();
    if (dims.size() != 4) {
        return ge::GRAPH_FAILED;
    }

    uint32_t n = static_cast<uint32_t>(dims[0]);
    uint32_t c = static_cast<uint32_t>(dims[1]);
    uint32_t h = static_cast<uint32_t>(dims[2]);
    uint32_t w = static_cast<uint32_t>(dims[3]);
    uint32_t planes = n * c;

    // 为Sum API计算对齐后的inner
    const uint32_t typeSize = 4; // float32
    uint32_t innerW = AlignUp32BytesElems(w, typeSize);
    uint32_t innerH = AlignUp32BytesElems(h, typeSize);

    // 计算并预留Sum API需要的临时空间（取两次归约所需临时空间的较大值）
    uint32_t maxTmpW = 0, minTmpW = 0;
    uint32_t maxTmpH = 0, minTmpH = 0;
    AscendC::GetSumMaxMinTmpSize(w, typeSize, false, maxTmpW, minTmpW);
    AscendC::GetSumMaxMinTmpSize(h, typeSize, false, maxTmpH, minTmpH);
    uint32_t tmpSize = minTmpW > minTmpH ? minTmpW : minTmpH;

    ConvTranspose2dMaxPoolHardtanhMeanTanhCustomTilingData tiling;
    tiling.set_planes(planes);
    tiling.set_height(h);
    tiling.set_width(w);
    tiling.set_innerW(innerW);
    tiling.set_innerH(innerH);
    tiling.set_tmpSize(tmpSize);

    // 设置BlockDim，样例取8
    context->SetBlockDim(8);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无额外workspace
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;

    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose2d_max_pool_hardtanh_mean_tanh_custom_tiling_data.h"

using T = DTYPE_X; // float32

constexpr int32_t QUE_NUM = 1;

inline __aicore__ uint32_t AlignUp32BytesElems(uint32_t elems) {
    uint32_t bytes = elems * sizeof(T);
    uint32_t aligned = (bytes + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE;
    return aligned / sizeof(T);
}

class KernelMeanTanh {
public:
    __aicore__ inline KernelMeanTanh() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t planes, uint32_t height, uint32_t width,
                                uint32_t innerW, uint32_t innerH, uint32_t tmpSize)
    {
        this->planes = planes;
        this->height = height;
        this->width = width;
        this->innerW = innerW;
        this->innerH = innerH;

        // 输入总元素数 = planes * height * width
        xGm.SetGlobalBuffer((__gm__ T*)x, static_cast<uint64_t>(planes) * height * width);
        yGm.SetGlobalBuffer((__gm__ T*)y, planes);

        // 为每个平面准备缓冲：整面缓冲（height * innerW），行和缓冲（对齐到32B），标量缓冲（1元素对齐）
        pipe.InitBuffer(inPlaneQueue, QUE_NUM, static_cast<uint64_t>(height) * innerW * sizeof(T));
        pipe.InitBuffer(rowSumQueue, QUE_NUM, AlignUp32BytesElems(height) * sizeof(T));
        pipe.InitBuffer(outScalarQueue, QUE_NUM, AscendC::ONE_BLK_SIZE); // 至少32字节
        pipe.InitBuffer(workQueue, QUE_NUM, tmpSize);
    }

    __aicore__ inline void Process()
    {
        // 简洁处理：将planes平均分配到block
        uint32_t blocks = AscendC::GetBlockNum();
        uint32_t bid = AscendC::GetBlockIdx();
        uint32_t planesPerBlock = planes / blocks;
        uint32_t tail = planes % blocks;
        uint32_t start = bid * planesPerBlock + AscendC::Min(bid, tail);
        uint32_t count = planesPerBlock + (bid < tail ? 1 : 0);

        for (uint32_t p = 0; p < count; ++p) {
            uint32_t planeIdx = start + p;
            CopyIn(planeIdx);
            Compute(planeIdx);
            CopyOut(planeIdx);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t planeIdx)
    {
        // 将该plane的数据按行拷入到对齐缓冲中，多余的对齐空间置0
        AscendC::LocalTensor<T> planeLocal = inPlaneQueue.AllocTensor<T>();
        uint32_t planeSizeAligned = height * innerW;
        T zero(0);
        AscendC::Duplicate<T>(planeLocal, zero, planeSizeAligned);

        uint64_t base = static_cast<uint64_t>(planeIdx) * height * width;
        for (uint32_t h = 0; h < height; ++h) {
            AscendC::DataCopy(planeLocal[h * innerW], xGm[base + static_cast<uint64_t>(h) * width], width);
        }
        inPlaneQueue.EnQue(planeLocal);
    }

    __aicore__ inline void Compute(uint32_t /*planeIdx*/)
    {
        // 两次Sum归约：先按行对W求和，再对行向量求和 -> 得到标量；随后除以元素数并Tanh
        AscendC::LocalTensor<T> planeLocal = inPlaneQueue.DeQue<T>();
        AscendC::LocalTensor<uint8_t> workLocal = workQueue.AllocTensor<uint8_t>();
        AscendC::LocalTensor<T> rowSumLocal = rowSumQueue.AllocTensor<T>();
        // 先对每行(width)做Sum归约
        AscendC::SumParams paramsW { height, innerW, width };
        AscendC::Sum(rowSumLocal, planeLocal, workLocal, paramsW);

        // 将行和向量拷入到对齐的一维缓冲再做第二次Sum
        AscendC::LocalTensor<T> scalarLocal = outScalarQueue.AllocTensor<T>();
        // 准备对齐的H缓冲并清零
        uint32_t innerHAligned = AlignUp32BytesElems(height);
        // 为简化，复用planeLocal的前innerHAligned位置作为中间一维缓冲
        AscendC::Duplicate<T>(planeLocal, T(0), innerHAligned);
        AscendC::DataCopy(planeLocal, rowSumLocal, height);

        AscendC::SumParams paramsH { 1, innerHAligned, height };
        AscendC::Sum(scalarLocal, planeLocal, workLocal, paramsH);

        // 求均值并tanh
        float denom = static_cast<float>(height) * static_cast<float>(width);
        float inv = 1.0f / denom;
        AscendC::Muls(scalarLocal, scalarLocal, inv, 1);
        AscendC::Tanh(scalarLocal, scalarLocal, 1);

        outScalarQueue.EnQue<T>(scalarLocal);
        workQueue.FreeTensor(workLocal);
        rowSumQueue.FreeTensor(rowSumLocal);
        inPlaneQueue.FreeTensor(planeLocal);
    }

    __aicore__ inline void CopyOut(uint32_t planeIdx)
    {
        AscendC::LocalTensor<T> scalarLocal = outScalarQueue.DeQue<T>();
        AscendC::DataCopy(yGm[planeIdx], scalarLocal, 1);
        outScalarQueue.FreeTensor(scalarLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  QUE_NUM> inPlaneQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, QUE_NUM> rowSumQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, QUE_NUM> outScalarQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, QUE_NUM> workQueue;

    AscendC::GlobalTensor<T> xGm;
    AscendC::GlobalTensor<T> yGm;

    uint32_t planes;
    uint32_t height;
    uint32_t width;
    uint32_t innerW;
    uint32_t innerH;
};

extern "C" __global__ __aicore__ void conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMeanTanh op;
    op.Init(x, y,
            tiling_data.planes, tiling_data.height, tiling_data.width,
            tiling_data.innerW, tiling_data.innerH, tiling_data.tmpSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    仅将 mean + tanh 替换为自定义AscendC算子；conv_transpose2d / max_pool / hardtanh仍使用原生模块。
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding,
                 maxpool_kernel_size, maxpool_stride, hardtanh_min, hardtanh_max):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.maxpool = torch.nn.MaxPool2d(kernel_size=maxpool_kernel_size, stride=maxpool_stride)
        self.hardtanh = torch.nn.Hardtanh(min_val=hardtanh_min, max_val=hardtanh_max)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.maxpool(x)
        x = self.hardtanh(x)
        # 自定义算子：对(H,W)进行均值并tanh，输出形状(N,C,1,1)
        x = custom_ops_lib.conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(x)
        return x
