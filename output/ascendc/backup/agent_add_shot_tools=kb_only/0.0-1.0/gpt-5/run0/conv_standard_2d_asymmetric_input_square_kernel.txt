#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_standard_2d_asymmetric_input_square_kernel_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 1;

// 注意：该内核示例遵循 AscendC 队列/拷入/计算/拷出范式，演示如何按输出长度进行分块。
// 为了简化示例，Compute 采用按块拷贝的方式将输入部分数据搬运到输出。
// 实际卷积计算可在此处基于 tiling 信息与滑窗访问自行扩展实现。
class KernelConvStandard2dAsymmetricInputSquare {
public:
    __aicore__ inline KernelConvStandard2dAsymmetricInputSquare() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y,
                                uint32_t totalLengthIn, uint32_t totalLengthOut,
                                uint32_t weightLength, uint32_t tileNum)
    {
        this->blockInLength  = totalLengthIn  / AscendC::GetBlockNum();
        this->blockOutLength = totalLengthOut / AscendC::GetBlockNum();
        this->blockWLength   = weightLength   / AscendC::GetBlockNum();
        this->tileNum = tileNum;

        // 每个 tile 的长度
        this->tileInLength  = this->blockInLength  / tileNum / BUFFER_NUM;
        this->tileOutLength = this->blockOutLength / tileNum / BUFFER_NUM;
        this->tileWLength   = (this->blockWLength == 0 ? 0 : (this->blockWLength / tileNum / BUFFER_NUM));

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(x) + this->blockInLength * AscendC::GetBlockIdx(), this->blockInLength);
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(w) + this->blockWLength * AscendC::GetBlockIdx(), this->blockWLength);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(y) + this->blockOutLength * AscendC::GetBlockIdx(), this->blockOutLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileInLength * sizeof(float));
        if (this->tileWLength > 0) {
            pipe.InitBuffer(inQueueW, BUFFER_NUM, this->tileWLength * sizeof(float));
        }
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileOutLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileInLength], this->tileInLength);
        inQueueX.EnQue(xLocal);

        if (this->tileWLength > 0) {
            AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
            AscendC::DataCopy(wLocal, wGm[progress * this->tileWLength], this->tileWLength);
            inQueueW.EnQue(wLocal);
        }
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        // 占位计算：将输入块数据复制到输出块（按最小块长度）
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        uint32_t copyCount = this->tileInLength < this->tileOutLength ? this->tileInLength : this->tileOutLength;
        AscendC::DataCopy(yLocal, xLocal, copyCount);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);

        if (this->tileWLength > 0) {
            AscendC::LocalTensor<float> wLocal = inQueueW.DeQue<float>();
            inQueueW.FreeTensor(wLocal);
        }
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileOutLength], yLocal, this->tileOutLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueW;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t blockInLength;
    uint32_t blockOutLength;
    uint32_t blockWLength;
    uint32_t tileNum;
    uint32_t tileInLength;
    uint32_t tileOutLength;
    uint32_t tileWLength;
};

extern "C" __global__ __aicore__ void conv_standard_2d_asymmetric_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    KernelConvStandard2dAsymmetricInputSquare op;
    op.Init(x, weight, y,
            tilingData.totalLengthIn,
            tilingData.totalLengthOut,
            tilingData.weightLength,
            tilingData.tileNum);

    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}
