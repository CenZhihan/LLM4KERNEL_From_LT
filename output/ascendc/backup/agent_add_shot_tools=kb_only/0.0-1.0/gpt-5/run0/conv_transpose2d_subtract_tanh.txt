kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose2d_subtract_tanh_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

// This kernel expects:
// x: [N, C, H, W] (float32), contiguous
// bias: [C, H, W] (float32), contiguous (expanded from [C,1,1])
// y: [N, C, H, W] (float32)
// It performs: y = tanh(x - bias), broadcasting bias per-channel over H*W, for all batches.
class KernelConvTranspose2dSubtractTanh {
public:
    __aicore__ inline KernelConvTranspose2dSubtractTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                uint32_t totalLength, uint32_t hw, uint32_t c, uint32_t tileNum)
    {
        this->hw = hw;                   // H*W
        this->c = c;                     // channels
        this->tileNum = tileNum;         // (N*C)/blockDim
        this->tileLength = hw;           // per tile = H*W
        this->blockLength = this->tileNum * this->tileLength; // number of elements in this block
        this->startNc = this->tileNum * AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        // bias is shared across batch dim; length = C*H*W
        biasGm.SetGlobalBuffer((__gm__ DTYPE_X *)bias, this->c * this->hw);
        yGm.SetGlobalBuffer((__gm__ DTYPE_X *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = static_cast<int32_t>(this->tileNum);
        // Double-buffering
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        // Determine channel index for this tile
        uint32_t globalNcIdx = this->startNc + static_cast<uint32_t>(progress);
        uint32_t chIdx = globalNcIdx % this->c;

        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> bLocal = inQueueB.AllocTensor<DTYPE_X>();

        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(bLocal, biasGm[chIdx * this->tileLength], this->tileLength);

        inQueueX.EnQue(xLocal);
        inQueueB.EnQue(bLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> bLocal = inQueueB.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> yLocal = outQueueY.AllocTensor<DTYPE_X>();

        // y = x - bias
        AscendC::Sub(yLocal, xLocal, bLocal, this->tileLength);
        // y = tanh(y)
        AscendC::Tanh(yLocal, yLocal, this->tileLength);

        outQueueY.EnQue<DTYPE_X>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueB.FreeTensor(bLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> yLocal = outQueueY.DeQue<DTYPE_X>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM>  inQueueX, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_X> biasGm;
    AscendC::GlobalTensor<DTYPE_X> yGm;

    uint32_t hw;
    uint32_t c;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t blockLength;
    uint32_t startNc;
};

extern "C" __global__ __aicore__ void conv_transpose2d_subtract_tanh_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose2dSubtractTanh op;
    op.Init(x, bias, y, tiling_data.totalLength, tiling_data.hw, tiling_data.c, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model: keep ConvTranspose2d from PyTorch, fuse bias-sub and tanh into a custom AscendC op.
    The custom op expects bias expanded to [C, H, W] and contiguous.
    """
    def __init__(self, in_channels, out_channels, kernel_size, bias_shape, stride=2, padding=1, output_padding=1):
        super().__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size,
                                                       stride=stride, padding=padding, output_padding=output_padding)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv_transpose(x)  # [N, C, H, W]
        # Expand bias from [C,1,1] to [C,H,W] for per-channel broadcast and ensure contiguous
        b = self.bias.expand(x.size(1), x.size(2), x.size(3)).contiguous()
        x = custom_ops_lib.conv_transpose2d_subtract_tanh_custom(x, b)
        return x
