host_operator_src="""
#include "register/op_def_registry.h"
#include "graph/tensor.h"
#include "graph/utils/type_utils.h"
#include "matmul_tiling.h"
#include "platform/platform_ascendc.h"

namespace {
inline int64_t GetDimSafe(const gert::Shape& s, size_t idx) {
    if (idx >= s.GetDimNum()) return 0;
    return static_cast<int64_t>(s.GetDim(idx));
}
}

namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 读取输入形状: A[B, M, K], B[B, K, N]
    const gert::Shape* a_shape = context->GetInputShape(0);
    const gert::Shape* b_shape = context->GetInputShape(1);
    const auto a_origin = a_shape->GetOriginShape();
    const auto b_origin = b_shape->GetOriginShape();

    int64_t B = GetDimSafe(a_origin, 0);
    int64_t M = GetDimSafe(a_origin, 1);
    int64_t K = GetDimSafe(a_origin, 2);
    int64_t N = GetDimSafe(b_origin, 2);

    // 平台与tiler初始化
    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    matmul_tiling::MultiCoreMatmulTiling tiling(ascendcPlatform);

    tiling->SetDim(1);
    tiling->SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling->SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    tiling->SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);

    tiling->SetShape((int32_t)M, (int32_t)N, (int32_t)K);
    tiling->SetOrgShape((int32_t)M, (int32_t)N, (int32_t)K);
    tiling->SetBufferSpace(-1, -1, -1);

    // NORMAL(BMNK)布局下设置批次与轴信息
    tiling->SetBatchInfoForNormal((int32_t)B, (int32_t)B, (int32_t)M, (int32_t)N, (int32_t)K);

    tiling->SetBufferSpace(-1, -1, -1);

    optiling::TCubeTiling tilingData;
    int ret = tiling.GetTiling(tilingData);
    if (ret != 0) {
        return ge::GRAPH_FAILED;
    }

    // block dim 选择固定值或依据平台设置。这里使用8
    const uint32_t BLOCK_DIM = 8;
    context->SetBlockDim(BLOCK_DIM);

    // workspace 设为0
    size_t* ws = context->GetWorkspaceSizes(1);
    ws[0] = 0;

    // 将TCubeTiling直接写入raw tiling buffer
    auto* raw = context->GetRawTilingData();
    auto* buf = raw->GetData();
    auto cap = raw->GetCapacity();
    if (cap < sizeof(optiling::TCubeTiling)) {
        return ge::GRAPH_FAILED;
    }
    // 以内存形式保存
    auto* src = reinterpret_cast<const uint8_t*>(&tilingData);
    for (size_t i = 0; i < sizeof(optiling::TCubeTiling); ++i) {
        buf[i] = src[i];
    }
    raw->SetDataSize(sizeof(optiling::TCubeTiling));
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* a_shape = context->GetInputShape(0);
    const gert::Shape* b_shape = context->GetInputShape(1);
    const auto a_origin = a_shape->GetOriginShape();
    const auto b_origin = b_shape->GetOriginShape();

    int64_t B = (int64_t)a_origin.GetDim(0);
    int64_t M = (int64_t)a_origin.GetDim(1);
    int64_t K1 = (int64_t)a_origin.GetDim(2);
    int64_t B2 = (int64_t)b_origin.GetDim(0);
    int64_t K2 = (int64_t)b_origin.GetDim(1);
    int64_t N = (int64_t)b_origin.GetDim(2);

    if (B != B2 || K1 != K2) {
        return GRAPH_FAILED;
    }

    gert::Shape* c_shape = context->GetOutputShape(0);
    gert::Shape out;
    out.SetDimNum(3);
    out.SetDim(0, B);
    out.SetDim(1, M);
    out.SetDim(2, N);
    *c_shape = out;
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

extern "C" __global__ __aicore__ void batched_matrix_multiplication_custom(GM_ADDR aGM, GM_ADDR bGM, GM_ADDR cGM, GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    // 读取TCubeTiling
    optiling::TCubeTiling tiling;
    {
        auto tempTilingGM = (__gm__ uint32_t*)tilingGM;
        auto tempTiling = (uint32_t*)&tiling;
        for (int i = 0; i < (int)(sizeof(optiling::TCubeTiling) / sizeof(uint32_t)); ++i, ++tempTilingGM, ++tempTiling) {
            *tempTiling = *tempTilingGM;
        }
    }

    // 定义Matmul类型：NORMAL(BMNK)布局，GM-ND，float
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> aType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> bType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::NORMAL> cType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float> biasType;

    set_atomic_none();

    // GM张量包装
    AscendC::GlobalTensor<float> aGlobal;
    AscendC::GlobalTensor<float> bGlobal;
    AscendC::GlobalTensor<float> cGlobal;

    // 从tiling中获取 NORMAL 模式的基本轴信息（示意使用，具体字段由TCubeTiling内部定义）
    // 假设已通过SetBatchInfoForNormal设置了batchA/batchB与m/n/k
    int32_t batchA = tiling.batchA;
    int32_t batchB = tiling.batchB;
    int32_t M = tiling.m;
    int32_t N = tiling.n;
    int32_t K = tiling.k;

    int64_t sizeA = (int64_t)batchA * M * K;
    int64_t sizeB = (int64_t)batchB * K * N;
    int64_t sizeC = (int64_t)batchA * M * N;

    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(aGM), (uint32_t)sizeA);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(bGM), (uint32_t)sizeB);
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(cGM), (uint32_t)sizeC);

    // 创建Matmul实例
    matmul::Matmul<aType, bType, cType, biasType> mm;
    AscendC::TPipe pipe;
    g_cubeTPipePtr = &pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm);
    mm.Init(&tiling);

    // 设置A, B, C
    mm.SetTensorA(aGlobal[0], false);
    mm.SetTensorB(bGlobal[0], false);
    mm.SetTensorC(cGlobal[0]);
    mm.SetWorkspace(workspaceGM, 0);

    // 计算跨批次的步长（相邻nd矩阵起始地址偏移，单位：元素）
    uint32_t matrixStrideA = (uint32_t)(M * K);
    uint32_t matrixStrideB = (uint32_t)(K * N);
    uint32_t matrixStrideC = (uint32_t)(M * N);

    // 批计算：一次调用内部执行N次IterateBatch
    // batchLoop设为1，按传入的batchA/batchB跨批迭代
    bool enSequentialWrite = true;
    mm.IterateNBatch(1, (uint32_t)batchA, (uint32_t)batchB, enSequentialWrite, matrixStrideA, matrixStrideB, matrixStrideC);
}
