#include "max_pooling_2d_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/platform/platform_ascendc.h"

namespace optiling {
static inline uint32_t DivDown(int64_t a, int64_t b) {
    return static_cast<uint32_t>(a / b);
}

static inline uint32_t FloorDiv(int64_t a, int64_t b) {
    return static_cast<uint32_t>(a / b);
}

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    MaxPooling2dCustomTilingData tiling;

    const gert::StorageShape *in_shape = context->GetInputShape(0);
    // Expect NCHW
    uint32_t n = in_shape->GetStorageShape().GetDim(0);
    uint32_t c = in_shape->GetStorageShape().GetDim(1);
    uint32_t h = in_shape->GetStorageShape().GetDim(2);
    uint32_t w = in_shape->GetStorageShape().GetDim(3);

    const gert::RuntimeAttrs *attrs = context->GetAttrs();
    const uint32_t kernel_size = *(attrs->GetAttrPointer<uint32_t>(0));
    const uint32_t stride      = *(attrs->GetAttrPointer<uint32_t>(1));
    const uint32_t padding     = *(attrs->GetAttrPointer<uint32_t>(2));

    // Output shape formula: floor((H + 2P - K)/S) + 1
    uint32_t out_h = 0;
    uint32_t out_w = 0;
    if (h + 2 * padding >= kernel_size) {
        out_h = static_cast<uint32_t>((static_cast<int64_t>(h) + 2 * padding - kernel_size) / stride + 1);
    } else {
        out_h = 0;
    }
    if (w + 2 * padding >= kernel_size) {
        out_w = static_cast<uint32_t>((static_cast<int64_t>(w) + 2 * padding - kernel_size) / stride + 1);
    } else {
        out_w = 0;
    }

    tiling.set_n(n);
    tiling.set_c(c);
    tiling.set_h(h);
    tiling.set_w(w);
    tiling.set_out_h(out_h);
    tiling.set_out_w(out_w);
    tiling.set_kernel_size(kernel_size);
    tiling.set_stride(stride);
    tiling.set_padding(padding);

    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    auto aicNum = ascendcPlatform.GetCoreNumAic();
    if (aicNum == 0) { aicNum = 1; }
    context->SetBlockDim(aicNum);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    const auto *attrs = context->GetAttrs();

    int64_t n = x_shape->GetOriginShape().GetDim(0);
    int64_t c = x_shape->GetOriginShape().GetDim(1);
    int64_t h = x_shape->GetOriginShape().GetDim(2);
    int64_t w = x_shape->GetOriginShape().GetDim(3);

    int64_t k = *(attrs->GetAttrPointer<int64_t>(0));
    int64_t s = *(attrs->GetAttrPointer<int64_t>(1));
    int64_t p = *(attrs->GetAttrPointer<int64_t>(2));

    int64_t out_h = (h + 2 * p - k) / s + 1;
    int64_t out_w = (w + 2 * p - k) / s + 1;
    if (out_h < 0) out_h = 0;
    if (out_w < 0) out_w = 0;

    gert::Shape* y_shape = context->GetOutputShape(0);
    y_shape->SetDimNum(4);
    y_shape->SetDim(0, n);
    y_shape->SetDim(1, c);
    y_shape->SetDim(2, out_h);
    y_shape->SetDim(3, out_w);
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "max_pooling_2d_custom_tiling_data.h"

class KernelMaxPool2D {
public:
    __aicore__ inline KernelMaxPool2D() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w,
                                uint32_t out_h, uint32_t out_w,
                                uint32_t kernel_size, uint32_t stride, uint32_t padding)
    {
        this->n = n; this->c = c; this->h = h; this->w = w;
        this->out_h = out_h; this->out_w = out_w;
        this->k = kernel_size; this->s = stride; this->p = padding;

        xGm.SetGlobalBuffer((__gm__ float *)x, static_cast<uint64_t>(n) * c * h * w);
        yGm.SetGlobalBuffer((__gm__ float *)y, static_cast<uint64_t>(n) * c * out_h * out_w);

        // Per (n,c) slice buffer: full H*W in, full out_h*out_w out
        uint64_t inSliceSize = static_cast<uint64_t>(h) * w * sizeof(float);
        uint64_t outSliceSize = static_cast<uint64_t>(out_h) * out_w * sizeof(float);
        pipe.InitBuffer(inQueue, 1, inSliceSize);
        pipe.InitBuffer(outQueue, 1, outSliceSize);
    }

    __aicore__ inline void Process()
    {
        uint32_t nc = n * c;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t chunk = (nc + blockNum - 1) / blockNum;
        uint32_t start = blockIdx * chunk;
        uint32_t end = start + chunk;
        if (end > nc) end = nc;

        for (uint32_t idx = start; idx < end; ++idx) {
            CopyIn(idx);
            Compute();
            CopyOut(idx);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t ncIndex)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        uint64_t inOffset = static_cast<uint64_t>(ncIndex) * h * w;
        AscendC::DataCopy(xLocal, xGm[inOffset], h * w);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueue.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueue.AllocTensor<float>();

        // Naive max pooling over local buffer
        for (uint32_t oh = 0; oh < out_h; ++oh) {
            int32_t hstart = static_cast<int32_t>(oh * s) - static_cast<int32_t>(p);
            for (uint32_t ow = 0; ow < out_w; ++ow) {
                int32_t wstart = static_cast<int32_t>(ow * s) - static_cast<int32_t>(p);
                float maxv = -3.402823466e+38F; // -FLT_MAX
                for (uint32_t kh = 0; kh < k; ++kh) {
                    int32_t ih = hstart + static_cast<int32_t>(kh);
                    if (ih < 0 || ih >= static_cast<int32_t>(h)) continue;
                    for (uint32_t kw = 0; kw < k; ++kw) {
                        int32_t iw = wstart + static_cast<int32_t>(kw);
                        if (iw < 0 || iw >= static_cast<int32_t>(w)) continue;
                        uint32_t inIdx = static_cast<uint32_t>(ih) * w + static_cast<uint32_t>(iw);
                        // Assume LocalTensor supports element access
                        float val = xLocal.GetValue(inIdx);
                        if (val > maxv) { maxv = val; }
                    }
                }
                uint32_t outIdx = oh * out_w + ow;
                yLocal.SetValue(outIdx, maxv);
            }
        }

        outQueue.EnQue<float>(yLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t ncIndex)
    {
        AscendC::LocalTensor<float> yLocal = outQueue.DeQue<float>();
        uint64_t outOffset = static_cast<uint64_t>(ncIndex) * out_h * out_w;
        AscendC::DataCopy(yGm[outOffset], yLocal, out_h * out_w);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t n, c, h, w;
    uint32_t out_h, out_w;
    uint32_t k, s, p;
};

extern "C" __global__ __aicore__ void max_pooling_2d_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMaxPool2D op;
    op.Init(x, y,
            tiling_data.n(), tiling_data.c(), tiling_data.h(), tiling_data.w(),
            tiling_data.out_h(), tiling_data.out_w(),
            tiling_data.kernel_size(), tiling_data.stride(), tiling_data.padding());
    op.Process();
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

static inline int64_t calc_out(int64_t in, int64_t k, int64_t s, int64_t p) {
    return (in + 2 * p - k) / s + 1;
}

at::Tensor max_pooling_2d_custom_impl_npu(const at::Tensor& x, int64_t kernel_size, int64_t stride, int64_t padding) {
    TORCH_CHECK(x.dim() == 4, "Input must be NCHW 4D tensor");
    auto N = x.size(0);
    auto C = x.size(1);
    auto H = x.size(2);
    auto W = x.size(3);
    int64_t outH = calc_out(H, kernel_size, stride, padding);
    int64_t outW = calc_out(W, kernel_size, stride, padding);
    at::Tensor result = at::empty({N, C, outH, outW}, x.options());
    EXEC_NPU_CMD(aclnnMaxPooling2dCustom, x, kernel_size, stride, padding, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("max_pooling_2d_custom", &max_pooling_2d_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("max_pooling_2d_custom", &max_pooling_2d_custom_impl_npu, "MaxPooling2dCustom");
}
