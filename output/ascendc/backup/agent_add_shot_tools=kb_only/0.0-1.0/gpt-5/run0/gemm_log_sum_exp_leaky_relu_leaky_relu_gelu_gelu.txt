kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "gemm_log_sum_exp_leaky_relu_leaky_relu_gelu_gelu_custom_tiling_data.h"

using DTYPE_X = float;
using DTYPE_Y = float;

constexpr int32_t IN_BUFF_NUM = 2;
constexpr int32_t OUT_BUFF_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t rows, uint32_t cols, uint32_t tileRows)
    {
        this->rows = rows;
        this->cols = cols;
        this->tileRows = tileRows;

        // 基于block划分行区间
        uint32_t blocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (rows + blocks - 1) / blocks; // ceil
        this->startRow = this->rowsPerBlock * blockIdx;
        // 实际有效行数
        this->validRows = (startRow + rowsPerBlock <= rows) ? rowsPerBlock : (rows > startRow ? rows - startRow : 0);

        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x + (uint64_t)startRow * cols, (uint64_t)validRows * cols);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y + startRow, validRows);

        pipe.InitBuffer(inQueueX, IN_BUFF_NUM, (uint32_t)(tileRows * cols * sizeof(DTYPE_X)));
        pipe.InitBuffer(outQueueY, OUT_BUFF_NUM, (uint32_t)(tileRows * sizeof(DTYPE_Y)));

        // 计算用中间缓冲：exp输入块大小
        pipe.InitBuffer(calcBufs, (uint32_t)(tileRows * cols * sizeof(DTYPE_X)));
    }

    __aicore__ inline void Process()
    {
        if (validRows == 0) { return; }
        uint32_t tiles = (validRows + tileRows - 1) / tileRows;
        for (uint32_t t = 0; t < tiles; ++t) {
            uint32_t rowsThisTile = GetRowsThisTile(t);
            CopyIn(t, rowsThisTile);
            Compute(t, rowsThisTile);
            CopyOut(t, rowsThisTile);
        }
    }

private:
    __aicore__ inline uint32_t GetRowsThisTile(uint32_t t) const
    {
        uint32_t rowStart = t * tileRows;
        uint32_t remain = (rowStart < validRows) ? (validRows - rowStart) : 0;
        return remain > tileRows ? tileRows : remain;
    }

    __aicore__ inline void CopyIn(uint32_t t, uint32_t rowsThisTile)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        uint64_t offset = (uint64_t)t * tileRows * cols;
        AscendC::DataCopy(xLocal, xGm[offset], (uint32_t)(rowsThisTile * cols));
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t t, uint32_t rowsThisTile)
    {
        // 取出输入tile
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();

        // 中间缓冲：存放exp(x)
        AscendC::LocalTensor<DTYPE_X> expLocal = calcBufs.Get<DTYPE_X>();
        // 逐元素 exp
        AscendC::Exp(expLocal, xLocal, (uint32_t)(rowsThisTile * cols));

        // 对最后一维（列）求和：得到每行的 sum(exp(x))
        AscendC::LocalTensor<DTYPE_Y> lseLocal = outQueueY.AllocTensor<DTYPE_Y>();
        // Sum: 按最后一维汇总，参考《Ascend C API Reference》中 Table 1-14 Sum
        AscendC::Sum(lseLocal, expLocal, rowsThisTile, cols);

        // log(sum(exp(x)))
        AscendC::Log(lseLocal, lseLocal, rowsThisTile);

        // 两次 LeakyReLU，负斜率 0.01
        const DTYPE_Y slope = (DTYPE_Y)0.01f;
        AscendC::LeakyRelu(lseLocal, lseLocal, slope, rowsThisTile);
        AscendC::LeakyRelu(lseLocal, lseLocal, slope, rowsThisTile);

        // 两次 GELU（使用 FasterGeluV2 高级API，参考文档 5.3.3.3）
        AscendC::FasterGeluV2(lseLocal, lseLocal, rowsThisTile);
        AscendC::FasterGeluV2(lseLocal, lseLocal, rowsThisTile);

        // 推出
        outQueueY.EnQue<DTYPE_Y>(lseLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t t, uint32_t rowsThisTile)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        uint64_t offset = (uint64_t)t * tileRows;
        AscendC::DataCopy(yGm[offset], yLocal, rowsThisTile);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, IN_BUFF_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, OUT_BUFF_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> calcBufs;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t rows;
    uint32_t cols;
    uint32_t tileRows;

    uint32_t rowsPerBlock;
    uint32_t startRow;
    uint32_t validRows;
};

extern "C" __global__ __aicore__
void gemm_log_sum_exp_leaky_relu_leaky_relu_gelu_gelu_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, y, tiling_data.rows, tiling_data.cols, tiling_data.tileRows);
    op.Process();
}
