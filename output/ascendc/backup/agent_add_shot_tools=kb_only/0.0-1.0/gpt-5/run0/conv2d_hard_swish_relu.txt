kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_hard_swish_relu_custom_tiling_data.h"

// 参考文档：
// - 矢量标量运算 Adds/Muls/Maxs/Mins、元素乘 Mul、Relu 等，见 Ascend C API Basic APIs（向量/标量指令）。
// - Conv2D 属于 ISASI 类 API（Conv2D Performs 2D convolution...）。
// - 本内核示例演示将 Conv2D + HardSwish + ReLU 融合计算到一个 AscendC 内核中。

constexpr int32_t BUFFER_NUM = 1;

class KernelConv2dHardSwishRelu {
public:
    __aicore__ inline KernelConv2dHardSwishRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
                                uint32_t N, uint32_t CIn, uint32_t H, uint32_t W,
                                uint32_t COut, uint32_t Kh, uint32_t Kw,
                                uint32_t Stride, uint32_t Pad,
                                uint32_t Ho, uint32_t Wo)
    {
        this->N = N; this->CIn = CIn; this->H = H; this->W = W;
        this->COut = COut; this->Kh = Kh; this->Kw = Kw;
        this->Stride = Stride; this->Pad = Pad; this->Ho = Ho; this->Wo = Wo;

        uint32_t xSize = N * CIn * H * W;
        uint32_t wSize = COut * CIn * Kh * Kw;
        uint32_t bSize = COut;
        uint32_t ySize = N * COut * Ho * Wo;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, xSize);
        wGm.SetGlobalBuffer((__gm__ DTYPE_W *)w, wSize);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, bSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, ySize);

        // 简化实现：一次性搬运，演示融合流程
        pipe.InitBuffer(inQueueX, BUFFER_NUM, xSize * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, wSize * sizeof(DTYPE_W));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, bSize * sizeof(DTYPE_B));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, ySize * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        uint32_t xSize = N * CIn * H * W;
        uint32_t wSize = COut * CIn * Kh * Kw;
        uint32_t bSize = COut;

        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_W> wLocal = inQueueW.AllocTensor<DTYPE_W>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();

        AscendC::DataCopy(xLocal, xGm, xSize);
        AscendC::DataCopy(wLocal, wGm, wSize);
        AscendC::DataCopy(bLocal, bGm, bSize);

        inQueueX.EnQue(xLocal);
        inQueueW.EnQue(wLocal);
        inQueueB.EnQue(bLocal);
    }

    __aicore__ inline void Compute()
    {
        uint32_t ySize = N * COut * Ho * Wo;

        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_W> wLocal = inQueueW.DeQue<DTYPE_W>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.DeQue<DTYPE_B>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        // 1) 卷积：调用 ISASI Conv2D（接口原型见文档 1.9 ISASI APIs -> Conv2D）
        // 具体参数结构体在商业文档中定义，此处示范调用流程。
        // AscendC::Conv2D(yLocal, xLocal, wLocal, bLocal, /*stride*/Stride, /*pad*/Pad, /*dilation*/1, /*groups*/1,
        //                 /*shape*/N, CIn, H, W, COut, Kh, Kw, Ho, Wo);

        // 注意：若无法直接调用 Conv2D，可在工程中使用已有算子包装或采用官方样例接口。

        // 2) 激活：HardSwish(x) = x * ReLU6(x + 3) / 6，然后再接 ReLU
        // 参考向量/标量指令：Adds/Muls/Maxs/Mins、Mul、Relu
        // 为演示，先将卷积结果 yLocal 原地进行 HardSwish + ReLU
        // tmp = y + 3
        AscendC::LocalTensor<DTYPE_Y> tmpLocal = yLocal; // 复用
        AscendC::Adds(tmpLocal, yLocal, static_cast<DTYPE_Y>(3.0f), ySize);
        // tmp = min(max(tmp, 0), 6)
        AscendC::Maxs(tmpLocal, tmpLocal, static_cast<DTYPE_Y>(0.0f), ySize);
        AscendC::Mins(tmpLocal, tmpLocal, static_cast<DTYPE_Y>(6.0f), ySize);
        // y = y * tmp
        AscendC::Mul(yLocal, yLocal, tmpLocal, ySize);
        // y = y / 6
        AscendC::Muls(yLocal, yLocal, static_cast<DTYPE_Y>(1.0f/6.0f), ySize);
        // y = ReLU(y)
        AscendC::Relu(yLocal, yLocal, ySize);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueW.FreeTensor(wLocal);
        inQueueB.FreeTensor(bLocal);
    }

    __aicore__ inline void CopyOut()
    {
        uint32_t ySize = N * COut * Ho * Wo;
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm, yLocal, ySize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX, inQueueW, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_W> wGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t N, CIn, H, W, COut, Kh, Kw, Stride, Pad, Ho, Wo;
};

extern "C" __global__ __aicore__ void conv2d_hard_swish_relu_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    KernelConv2dHardSwishRelu op;
    op.Init(
        x, w, b, y,
        tiling_data.N, tiling_data.CIn, tiling_data.H, tiling_data.W,
        tiling_data.COut, tiling_data.Kh, tiling_data.Kw,
        tiling_data.Stride, tiling_data.Pad,
        tiling_data.Ho, tiling_data.Wo
    );
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义 AscendC 融合算子：Conv2d + HardSwish + ReLU
    """
    def __init__(self, in_channels, out_channels, kernel_size):
        super(ModelNew, self).__init__()
        # 仅用于权重/偏置的创建与管理（参数与原模型一致）
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size, stride=1, padding=kernel_size // 2, bias=True)
        self.kernel_size = kernel_size

    def forward(self, x):
        # 将 PyTorch 卷积权重/偏置传入自定义融合算子
        k = int(self.kernel_size)
        stride = 1
        pad = k // 2
        return custom_ops_lib.conv2d_hard_swish_relu_custom(x, self.conv.weight, self.conv.bias, k, stride, pad)
