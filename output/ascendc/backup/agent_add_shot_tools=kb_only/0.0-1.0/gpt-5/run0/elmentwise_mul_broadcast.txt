static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    ElmentwiseMulBroadcastCustomTilingData tiling;

    const gert::StorageShape* x_shape = context->GetInputShape(0);
    const gert::StorageShape* y_shape = context->GetInputShape(1);

    // 假设为3D张量 [d0, d1, d2]
    uint32_t x0 = x_shape->GetOriginShape().GetDim(0);
    uint32_t x1 = x_shape->GetOriginShape().GetDim(1);
    uint32_t x2 = x_shape->GetOriginShape().GetDim(2);

    uint32_t y0 = y_shape->GetOriginShape().GetDim(0);
    uint32_t y1 = y_shape->GetOriginShape().GetDim(1);
    uint32_t y2 = y_shape->GetOriginShape().GetDim(2);

    // 广播后的维度
    uint32_t out0 = (x0 == 1) ? y0 : ((y0 == 1) ? x0 : x0);
    uint32_t out1 = (x1 == 1) ? y1 : ((y1 == 1) ? x1 : x1);
    uint32_t inner = (x2 == 1) ? y2 : ((y2 == 1) ? x2 : x2);

    context->SetBlockDim(BLOCK_DIM);

    tiling.set_out0(out0);
    tiling.set_out1(out1);
    tiling.set_inner(inner);
    tiling.set_x0(x0);
    tiling.set_x1(x1);
    tiling.set_y0(y0);
    tiling.set_y1(y1);
    tiling.set_tileNum(TILE_NUM);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

    const gert::Shape* y_shape = context->GetInputShape(1);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "elmentwise_mul_broadcast_custom_tiling_data.h"

// 参考资料 4.1.2.3 Mul，使用AscendC::Mul实现按元素乘法
constexpr int32_t BUFFER_NUM = 2; // 每个队列使用的tensor数量（双缓冲）

class KernelMulBrc {
public:
    __aicore__ inline KernelMulBrc() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t out0, uint32_t out1, uint32_t inner,
                                uint32_t x0, uint32_t x1, uint32_t y0, uint32_t y1,
                                uint32_t tileNum)
    {
        this->out0 = out0;
        this->out1 = out1;
        this->inner = inner;
        this->x0 = x0;
        this->x1 = x1;
        this->y0 = y0;
        this->y1 = y1;
        this->tileNum = tileNum;

        // 每个pair(outer元素对)上的长度切分
        uint32_t denom = this->tileNum * BUFFER_NUM;
        this->tileLength = (this->inner + denom - 1) / denom;
        if (this->tileLength == 0) { this->tileLength = this->inner; }

        uint32_t xTotal = x0 * x1 * inner;
        uint32_t yTotal = y0 * y1 * inner;
        uint32_t zTotal = out0 * out1 * inner;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, xTotal);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, yTotal);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, zTotal);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        uint32_t outter = this->out0 * this->out1;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t pairsPerBlock = (outter + blockNum - 1) / blockNum;
        uint32_t startPair = blockIdx * pairsPerBlock;
        uint32_t endPair = AscendC::Min(startPair + pairsPerBlock, outter);

        for (uint32_t p = startPair; p < endPair; ++p) {
            // 双缓冲分tile跑完inner维度
            uint32_t loopCount = this->tileNum * BUFFER_NUM;
            for (uint32_t t = 0; t < loopCount; ++t) {
                uint32_t tileBase = t * this->tileLength;
                if (tileBase >= this->inner) break;
                uint32_t curLen = AscendC::Min(this->tileLength, this->inner - tileBase);

                CopyIn(p, tileBase, curLen);
                Compute(curLen);
                CopyOut(p, tileBase, curLen);
            }
        }
    }

private:
    __aicore__ inline void PairToMN(uint32_t pairIdx, uint32_t &m, uint32_t &n) const
    {
        m = pairIdx / this->out1;
        n = pairIdx % this->out1;
    }

    __aicore__ inline uint32_t XOffset(uint32_t m, uint32_t n, uint32_t innerOffset) const
    {
        uint32_t mm = (this->x0 == 1) ? 0 : m;
        uint32_t nn = (this->x1 == 1) ? 0 : n;
        return ((mm * this->x1 + nn) * this->inner) + innerOffset;
    }

    __aicore__ inline uint32_t YOffset(uint32_t m, uint32_t n, uint32_t innerOffset) const
    {
        uint32_t mm = (this->y0 == 1) ? 0 : m;
        uint32_t nn = (this->y1 == 1) ? 0 : n;
        return ((mm * this->y1 + nn) * this->inner) + innerOffset;
    }

    __aicore__ inline uint32_t ZOffset(uint32_t m, uint32_t n, uint32_t innerOffset) const
    {
        return ((m * this->out1 + n) * this->inner) + innerOffset;
    }

    __aicore__ inline void CopyIn(uint32_t pairIdx, uint32_t innerOffset, uint32_t curLen)
    {
        uint32_t m, n;
        PairToMN(pairIdx, m, n);
        uint32_t xOff = XOffset(m, n, innerOffset);
        uint32_t yOff = YOffset(m, n, innerOffset);

        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.AllocTensor<DTYPE_Y>();

        AscendC::DataCopy(xLocal, xGm[xOff], curLen);
        AscendC::DataCopy(yLocal, yGm[yOff], curLen);

        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();

        AscendC::Mul(zLocal, xLocal, yLocal, (int32_t)curLen); // 参考资料 4.1.2.3 Mul

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t pairIdx, uint32_t innerOffset, uint32_t curLen)
    {
        uint32_t m, n;
        PairToMN(pairIdx, m, n);
        uint32_t zOff = ZOffset(m, n, innerOffset);

        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[zOff], zLocal, curLen);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;

    uint32_t out0;
    uint32_t out1;
    uint32_t inner;
    uint32_t x0;
    uint32_t x1;
    uint32_t y0;
    uint32_t y1;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void elmentwise_mul_broadcast_custom(GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMulBrc op;
    op.Init(x, y, z,
            tiling_data.out0, tiling_data.out1, tiling_data.inner,
            tiling_data.x0, tiling_data.x1, tiling_data.y0, tiling_data.y1,
            tiling_data.tileNum);
    op.Process();
}
