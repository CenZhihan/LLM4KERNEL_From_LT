// 参考 Matmul 高阶 API 说明：可通过 GetMatmulApiTiling 在 Host 获取 TCubeTiling，

// Tiling：将 Matmul 的 TCubeTiling 直接写入 RawTilingData，Kernel 侧按字节拷贝解析。
namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    const auto* x_shape = context->GetInputShape(0); // [B, K]
    const auto* w_shape = context->GetInputShape(1); // [K, N]（模型中传入已转置的权重）
    if (x_shape == nullptr || w_shape == nullptr) {
        return ge::GRAPH_FAILED;
    }

    uint64_t B = x_shape->GetOriginShape().GetDim(0);
    uint64_t K = x_shape->GetOriginShape().GetDim(1);
    uint64_t K_w = w_shape->GetOriginShape().GetDim(0);
    uint64_t N = w_shape->GetOriginShape().GetDim(1);
    if (K != K_w) {
        return ge::GRAPH_FAILED;
    }

    // 设置 block 数
    context->SetBlockDim(BLOCK_DIM);

    // 生成 Matmul 的 tiling（静态/常量 tiling）。
    // 参考文档：可通过 GetMatmulApiTiling 获取常量 tiling 参数。
    // 这里以 NORMAL 布局为例。
    TCubeTiling tiling = {};
    {
        // 配置 NORMAL 模式（示例，按文档说明）
        constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);
        // 获取 tiling（示例接口，具体实现由库提供）
        bool ok = GetMatmulApiTiling((int)B, (int)N, (int)K, MM_CFG, tiling);
        if (!ok) {
            return ge::GRAPH_FAILED;
        }
    }

    // 将 tiling 原样写入 RawTilingData
    auto* raw = context->GetRawTilingData();
    auto* buf = reinterpret_cast<uint8_t*>(raw->GetData());
    size_t cap = raw->GetCapacity();
    if (cap < sizeof(TCubeTiling)) {
        return ge::GRAPH_FAILED;
    }
    std::memcpy(buf, &tiling, sizeof(TCubeTiling));
    raw->SetDataSize(sizeof(TCubeTiling));

    // workspace 按需设置（此处示例设为 0）
    size_t* ws = context->GetWorkspaceSizes(1);
    ws[0] = 0;
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

// 使用 Matmul 高阶 API（参考文档 5.2.x），并融合 Bias 与 ReLU（ReLU 在 Matmul 后单独 VEC 处理）。
// Tiling：Host 通过 GetMatmulApiTiling 生成的 TCubeTiling 以原始字节传入，本 Kernel 直接解析。

extern "C" __global__ __aicore__ void gemm_add_relu_custom(GM_ADDR xGM, GM_ADDR wGM, GM_ADDR biasGM, GM_ADDR yGM, GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    if ASCEND_IS_AIC {
        return;
    }

    // 解析 tiling（参考示例：将 GM 中的 TCubeTiling 原样拷贝到本地结构体）
    TCubeTiling tiling;
    {
        auto tgm = reinterpret_cast<__gm__ uint32_t*>(tilingGM);
        auto tlc = reinterpret_cast<uint32_t*>(&tiling);
        for (int i = 0; i < (int)(sizeof(TCubeTiling) / sizeof(uint32_t)); ++i, ++tgm, ++tlc) {
            *tlc = *tgm;
        }
        // 共享资源示例配置（可按需求调整）
        tiling.shareMode = 0;
        tiling.shareL1Size = 512 * 1024;
        tiling.shareL0CSize = 128 * 1024;
        tiling.shareUbSize = 0;
    }

    // GM Tensor 绑定（类型使用 float）
    using A_T = float;
    using B_T = float;
    using C_T = float;
    using BiasT = float;

    AscendC::GlobalTensor<A_T> aGlobal;
    AscendC::GlobalTensor<B_T> bGlobal;
    AscendC::GlobalTensor<C_T> cGlobal;
    AscendC::GlobalTensor<BiasT> biasGlobal;

    // 从 tiling 推导本 Core 处理块的大小（参考示例）
    int32_t sizeA = tiling.ALayoutInfoB * tiling.singleCoreM * tiling.singleCoreK * sizeof(A_T);
    int32_t sizeB = tiling.BLayoutInfoB * tiling.singleCoreK * tiling.singleCoreN * sizeof(B_T);
    int32_t sizeC = tiling.CLayoutInfoB * tiling.singleCoreM * tiling.singleCoreN * sizeof(C_T);
    int32_t sizeBias = tiling.CLayoutInfoB * tiling.singleCoreN * sizeof(C_T);

    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(xGM), sizeA);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(wGM), sizeB);
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(yGM), sizeC);
    biasGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ BiasT*>(biasGM), sizeBias);

    // 为单 Core 取出分块地址（与示例一致）
    int offset_a = 0, offset_b = 0, offset_c = 0, offset_bias = 0;
    AscendC::GlobalTensor<A_T> gm_a;
    gm_a.SetGlobalBuffer(const_cast<__gm__ A_T*>(aGlobal[offset_a].GetPhyAddr()),
                         tiling.singleCoreM * tiling.singleCoreK);
    AscendC::GlobalTensor<B_T> gm_b;
    gm_b.SetGlobalBuffer(const_cast<__gm__ B_T*>(bGlobal[offset_b].GetPhyAddr()),
                         tiling.singleCoreK * tiling.singleCoreN);
    AscendC::GlobalTensor<C_T> gm_c;
    gm_c.SetGlobalBuffer(const_cast<__gm__ C_T*>(cGlobal[offset_c].GetPhyAddr()),
                         tiling.singleCoreM * tiling.singleCoreN);
    AscendC::GlobalTensor<BiasT> gm_bias;
    gm_bias.SetGlobalBuffer(const_cast<__gm__ BiasT*>(biasGlobal[offset_bias].GetPhyAddr()),
                            tiling.singleCoreN);

    // 创建 Matmul 实例并注册（参考 REGIST_MATMUL_OBJ 示例）
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, A_T> aType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, B_T> bType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, C_T> cType;
    typedef matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, BiasT> biasType;

    // NORMAL 模式配置示例
    constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);
    matmul::Matmul<aType, bType, cType, biasType, MM_CFG> mm;

    TPipe pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm);
    mm.Init(&tiling);

    // 设置 A/B/Bias；此处假设输入 X:[B,K]、权重 Wt:[K,N]（模型侧已转置权重）
    mm.SetTensorA(gm_a, false); // isTransposeA=false
    mm.SetTensorB(gm_b, false); // isTransposeB=false
    mm.SetWorkspace(workspaceGM, 0);
    mm.SetBias(gm_bias);

    // 执行 Matmul；单 batch 场景
    mm.IterateBatch(gm_c, 1, 1, false);

    // Matmul 完成后对输出执行 ReLU：y = max(y, 0)
    // 使用 VEC 流水线分片做 ReLU，按元素处理
    {
        constexpr int32_t BUF_NUM = 1;
        AscendC::TQue<AscendC::TPosition::VECIN, BUF_NUM> inQ;
        AscendC::TQue<AscendC::TPosition::VECOUT, BUF_NUM> outQ;

        // 以 UB 大小估算一次处理的元素数；这里简单按 singleCoreM*singleCoreN 为一块
        uint32_t totalElems = tiling.CLayoutInfoB * tiling.singleCoreM * tiling.singleCoreN;
        uint32_t tileElems = tiling.singleCoreM * tiling.singleCoreN;

        pipe.InitBuffer(inQ, BUF_NUM, tileElems * sizeof(C_T));
        pipe.InitBuffer(outQ, BUF_NUM, tileElems * sizeof(C_T));

        // 逐块处理（如果 totalElems == tileElems，这里循环一次）
        for (uint32_t off = 0; off < totalElems; off += tileElems) {
            AscendC::LocalTensor<C_T> yLocalIn = inQ.AllocTensor<C_T>();
            AscendC::DataCopy(yLocalIn, gm_c[off], tileElems);
            inQ.EnQue(yLocalIn);

            AscendC::LocalTensor<C_T> yLocal = inQ.DeQue<C_T>();
            AscendC::LocalTensor<C_T> yLocalOut = outQ.AllocTensor<C_T>();

            // ReLU 操作（示例接口）
            AscendC::Relu(yLocalOut, yLocal, tileElems);

            outQ.EnQue(yLocalOut);
            inQ.FreeTensor(yLocal);

            AscendC::LocalTensor<C_T> yLocalWrite = outQ.DeQue<C_T>();
            AscendC::DataCopy(gm_c[off], yLocalWrite, tileElems);
            outQ.FreeTensor(yLocalWrite);
        }
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用自定义 AscendC 算子 GemmAddReluCustom：
    y = ReLU(x @ W + bias)，其中 W 需转置为 [K, N] 传入 Kernel。
    """
    def __init__(self, in_features, out_features, bias_shape):
        super().__init__()
        self.gemm = torch.nn.Linear(in_features, out_features, bias=False)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        # nn.Linear.weight 形状为 [out_features, in_features]，需转置为 [in_features, out_features]
        w_t = self.gemm.weight.t().contiguous()
        return custom_ops_lib.gemm_add_relu_custom(x, w_t, self.bias)

def get_inputs():
    # 与原模型一致
    batch_size = 1024
    in_features = 8192
    return [torch.rand(batch_size, in_features)]

def get_init_inputs():
    batch_size = 1024
    in_features = 8192
    out_features = 8192
    bias_shape = (out_features,)
    return [in_features, out_features, bias_shape]
