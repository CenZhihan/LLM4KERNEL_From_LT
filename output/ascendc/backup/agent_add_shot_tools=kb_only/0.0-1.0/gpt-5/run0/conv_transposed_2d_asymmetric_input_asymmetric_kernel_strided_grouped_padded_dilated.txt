    const gert::Shape* xShape = context->GetInputShape(0);   // x: [N, C_in, H_in, W_in]

static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    if (context == nullptr) { return GRAPH_FAILED; }

    const gert::Shape* wShape = context->GetInputShape(1);
    if (xShape == nullptr || wShape == nullptr) { return GRAPH_FAILED; }

    auto attrs = context->GetAttrs();
    if (attrs != nullptr) {
        attrs->GetInt("stride_h", strideH);
        attrs->GetInt("stride_w", strideW);
        attrs->GetInt("pad_h", padH);
        attrs->GetInt("pad_w", padW);
        attrs->GetInt("dilation_h", dilationH);
        attrs->GetInt("dilation_w", dilationW);
        attrs->GetInt("groups", groups);
    }

    const auto& xs = xShape->GetStorageShape();
    const auto& ws = wShape->GetStorageShape();

    int64_t n   = xs.GetDim(0);
    int64_t cIn = xs.GetDim(1);
    int64_t hIn = xs.GetDim(2);
    int64_t wIn = xs.GetDim(3);

    int64_t kIn  = ws.GetDim(0); // == cIn
    int64_t kPer = ws.GetDim(1); // out_channels / groups
    int64_t kH   = ws.GetDim(2);
    int64_t kW   = ws.GetDim(3);
    int64_t cOut = kPer * groups;

    int64_t hOut = (hIn - 1) * strideH - 2 * padH + dilationH * (kH - 1) + 1;
    int64_t wOut = (wIn - 1) * strideW - 2 * padW + dilationW * (kW - 1) + 1;

    gert::Shape* yShape = context->GetOutputShape(0);
    yShape->SetDimNum(4);
    yShape->SetDim(0, n);
    yShape->SetDim(1, cOut);
    yShape->SetDim(2, hOut);
    yShape->SetDim(3, wOut);
    return GRAPH_SUCCESS;
}
