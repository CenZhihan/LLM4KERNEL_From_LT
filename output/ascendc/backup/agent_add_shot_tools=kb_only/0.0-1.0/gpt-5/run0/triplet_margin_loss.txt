static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 获取输入shape: [B, H]
    const ge::Shape& shape = context->GetInputShape(0)->GetOriginShape();
    std::vector<int64_t> dims = shape.GetDims();
    uint32_t bLength = 1;
    uint32_t hLength = 1;
    if (dims.size() == 1) {
        bLength = dims[0];
        hLength = 1;
    } else if (dims.size() >= 2) {
        bLength = dims[0];
        hLength = 1;
        for (size_t i = 1; i < dims.size(); ++i) {
            hLength *= dims[i];
        }
    }

    // 读取margin属性（若框架未传入，则用1.0f）
    float margin = 1.0f;
    if (context->GetAttrs() != nullptr) {
        (void)context->GetAttrs()->GetFloat("margin", margin);
    }

    // 单block执行，便于在核内完成全局归约
    context->SetBlockDim(1);

    TripletMarginLossCustomTilingData tiling;
    tiling.set_bLength(bLength);
    tiling.set_hLength(hLength);
    tiling.set_margin(margin);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无workspace
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "triplet_margin_loss_custom_tiling_data.h"

// 参考文档示例：使用ReduceSum与GetAccVal获取和（4.1.8.12 GetAccVal）
// 以及DataCopy/Sub/Mul等基础算子用法
// 为避免跨block归约问题，BlockDim在host侧固定为1

constexpr int32_t BUFFER_NUM = 1;

using DTYPE = float;

class KernelTripletMarginLoss {
public:
    __aicore__ inline KernelTripletMarginLoss() {}

    __aicore__ inline void Init(GM_ADDR anchor, GM_ADDR positive, GM_ADDR negative, GM_ADDR loss,
                                uint32_t bLength, uint32_t hLength, float margin) {
        this->bLength = bLength;
        this->hLength = hLength;
        this->margin = margin;

        anchorGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(anchor), bLength * hLength);
        positiveGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(positive), bLength * hLength);
        negativeGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(negative), bLength * hLength);
        lossGm = reinterpret_cast<__gm__ DTYPE *>(loss);

        // 三个输入行缓冲 + 临时计算缓冲 + ReduceSum工作缓冲 + 一个dummy dst
        pipe.InitBuffer(inQueueA, BUFFER_NUM, hLength * sizeof(DTYPE));
        pipe.InitBuffer(inQueueP, BUFFER_NUM, hLength * sizeof(DTYPE));
        pipe.InitBuffer(inQueueN, BUFFER_NUM, hLength * sizeof(DTYPE));
        pipe.InitBuffer(calcQueue1, BUFFER_NUM, hLength * sizeof(DTYPE));
        pipe.InitBuffer(calcQueue2, BUFFER_NUM, hLength * sizeof(DTYPE));
        pipe.InitBuffer(workQueue, BUFFER_NUM, hLength * sizeof(DTYPE));
        pipe.InitBuffer(dstQueue, BUFFER_NUM, hLength * sizeof(DTYPE)); // 占位
    }

    __aicore__ inline void Process() {
        float sumLoss = 0.0f;

        for (uint32_t i = 0; i < bLength; ++i) {
            CopyInRow(i);
            sumLoss += ComputeRow();
            FreeRow();
        }

        float meanLoss = sumLoss / static_cast<float>(bLength);

        // 写回标量到GM（直接写GM指针）
        lossGm[0] = meanLoss;
    }

private:
    __aicore__ inline void CopyInRow(uint32_t rowIndex) {
        AscendC::LocalTensor<DTYPE> aLocal = inQueueA.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> pLocal = inQueueP.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> nLocal = inQueueN.AllocTensor<DTYPE>();

        uint32_t offset = rowIndex * hLength;
        AscendC::DataCopy(aLocal, anchorGm[offset], hLength);
        AscendC::DataCopy(pLocal, positiveGm[offset], hLength);
        AscendC::DataCopy(nLocal, negativeGm[offset], hLength);

        inQueueA.EnQue(aLocal);
        inQueueP.EnQue(pLocal);
        inQueueN.EnQue(nLocal);
    }

    __aicore__ inline float ComputeRow() {
        AscendC::LocalTensor<DTYPE> aLocal = inQueueA.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> pLocal = inQueueP.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> nLocal = inQueueN.DeQue<DTYPE>();

        AscendC::LocalTensor<DTYPE> tmp1 = calcQueue1.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> tmp2 = calcQueue2.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> work = workQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> dstDummy = dstQueue.AllocTensor<DTYPE>();

        // dist_pos^2
        AscendC::Sub(tmp1, aLocal, pLocal, hLength);
        AscendC::Mul(tmp2, tmp1, tmp1, hLength);
        AscendC::ReduceSum(dstDummy, tmp2, work, hLength);
        float dpos2 = AscendC::GetAccVal<float>();

        // dist_neg^2
        AscendC::Sub(tmp1, aLocal, nLocal, hLength);
        AscendC::Mul(tmp2, tmp1, tmp1, hLength);
        AscendC::ReduceSum(dstDummy, tmp2, work, hLength);
        float dneg2 = AscendC::GetAccVal<float>();

        // L2距离（sqrt）
        float dpos = sqrtf(dpos2);
        float dneg = sqrtf(dneg2);

        float hinge = dpos - dneg + margin;
        if (hinge < 0.0f) hinge = 0.0f;

        // 释放本行资源
        return hinge;
    }

    __aicore__ inline void FreeRow() {
        AscendC::LocalTensor<DTYPE> aLocal = inQueueA.DeQue<DTYPE>(); // 实际不会进入，这里只为占位
    }

private:
    AscendC::TPipe pipe;

    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM>  inQueueA;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM>  inQueueP;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM>  inQueueN;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> calcQueue1;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> calcQueue2;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> workQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> dstQueue;

    AscendC::GlobalTensor<DTYPE> anchorGm;
    AscendC::GlobalTensor<DTYPE> positiveGm;
    AscendC::GlobalTensor<DTYPE> negativeGm;
    __gm__ DTYPE* lossGm = nullptr;

    uint32_t bLength;
    uint32_t hLength;
    float margin;
};

extern "C" __global__ __aicore__ void triplet_margin_loss_custom(GM_ADDR anchor, GM_ADDR positive, GM_ADDR negative,
                                                                 GM_ADDR loss, GM_ADDR tiling) {
    GET_TILING_DATA(tilingData, tiling);
    KernelTripletMarginLoss op;
    op.Init(anchor, positive, negative, loss,
            tilingData.bLength, tilingData.hLength, tilingData.margin);
    op.Process();
}
