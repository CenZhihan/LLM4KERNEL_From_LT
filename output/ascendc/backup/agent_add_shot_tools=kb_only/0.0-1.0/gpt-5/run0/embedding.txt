host_operator_src="""
#include "embedding_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    EmbeddingCustomTilingData tiling;

    // 输入0: weight, 输入1: indices
    // 计算 totalRows = indices 元素总数
    uint32_t totalRows = context->GetInputShape(1)->GetOriginShape().GetShapeSize();

    // 计算 embDim = weight 的最后一维
    auto weightOrigin = context->GetInputShape(0)->GetOriginShape();
    int64_t dimNum = weightOrigin.GetDimNum();
    int64_t embDim = weightOrigin.GetDim(dimNum - 1);

    context->SetBlockDim(BLOCK_DIM);
    tiling.set_totalRows(totalRows);
    tiling.set_embDim(static_cast<uint32_t>(embDim));
    tiling.set_tileNum(TILE_NUM);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    // y.shape = indices.shape + [embDim]
    const gert::Shape* weight_shape = context->GetInputShape(0);
    const gert::Shape* indices_shape = context->GetInputShape(1);

    auto weight_origin = weight_shape->GetOriginShape();
    auto indices_origin = indices_shape->GetOriginShape();

    int64_t w_dim_num = weight_origin.GetDimNum();
    int64_t emb_dim = weight_origin.GetDim(w_dim_num - 1);

    std::vector<int64_t> out_dims;
    out_dims.reserve(indices_origin.GetDimNum() + 1);
    for (int64_t i = 0; i < indices_origin.GetDimNum(); ++i) {
        out_dims.push_back(indices_origin.GetDim(i));
    }
    out_dims.push_back(emb_dim);

    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = gert::Shape(out_dims);
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    // 输出 dtype 与 weight 一致
    const auto weight_dt = context->GetInputDataType(0);
    context->SetOutputDataType(0, weight_dt);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class EmbeddingCustom : public OpDef {
public:
    explicit EmbeddingCustom(const char* name) : OpDef(name)
    {
        this->Input("weight")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})

        this->AICore().SetTiling(optiling::TilingFunc);

    }
};
OP_ADD(EmbeddingCustom);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "embedding_custom_tiling_data.h"

// 每个 tile 处理的行数由 tiling 决定，按 rows 粒度复制 embDim 列
constexpr int32_t BUFFER_NUM = 2;

class KernelEmbedding {
public:
    __aicore__ inline KernelEmbedding() {}
    __aicore__ inline void Init(GM_ADDR weight, GM_ADDR indices, GM_ADDR y,
                                uint32_t totalRows, uint32_t embDim, uint32_t tileNum)
    {
        this->embDim = embDim;
        this->tileNum = tileNum;

        // 当前核负责的 rows
        this->blockRows = totalRows / AscendC::GetBlockNum();
        this->rowsPerTile = this->blockRows / tileNum / BUFFER_NUM;
        if (this->rowsPerTile == 0) {
            this->rowsPerTile = 1; // 兜底，防止极端情况下为0
        }

        weightGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(weight));
        indicesGm.SetGlobalBuffer(reinterpret_cast<__gm__ int32_t *>(indices) + this->blockRows * AscendC::GetBlockIdx(), this->blockRows);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(y) + static_cast<uint64_t>(this->blockRows * this->embDim) * AscendC::GetBlockIdx(),
                            this->blockRows * this->embDim);

        pipe.InitBuffer(inQueueIdx, BUFFER_NUM, this->rowsPerTile * sizeof(int32_t));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->rowsPerTile * this->embDim * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t loopCount = this->tileNum * BUFFER_NUM;
        for (uint32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
        // 处理可能的尾块
        uint32_t rowsDone = loopCount * this->rowsPerTile;
        if (rowsDone < this->blockRows) {
            CopyInTail(rowsDone);
            ComputeTail(rowsDone);
            CopyOutTail(rowsDone);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<int32_t> idxLocal = inQueueIdx.AllocTensor<int32_t>();
        uint32_t rowOffset = progress * this->rowsPerTile;
        AscendC::DataCopy(idxLocal, indicesGm[rowOffset], this->rowsPerTile);
        inQueueIdx.EnQue(idxLocal);
    }

    __aicore__ inline void Compute(uint32_t progress)
    {
        AscendC::LocalTensor<int32_t> idxLocal = inQueueIdx.DeQue<int32_t>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // 对本 tile 的每一行，执行一次查表行拷贝
        for (uint32_t r = 0; r < this->rowsPerTile; ++r) {
            int32_t id = idxLocal.GetValue(r);
            if (id >= 0) {
                AscendC::DataCopy(yLocal[r * this->embDim], weightGm[static_cast<uint32_t>(id) * this->embDim], this->embDim);
            } else {
                // 负索引兜底：填零
                AscendC::Duplicate<float>(yLocal[r * this->embDim], 0.0f, this->embDim);
            }
        }

        outQueueY.EnQue<float>(yLocal);
        inQueueIdx.FreeTensor(idxLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        uint32_t rowOffset = progress * this->rowsPerTile;
        AscendC::DataCopy(yGm[rowOffset * this->embDim], yLocal, this->rowsPerTile * this->embDim);
        outQueueY.FreeTensor(yLocal);
    }

    // 处理尾块（不足 rowsPerTile 的剩余行）
    __aicore__ inline void CopyInTail(uint32_t rowOffset)
    {
        tailRows = this->blockRows - rowOffset;
        AscendC::LocalTensor<int32_t> idxLocal = inQueueIdx.AllocTensor<int32_t>();
        AscendC::DataCopy(idxLocal, indicesGm[rowOffset], tailRows);
        inQueueIdx.EnQue(idxLocal);
    }

    __aicore__ inline void ComputeTail(uint32_t /*rowOffset*/)
    {
        AscendC::LocalTensor<int32_t> idxLocal = inQueueIdx.DeQue<int32_t>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        for (uint32_t r = 0; r < tailRows; ++r) {
            int32_t id = idxLocal.GetValue(r);
            if (id >= 0) {
                AscendC::DataCopy(yLocal[r * this->embDim], weightGm[static_cast<uint32_t>(id) * this->embDim], this->embDim);
            } else {
                AscendC::Duplicate<float>(yLocal[r * this->embDim], 0.0f, this->embDim);
            }
        }

        outQueueY.EnQue<float>(yLocal);
        inQueueIdx.FreeTensor(idxLocal);
    }

    __aicore__ inline void CopyOutTail(uint32_t rowOffset)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[rowOffset * this->embDim], yLocal, tailRows * this->embDim);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueIdx;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<int32_t> indicesGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t embDim = 0;
    uint32_t tileNum = 0;
    uint32_t blockRows = 0;
    uint32_t rowsPerTile = 0;
    uint32_t tailRows = 0;
};

extern "C" __global__ __aicore__ void embedding_custom(GM_ADDR weight, GM_ADDR indices, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelEmbedding op;
    op.Init(weight, indices, y, tiling_data.totalRows, tiling_data.embDim, tiling_data.tileNum);
    op.Process();
}
"""

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor embedding_custom_impl_npu(const at::Tensor& weight, const at::Tensor& indices) {
    TORCH_CHECK(weight.device().type() == c10::kPrivateUse1, "weight must be on NPU");
    TORCH_CHECK(indices.device().type() == c10::kPrivateUse1, "indices must be on NPU");
    TORCH_CHECK(weight.dtype() == at::kFloat, "weight dtype must be float32");

    // 输出形状 = indices.shape + [emb_dim]
    auto out_sizes = indices.sizes().vec();
    out_sizes.push_back(weight.size(-1));
    at::Tensor result = at::empty(out_sizes, weight.options());

    // indices 使用 int32
    at::Tensor indices_i32 = indices.to(at::kInt);

    EXEC_NPU_CMD(aclnnEmbeddingCustom, weight, indices_i32, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("embedding_custom", &embedding_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("embedding_custom", &embedding_custom_impl_npu, "embedding lookup");
}
