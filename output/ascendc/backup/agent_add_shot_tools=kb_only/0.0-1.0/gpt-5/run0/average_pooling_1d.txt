namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    AveragePooling1dCustomTilingData tiling;

    const gert::StorageShape* x_shape = context->GetInputShape(0);
    auto dims = x_shape->GetStorageShape().GetDims();
    if (dims.size() != 3) {
        return ge::GRAPH_FAILED;
    }
    uint32_t batch = static_cast<uint32_t>(dims[0]);
    uint32_t channels = static_cast<uint32_t>(dims[1]);
    uint32_t lengthIn = static_cast<uint32_t>(dims[2]);

    const gert::RuntimeAttrs* attrs = context->GetAttrs();
    uint32_t kernel_size = *(attrs->GetAttrPointer<uint32_t>(0));
    uint32_t stride = *(attrs->GetAttrPointer<uint32_t>(1));
    uint32_t padding = *(attrs->GetAttrPointer<uint32_t>(2));

    // PyTorch AvgPool1d default: ceil_mode=False, count_include_pad=True
    uint32_t lengthOut = 0;
    if (stride == 0) {
        return ge::GRAPH_FAILED;
    } else {
        int64_t tmp = static_cast<int64_t>(lengthIn) + 2 * static_cast<int64_t>(padding) - static_cast<int64_t>(kernel_size);
        if (tmp < 0) tmp = -1; // ensures lengthOut becomes 0 after +1 then max with 0
        lengthOut = static_cast<uint32_t>(tmp / static_cast<int64_t>(stride) + 1);
    }

    // Use single core for simplicity in this sample
    context->SetBlockDim(1);

    tiling.set_batch(batch);
    tiling.set_channels(channels);
    tiling.set_lengthIn(lengthIn);
    tiling.set_kernelSize(kernel_size);
    tiling.set_stride(stride);
    tiling.set_padding(padding);
    tiling.set_lengthOut(lengthOut);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t* currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}

namespace ge {

    const gert::Shape* x_shape = context->GetInputShape(0);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "average_pooling_1d_custom_tiling_data.h"

// This kernel computes 1D Average Pooling along the last axis with count_include_pad=True.
// Strategy: partition outer = N*C lines; each line processes full length, generate Lout outputs.
// For each output position, we copy valid window segment to UB and ReduceSum, then divide by K.

class KernelAveragePooling1d {
public:
    __aicore__ inline KernelAveragePooling1d() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batch, uint32_t channels, uint32_t lengthIn,
                                uint32_t kernelSize, uint32_t stride, uint32_t padding, uint32_t lengthOut)
    {
        this->N = batch;
        this->C = channels;
        this->Lin = lengthIn;
        this->K = kernelSize;
        this->S = stride;
        this->P = padding;
        this->Lout = lengthOut;

        const uint32_t outer = N * C;
        xGm.SetGlobalBuffer((__gm__ float*)x, outer * Lin);
        yGm.SetGlobalBuffer((__gm__ float*)y, outer * Lout);

        // Buffers:
        // 1) UB buffer to hold up to K elements for ReduceSum source
        pipe.InitBuffer(inQueueSrc, 1, K * sizeof(float));
        // 2) Work buffer for ReduceSum (use 16 floats as minimal workspace, referring to ReduceSum example)
        pipe.InitBuffer(workQueue, 1, 16 * sizeof(float));
        // 3) Output buffer of ReduceSum result (length 16, first element is valid sum)
        pipe.InitBuffer(outQueueDst, 1, 16 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        const uint32_t outer = N * C;
        float invK = 1.0f / static_cast<float>(K);

        for (uint32_t oc = 0; oc < outer; ++oc) {
            const uint32_t xBase = oc * Lin;
            const uint32_t yBase = oc * Lout;

            for (uint32_t t = 0; t < Lout; ++t) {
                // window start in padded coordinates
                int32_t startPad = static_cast<int32_t>(t) * static_cast<int32_t>(S) - static_cast<int32_t>(P);
                int32_t endPad = startPad + static_cast<int32_t>(K); // exclusive

                // map to valid [0, Lin)
                uint32_t validStart = static_cast<uint32_t>(startPad < 0 ? 0 : startPad);
                uint32_t validEnd = static_cast<uint32_t>(endPad < 0 ? 0 : (endPad > static_cast<int32_t>(Lin) ? static_cast<int32_t>(Lin) : endPad));
                uint32_t validLen = (validEnd > validStart) ? (validEnd - validStart) : 0;

                AscendC::LocalTensor<float> dstLocal = outQueueDst.AllocTensor<float>();
                if (validLen == 0) {
                    // Entire window falls in padding -> average is 0
                    float zero = 0.0f;
                    AscendC::Duplicate<float>(dstLocal, zero, 16);
                } else {
                    // Copy valid segment
                    AscendC::LocalTensor<float> srcLocal = inQueueSrc.AllocTensor<float>();
                    AscendC::DataCopy(srcLocal, xGm[xBase + validStart], validLen);
                    inQueueSrc.EnQue(srcLocal);

                    AscendC::LocalTensor<float> srcLocal2 = inQueueSrc.DeQue<float>();
                    AscendC::LocalTensor<float> workLocal = workQueue.AllocTensor<float>();
                    // ReduceSum first n elements (validLen)
                    AscendC::ReduceSum<float>(dstLocal, srcLocal2, workLocal, validLen);
                    workQueue.FreeTensor(workLocal);
                    inQueueSrc.FreeTensor(srcLocal2);
                }

                // Divide by kernel size (count_include_pad=True)
                AscendC::Muls(dstLocal, dstLocal, invK, 16);

                // Write first element to GM
                AscendC::DataCopy(yGm[yBase + t], dstLocal, 1);
                outQueueDst.FreeTensor(dstLocal);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueSrc;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> workQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueDst;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t N;
    uint32_t C;
    uint32_t Lin;
    uint32_t K;
    uint32_t S;
    uint32_t P;
    uint32_t Lout;
};

extern "C" __global__ __aicore__ void average_pooling_1d_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelAveragePooling1d op;
    op.Init(x, y,
            tiling_data.batch(),
            tiling_data.channels(),
            tiling_data.lengthIn(),
            tiling_data.kernelSize(),
            tiling_data.stride(),
            tiling_data.padding(),
            tiling_data.lengthOut());
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor average_pooling_1d_custom_impl_npu(const at::Tensor& x, int64_t kernel_size, int64_t stride, int64_t padding) {
    TORCH_CHECK(x.dim() == 3, "average_pooling_1d_custom expects input of shape (N, C, L)");
    at::Tensor y = at::empty({x.size(0), x.size(1),
                              (x.size(2) + 2 * padding - kernel_size) / stride + 1},
                              x.options());
    EXEC_NPU_CMD(aclnnAveragePooling1dCustom, x, (uint32_t)kernel_size, (uint32_t)stride, (uint32_t)padding, y);
    return y;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("average_pooling_1d_custom", &average_pooling_1d_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("average_pooling_1d_custom", &average_pooling_1d_custom_impl_npu, "AvgPool1d custom");
