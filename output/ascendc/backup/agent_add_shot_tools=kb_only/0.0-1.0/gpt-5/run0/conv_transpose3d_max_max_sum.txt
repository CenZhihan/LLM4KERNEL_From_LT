kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose3d_max_max_sum_custom_tiling_data.h"

// 该自定义核实现对输入张量最后一维做求和（Sum），用于配合上层将通道维(C)置于最后一维的场景。
// 参考了AscendC高阶API Sum的使用方法与临时缓冲获取方式（检索结果：Sum API与More Examples）。

template <typename T>
class KernelSumLastDim
{
public:
    __aicore__ inline KernelSumLastDim() {}

    __aicore__ inline void Init(GM_ADDR xGm, GM_ADDR yGm, uint32_t outter, uint32_t inner, uint32_t n, uint32_t tmpBufferSize)
    {
        height = outter; // 外轴长度：N*D*H*W
        width = inner;   // 对齐后的内轴长度：C对齐到32字节
        nNum = n;        // 实际内轴长度：C
        srcGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(xGm), height * width);
        dstGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(yGm), height); // 仅保存归约后的结果(每个外轴元素一个值)

        // 输入拷入UB的缓冲
        pipe.InitBuffer(inQueueSrc, 1, height * width * sizeof(T));
        // 输出缓冲（32B对齐长度）
        uint32_t outAlignedCnt = (height * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(T);
        pipe.InitBuffer(outQueueDst, 1, outAlignedCnt * sizeof(T));
        // 临时缓冲（Sum内部复杂计算所需）
        pipe.InitBuffer(tmpQueue, 1, tmpBufferSize);
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<T> srcLocal = inQueueSrc.AllocTensor<T>();
        AscendC::DataCopy(srcLocal, srcGlobal, height * width);
        inQueueSrc.EnQue(srcLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<T> srcLocal = inQueueSrc.DeQue<T>();
        AscendC::LocalTensor<uint8_t> sharedTmp = tmpQueue.AllocTensor<uint8_t>();
        AscendC::LocalTensor<T> dstLocal = outQueueDst.AllocTensor<T>();

        // 构造Sum参数：以最后一维为归约维度
        AscendC::SumParams params{height, width, nNum};
        // 执行Sum（参考文档中Sum的带sharedTmpBuffer原型）
        AscendC::Sum(dstLocal, srcLocal, sharedTmp, params);

        outQueueDst.EnQue<T>(dstLocal);
        inQueueSrc.FreeTensor(srcLocal);
        tmpQueue.FreeTensor(sharedTmp);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<T> dstLocal = outQueueDst.DeQue<T>();
        uint32_t outAlignedCnt = (height * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(T);
        AscendC::DataCopy(dstGlobal, dstLocal, outAlignedCnt);
        outQueueDst.FreeTensor(dstLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueSrc;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueDst;

    AscendC::GlobalTensor<T> srcGlobal, dstGlobal;
    uint32_t height = 0;
    uint32_t width = 0;
    uint32_t nNum = 0;
};

extern "C" __global__ __aicore__ void conv_transpose3d_max_max_sum_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelSumLastDim<DTYPE_X> op; // 输入与输出同类型
    op.Init(x, y, tiling_data.outter, tiling_data.inner, tiling_data.n, tiling_data.tmpBufferSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
import torch.nn as nn

class ModelNew(torch.nn.Module):
    """
    优化版：保留PyTorch的ConvTranspose3d与MaxPool3d，使用自定义AscendC内核对通道维求和。
    内核要求最后一维为被归约维度，因此在调用前后做维度置换。
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.max_pool1 = nn.MaxPool3d(kernel_size=2)
        self.max_pool2 = nn.MaxPool3d(kernel_size=3)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.max_pool1(x)
        x = self.max_pool2(x)
        # [N, C, D, H, W] -> [N, D, H, W, C]
        x_last = x.permute(0, 2, 3, 4, 1).contiguous()
        # 调用自定义核（在最后一维求和），返回[N, D, H, W, 1]
        y_last1 = custom_ops_lib.conv_transpose3d_max_max_sum_custom(x_last)
        # [N, D, H, W, 1] -> [N, 1, D, H, W]
        y = y_last1.permute(0, 4, 1, 2, 3).contiguous()
        return y
'''
