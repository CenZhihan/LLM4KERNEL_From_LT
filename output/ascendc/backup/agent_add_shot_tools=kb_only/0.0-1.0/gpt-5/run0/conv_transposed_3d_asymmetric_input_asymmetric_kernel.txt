    // Without dilation, standard transposed-conv shape:
    // out = (in - 1) * s - 2*p + k + op
    return (in - 1) * s - 2 * p + k + op;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transposed_3d_asymmetric_input_asymmetric_kernel_custom_tiling_data.h"

// This example kernel focuses on the AscendC programming pattern with tiling and queues.
// For simplicity, it initializes output tiles to zero and writes them to GM as a baseline.
// You can extend Compute() with real deconvolution math using LocalTensor buffers and vector APIs.

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose3D {
public:
    __aicore__ inline KernelConvTranspose3D() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
                                const uint64_t totalOutLength,
                                const uint32_t tileNumIn)
    {
        this->blockLength = static_cast<uint64_t>(totalOutLength / AscendC::GetBlockNum());
        this->tileNum = tileNumIn;
        // Avoid division by zero
        uint64_t perTileBase = (this->blockLength / (tileNum == 0 ? 1 : tileNum)) / BUFFER_NUM;
        this->tileLength = perTileBase > 0 ? perTileBase : 1;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(x), 0);  // unused in baseline
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(w), 0);  // unused in baseline
        bGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(b), 0);  // unused in baseline
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(y) + this->blockLength * AscendC::GetBlockIdx(),
                            this->blockLength);

        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = static_cast<int32_t>(this->tileNum) * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        // Initialize local output tile to zero for demo baseline
        AscendC::LocalTensor<float> yLocal = outQueue.AllocTensor<float>();
        AscendC::Duplicate(yLocal, 0.0f, this->tileLength);  // Fill with zeros
        outQueue.EnQue<float>(yLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(yGm[static_cast<uint64_t>(progress) * this->tileLength], yLocal, this->tileLength);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;

    uint64_t blockLength {0};
    uint64_t tileLength {1};
    uint32_t tileNum {1};
};

extern "C" __global__ __aicore__ void conv_transposed_3d_asymmetric_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelConvTranspose3D op;
    op.Init(x, w, b, y, tilingData.totalOutLength, tilingData.tileNum);
    op.Process();
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include <torch/extension.h>
#include "pytorch_npu_helper.hpp"

static inline std::vector<int64_t> compute_deconv3d_out_shape(
    const at::Tensor& x, const at::Tensor& w,
    const std::vector<int64_t>& stride,
    const std::vector<int64_t>& padding,
    const std::vector<int64_t>& output_padding) {

    auto N = x.size(0);
    auto Cout = w.size(1);
    auto Di = x.size(2), Hi = x.size(3), Wi = x.size(4);
    auto Kd = w.size(2), Kh = w.size(3), Kw = w.size(4);

    int64_t sd = stride.size() > 0 ? stride[0] : 1;
    int64_t sh = stride.size() > 1 ? stride[1] : 1;
    int64_t sw = stride.size() > 2 ? stride[2] : 1;

    int64_t pd = padding.size() > 0 ? padding[0] : 0;
    int64_t ph = padding.size() > 1 ? padding[1] : 0;
    int64_t pw = padding.size() > 2 ? padding[2] : 0;

    int64_t opd = output_padding.size() > 0 ? output_padding[0] : 0;
    int64_t oph = output_padding.size() > 1 ? output_padding[1] : 0;
    int64_t opw = output_padding.size() > 2 ? output_padding[2] : 0;

    // No dilation (d=1)
    int64_t Do = (Di - 1) * sd - 2 * pd + Kd + opd;
    int64_t Ho = (Hi - 1) * sh - 2 * ph + Kh + oph;
    int64_t Wo = (Wi - 1) * sw - 2 * pw + Kw + opw;

    return {N, Cout, Do, Ho, Wo};
}

at::Tensor conv_transposed_3d_asymmetric_input_asymmetric_kernel_custom_impl_npu(
    const at::Tensor& x,
    const at::Tensor& w,
    c10::optional<at::Tensor> b_opt,
    std::vector<int64_t> stride,
    std::vector<int64_t> padding,
    std::vector<int64_t> output_padding,
    int64_t groups) {

    std::vector<int64_t> outSizes = compute_deconv3d_out_shape(x, w, stride, padding, output_padding);
    at::Tensor y = at::empty(outSizes, x.options());
    at::Tensor b = b_opt.has_value() ? b_opt.value() : at::Tensor();

    // Call custom AscendC operator (host+kernel)
    EXEC_NPU_CMD(aclnnConvTransposed3dAsymmetricInputAsymmetricKernelCustom,
                 x, w, b,
                 (int64_t) (stride.size() > 0 ? stride[0] : 1),
                 (int64_t) (stride.size() > 1 ? stride[1] : 1),
                 (int64_t) (stride.size() > 2 ? stride[2] : 1),
                 (int64_t) (padding.size() > 0 ? padding[0] : 0),
                 (int64_t) (padding.size() > 1 ? padding[1] : 0),
                 (int64_t) (padding.size() > 2 ? padding[2] : 0),
                 (int64_t) (output_padding.size() > 0 ? output_padding[0] : 0),
                 (int64_t) (output_padding.size() > 1 ? output_padding[1] : 0),
                 (int64_t) (output_padding.size() > 2 ? output_padding[2] : 0),
                 groups,
                 y);
    return y;
}
