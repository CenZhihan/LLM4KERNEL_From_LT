    uint32_t outter = *(attrs->GetAttrPointer<uint32_t>(0)); // "outter"

    uint32_t dtypesize = (dt == ge::DT_FLOAT16) ? 2 : 4;

    // Compute padded inner according to Sum API requirement: inner*sizeof(T) is 32B-aligned
    uint32_t inner = ((n * dtypesize + 32 - 1) / 32) * (32 / dtypesize);

    uint32_t maxValue = 0;
    uint32_t minValue = 0;
    AscendC::GetSumMaxMinTmpSize(n, dtypesize, false, maxValue, minValue);

    tiling.set_inner(inner);
    tiling.set_n(n);
    tiling.set_size(minValue);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "sum_reduction_over_a_dimension_custom_tiling_data.h"

template <typename T>
class KernelSumReductionOverADimension {
public:
    __aicore__ inline KernelSumReductionOverADimension() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t tmpSize, uint32_t outter, uint32_t inner, uint32_t n)
    {
        height = outter;
        width = inner;
        nNum = n;

        srcGlobal.SetGlobalBuffer((__gm__ T *)x);
        dstGlobal.SetGlobalBuffer((__gm__ T *)y);

        // Initialize UB queues. Copy entire [outter x inner] once as in official example.
        pipe.InitBuffer(inQueueSrc, 1, height * width * sizeof(T));
        // Output buffer: align to ONE_BLK_SIZE by height elements
        uint32_t outAlignElems = (height * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(T);
        pipe.InitBuffer(outQueueDst, 1, outAlignElems * sizeof(T));
        // Tmp buffer reserved for Sum internal computation
        pipe.InitBuffer(workQueue, 1, tmpSize);
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<T> srcLocal = inQueueSrc.AllocTensor<T>();
        AscendC::DataCopy(srcLocal, srcGlobal, height * width);
        inQueueSrc.EnQue(srcLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<T> srcLocal = inQueueSrc.DeQue<T>();
        AscendC::LocalTensor<uint8_t> workLocal = workQueue.AllocTensor<uint8_t>();
        AscendC::LocalTensor<T> dstLocal = outQueueDst.AllocTensor<T>();

        AscendC::SumParams params{height, width, nNum};

        // Zero init dst buffer for aligned region
        uint32_t outAlignElems = (height * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(T);
        T zeroVal(0);
        AscendC::Duplicate<T>(dstLocal, zeroVal, outAlignElems);

        // Call Sum API (reduce along last axis), framework-managed tmp buffer reserved via tiling
        AscendC::Sum(dstLocal, srcLocal, params);

        outQueueDst.EnQue<T>(dstLocal);
        workQueue.FreeTensor(workLocal);
        inQueueSrc.FreeTensor(srcLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<T> dstLocal = outQueueDst.DeQue<T>();
        uint32_t outAlignElems = (height * sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE * AscendC::ONE_BLK_SIZE / sizeof(T);
        AscendC::DataCopy(dstGlobal, dstLocal, outAlignElems);
        outQueueDst.FreeTensor(dstLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueSrc;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> workQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueDst;
    AscendC::GlobalTensor<T> srcGlobal, dstGlobal;

    uint32_t width = 0;   // inner (padded)
    uint32_t height = 0;  // outter
    uint32_t nNum = 0;    // actual inner
};

extern "C" __global__ __aicore__ void sum_reduction_over_a_dimension_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
#ifdef DTYPE_X
    using TypeT = DTYPE_X;
#else
    using TypeT = float;
#endif
    KernelSumReductionOverADimension<TypeT> op;
    op.Init(x, y, tiling_data.size, tiling_data.outter, tiling_data.inner, tiling_data.n);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model: reduce over arbitrary dim by
    1) permuting target dim to last,
    2) reshaping to 2D [outter, n],
    3) calling custom AscendC Sum (reduces last axis),
    4) reshaping back with keepdim=True.
    """
    def __init__(self, dim: int):
        super().__init__()
        self.dim = dim

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        d = self.dim
        ndim = x.dim()
        if d < 0:
            d += ndim
        # Move reduce dim to last
        perm = [i for i in range(ndim) if i != d] + [d]
        x_perm = x.permute(perm).contiguous()
        n = x_perm.size(-1)
        others_shape = [x_perm.size(i) for i in range(x_perm.dim() - 1)]
        outter = 1
        for s in others_shape:
            outter *= s
        # Flatten to 2D
        x_2d = x_perm.reshape(outter, n)
        # Call custom op: returns same shape as input; take first column as reduced result
        y_2d_full = custom_ops_lib.sum_reduction_over_a_dimension_custom(x_2d)
        y_2d = y_2d_full[:, :1]  # keepdim=True -> [outter, 1]
        # Restore shape and permute back
        y_perm = y_2d.reshape(*others_shape, 1)
        inv_perm = [0] * ndim
        for i, p in enumerate(perm):
            inv_perm[p] = i
        y = y_perm.permute(inv_perm).contiguous()
        return y
'''
