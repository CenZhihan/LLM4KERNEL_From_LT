kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "kv_cached_chat_batch_attention_custom_tiling_data.h"

// 基于提供的AscendC文档示例，使用TPipe进行缓冲区管理。
// 参考：InitBuffer/Que/Buf分配（CANN API Reference 4.3.1.2）
// 参考：Softmax（5.3.1.9）及示例，使用局部张量进行按行softmax。
// 注意：本示例实现面向形如B×q_len×d 和 B×kv_len×d的自回归Attention，适配q_len较小（例如1）。

using DTYPE = float;

constexpr int32_t BUFFER_NUM = 1;

class KernelKvBatchAttn {
public:
    __aicore__ inline KernelKvBatchAttn() {}

    __aicore__ inline void Init(
        GM_ADDR qGm,
        GM_ADDR kGm,
        GM_ADDR vGm,
        GM_ADDR oGm,
        GM_ADDR workspaceGm,
        uint32_t batch,
        uint32_t qLen,
        uint32_t kvLen,
        uint32_t dModel,
        float scale)
    {
        this->B = batch;
        this->Q = qLen;
        this->T = kvLen;
        this->D = dModel;
        this->scale = scale;

        qGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(qGm));
        kGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(kGm));
        vGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(vGm));
        oGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(oGm));

        // workspace布局：
        // logits: [B*Q, T] 后接 probs: [B*Q, T]
        uint64_t attnCount = static_cast<uint64_t>(B) * static_cast<uint64_t>(Q) * static_cast<uint64_t>(T);
        logitsGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(workspaceGm), attnCount);
        probsGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE*>(workspaceGm) + attnCount, attnCount);

        // 队列缓冲：Q向量、K/V行向量、logits/probs行、输出向量
        pipe.InitBuffer(qVecQueue, BUFFER_NUM, D * sizeof(DTYPE));
        pipe.InitBuffer(kvRowQueue, BUFFER_NUM, D * sizeof(DTYPE));
        pipe.InitBuffer(attnRowQueue, BUFFER_NUM, T * sizeof(DTYPE));
        pipe.InitBuffer(outVecQueue, BUFFER_NUM, D * sizeof(DTYPE));

        // Softmax临时sum/max缓存，最后轴固定32B对齐（见5.3.1.9）
        elementNumPerBlk = 32 / sizeof(DTYPE);
        // height=1（按单行softmax），分配sum/max各1*elementNumPerBlk
        pipe.InitBuffer(sumQueue, 1, elementNumPerBlk * sizeof(DTYPE));
        pipe.InitBuffer(maxQueue, 1, elementNumPerBlk * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        // 遍历每个批次与每个查询位置
        for (uint32_t b = 0; b < B; ++b) {
            for (uint32_t q = 0; q < Q; ++q) {
                ComputeOne(b, q);
            }
        }
    }

private:
    __aicore__ inline void ComputeOne(uint32_t b, uint32_t qIdx)
    {
        // 局部缓冲
        AscendC::LocalTensor<DTYPE> qLocal = qVecQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> outLocal = outVecQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> logitsLocal = attnRowQueue.AllocTensor<DTYPE>();

        // 拷贝Q[b, qIdx, :]
        uint64_t qOffset = (static_cast<uint64_t>(b) * Q + qIdx) * D;
        AscendC::DataCopy(qLocal, qGlobal[qOffset], D);

        // outLocal清零
        for (uint32_t i = 0; i < D; ++i) {
            outLocal.GetData()[i] = static_cast<DTYPE>(0.0f);
        }

        // 逐行计算logits = (K[b, t, :] · Q[b, qIdx, :]) * scale
        for (uint32_t t = 0; t < T; ++t) {
            AscendC::LocalTensor<DTYPE> kRowLocal = kvRowQueue.AllocTensor<DTYPE>();
            uint64_t kOffset = (static_cast<uint64_t>(b) * T + t) * D;
            AscendC::DataCopy(kRowLocal, kGlobal[kOffset], D);

            DTYPE acc = static_cast<DTYPE>(0.0f);
            // 向量点乘
            for (uint32_t j = 0; j < D; ++j) {
                acc += kRowLocal.GetData()[j] * qLocal.GetData()[j];
            }
            acc = acc * static_cast<DTYPE>(scale);
            logitsLocal.GetData()[t] = acc;

            kvRowQueue.FreeTensor(kRowLocal);
        }

        // Softmax over logitsLocal[0..T-1]，输出到probsLocal
        AscendC::LocalTensor<DTYPE> probsLocal = attnRowQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> sumLocal = sumQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> maxLocal = maxQueue.AllocTensor<DTYPE>();

        // 形状：height=1, width=T
        AscendC::SoftMaxShapeInfo shapeInfo = {1, T, 1, T};
        // 使用库SoftMax实现（见5.3.1.9示例）
        AscendC::SoftMax<DTYPE>(probsLocal, sumLocal, maxLocal, logitsLocal, /*tiling*/{}, shapeInfo);

        // 将probs写回GM以便保留（可选）
        uint64_t rowIndex = static_cast<uint64_t>(b) * Q + qIdx;
        AscendC::DataCopy(probsGlobal[rowIndex * T], probsLocal, T);
        AscendC::DataCopy(logitsGlobal[rowIndex * T], logitsLocal, T);

        // 基于probs与V按行加权求和：out = sum_t probs[t] * V[b, t, :]
        for (uint32_t t = 0; t < T; ++t) {
            DTYPE p = probsLocal.GetData()[t];
            AscendC::LocalTensor<DTYPE> vRowLocal = kvRowQueue.AllocTensor<DTYPE>();
            uint64_t vOffset = (static_cast<uint64_t>(b) * T + t) * D;
            AscendC::DataCopy(vRowLocal, vGlobal[vOffset], D);
            for (uint32_t j = 0; j < D; ++j) {
                outLocal.GetData()[j] += p * vRowLocal.GetData()[j];
            }
            kvRowQueue.FreeTensor(vRowLocal);
        }

        // 写回输出 O[b, qIdx, :]
        uint64_t oOffset = (static_cast<uint64_t>(b) * Q + qIdx) * D;
        AscendC::DataCopy(oGlobal[oOffset], outLocal, D);

        // 释放本次局部张量
        sumQueue.FreeTensor(sumLocal);
        maxQueue.FreeTensor(maxLocal);
        attnRowQueue.FreeTensor(probsLocal);
        attnRowQueue.FreeTensor(logitsLocal);
        outVecQueue.FreeTensor(outLocal);
        qVecQueue.FreeTensor(qLocal);
    }

private:
    AscendC::TPipe pipe;

    // 输入/输出GM
    AscendC::GlobalTensor<DTYPE> qGlobal;
    AscendC::GlobalTensor<DTYPE> kGlobal;
    AscendC::GlobalTensor<DTYPE> vGlobal;
    AscendC::GlobalTensor<DTYPE> oGlobal;

    // 中间GM（workspace）
    AscendC::GlobalTensor<DTYPE> logitsGlobal;
    AscendC::GlobalTensor<DTYPE> probsGlobal;

    // 队列
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> qVecQueue;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> kvRowQueue;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> attnRowQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outVecQueue;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> sumQueue;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> maxQueue;

    // 形状
    uint32_t B = 0;   // batch
    uint32_t Q = 0;   // q_len
    uint32_t T = 0;   // kv_len
    uint32_t D = 0;   // d_model
    float scale = 1.0f;

    uint32_t elementNumPerBlk = 0;
};

extern "C" __global__ __aicore__ void kv_cached_chat_batch_attention_custom(
    GM_ADDR qGm,
    GM_ADDR kGm,
    GM_ADDR vGm,
    GM_ADDR oGm,
    GM_ADDR workspaceGm,
    GM_ADDR tilingGm)
{
    GET_TILING_DATA(tilingData, tilingGm);

    KernelKvBatchAttn op;
    op.Init(
        qGm, kGm, vGm, oGm, workspaceGm,
        tilingData.batch(),
        tilingData.qLen(),
        tilingData.kvLen(),
        tilingData.dModel(),
        tilingData.scale());
    op.Process();
}
