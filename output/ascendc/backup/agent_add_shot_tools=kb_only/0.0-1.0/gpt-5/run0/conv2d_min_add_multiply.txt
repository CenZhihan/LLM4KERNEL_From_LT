kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_min_add_multiply_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelConv2dMinAddMultiply {
public:
    __aicore__ inline KernelConv2dMinAddMultiply() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR const_t, GM_ADDR scale_t, GM_ADDR z,
                                uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        biasGm.SetGlobalBuffer((__gm__ float *)bias + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        constGm.SetGlobalBuffer((__gm__ float *)const_t + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        scaleGm.SetGlobalBuffer((__gm__ float *)scale_t + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueConst, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueScale, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        AscendC::LocalTensor<float> constLocal = inQueueConst.AllocTensor<float>();
        AscendC::LocalTensor<float> scaleLocal = inQueueScale.AllocTensor<float>();

        AscendC::DataCopy(xLocal,    xGm[progress * this->tileLength],    this->tileLength);
        AscendC::DataCopy(biasLocal, biasGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(constLocal,constGm[progress * this->tileLength],this->tileLength);
        AscendC::DataCopy(scaleLocal,scaleGm[progress * this->tileLength],this->tileLength);

        inQueueX.EnQue(xLocal);
        inQueueBias.EnQue(biasLocal);
        inQueueConst.EnQue(constLocal);
        inQueueScale.EnQue(scaleLocal);
    }
    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.DeQue<float>();
        AscendC::LocalTensor<float> constLocal = inQueueConst.DeQue<float>();
        AscendC::LocalTensor<float> scaleLocal = inQueueScale.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        // z = min(x, const) + bias
        AscendC::Min(zLocal, xLocal, constLocal, this->tileLength);
        AscendC::Add(zLocal, zLocal, biasLocal, this->tileLength);
        // z = z * scale
        AscendC::Mul(zLocal, zLocal, scaleLocal, this->tileLength);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueBias.FreeTensor(biasLocal);
        inQueueConst.FreeTensor(constLocal);
        inQueueScale.FreeTensor(scaleLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX, inQueueBias, inQueueConst, inQueueScale;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> constGm;
    AscendC::GlobalTensor<float> scaleGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv2d_min_add_multiply_custom(GM_ADDR x,
                                                                      GM_ADDR bias,
                                                                      GM_ADDR const_tensor,
                                                                      GM_ADDR scale_tensor,
                                                                      GM_ADDR z,
                                                                      GM_ADDR workspace,
                                                                      GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dMinAddMultiply op;
    op.Init(x, bias, const_tensor, scale_tensor, z, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, constant_value, bias_shape, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.constant_value = constant_value
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.conv(x)
        # 预先在 PyTorch 侧进行广播，传给自定义算子做逐元素融合计算
        bias_broadcast = self.bias.expand_as(x)
        const_tensor = torch.full_like(x, self.constant_value)
        scale_tensor = torch.full_like(x, self.scaling_factor)
        x = custom_ops_lib.conv2d_min_add_multiply_custom(x, bias_broadcast, const_tensor, scale_tensor)
        return x
'''
