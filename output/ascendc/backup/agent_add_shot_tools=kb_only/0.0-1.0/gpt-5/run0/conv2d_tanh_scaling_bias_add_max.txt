kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_tanh_scaling_bias_add_max_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedTanhScale {
public:
    __aicore__ inline KernelFusedTanhScale() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum,
                                uint32_t hw, uint32_t channels, float scale)
    {
        this->blockNum = AscendC::GetBlockNum();
        this->blockIdx = AscendC::GetBlockIdx();
        this->blockLength = totalLength / this->blockNum;
        this->tileNum = tileNum;
        this->tileLength = hw;           // 每个tile处理一个channel平面
        this->channels = channels;
        this->hw = hw;
        this->scaleVal = scale;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_X *>(x) + this->blockLength * this->blockIdx, this->blockLength);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_Y *>(y) + this->blockLength * this->blockIdx, this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        // 双缓冲流水
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        // 1) tanh
        AscendC::Tanh(yLocal, xLocal, this->tileLength);
        // 2) scaling
        AscendC::Muls(yLocal, yLocal, static_cast<DTYPE_Y>(this->scaleVal), this->tileLength);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t blockNum;
    uint32_t blockIdx;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t channels;
    uint32_t hw;
    float     scaleVal;
};

extern "C" __global__ __aicore__ void conv2d_tanh_scaling_bias_add_max_custom(GM_ADDR x, GM_ADDR y,
                                                                              GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedTanhScale op;
    op.Init(x, y,
            tiling_data.totalLength,
            tiling_data.tileNum,
            tiling_data.hw,
            tiling_data.channels,
            tiling_data.scale);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    将原模型中的 tanh 与 scaling 融合为 AscendC 自定义算子；
    卷积与最大池化依然使用 PyTorch 原生（也可进一步融合/替换）。
    """
    def __init__(self, in_channels, out_channels, kernel_size, scaling_factor, bias_shape, pool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.scaling_factor = scaling_factor
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.max_pool = torch.nn.MaxPool2d(pool_kernel_size)

    def forward(self, x):
        # Convolution
        x = self.conv(x)
        # Fused: tanh + scaling (AscendC)
        x = custom_ops_lib.conv2d_tanh_scaling_bias_add_max_custom(x, float(self.scaling_factor))
        # Bias addition
        x = x + self.bias
        # Max-pooling
        x = self.max_pool(x)
        return x
