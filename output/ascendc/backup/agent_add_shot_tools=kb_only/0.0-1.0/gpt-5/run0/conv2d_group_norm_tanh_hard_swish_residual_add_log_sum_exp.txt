kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_group_norm_tanh_hard_swish_residual_add_log_sum_exp_custom_tiling_data.h"

// 本算子输入：x_conv（NCHW）、x_norm（NCHW）
// 输出：在C维做LogSumExp后的张量（N,1,H,W）
// 内部计算：res = x_conv + hardswish(tanh(x_norm))
// y = logsumexp(res, axis=C, keepdim=True)

constexpr int32_t BUFFER_NUM = 2; // 每个队列的tensor数

class KernelConv2dGroupNormTanhHardSwishResidualAddLogSumExp {
public:
    __aicore__ inline KernelConv2dGroupNormTanhHardSwishResidualAddLogSumExp() {}

    __aicore__ inline void Init(GM_ADDR xConv, GM_ADDR xNorm, GM_ADDR yOut,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w,
                                uint32_t hw, uint32_t tileNum)
    {
        // 简化：BLOCK_DIM=1，按每个batch内的H*W分块
        this->N = n; this->C = c; this->H = h; this->W = w; this->HW = hw;
        this->tileNum = tileNum;
        this->tileLength = (this->HW / tileNum) / BUFFER_NUM;
        if (this->tileLength == 0) { this->tileLength = this->HW; }

        xConvGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(xConv), static_cast<uint32_t>(N*C*HW));
        xNormGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(xNorm), static_cast<uint32_t>(N*C*HW));
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(yOut), static_cast<uint32_t>(N*HW)); // N*1*H*W

        // 输入两个队列，输出一个队列，另配一个计算队列做中间缓冲
        pipe.InitBuffer(inQueueConv, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueNorm, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpQue, BUFFER_NUM * 2, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t nIdx = 0; nIdx < N; ++nIdx) {
            // 对单个batch内的H*W按tile处理
            uint32_t loopCount = (this->HW + this->tileLength - 1) / this->tileLength;
            for (uint32_t tileIdx = 0; tileIdx < loopCount; ++tileIdx) {
                uint32_t posStart = tileIdx * this->tileLength;
                uint32_t curLen = AscendC::min(this->tileLength, this->HW - posStart);
                ComputeTile(nIdx, posStart, curLen);
            }
        }
    }

private:
    __aicore__ inline void ComputeTile(uint32_t nIdx, uint32_t posStart, uint32_t curLen)
    {
        // 分配局部buffer
        AscendC::LocalTensor<float> maxLocal = tmpQue.AllocTensor<float>();
        AscendC::LocalTensor<float> sumLocal = tmpQue.AllocTensor<float>();

        // 1) 第一轮：计算每个位置在C上的最大值 maxLocal
        bool firstCh = true;
        for (uint32_t cIdx = 0; cIdx < C; ++cIdx) {
            // 从GM取当前(n,c)在[posStart, posStart+curLen)的切片（连续）
            AscendC::LocalTensor<float> convLocal = inQueueConv.AllocTensor<float>();
            AscendC::LocalTensor<float> normLocal = inQueueNorm.AllocTensor<float>();
            uint32_t base = (nIdx * C + cIdx) * HW + posStart;
            AscendC::DataCopy(convLocal, xConvGm[base], curLen);
            AscendC::DataCopy(normLocal, xNormGm[base], curLen);
            inQueueConv.EnQue(convLocal);
            inQueueNorm.EnQue(normLocal);

            // 计算 tanh -> hardswish -> residual
            AscendC::LocalTensor<float> convLocalD = inQueueConv.DeQue<float>();
            AscendC::LocalTensor<float> normLocalD = inQueueNorm.DeQue<float>();
            AscendC::LocalTensor<float> resLocal = outQueueY.AllocTensor<float>(); // 复用输出队列做中间计算

            // tanh
            AscendC::Tanh(resLocal, normLocalD, curLen);
            // hardswish(x) = x * relu6(x + 3) / 6
            AscendC::HardSwish(resLocal, resLocal, curLen);
            // residual add
            AscendC::Add(resLocal, resLocal, convLocalD, curLen);

            // 更新max
            if (firstCh) {
                AscendC::DataCopy(maxLocal, resLocal, curLen);
                firstCh = false;
            } else {
                AscendC::Maximum(maxLocal, maxLocal, resLocal, curLen);
            }

            outQueueY.EnQue(resLocal);
            outQueueY.DeQue<float>(); // 释放resLocal
            inQueueConv.FreeTensor(convLocalD);
            inQueueNorm.FreeTensor(normLocalD);
        }

        // 2) 第二轮：sumLocal = sum(exp(res - maxLocal))
        bool firstSum = true;
        for (uint32_t cIdx = 0; cIdx < C; ++cIdx) {
            AscendC::LocalTensor<float> convLocal = inQueueConv.AllocTensor<float>();
            AscendC::LocalTensor<float> normLocal = inQueueNorm.AllocTensor<float>();
            uint32_t base = (nIdx * C + cIdx) * HW + posStart;
            AscendC::DataCopy(convLocal, xConvGm[base], curLen);
            AscendC::DataCopy(normLocal, xNormGm[base], curLen);
            inQueueConv.EnQue(convLocal);
            inQueueNorm.EnQue(normLocal);

            AscendC::LocalTensor<float> convLocalD = inQueueConv.DeQue<float>();
            AscendC::LocalTensor<float> normLocalD = inQueueNorm.DeQue<float>();
            AscendC::LocalTensor<float> resLocal = outQueueY.AllocTensor<float>();
            AscendC::LocalTensor<float> tmpLocal = tmpQue.AllocTensor<float>();

            // tanh -> hardswish -> residual
            AscendC::Tanh(resLocal, normLocalD, curLen);
            AscendC::HardSwish(resLocal, resLocal, curLen);
            AscendC::Add(resLocal, resLocal, convLocalD, curLen);

            // exp(res - maxLocal)
            AscendC::Sub(tmpLocal, resLocal, maxLocal, curLen);
            AscendC::Exp(tmpLocal, tmpLocal, curLen);

            if (firstSum) {
                AscendC::DataCopy(sumLocal, tmpLocal, curLen);
                firstSum = false;
            } else {
                AscendC::Add(sumLocal, sumLocal, tmpLocal, curLen);
            }

            tmpQue.FreeTensor(tmpLocal);
            outQueueY.EnQue(resLocal);
            outQueueY.DeQue<float>(); // 释放resLocal
            inQueueConv.FreeTensor(convLocalD);
            inQueueNorm.FreeTensor(normLocalD);
        }

        // 3) y = log(sumLocal) + maxLocal
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::Log(yLocal, sumLocal, curLen);
        AscendC::Add(yLocal, yLocal, maxLocal, curLen);

        // 写回 (N,1,H,W) 的对应位置
        uint32_t outBase = nIdx * HW + posStart;
        AscendC::DataCopy(yGm[outBase], yLocal, curLen);

        // 释放
        outQueueY.EnQue(yLocal);
        outQueueY.DeQue<float>();
        tmpQue.FreeTensor(sumLocal);
        tmpQue.FreeTensor(maxLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueConv, inQueueNorm;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM * 2> tmpQue;

    AscendC::GlobalTensor<float> xConvGm;
    AscendC::GlobalTensor<float> xNormGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t N, C, H, W, HW;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv2d_group_norm_tanh_hard_swish_residual_add_log_sum_exp_custom(
    GM_ADDR xConv, GM_ADDR xNorm, GM_ADDR yOut, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelConv2dGroupNormTanhHardSwishResidualAddLogSumExp op;
    op.Init(xConv, xNorm, yOut,
            tilingData.n, tilingData.c, tilingData.h, tilingData.w,
            tilingData.hw, tilingData.tileNum);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Fused custom NPU op replaces: Tanh -> HardSwish -> Residual Add -> LogSumExp.
    Conv2d and GroupNorm remain as native modules.
    """
    def __init__(self, in_channels, out_channels, kernel_size, groups, eps=1e-5):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.group_norm = torch.nn.GroupNorm(groups, out_channels, eps=eps)

    def forward(self, x):
        x_conv = self.conv(x)           # [N,C,H,W]
        x_norm = self.group_norm(x_conv)
        # Fused custom op expects x_conv and x_norm, outputs [N,1,H,W]
        y = custom_ops_lib.conv2d_group_norm_tanh_hard_swish_residual_add_log_sum_exp_custom(x_conv, x_norm)
        return y
