#include "masked_cumsum_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    MaskedCumsumCustomTilingData tiling;

    // 仅支持沿最后一维做前缀和（与AscendC文档中Sum/Reduce等API基于最后一维一致）
    const gert::Shape* inShape = context->GetInputShape(0);
    auto origin = inShape->GetOriginShape();
    std::vector<int64_t> dims = origin.GetDimSizes();
    if (dims.empty()) {
        return ge::GRAPH_FAILED;
    }
    uint32_t lastAxis = static_cast<uint32_t>(dims.back());
    uint32_t firstAxis = 1U;
    for (size_t i = 0; i + 1 < dims.size(); ++i) {
        firstAxis *= static_cast<uint32_t>(dims[i]);
    }

    // 按UB能力选择一个保守的tile大小。考虑到需要x/mask/y三份本地缓存（参考AscendC示例的本地队列使用方式），
    // 对float类型每元素4B，tileSize=2048时占用约 2048*4*3 = 24KB，较为安全。
    const uint32_t TILE_SIZE = 2048;

    tiling.set_firstAxis(firstAxis);
    tiling.set_lastAxis(lastAxis);
    tiling.set_tileSize(TILE_SIZE);

    // 设置Block数（参考文档示例设置8）
    context->SetBlockDim(8);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 无需workspace
    size_t* workspace = context->GetWorkspaceSizes(1);
    workspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    const gert::Shape* m_shape = context->GetInputShape(1);
    // 简单校验：形状一致
    if (x_shape->GetOriginShape().GetDimSizes() != m_shape->GetOriginShape().GetDimSizes()) {
        return ge::GRAPH_FAILED;
    }
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "masked_cumsum_custom_tiling_data.h"

// 参考官方示例的队列和拷贝模式（Add/Sum示例）
constexpr int32_t BUFFER_NUM = 1;

class KernelMaskedCumsum {
public:
    __aicore__ inline KernelMaskedCumsum() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mask, GM_ADDR y,
                                uint32_t firstAxis, uint32_t lastAxis, uint32_t tileSize)
    {
        this->firstAxis = firstAxis;
        this->lastAxis = lastAxis;
        this->tileSize = tileSize;

        // 行划分到不同block（参考示例使用GetBlockNum/GetBlockIdx）
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowStart = (firstAxis * blockIdx) / blockNum;
        uint32_t rowEnd = (firstAxis * (blockIdx + 1)) / blockNum;
        this->rowsThisBlock = rowEnd - rowStart;

        // 每行长度为lastAxis
        uint32_t baseOffset = rowStart * lastAxis;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + baseOffset, (uint32_t)(rowsThisBlock * lastAxis));
        mGm.SetGlobalBuffer((__gm__ DTYPE_Y *)mask + baseOffset, (uint32_t)(rowsThisBlock * lastAxis));
        yGm.SetGlobalBuffer((__gm__ DTYPE_Z *)y + baseOffset, (uint32_t)(rowsThisBlock * lastAxis));

        // 初始化本地队列缓冲，三路(x, mask, y)，按tileSize配置
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileSize * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueM, BUFFER_NUM, this->tileSize * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileSize * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        // 逐行处理（沿最后一维做前缀和，参考文档中Reduce/Sum等均基于最后一维）
        for (uint32_t row = 0; row < rowsThisBlock; ++row) {
            uint32_t processed = 0;
            DTYPE_Z carry = static_cast<DTYPE_Z>(0);

            while (processed < lastAxis) {
                uint32_t curLen = lastAxis - processed;
                if (curLen > tileSize) curLen = tileSize;

                // CopyIn
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::LocalTensor<DTYPE_Y> mLocal = inQueueM.AllocTensor<DTYPE_Y>();
                AscendC::DataCopy(xLocal, xGm[row * lastAxis + processed], curLen);
                AscendC::DataCopy(mLocal, mGm[row * lastAxis + processed], curLen);
                inQueueX.EnQue(xLocal);
                inQueueM.EnQue(mLocal);

                // Compute
                AscendC::LocalTensor<DTYPE_X> xLoc = inQueueX.DeQue<DTYPE_X>();
                AscendC::LocalTensor<DTYPE_Y> mLoc = inQueueM.DeQue<DTYPE_Y>();
                AscendC::LocalTensor<DTYPE_Z> yLocal = outQueueY.AllocTensor<DTYPE_Z>();

                // 逐元素累加（前缀和），mask为0/1浮点，与x逐元素相乘（参考高阶APIAdd/Asin示例的本地张量访问方式）
                // 注意：此处展示逻辑实现，依赖编译器对本地张量的元素访问支持。
                for (uint32_t i = 0; i < curLen; ++i) {
                    DTYPE_Z xv = static_cast<DTYPE_Z>(xLoc.GetValue(i));
                    DTYPE_Z mv = static_cast<DTYPE_Z>(mLoc.GetValue(i));
                    carry = carry + xv * mv;
                    yLocal.SetValue(i, carry);
                }

                outQueueY.EnQue<DTYPE_Z>(yLocal);
                inQueueX.FreeTensor(xLoc);
                inQueueM.FreeTensor(mLoc);

                // CopyOut
                AscendC::LocalTensor<DTYPE_Z> yLoc = outQueueY.DeQue<DTYPE_Z>();
                AscendC::DataCopy(yGm[row * lastAxis + processed], yLoc, curLen);
                outQueueY.FreeTensor(yLoc);

                processed += curLen;
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueM;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> mGm;
    AscendC::GlobalTensor<DTYPE_Z> yGm;

    uint32_t firstAxis;
    uint32_t lastAxis;
    uint32_t tileSize;
    uint32_t rowsThisBlock;
};

extern "C" __global__ __aicore__ void masked_cumsum_custom(GM_ADDR x, GM_ADDR mask, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMaskedCumsum op;
    op.Init(x, mask, y, tiling_data.firstAxis, tiling_data.lastAxis, tiling_data.tileSize);
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor masked_cumsum_custom_impl_npu(const at::Tensor& x, const at::Tensor& mask) {
    // 内核期望float输入，且沿最后一维做前缀和
    auto mask_f = mask.to(x.dtype());
    at::Tensor y = at::empty_like(x);
    EXEC_NPU_CMD(aclnnMaskedCumsumCustom, x, mask_f, y);
    return y;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("masked_cumsum_custom", &masked_cumsum_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("masked_cumsum_custom", &masked_cumsum_custom_impl_npu, "masked cumsum along last axis");
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    优化版：使用自定义AscendC算子实现带mask的累加和，沿最后一维执行。
    注意：dim需等于最后一维。
    """
    def __init__(self, dim):
        super().__init__()
        self.dim = dim  # 仅支持 dim == x.dim() - 1

    def forward(self, x, mask):
        # 自定义算子输入为float，沿最后一维执行前缀和
        assert self.dim == x.dim() - 1, "MaskedCumsumCustom 仅支持沿最后一维扫描"
        return custom_ops_lib.masked_cumsum_custom(x, mask.to(dtype=x.dtype))
