kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transposed_1d_custom_tiling_data.h"

// 计算公式（stride=1，padding=0，output_padding=0，groups=1）：
// y[n, oc, p] = sum_{ic=0..Cin-1} sum_{k=0..K-1} x[n, ic, l] * w[ic, oc, k]
// 其中 p = l - 0 + k = l + k，且 0 <= p < Lout，等价于对x做反卷积的直接计算

extern "C" __global__ __aicore__ void conv_transposed_1d_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tilingData, tiling);

    const uint32_t N     = tilingData.batch;
    const uint32_t Cin   = tilingData.inC;
    const uint32_t Cout  = tilingData.outC;
    const uint32_t Lin   = tilingData.inL;
    const uint32_t K     = tilingData.kSize;
    const uint32_t stride = tilingData.stride;         // 默认1
    const uint32_t pad    = tilingData.padding;        // 默认0
    const uint32_t outPad = tilingData.outputPadding;  // 默认0
    const uint32_t groups = tilingData.groups;         // 默认1
    const uint32_t Lout   = tilingData.outL;

    // 简单按batch维划分block
    uint32_t blockNum = AscendC::GetBlockNum();
    uint32_t blockIdx = AscendC::GetBlockIdx();
    uint32_t bsPerBlock = (N + blockNum - 1) / blockNum;
    uint32_t nStart = blockIdx * bsPerBlock;
    uint32_t nEnd = nStart + bsPerBlock;
    if (nStart >= N) return;
    if (nEnd > N) nEnd = N;

    __gm__ DTYPE_X* x_ptr = reinterpret_cast<__gm__ DTYPE_X*>(x);
    __gm__ DTYPE_Y* y_ptr = reinterpret_cast<__gm__ DTYPE_Y*>(y);
    __gm__ DTYPE_WEIGHT* w_ptr = reinterpret_cast<__gm__ DTYPE_WEIGHT*>(weight);

    // 初始化输出为0
    for (uint32_t n = nStart; n < nEnd; ++n) {
        uint64_t yBase = (uint64_t)n * Cout * Lout;
        for (uint32_t oc = 0; oc < Cout; ++oc) {
            uint64_t yOff = yBase + (uint64_t)oc * Lout;
            for (uint32_t p = 0; p < Lout; ++p) {
                y_ptr[yOff + p] = static_cast<DTYPE_Y>(0);
            }
        }
    }

    // 直接计算（groups=1）
    for (uint32_t n = nStart; n < nEnd; ++n) {
        uint64_t xBase = (uint64_t)n * Cin * Lin;
        uint64_t yBase = (uint64_t)n * Cout * Lout;
        for (uint32_t ic = 0; ic < Cin; ++ic) {
            uint64_t xOff = xBase + (uint64_t)ic * Lin;
            for (uint32_t oc = 0; oc < Cout; ++oc) {
                // 权重布局: [Cin, Cout, K]
                uint64_t wOff = ((uint64_t)ic * Cout + oc) * K;
                for (uint32_t l = 0; l < Lin; ++l) {
                    // 输出起点位置
                    // p = l * stride - pad + k
                    int64_t p0 = (int64_t)l * (int64_t)stride - (int64_t)pad;
                    DTYPE_X xval = x_ptr[xOff + l];
                    for (uint32_t k = 0; k < K; ++k) {
                        int64_t p = p0 + (int64_t)k + (int64_t)outPad; // outPad默认为0
                        if (p >= 0 && (uint64_t)p < Lout) {
                            uint64_t yOff = yBase + (uint64_t)oc * Lout + (uint64_t)p;
                            DTYPE_WEIGHT wval = w_ptr[wOff + k];
                            y_ptr[yOff] = static_cast<DTYPE_Y>(y_ptr[yOff] + xval * wval);
                        }
                    }
                }
            }
        }
    }
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor conv_transposed_1d_custom_impl_npu(const at::Tensor& x, const at::Tensor& weight) {
    // 输出形状推导：Lout = (Lin - 1) * stride - 2*padding + K + output_padding，默认stride=1,pad=0,out_pad=0
    TORCH_CHECK(x.dim() == 3, "x must be 3D [N, Cin, Lin]");
    TORCH_CHECK(weight.dim() == 3, "weight must be 3D [Cin, Cout, K]");
    int64_t N = x.size(0);
    int64_t Cin = x.size(1);
    int64_t Lin = x.size(2);
    TORCH_CHECK(weight.size(0) == Cin, "weight.size(0) must equal Cin");
    int64_t Cout = weight.size(1);
    int64_t K = weight.size(2);

    int64_t Lout = (Lin - 1) * 1 - 2 * 0 + K + 0;
    at::Tensor y = at::empty({N, Cout, Lout}, x.options());

    EXEC_NPU_CMD(aclnnConvTransposed1dCustom, x, weight, y);
    return y;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("conv_transposed_1d_custom", &conv_transposed_1d_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("conv_transposed_1d_custom", &conv_transposed_1d_custom_impl_npu, "conv_transposed_1d_custom");
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    使用AscendC自定义算子实现的1D反卷积（仅支持stride=1, padding=0, output_padding=0, groups=1, bias=False）。
    权重张量布局: [in_channels, out_channels, kernel_size]
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        # 采用与ConvTranspose1d相同的权重语义（groups=1）：[Cin, Cout, K]
        w = torch.empty(in_channels, out_channels, kernel_size)
        torch.nn.init.kaiming_uniform_(w, a=5 ** 0.5)
        self.weight = torch.nn.Parameter(w)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return custom_ops_lib.conv_transposed_1d_custom(x, self.weight)
