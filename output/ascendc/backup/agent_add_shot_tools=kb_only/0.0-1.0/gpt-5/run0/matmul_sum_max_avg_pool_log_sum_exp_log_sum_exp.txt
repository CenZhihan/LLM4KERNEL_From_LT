kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matmul_sum_max_avg_pool_log_sum_exp_log_sum_exp_custom_tiling_data.h"

using DTYPE = float;

class KernelFusedReduceSum {
public:
    __aicore__ inline KernelFusedReduceSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t outter, uint32_t inner, uint32_t n, uint32_t tmpSize)
    {
        this->outter = outter;
        this->inner = inner;
        this->n = n;
        this->tmpSize = tmpSize;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(x), outter * inner);
        zGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(z), outter);

        // one row buffer
        pipe.InitBuffer(inQueue, 1, inner * sizeof(DTYPE));
        // result buffer (one value per row)
        pipe.InitBuffer(outQueue, 1, sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        // Reduce across dim=1 by summation:
        // Implemented via Mean (docs example) then scaled by n using repeated Add (docs Add API usage).
        for (uint32_t b = 0; b < outter; ++b) {
            CopyIn(b);
            Compute(b);
            CopyOut(b);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t b)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueue.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm[b * inner], inner);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t)
    {
        // Dequeue input row
        AscendC::LocalTensor<DTYPE> xLocal = inQueue.DeQue<DTYPE>();

        // Prepare MeanParams like in docs (Mean API)
        AscendC::MeanParams meanParams;
        meanParams.outter = 1;
        meanParams.inner = inner;
        meanParams.n = n;

        AscendC::LocalTensor<DTYPE> meanLocal = outQueue.AllocTensor<DTYPE>();

        if (tmpSize != 0) {
            pipe.InitBuffer(tmpBuf, tmpSize);
            AscendC::LocalTensor<uint8_t> tmpLocal = tmpBuf.Get<uint8_t>();
            AscendC::Mean<DTYPE, DTYPE>(meanLocal, xLocal, tmpLocal, meanParams);
        } else {
            AscendC::Mean<DTYPE, DTYPE>(meanLocal, xLocal, meanParams);
        }

        // Scale mean by n to get sum: sum = mean * n
        // Use Add repeatedly with count=1 as per available high-level API pattern.
        AscendC::LocalTensor<DTYPE> sumLocal = meanLocal; // start from mean
        for (uint32_t i = 1; i < n; ++i) {
            AscendC::Add(sumLocal, sumLocal, meanLocal, 1);
            AscendC::PipeBarrier<PIPE_V>();
        }

        // enqueue and free tensors
        outQueue.EnQue<DTYPE>(sumLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t b)
    {
        AscendC::LocalTensor<DTYPE> zLocal = outQueue.DeQue<DTYPE>();
        AscendC::DataCopy(zGm[b], zLocal, 1);
        outQueue.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> zGm;

    uint32_t outter;
    uint32_t inner;
    uint32_t n;
    uint32_t tmpSize;
};

extern "C" __global__ __aicore__ void matmul_sum_max_avg_pool_log_sum_exp_log_sum_exp_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    if (TILING_KEY_IS(1)) {
        KernelFusedReduceSum op;
        op.Init(x, z, tiling_data.outter, tiling_data.inner, tiling_data.n, tiling_data.tmpSize);
        op.Process();
    }
}
