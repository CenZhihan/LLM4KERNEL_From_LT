#include "clamp_broadcast_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 输入0为x，按其形状做切分
    const gert::Shape* in_shape = context->GetInputShape(0);
    ge::Shape shape = in_shape->GetOriginShape();
    uint32_t totalLength = static_cast<uint32_t>(shape.GetShapeSize());

    // 查询Clamp所需的临时空间范围（基于文档5.1.22.3 GetClampMaxMinTmpSize）
    uint32_t maxValue = 0;
    uint32_t minValue = 0;
    uint32_t typeSize = 4; // float
    AscendC::GetClampMaxMinTmpSize(shape, typeSize, false, maxValue, minValue);

    // 根据平台剩余UB空间选择实际的tmpBufferSize（参考5.1.31 More Examples）
    auto platformInfo = context->GetPlatformInfo();
    auto ascendcPlatform = platform_ascendc::PlatformAscendC(platformInfo);
    uint64_t tailSize = 0; // UB剩余空间
    ascendcPlatform.GetCoreMemSize(platform_ascendc::CoreMemType::UB, tailSize);
    uint32_t tmpSize = static_cast<uint32_t>(tailSize >= maxValue ? maxValue : tailSize);

    ClampBroadcastCustomTilingData tiling;
    context->SetBlockDim(BLOCK_DIM);
    tiling.set_totalLength(totalLength);
    tiling.set_tileNum(TILE_NUM);
    tiling.set_tmpBufferSize(tmpSize);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t* currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}
}


namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* z_shape = context->GetOutputShape(0);
    *z_shape = *x_shape;
    return GRAPH_SUCCESS;
}

{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "clamp_broadcast_custom_tiling_data.h"

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 2; // 双缓冲
// 钳位阈值：示例中min=-2.0, max=2.0（标量阈值，结合文档5.1.22.1/5.1.22.2）
constexpr DTYPE CLAMP_MIN_SCALAR = static_cast<DTYPE>(-2.0f);
constexpr DTYPE CLAMP_MAX_SCALAR = static_cast<DTYPE>(2.0f);

class KernelClampBroadcast {
public:
    __aicore__ inline KernelClampBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR minv, GM_ADDR maxv, GM_ADDR z,
                                uint32_t totalLength, uint32_t tileNum, uint32_t tmpBufferSize)
    {
        // 基于block划分
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        // 输入输出队列
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(midQueue, BUFFER_NUM, this->tileLength * sizeof(DTYPE)); // 中间结果
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        // Clamp所需临时空间（文档5.1.22.1/5.1.22.2及5.1.22.3）
        pipe.InitBuffer(tmpQueue, 1, tmpBufferSize);
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> midLocal = midQueue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.AllocTensor<DTYPE>();
        AscendC::LocalTensor<uint8_t> sharedTmpBuffer = tmpQueue.AllocTensor<uint8_t>();

        // 先执行ClampMax，再执行ClampMin（文档5.1.22.1与5.1.22.2）
        AscendC::ClampMax<DTYPE>(midLocal, xLocal, sharedTmpBuffer, CLAMP_MAX_SCALAR, this->tileLength);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::ClampMin<DTYPE>(zLocal, midLocal, sharedTmpBuffer, CLAMP_MIN_SCALAR, this->tileLength);
        AscendC::PipeBarrier<PIPE_V>();

        outQueueZ.EnQue<DTYPE>(zLocal);
        inQueueX.FreeTensor(xLocal);
        midQueue.FreeTensor(midLocal);
        tmpQueue.FreeTensor(sharedTmpBuffer);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE> zLocal = outQueueZ.DeQue<DTYPE>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> midQueue;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> zGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void clamp_broadcast_custom(GM_ADDR x, GM_ADDR minv, GM_ADDR maxv, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    (void)minv; (void)maxv; // 本实现按标量阈值路径使用
    GET_TILING_DATA(tiling_data, tiling);
    KernelClampBroadcast op;
    op.Init(x, minv, maxv, z, tiling_data.totalLength, tiling_data.tileNum, tiling_data.tmpBufferSize);
    op.Process();
}
