static ge::graphStatus TilingFunc(gert::TilingContext* context)

    // 获取输入形状 (M,K) 和 (K,N)
    auto a_shape = context->GetInputShape(0)->GetOriginShape();
    auto b_shape = context->GetInputShape(1)->GetOriginShape();
    if (a_shape.GetDimNum() != 2 || b_shape.GetDimNum() != 2) {
        return ge::GRAPH_FAILED;
    }
    uint32_t M = static_cast<uint32_t>(a_shape.GetDim(0));
    uint32_t K = static_cast<uint32_t>(a_shape.GetDim(1));
    uint32_t Kb = static_cast<uint32_t>(b_shape.GetDim(0));
    uint32_t N = static_cast<uint32_t>(b_shape.GetDim(1));
    if (K != Kb) {
        return ge::GRAPH_FAILED;

    // 简单的分块参数（可根据平台特性调优）

    const uint32_t tileK = 128; // K方向分块

    tiling.set_M(M);
    tiling.set_N(N);
    tiling.set_K(K);
    tiling.set_tileK(tileK);
    tiling.set_tileN(tileN);

    // 保存tiling数据
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // 本算子不需要额外workspace
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}

        return GRAPH_FAILED;

    int64_t M = a_shape->GetOriginShape().GetDim(0);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "square_matrix_multiplication_custom_tiling_data.h"

// 简单双缓冲实现的向量化分块GEMM示例：按行切分到不同block，每个block计算部分行。
// 参考文档中LocalTensor/DataCopy/队列等用法（见LayerNormGrad与RmsNorm示例）。

using DTYPE_A = float;
using DTYPE_B = float;
using DTYPE_C = float;

constexpr int32_t BUFFER_NUM = 2;

class KernelSquareMatmul {
public:
    __aicore__ inline KernelSquareMatmul() {}

    __aicore__ inline void Init(GM_ADDR aGm, GM_ADDR bGm, GM_ADDR cGm,
                                uint32_t M, uint32_t N, uint32_t K,
                                uint32_t tileK, uint32_t tileN)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        this->tileK = tileK;
        this->tileN = tileN;

        // 按block将行均分
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (M + blockNum - 1) / blockNum;
        this->rowStart = blockIdx * rowsPerBlock;
        this->rowEnd = AscendC::Min(rowStart + rowsPerBlock, M);

        // 绑定GM
        aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_A*>(aGm), (uint64_t)M * K);
        bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_B*>(bGm), (uint64_t)K * N);
        cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE_C*>(cGm), (uint64_t)M * N);

        // 队列缓冲，按最大列分块tileN，以及K分块tileK 设计本地缓存
        // 每次计算：1行C的一个列块（长度tileN），累加遍历K方向
        pipe.InitBuffer(inQueueA, BUFFER_NUM, tileK * sizeof(DTYPE_A));        // A的分块（1行 tileK段）
        pipe.InitBuffer(inQueueB, BUFFER_NUM, tileN * sizeof(DTYPE_B));        // B的分块（1行 tileN段）
        pipe.InitBuffer(outQueueC, BUFFER_NUM, tileN * sizeof(DTYPE_C));       // C的输出分块（1行 tileN段）
        pipe.InitBuffer(tmpQueue, BUFFER_NUM, tileN * sizeof(DTYPE_C));        // 累加缓冲
    }

    __aicore__ inline void Process()
    {
        // 遍历分配到本block的行
        for (uint32_t r = rowStart; r < rowEnd; ++r) {
            // 列方向分块
            for (uint32_t cBase = 0; cBase < N; cBase += tileN) {
                uint32_t curN = AscendC::Min(tileN, N - cBase);

                // 分配并清零一个输出分块
                AscendC::LocalTensor<DTYPE_C> cTile = outQueueC.AllocTensor<DTYPE_C>();
                AscendC::Duplicate(cTile, static_cast<DTYPE_C>(0), curN);

                // K方向分块累加
                for (uint32_t kBase = 0; kBase < K; kBase += tileK) {
                    uint32_t curK = AscendC::Min(tileK, K - kBase);

                    // 读入A当前行的一段（长度curK）
                    AscendC::LocalTensor<DTYPE_A> aSeg = inQueueA.AllocTensor<DTYPE_A>();
                    AscendC::DataCopy(aSeg, aGlobal[r * K + kBase], curK);
                    inQueueA.EnQue(aSeg);

                    // 对于该curK段中的每个k，读取B[k, cBase:cBase+curN]并做向量FMA到cTile
                    AscendC::LocalTensor<DTYPE_A> aSegLocal = inQueueA.DeQue<DTYPE_A>();
                    for (uint32_t kk = 0; kk < curK; ++kk) {
                        // 读取B的一个向量块
                        AscendC::LocalTensor<DTYPE_B> bVec = inQueueB.AllocTensor<DTYPE_B>();
                        AscendC::DataCopy(bVec, bGlobal[(kBase + kk) * N + cBase], curN);
                        inQueueB.EnQue(bVec);

                        AscendC::LocalTensor<DTYPE_B> bVecLocal = inQueueB.DeQue<DTYPE_B>();

                        // 临时向量：bVec * scalar(aSeg[kk])
                        AscendC::LocalTensor<DTYPE_C> tmp = tmpQueue.AllocTensor<DTYPE_C>();
                        // 标量广播乘法：使用Muls将b向量按标量缩放
                        AscendC::Muls(tmp, bVecLocal, static_cast<DTYPE_C>(aSegLocal.GetValue(kk)), curN);
                        // cTile += tmp
                        AscendC::Add(cTile, cTile, tmp, curN);

                        tmpQueue.FreeTensor(tmp);
                        inQueueB.FreeTensor(bVecLocal);
                    }
                    inQueueA.FreeTensor(aSegLocal);
                }

                // 写回C
                outQueueC.EnQue(cTile);
                AscendC::LocalTensor<DTYPE_C> cTileOut = outQueueC.DeQue<DTYPE_C>();
                AscendC::DataCopy(cGlobal[r * N + cBase], cTileOut, curN);
                outQueueC.FreeTensor(cTileOut);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpQueue;

    AscendC::GlobalTensor<DTYPE_A> aGlobal;
    AscendC::GlobalTensor<DTYPE_B> bGlobal;
    AscendC::GlobalTensor<DTYPE_C> cGlobal;

    uint32_t M, N, K;
    uint32_t tileK, tileN;

    uint32_t rowsPerBlock;
    uint32_t rowStart, rowEnd;
};

extern "C" __global__ __aicore__ void square_matrix_multiplication_custom(GM_ADDR aGm, GM_ADDR bGm, GM_ADDR cGm, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSquareMatmul op;
    op.Init(aGm, bGm, cGm,
            tiling_data.M, tiling_data.N, tiling_data.K,
            tiling_data.tileK, tiling_data.tileN);
    op.Process();
}
"""
