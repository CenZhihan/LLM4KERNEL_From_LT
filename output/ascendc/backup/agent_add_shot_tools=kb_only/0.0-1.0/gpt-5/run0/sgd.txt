host_operator_src="""
#include "sgd_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    SgdCustomTilingData tiling;
    uint32_t totalLength = context->GetInputShape(0)->GetOriginShape().GetShapeSize();

    // Read attributes (momentum, lr) from context, with fallbacks
    float momentum = 0.9f;
    float lr = 1e-2f;
    if (context->GetAttrs() != nullptr) {
        auto attrs = context->GetAttrs();
        (void)attrs->GetFloat("momentum", momentum);
        (void)attrs->GetFloat("lr", lr);
    }

    context->SetBlockDim(BLOCK_DIM);
    tiling.set_totalLength(totalLength);
    tiling.set_tileNum(TILE_NUM);
    tiling.set_momentum(momentum);
    tiling.set_lr(lr);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* p_shape = context->GetInputShape(0);
    gert::Shape* out_shape = context->GetOutputShape(0);
    *out_shape = *p_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class SgdCustom : public OpDef {
public:
    explicit SgdCustom(const char* name) : OpDef(name)
    {
        this->Input("param")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("grad")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Input("velocity")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});
        this->Output("out")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_ND});

        // Attributes
        this->Attr("momentum").Float(0.9f);
        this->Attr("lr").Float(1e-2f);

        this->SetInferShape(ge::InferShape).SetInferDataType(ge::InferDataType);
        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(SgdCustom);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "sgd_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSgd {
public:
    __aicore__ inline KernelSgd() {}
    __aicore__ inline void Init(GM_ADDR param, GM_ADDR grad, GM_ADDR velocity, GM_ADDR out,
                                uint32_t totalLength, uint32_t tileNum,
                                float momentum, float lr)
    {
        ASSERT(AscendC::GetBlockNum() != 0 && "block dim can not be zero!");
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        ASSERT(tileNum != 0 && "tile num can not be zero!");
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->momentum = momentum;
        this->lr = lr;

        paramGm.SetGlobalBuffer((__gm__ DTYPE_X*)param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        gradGm.SetGlobalBuffer((__gm__ DTYPE_X*)grad + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        velGm.SetGlobalBuffer((__gm__ DTYPE_X*)velocity + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outGm.SetGlobalBuffer((__gm__ DTYPE_Z*)out + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueParam, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueGrad,  BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueVel,   BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueOut,  BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpQueue,     BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> pLocal = inQueueParam.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> gLocal = inQueueGrad.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> vLocal = inQueueVel.AllocTensor<DTYPE_X>();

        AscendC::DataCopy(pLocal, paramGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(gLocal, gradGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(vLocal, velGm[progress * this->tileLength], this->tileLength);

        inQueueParam.EnQue(pLocal);
        inQueueGrad.EnQue(gLocal);
        inQueueVel.EnQue(vLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> pLocal = inQueueParam.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> gLocal = inQueueGrad.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> vLocal = inQueueVel.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueueOut.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_Z> tmpLocal = tmpQueue.AllocTensor<DTYPE_Z>();

        // tmp = momentum * velocity
        AscendC::Muls(tmpLocal, vLocal, (DTYPE_Z)momentum, this->tileLength);
        // tmp = tmp + grad
        AscendC::Add(tmpLocal, tmpLocal, gLocal, this->tileLength);
        // tmp = lr * tmp
        AscendC::Muls(tmpLocal, tmpLocal, (DTYPE_Z)lr, this->tileLength);
        // out = param - tmp
        AscendC::Sub(outLocal, pLocal, tmpLocal, this->tileLength);

        outQueueOut.EnQue(outLocal);
        tmpQueue.FreeTensor(tmpLocal);
        inQueueParam.FreeTensor(pLocal);
        inQueueGrad.FreeTensor(gLocal);
        inQueueVel.FreeTensor(vLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueueOut.DeQue<DTYPE_Z>();
        AscendC::DataCopy(outGm[progress * this->tileLength], outLocal, this->tileLength);
        outQueueOut.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueParam;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueGrad;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueVel;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueOut;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> tmpQueue;

    AscendC::GlobalTensor<DTYPE_X> paramGm;
    AscendC::GlobalTensor<DTYPE_X> gradGm;
    AscendC::GlobalTensor<DTYPE_X> velGm;
    AscendC::GlobalTensor<DTYPE_Z> outGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    DTYPE_Z momentum;
    DTYPE_Z lr;
};

extern "C" __global__ __aicore__ void sgd_custom(GM_ADDR param, GM_ADDR grad, GM_ADDR velocity, GM_ADDR out,
                                                 GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelSgd op;
    op.Init(param, grad, velocity, out,
            tilingData.totalLength, tilingData.tileNum,
            tilingData.momentum, tilingData.lr);
    op.Process();
}
"""

at::Tensor sgd_custom_impl_npu(const at::Tensor& param, const at::Tensor& grad, const at::Tensor& velocity,
                               double momentum, double lr) {
    at::Tensor out = at::empty_like(param);
    EXEC_NPU_CMD(aclnnSgdCustom, param, grad, velocity, static_cast<float>(momentum), static_cast<float>(lr), out);
    return out;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("sgd_custom", &sgd_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("sgd_custom", &sgd_custom_impl_npu, "SGD update: param - lr*(momentum*vel + grad)");
}
