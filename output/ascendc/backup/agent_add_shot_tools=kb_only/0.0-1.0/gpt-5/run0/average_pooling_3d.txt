kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "average_pooling_3d_custom_tiling_data.h"

// 本实现基于 AscendC 高级/基础 API：
// - DataCopy/Duplicate: 数据搬运与初始化（参考 CANN Ascend C API Reference 表1-3/数据填充Duplicate）
// - Add/Muls: 向量与标量计算（参考 基础API 两操作数与标量两操作数）
// - Sum: 高级API按最后一维归约求和，需满足inner 32B对齐并可通过GetSumMaxMinTmpSize预留临时空间（参考5.x Sum章节）

class KernelAveragePooling3d {
public:
    __aicore__ inline KernelAveragePooling3d() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t N, uint32_t C, uint32_t D, uint32_t H, uint32_t W,
                                uint32_t K, uint32_t S, uint32_t P,
                                uint32_t OD, uint32_t OH, uint32_t OW,
                                uint32_t TotalOut,
                                uint32_t WinElems, uint32_t WinInnerAligned,
                                uint32_t SumTmpBufSize)
    {
        this->N = N; this->C = C; this->D = D; this->H = H; this->W = W;
        this->K = K; this->S = S; this->P = P;
        this->OD = OD; this->OH = OH; this->OW = OW;
        this->TotalOut = TotalOut;
        this->WinElems = WinElems;
        this->WinInnerAligned = WinInnerAligned;

        // 多核分工：按输出元素均分
        this->blockOut = TotalOut / AscendC::GetBlockNum();
        this->blockStart = this->blockOut * AscendC::GetBlockIdx();

        // 绑定GM缓冲
        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x, (uint64_t)N * C * D * H * W);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Z*)y, (uint64_t)N * C * OD * OH * OW);

        // 工作队列：窗口缓冲（对齐到32B）
        pipe.InitBuffer(winQueue, 1, WinInnerAligned * sizeof(DTYPE_X));
        // 输出队列：至少分配一个数据块
        pipe.InitBuffer(outQueue, 1, 32);
        // Sum临时缓冲（由框架管理，需预留空间，参考Sum API说明）
        if (SumTmpBufSize > 0) {
            pipe.InitBuffer(tmpQueue, 1, SumTmpBufSize);
        }
    }

    __aicore__ inline void Process()
    {
        // 逐输出元素计算平均池化（包含padding计数）
        for (uint32_t i = 0; i < blockOut; ++i) {
            uint64_t outIndex = (uint64_t)blockStart + i;
            // 反解线性索引到(N, C, od, oh, ow)
            uint32_t ow = static_cast<uint32_t>(outIndex % OW);
            uint64_t t1 = outIndex / OW;
            uint32_t oh = static_cast<uint32_t>(t1 % OH);
            uint64_t t2 = t1 / OH;
            uint32_t od = static_cast<uint32_t>(t2 % OD);
            uint64_t t3 = t2 / OD;
            uint32_t c = static_cast<uint32_t>(t3 % C);
            uint32_t n = static_cast<uint32_t>(t3 / C);

            ComputeOne(n, c, od, oh, ow, outIndex);
        }
    }

private:
    __aicore__ inline void ComputeOne(uint32_t n, uint32_t c, uint32_t od, uint32_t oh, uint32_t ow, uint64_t outLinearIndex)
    {
        AscendC::LocalTensor<DTYPE_X> winLocal = winQueue.AllocTensor<DTYPE_X>();
        // 预留Sum的临时缓冲（按Sum API说明）
        AscendC::LocalTensor<uint8_t> sumTmp;
        if (tmpQueue.IsValid()) {
            sumTmp = tmpQueue.AllocTensor<uint8_t>();
        }
        // 窗口缓冲清零，长度按32B对齐
        DTYPE_X zeroVal(0);
        AscendC::Duplicate<DTYPE_X>(winLocal, zeroVal, WinInnerAligned);

        // 窗口左上后起点（含padding）
        int32_t d0 = static_cast<int32_t>(od) * static_cast<int32_t>(S) - static_cast<int32_t>(P);
        int32_t h0 = static_cast<int32_t>(oh) * static_cast<int32_t>(S) - static_cast<int32_t>(P);
        int32_t w0 = static_cast<int32_t>(ow) * static_cast<int32_t>(S) - static_cast<int32_t>(P);

        // 将有效输入拷入窗口缓冲，布局为 [kd, kh, kw] -> 紧密存储
        for (uint32_t kd = 0; kd < K; ++kd) {
            int32_t di = d0 + static_cast<int32_t>(kd);
            if (di < 0 || di >= static_cast<int32_t>(D)) continue;
            for (uint32_t kh = 0; kh < K; ++kh) {
                int32_t hi = h0 + static_cast<int32_t>(kh);
                if (hi < 0 || hi >= static_cast<int32_t>(H)) continue;

                int32_t wStartValid = w0 < 0 ? 0 : w0;
                int32_t wEndValid = (w0 + static_cast<int32_t>(K)) > static_cast<int32_t>(W)
                                        ? static_cast<int32_t>(W)
                                        : (w0 + static_cast<int32_t>(K));
                int32_t validLen = wEndValid - wStartValid;
                if (validLen <= 0) continue;

                // 输入起始偏移（N,C,D,H,W）
                uint64_t inBase = ((((uint64_t)n * C + c) * D + (uint32_t)di) * H + (uint32_t)hi) * W;
                uint64_t inOffset = inBase + (uint32_t)wStartValid;

                // 窗口缓冲起始偏移：把有效段拷贝到 [kd*K*K + kh*K + (wStartValid - w0)]
                uint32_t winRowBase = kd * K * K + kh * K;
                uint32_t winOffset = winRowBase + static_cast<uint32_t>(wStartValid - w0);

                AscendC::DataCopy(winLocal[winOffset], xGm[inOffset], static_cast<uint32_t>(validLen));
            }
        }

        // Sum按最后一维求和：outter=1, inner=WinInnerAligned(32B对齐), n=WinElems
        AscendC::LocalTensor<DTYPE_X> accLocal = outQueue.AllocTensor<DTYPE_X>();
        AscendC::SumParams sumParams{1u, WinInnerAligned, WinElems};
        AscendC::Sum(accLocal, winLocal, sumParams);

        // 标量乘以 1/(K^3) 得到平均值
        float invWin = 1.0f / static_cast<float>(WinElems);
        AscendC::Muls(accLocal, accLocal, invWin, 1);

        // 写回输出张量（N,C,OD,OH,OW）
        AscendC::LocalTensor<DTYPE_X> outLocal = accLocal;
        AscendC::DataCopy(yGm[outLinearIndex], outLocal, 1);

        outQueue.FreeTensor(accLocal);
        if (tmpQueue.IsValid()) {
            tmpQueue.FreeTensor(sumTmp);
        }
        winQueue.FreeTensor(winLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> winQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpQueue;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> yGm;

    uint32_t N, C, D, H, W;
    uint32_t K, S, P;
    uint32_t OD, OH, OW;
    uint32_t TotalOut;
    uint32_t WinElems, WinInnerAligned;

    uint32_t blockOut;
    uint32_t blockStart;
};

extern "C" __global__ __aicore__ void average_pooling_3d_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelAveragePooling3d op;
    op.Init(x, y,
            tiling_data.N, tiling_data.C, tiling_data.D, tiling_data.H, tiling_data.W,
            tiling_data.K, tiling_data.S, tiling_data.P,
            tiling_data.OD, tiling_data.OH, tiling_data.OW,
            tiling_data.TotalOut,
            tiling_data.WinElems, tiling_data.WinInnerAligned,
            tiling_data.SumTmpBufSize);
    op.Process();
}
