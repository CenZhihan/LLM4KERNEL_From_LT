#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose3d_log_sum_exp_hard_swish_subtract_clamp_max_custom_tiling_data.h"

using DTYPE = float;

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose3dLogSumExpHardSwishSubtractClampMax {
public:
    __aicore__ inline KernelConvTranspose3dLogSumExpHardSwishSubtractClampMax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y, uint32_t totalSpatial, uint32_t channels,
                                uint32_t tileNum, uint32_t tmpBufferSize)
    {
        this->totalSpatial = totalSpatial;
        this->channels = channels;
        this->tileNum = tileNum;
        this->blockSpatial = totalSpatial / AscendC::GetBlockNum();
        this->tileLen = this->blockSpatial / tileNum / BUFFER_NUM;
        this->blockStart = this->blockSpatial * AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(x), totalSpatial * channels);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(y) + this->blockStart, this->blockSpatial);
        biasGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(bias), 1);

        // 工作队列：用于运行时局部缓冲(runMax, sumExp, 中间tmp, 输入切片x等)
        pipe.InitBuffer(calcQue, 6, this->tileLen * sizeof(DTYPE));
        // 输出队列
        pipe.InitBuffer(outQue, BUFFER_NUM, this->tileLen * sizeof(DTYPE));
        // Clamp 临时缓冲（参考文档：sharedTmpBuffer）
        pipe.InitBuffer(tmpQue, 1, tmpBufferSize);
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            ComputeTile(i);
        }
    }

private:
    __aicore__ inline void ComputeTile(int32_t progress)
    {
        uint32_t spatialOffset = this->blockStart + progress * this->tileLen;

        // 分配运行中间缓冲
        AscendC::LocalTensor<DTYPE> runMax = calcQue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> sumExp = calcQue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> tmpA   = calcQue.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> tmpB   = calcQue.AllocTensor<DTYPE>();

        // 初始化 runMax = -1e30f, sumExp = 0
        AscendC::Duplicate(runMax, static_cast<DTYPE>(-1.0e30f), this->tileLen);
        AscendC::Duplicate(sumExp, static_cast<DTYPE>(0.0f), this->tileLen);

        // 逐通道累加 logsumexp: m = max_i x_i; lse = m + log(sum(exp(x_i - m)))
        for (uint32_t c = 0; c < channels; ++c) {
            AscendC::LocalTensor<DTYPE> xLocal = calcQue.AllocTensor<DTYPE>();
            AscendC::DataCopy(xLocal, xGm[c * this->totalSpatial + spatialOffset], this->tileLen);

            // tmpA = newMax = max(runMax, xLocal)
            AscendC::Max(tmpA, runMax, xLocal, this->tileLen);

            // tmpB = exp(runMax - newMax)
            AscendC::Sub(tmpB, runMax, tmpA, this->tileLen);
            AscendC::Exp(tmpB, tmpB, this->tileLen);

            // sumExp = sumExp * tmpB
            AscendC::Mul(sumExp, sumExp, tmpB, this->tileLen);

            // tmpB = exp(xLocal - newMax)
            AscendC::Sub(tmpB, xLocal, tmpA, this->tileLen);
            AscendC::Exp(tmpB, tmpB, this->tileLen);

            // sumExp = sumExp + tmpB
            AscendC::Add(sumExp, sumExp, tmpB, this->tileLen);

            // runMax = newMax (tmpA)
            AscendC::Copy(runMax, tmpA, this->tileLen);

            calcQue.FreeTensor(xLocal);
        }

        // lse = log(sumExp) + runMax
        AscendC::LocalTensor<DTYPE> lse = calcQue.AllocTensor<DTYPE>();
        AscendC::Log(sumExp, sumExp, this->tileLen);
        AscendC::Add(lse, sumExp, runMax, this->tileLen);

        // HardSwish: y = lse * sigmoid(lse + 3) / 6
        AscendC::LocalTensor<DTYPE> tmpC = calcQue.AllocTensor<DTYPE>(); // lse + 3
        AscendC::Adds(tmpC, lse, static_cast<DTYPE>(3.0f), this->tileLen);
        AscendC::LocalTensor<DTYPE> sig = calcQue.AllocTensor<DTYPE>();
        AscendC::Sigmoid(sig, tmpC, this->tileLen);
        AscendC::LocalTensor<DTYPE> hs = outQue.AllocTensor<DTYPE>(); // 复用为输出中间
        AscendC::Mul(hs, lse, sig, this->tileLen);
        AscendC::Muls(hs, hs, static_cast<DTYPE>(1.0f/6.0f), this->tileLen);

        // 读取 bias 标量，并执行 y = hs - bias
        AscendC::LocalTensor<DTYPE> biasLocal = calcQue.AllocTensor<DTYPE>();
        AscendC::DataCopy(biasLocal, biasGm, 1); // 读入 1 个元素
        // 使用 Duplicate 将标量扩展为向量（拷贝变量/立即数，参考 Duplicate 功能）
        AscendC::LocalTensor<DTYPE> biasVec = calcQue.AllocTensor<DTYPE>();
        // 这里假定 Duplicate 支持以变量值进行填充
        AscendC::Duplicate(biasVec, biasLocal.GetValue(0), this->tileLen);
        AscendC::Sub(hs, hs, biasVec, this->tileLen);

        // Clamp 到 [-1, 1]（使用共享临时缓冲，参考 5.1.22.2/5.1.22.3）
        AscendC::LocalTensor<uint8_t> sharedTmpBuffer = tmpQue.AllocTensor<uint8_t>();
        AscendC::LocalTensor<DTYPE> clamped = calcQue.AllocTensor<DTYPE>();
        AscendC::ClampMin<DTYPE>(clamped, hs, sharedTmpBuffer, static_cast<DTYPE>(-1.0f), this->tileLen);
        AscendC::ClampMax<DTYPE>(hs, clamped, sharedTmpBuffer, static_cast<DTYPE>(1.0f), this->tileLen);
        tmpQue.FreeTensor(sharedTmpBuffer);

        // 写回
        AscendC::DataCopy(yGm[spatialOffset - this->blockStart], hs, this->tileLen);

        // 释放局部张量
        outQue.FreeTensor(hs);
        calcQue.FreeTensor(runMax);
        calcQue.FreeTensor(sumExp);
        calcQue.FreeTensor(tmpA);
        calcQue.FreeTensor(tmpB);
        calcQue.FreeTensor(tmpC);
        calcQue.FreeTensor(sig);
        calcQue.FreeTensor(lse);
        calcQue.FreeTensor(biasLocal);
        calcQue.FreeTensor(biasVec);
    }

private:
    AscendC::TPipe pipe;

    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> yGm;
    AscendC::GlobalTensor<DTYPE> biasGm;

    AscendC::TQue<AscendC::TPosition::VECCALC, 6> calcQue;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpQue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQue;

    uint32_t totalSpatial;
    uint32_t channels;
    uint32_t tileNum;
    uint32_t blockSpatial;
    uint32_t tileLen;
    uint32_t blockStart;
};

extern "C" __global__ __aicore__ void conv_transpose3d_log_sum_exp_hard_swish_subtract_clamp_max_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{

    op.Init(x, bias, y, tiling_data.totalSpatial, tiling_data.channels, tiling_data.tileNum, tiling_data.tmpBufferSize);
    op.Process();
}
