    tiling.set_i(i);

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"
#define __NPU_TILING__
#include "gemm_relu_divide_custom_tiling_data.h"

// Fused kernel: GEMM (x * weight^T + bias) -> ReLU -> divide by scalar
// Data type: float

using AType = matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::BSNGD>;
using BType = matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, true,  LayoutMode::BSNGD>;
using CType = matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float, false, LayoutMode::BSNGS1S2>;
using BiasType = matmul::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float>;

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmReluDivide {
public:
    __aicore__ inline KernelGemmReluDivide() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR divisor, GM_ADDR y,
                                uint32_t B, uint32_t I, uint32_t O)
    {
        b = B; i = I; o = O;

        xGm.SetGlobalBuffer((__gm__ float*)x, b * i);
        wGm.SetGlobalBuffer((__gm__ float*)weight, o * i);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, o);
        divGm.SetGlobalBuffer((__gm__ float*)divisor, 1);
        yGm.SetGlobalBuffer((__gm__ float*)y, b * o);

        // Vector IO queues for post-ops
        tileLen = ((b * o + AscendC::ONE_BLK_SIZE/sizeof(float) - 1) / (AscendC::ONE_BLK_SIZE/sizeof(float))) * (AscendC::ONE_BLK_SIZE/sizeof(float));
        pipe.InitBuffer(inQueueC, BUFFER_NUM, tileLen * sizeof(float));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, tileLen * sizeof(float));
    }

    __aicore__ inline void ComputeGemm()
    {
        // Initialize TCubeTiling from tiling GM (if provided by framework) is skipped in this demo.
        // Use Matmul object and basic configuration pattern from CANN AscendC API Reference.
        matmul::Matmul<AType, BType, CType, BiasType> mm;
        AscendC::TPipe cubePipe;
        g_cubeTPipePtr = &cubePipe;
        REGIST_MATMUL_OBJ(&cubePipe, GetSysWorkSpacePtr(), mm);

        // Minimal tiling initialization (assumed provided by framework in real use)
        TCubeTiling t = {0};
        mm.Init(&t);

        // Bind tensors
        AscendC::GlobalTensor<float> gmA, gmB, gmBias, gmC;
        gmA.SetGlobalBuffer(xGm.GetPhyAddr(), i);         // shape [B, I]
        gmB.SetGlobalBuffer(wGm.GetPhyAddr(), i);         // shape [O, I]
        gmBias.SetGlobalBuffer(biasGm.GetPhyAddr(), o);
        gmC.SetGlobalBuffer(yGm.GetPhyAddr(), o);

        mm.SetTensorA(gmA[0], false);
        mm.SetTensorB(gmB[0], true);
        mm.SetWorkspace(nullptr, 0);
        mm.SetBias(gmBias[0]);

        // Single call path for simplicity
        mm.IterateNBatch(1, 1, 1, false);
    }

    __aicore__ inline void PostReluDivide()
    {
        // Load scalar divisor and compute its reciprocal
        AscendC::LocalTensor<float> divLocal = inQueueC.AllocTensor<float>();
        AscendC::DataCopy(divLocal, divGm, 1);
        inQueueC.EnQue(divLocal);
        AscendC::LocalTensor<float> divValLocal = inQueueC.DeQue<float>();
        float divisor_val = divValLocal.GetValue(0); // assume API to read first value
        inQueueC.FreeTensor(divValLocal);
        float inv_div = 1.0f / divisor_val;

        // Process y in tiles: ReLU then scaling by inv_div
        uint32_t total = b * o;
        uint32_t processed = 0;
        while (processed < total) {
            uint32_t cur = AscendC::Min(tileLen, total - processed);
            AscendC::LocalTensor<float> cLocal = inQueueC.AllocTensor<float>();
            AscendC::DataCopy(cLocal, yGm[processed], cur);
            inQueueC.EnQue(cLocal);

            AscendC::LocalTensor<float> inLocal = inQueueC.DeQue<float>();
            AscendC::LocalTensor<float> outLocal = outQueueC.AllocTensor<float>();

            // ReLU
            AscendC::Relu(outLocal, inLocal, cur);
            // Divide by scalar (scale with reciprocal)
            AscendC::Muls(outLocal, outLocal, inv_div, cur);

            outQueueC.EnQue(outLocal);
            inQueueC.FreeTensor(inLocal);

            AscendC::LocalTensor<float> outLocal2 = outQueueC.DeQue<float>();
            AscendC::DataCopy(yGm[processed], outLocal2, cur);
            outQueueC.FreeTensor(outLocal2);

            processed += cur;
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueC;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueC;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> divGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t b = 0, i = 0, o = 0;
    uint32_t tileLen = 0;
};

extern "C" __global__ __aicore__ void gemm_relu_divide_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
                                                               GM_ADDR divisor, GM_ADDR y,
                                                               GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmReluDivide op;
    op.Init(x, weight, bias, divisor, y, tiling_data.b, tiling_data.i, tiling_data.o);
    op.ComputeGemm();
    op.PostReluDivide();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor gemm_relu_divide_custom_impl_npu(const at::Tensor& x,
                                            const at::Tensor& weight,
                                            const at::Tensor& bias,
                                            const at::Tensor& divisor) {
    TORCH_CHECK(x.scalar_type() == at::kFloat, "Only float32 is supported");
    TORCH_CHECK(weight.scalar_type() == at::kFloat && bias.scalar_type() == at::kFloat, "Only float32 is supported");
    TORCH_CHECK(divisor.numel() == 1, "divisor must be a scalar tensor");

    // Output shape: [B, O]
    auto B = x.size(0);
    auto O = weight.size(0);
    at::Tensor result = at::empty({B, O}, x.options());
    EXEC_NPU_CMD(aclnnGemmReluDivideCustom, x, weight, bias, divisor, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("gemm_relu_divide_custom", &gemm_relu_divide_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("gemm_relu_divide_custom", &gemm_relu_divide_custom_impl_npu, "GEMM + ReLU + Divide");

    def __init__(self, in_features, out_features, divisor) -> None:

        self.divisor = float(divisor)
