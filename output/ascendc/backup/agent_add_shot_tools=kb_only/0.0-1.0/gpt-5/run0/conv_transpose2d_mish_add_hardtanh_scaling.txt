kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transpose2d_mish_add_hardtanh_scaling_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose2dMishAddHardtanhScaling {
public:
    __aicore__ inline KernelConvTranspose2dMishAddHardtanhScaling() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t totalLength, uint32_t tileNum,
                                float addValue, float minVal, float maxVal, float scale)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        this->addValue = addValue;
        this->minVal = minVal;
        this->maxVal = maxVal;
        this->scale = scale;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        // 1) Mish activation
        // 2) Add constant
        // 3) Hardtanh [-1, 1]
        // 4) Scale by constant
        // Following high-level APIs follow the same usage style as other elementwise ops (e.g., Add/Xor):
        //   Mish(dst, src, count)
        //   Adds(dst, src, scalar, count)
        //   Hardtanh(dst, src, minVal, maxVal, count)
        //   Muls(dst, src, scalar, count)
        AscendC::Mish(zLocal, xLocal, this->tileLength);
        AscendC::Adds(zLocal, zLocal, this->addValue, this->tileLength);
        AscendC::Hardtanh(zLocal, zLocal, this->minVal, this->maxVal, this->tileLength);
        AscendC::Muls(zLocal, zLocal, this->scale, this->tileLength);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;

    float addValue;
    float minVal;
    float maxVal;
    float scale;
};

extern "C" __global__ __aicore__ void conv_transpose2d_mish_add_hardtanh_scaling_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    if (TILING_KEY_IS(1)) {
        KernelConvTranspose2dMishAddHardtanhScaling op;
        op.Init(x, z,
                tilingData.totalLength, tilingData.tileNum,
                tilingData.addValue, tilingData.minVal, tilingData.maxVal, tilingData.scale);
        op.Process();
    }
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model:
    - Keep ConvTranspose2d on framework side.
    - Fuse Mish + Add(const) + Hardtanh[-1,1] + Scale(const) into a single custom AscendC operator.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, add_value, scale):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride, padding, output_padding)
        self.add_value = float(add_value)
        self.scale = float(scale)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = torch.nn.functional.mish(x)  # Keep on framework side, custom kernel fuses the rest.
        x = custom_ops_lib.conv_transpose2d_mish_add_hardtanh_scaling_custom(x, self.add_value, self.scale)
