kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_standard_2d_square_input_square_kernel_custom_tiling_data.h"

using f32 = float;

class KernelConv2dNaive {
public:
    __aicore__ inline KernelConv2dNaive() {}

    __aicore__ inline void Init(
        GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
        uint32_t N, uint32_t Cin, uint32_t H, uint32_t W,
        uint32_t Cout, uint32_t K, uint32_t stride, uint32_t pad,
        uint32_t outH, uint32_t outW, uint64_t totalOut)
    {
        // Store gm pointers
        xPtr = reinterpret_cast<__gm__ f32*>(x);
        wPtr = reinterpret_cast<__gm__ f32*>(w);
        bPtr = reinterpret_cast<__gm__ f32*>(b);
        yPtr = reinterpret_cast<__gm__ f32*>(y);

        this->N = N; this->Cin = Cin; this->H = H; this->W = W;
        this->Cout = Cout; this->K = K; this->stride = stride; this->pad = pad;
        this->outH = outH; this->outW = outW; this->totalOut = totalOut;

        // Per-block work distribution
        uint32_t blkNum = AscendC::GetBlockNum();
        uint32_t blkIdx = AscendC::GetBlockIdx();

        uint64_t basePerBlock = totalOut / blkNum;
        uint64_t remain = totalOut % blkNum;

        // Round-robin the tail
        blockStart = basePerBlock * blkIdx + (blkIdx < remain ? blkIdx : remain);
        blockLen = basePerBlock + (blkIdx < remain ? 1 : 0);
    }

    __aicore__ inline void Process()
    {
        // Naive NCHW convolution, single-core per-block loop
        for (uint64_t p = 0; p < blockLen; ++p) {
            uint64_t gIdx = blockStart + p;

            // Unravel gIdx -> (n, oc, oh, ow)
            uint64_t spanOW = outW;
            uint64_t spanOHOW = (uint64_t)outH * outW;
            uint64_t spanOC = (uint64_t)Cout * spanOHOW;

            uint32_t n  = (uint32_t)(gIdx / spanOC);
            uint64_t r1 = gIdx % spanOC;
            uint32_t oc = (uint32_t)(r1 / spanOHOW);
            uint64_t r2 = r1 % spanOHOW;
            uint32_t oh = (uint32_t)(r2 / spanOW);
            uint32_t ow = (uint32_t)(r2 % spanOW);

            int32_t ih0 = (int32_t)oh * (int32_t)stride - (int32_t)pad;
            int32_t iw0 = (int32_t)ow * (int32_t)stride - (int32_t)pad;

            f32 acc = 0.0f;

            for (uint32_t ci = 0; ci < Cin; ++ci) {
                for (uint32_t kh = 0; kh < K; ++kh) {
                    int32_t ih = ih0 + (int32_t)kh;
                    if (ih < 0 || ih >= (int32_t)H) continue;

                    for (uint32_t kw = 0; kw < K; ++kw) {
                        int32_t iw = iw0 + (int32_t)kw;
                        if (iw < 0 || iw >= (int32_t)W) continue;

                        // x index: ((n*Cin + ci)*H + ih)*W + iw
                        uint64_t xIndex = (((uint64_t)n * Cin + ci) * H + (uint32_t)ih) * W + (uint32_t)iw;

                        // w index: ((oc*Cin + ci)*K + kh)*K + kw
                        uint64_t wIndex = (((uint64_t)oc * Cin + ci) * K + kh) * K + kw;

                        acc += xPtr[xIndex] * wPtr[wIndex];
                    }
                }
            }

            acc += bPtr[oc];

            // y index: ((n*Cout + oc)*outH + oh)*outW + ow
            uint64_t yIndex = (((uint64_t)n * Cout + oc) * outH + oh) * outW + ow;
            yPtr[yIndex] = acc;
        }
    }

private:
    __gm__ f32* xPtr;
    __gm__ f32* wPtr;
    __gm__ f32* bPtr;
    __gm__ f32* yPtr;

    uint32_t N, Cin, H, W, Cout, K, stride, pad, outH, outW;
    uint64_t totalOut;
    uint64_t blockStart;
    uint64_t blockLen;
};

extern "C" __global__ __aicore__ void conv_standard_2d_square_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dNaive op;
    op.Init(
        x, w, b, y,
        tiling_data.N, tiling_data.Cin, tiling_data.H, tiling_data.W,
        tiling_data.Cout, tiling_data.K, tiling_data.stride, tiling_data.pad,
        tiling_data.outH, tiling_data.outW, tiling_data.totalOut);
    op.Process();
}
