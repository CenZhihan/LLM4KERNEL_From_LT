    uint32_t h_y  = y_shape->GetStorageShape().GetDim(2);
    uint32_t w_y  = y_shape->GetStorageShape().GetDim(3);

    tiling.set_c0(c0);
    tiling.set_c1(c1);

    tiling.set_w(w);
    tiling.set_tileNum(TILE_NUM);

    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());
    context->SetTilingKey(1);

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    const gert::Shape* y_shape = context->GetInputShape(1);
    gert::Shape* z_shape = context->GetOutputShape(0);

    if (x_shape->GetDimNum() != 4 || y_shape->GetDimNum() != 4) {
        return GRAPH_FAILED;
    }

    int64_t n = x_shape->GetDim(0);
    int64_t c0 = x_shape->GetDim(1);
    int64_t h = x_shape->GetDim(2);
    int64_t w = x_shape->GetDim(3);

    int64_t n_y = y_shape->GetDim(0);
    int64_t c1 = y_shape->GetDim(1);
    int64_t h_y = y_shape->GetDim(2);
    int64_t w_y = y_shape->GetDim(3);

    if (!(n == n_y && h == h_y && w == w_y)) {
        return GRAPH_FAILED;
    }

    gert::Shape out({n, c0 + c1, h, w});
    *z_shape = out;
    return GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "densenet121_dense_block_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelDenseConcat {
public:
    __aicore__ inline KernelDenseConcat() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z, uint32_t n, uint32_t c0, uint32_t c1, uint32_t h, uint32_t w, uint32_t tileNum)
    {
        this->n = n;
        this->c0 = c0;
        this->c1 = c1;
        this->h = h;
        this->w = w;
        this->tileNum = tileNum;

        this->spatial = h * w;
        this->sampleSizeX = c0 * spatial;
        this->sampleSizeY = c1 * spatial;
        this->sampleSizeZ = (c0 + c1) * spatial;

        xGm.SetGlobalBuffer((__gm__ float *)x, n * sampleSizeX);
        yGm.SetGlobalBuffer((__gm__ float *)y, n * sampleSizeY);
        zGm.SetGlobalBuffer((__gm__ float *)z, n * sampleSizeZ);

        // 依据样本大小简单设定tile长度，避免超出UB
        uint32_t totalLen = n * sampleSizeZ;
        this->tileLength = (totalLen / tileNum / BUFFER_NUM);
        if (this->tileLength == 0) {
            this->tileLength = 1024; // 保底
        }
        // 限制tileLength不超过一个样本大小，便于分块拷贝
        if (this->tileLength > sampleSizeZ) {
            this->tileLength = sampleSizeZ;
        }

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t bn = 0; bn < n; ++bn) {
            CopySample(bn);
        }
    }

private:
    __aicore__ inline void CopyChunk(const AscendC::GlobalTensor<float> &src, uint32_t srcBase, AscendC::GlobalTensor<float> &dst, uint32_t dstBase, uint32_t length)
    {
        for (uint32_t offset = 0; offset < length; offset += this->tileLength) {
            uint32_t cur = (offset + this->tileLength <= length) ? this->tileLength : (length - offset);
            AscendC::LocalTensor<float> buf = inQueue.AllocTensor<float>();
            AscendC::DataCopy(buf, src[srcBase + offset], cur);
            AscendC::DataCopy(dst[dstBase + offset], buf, cur);
            inQueue.FreeTensor(buf);
        }
    }

    __aicore__ inline void CopySample(uint32_t bn)
    {
        uint32_t xBase = bn * sampleSizeX;
        uint32_t yBase = bn * sampleSizeY;
        uint32_t zBase = bn * sampleSizeZ;

        // 先拷贝x到z的前半部分
        CopyChunk(xGm, xBase, zGm, zBase, sampleSizeX);
        // 再拷贝y到z的后半部分
        CopyChunk(yGm, yBase, zGm, zBase + sampleSizeX, sampleSizeY);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t n;
    uint32_t c0;
    uint32_t c1;
    uint32_t h;
    uint32_t w;

    uint32_t spatial;
    uint32_t sampleSizeX;
    uint32_t sampleSizeY;
    uint32_t sampleSizeZ;

    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void densenet121_dense_block_custom(GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelDenseConcat op;
    op.Init(x, y, z, tiling_data.n, tiling_data.c0, tiling_data.c1, tiling_data.h, tiling_data.w, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, num_layers: int, num_input_features: int, growth_rate: int):
        super().__init__()
        layers = []
        for i in range(num_layers):
            layers.append(self._make_layer(num_input_features + i * growth_rate, growth_rate))
        self.layers = torch.nn.ModuleList(layers)

    def _make_layer(self, in_features: int, growth_rate: int):
        return torch.nn.Sequential(
            torch.nn.BatchNorm2d(in_features),
            torch.nn.ReLU(inplace=True),
            torch.nn.Conv2d(in_features, growth_rate, kernel_size=3, padding=1, bias=False),
            torch.nn.Dropout(0.0)
        )

    def forward(self, x):
        for layer in self.layers:
            new_feature = layer(x)
            x = custom_ops_lib.densenet121_dense_block_custom(x, new_feature)
        return x
'''
