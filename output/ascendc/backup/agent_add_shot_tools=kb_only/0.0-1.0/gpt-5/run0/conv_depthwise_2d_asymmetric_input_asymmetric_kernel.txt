static ge::graphStatus InferShape(gert::InferShapeContext* context)

    int64_t dilW    = *(attrs->GetAttrPointer<uint32_t>(5));
    (void)attrs; // avoid unused in some builds

    auto CalcOut = [](int64_t in, int64_t pad, int64_t dil, int64_t k, int64_t s) -> int64_t {
        return (in + 2 * pad - dil * (k - 1) - 1) / s + 1;

    int64_t Ho = CalcOut(H, padH, dilH, KH, strideH);
    int64_t Wo = CalcOut(W, padW, dilW, KW, strideW);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_depthwise_2d_asymmetric_input_asymmetric_kernel_custom_tiling_data.h"

class KernelDWConv2D {
public:
    __aicore__ inline KernelDWConv2D() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w_in,
                                uint32_t ho, uint32_t wo,
                                uint32_t kh, uint32_t kw,
                                uint32_t strideH, uint32_t strideW,
                                uint32_t padH, uint32_t padW,
                                uint32_t dilH, uint32_t dilW,
                                uint32_t hasBias) {
        this->N = n; this->C = c; this->H = h; this->W = w_in;
        this->Ho = ho; this->Wo = wo;
        this->KH = kh; this->KW = kw;
        this->strideH = strideH; this->strideW = strideW;
        this->padH = padH; this->padW = padW;
        this->dilH = dilH; this->dilW = dilW;
        this->hasBias = hasBias;

        uint64_t outSize = (uint64_t)N * C * Ho * Wo;
        uint64_t inSize  = (uint64_t)N * C * H * W;
        uint64_t wSize   = (uint64_t)C * KH * KW;  // depthwise: [C, 1, KH, KW]
        xGm.SetGlobalBuffer((__gm__ float*)x, inSize);
        wGm.SetGlobalBuffer((__gm__ float*)w, wSize);
        bGm.SetGlobalBuffer((__gm__ float*)b, C);
        yGm.SetGlobalBuffer((__gm__ float*)y, outSize);

        totalOut = outSize;
        blockNum = AscendC::GetBlockNum();
        blockIdx = AscendC::GetBlockIdx();
        // split by blocks with ceil-div
        strideBlock = (totalOut + blockNum - 1) / blockNum;
        start = strideBlock * blockIdx;
        end = start + strideBlock;
        if (end > totalOut) end = totalOut;
    }

    __aicore__ inline void Process() {
        Compute();
    }

private:
    __aicore__ inline void Compute() {
        for (uint64_t outIndex = start; outIndex < end; ++outIndex) {
            // map flat index -> (n,c,ho,wo)
            uint64_t t = outIndex;
            uint32_t wo_idx = (uint32_t)(t % Wo); t /= Wo;
            uint32_t ho_idx = (uint32_t)(t % Ho); t /= Ho;
            uint32_t c_idx  = (uint32_t)(t % C);  t /= C;
            uint32_t n_idx  = (uint32_t)(t);

            int32_t in_h_base = (int32_t)ho_idx * (int32_t)strideH - (int32_t)padH;
            int32_t in_w_base = (int32_t)wo_idx * (int32_t)strideW - (int32_t)padW;

            float acc = 0.f;
            for (uint32_t r = 0; r < KH; ++r) {
                int32_t ih = in_h_base + (int32_t)r * (int32_t)dilH;
                if ((uint32_t)ih >= H) continue;
                for (uint32_t s = 0; s < KW; ++s) {
                    int32_t iw = in_w_base + (int32_t)s * (int32_t)dilW;
                    if ((uint32_t)iw >= W) continue;
                    uint64_t xIdx = (((uint64_t)n_idx * C + c_idx) * H + (uint32_t)ih) * W + (uint32_t)iw;
                    uint64_t wIdx = ((uint64_t)c_idx * KH + r) * KW + s; // [C,KH,KW]
                    float xv = xGm.GetValue(xIdx);
                    float wv = wGm.GetValue(wIdx);
                    acc += xv * wv;
                }
            }
            if (hasBias) {
                acc += bGm.GetValue((uint64_t)c_idx);
            }
            uint64_t yIdx = (((uint64_t)n_idx * C + c_idx) * Ho + ho_idx) * Wo + wo_idx;
            yGm.SetValue(yIdx, acc);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t N, C, H, W, Ho, Wo, KH, KW;
    uint32_t strideH, strideW, padH, padW, dilH, dilW;
    uint32_t hasBias;

    uint64_t totalOut;
    uint32_t blockNum;
    uint32_t blockIdx;
    uint64_t strideBlock;
    uint64_t start, end;
};

extern "C" __global__ __aicore__ void conv_depthwise_2d_asymmetric_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelDWConv2D op;
    op.Init(x, w, b, y,
            tilingData.n(), tilingData.c(), tilingData.h(), tilingData.w(),
            tilingData.ho(), tilingData.wo(),
            tilingData.kh(), tilingData.kw(),
            tilingData.strideH(), tilingData.strideW(),
            tilingData.padH(), tilingData.padW(),
            tilingData.dilH(), tilingData.dilW(),
            tilingData.hasBias());
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

                                  (kernel_size_h, kernel_size_w),
                                  stride=(stride_h, stride_w),
                                  padding=(padding_h, padding_w),
                                  dilation=(dilation_h, dilation_w),
                                  groups=in_channels,
                                  bias=bias)
        self.stride_h = stride_h
        self.stride_w = stride_w
        self.padding_h = padding_h
        self.padding_w = padding_w
        self.dilation_h = dilation_h
        self.dilation_w = dilation_w
        self.groups = groups

        bias_tensor = self.dw.bias if self.dw.bias is not None else None
        return custom_ops_lib.conv_depthwise_2d_asymmetric_input_asymmetric_kernel_custom(
            x, self.dw.weight, bias_tensor,
