namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    ScatterCustomTilingData tiling;

    const gert::Shape* xShape = context->GetInputShape(0);
    const gert::Shape* idxShape = context->GetInputShape(1);

    // 假设输入为2D: [rows, cols] 与 [rows, updatesCols]
    uint32_t rows = static_cast<uint32_t>(xShape->GetOriginShape().GetDim(0));
    uint32_t cols = static_cast<uint32_t>(xShape->GetOriginShape().GetDim(1));
    uint32_t updatesCols = static_cast<uint32_t>(idxShape->GetOriginShape().GetDim(1));

    // 以行并行：每个block处理一行
    uint32_t blockDim = rows;
    context->SetBlockDim(blockDim);

    tiling.set_rows(rows);
    tiling.set_cols(cols);
    tiling.set_updatesCols(updatesCols);
    tiling.set_blockDim(blockDim);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());
    context->SetTilingKey(1);

    size_t* currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "scatter_custom_tiling_data.h"

// 简化的行并行scatter：每个Block处理一整行
extern "C" __global__ __aicore__ void scatter_custom(GM_ADDR x, GM_ADDR idx, GM_ADDR updates, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    const uint32_t rows = tilingData.rows;
    const uint32_t cols = tilingData.cols;
    const uint32_t updatesCols = tilingData.updatesCols;

    const uint32_t rowId = AscendC::GetBlockIdx();
    if (rowId >= rows) {
        return;
    }

    // 全局内存指针
    __gm__ float* xGm = reinterpret_cast<__gm__ float*>(x);
    __gm__ int64_t* idxGm = reinterpret_cast<__gm__ int64_t*>(idx);
    __gm__ float* updGm = reinterpret_cast<__gm__ float*>(updates);
    __gm__ float* zGm = reinterpret_cast<__gm__ float*>(z);

    // 计算每个row的偏移
    uint64_t baseX = static_cast<uint64_t>(rowId) * cols;
    uint64_t baseZ = static_cast<uint64_t>(rowId) * cols;
    uint64_t baseI = static_cast<uint64_t>(rowId) * updatesCols;
    uint64_t baseU = static_cast<uint64_t>(rowId) * updatesCols;

    // 先复制整行 x -> z
    for (uint32_t c = 0; c < cols; ++c) {
        zGm[baseZ + c] = xGm[baseX + c];
    }

    // 按index更新对应位置
    if (TILING_KEY_IS(1)) {
        for (uint32_t j = 0; j < updatesCols; ++j) {
            int64_t col = idxGm[baseI + j];
            if (col >= 0 && col < static_cast<int64_t>(cols)) {
                zGm[baseZ + static_cast<uint64_t>(col)] = updGm[baseU + j];
            }
        }
    }
}
