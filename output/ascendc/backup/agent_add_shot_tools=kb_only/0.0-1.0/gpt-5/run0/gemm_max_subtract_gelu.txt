static ge::graphStatus TilingFunc(gert::TilingContext* context)

    context->SetBlockDim(BLOCK_DIM);

    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    matmul_tiling::MatmulApiTiling mm(ascendcPlatform);

    mm.SetBiasType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND, matmul_tiling::DataType::DT_FLOAT);
    mm.SetShape(B, N, K);
    mm.SetOrgShape(B, N, K);
    mm.SetBias(true);

    mm.SetBufferSpace(-1, -1, -1);

    tiling.set_inFeat(K);
    tiling.set_outFeat(N);
    tiling.set_maxDim(1);      // this model uses dim=1
    tiling.set_hasBias(true);
    tiling.set_mmTiling(cubeTiling);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());

    const gert::Shape* w = context->GetInputShape(1);

        return GRAPH_FAILED;

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"
#define __NPU_TILING__
#include "gemm_max_subtract_gelu_custom_tiling_data.h"

using A_T = float;
using B_T = float;
using C_T = float;
using BiasT = float;

extern "C" __global__ __aicore__ void gemm_max_subtract_gelu_custom(GM_ADDR xGM, GM_ADDR wGM, GM_ADDR bGM,
                                                                    GM_ADDR yGM, GM_ADDR workspaceGM, GM_ADDR tilingGM)
{
    // Parse tiling
    GET_TILING_DATA(tiling, tilingGM);
    const uint32_t B = tiling.batch();
    const uint32_t K = tiling.inFeat();
    const uint32_t N = tiling.outFeat();

    // Prepare GM tensors
    AscendC::GlobalTensor<A_T> aGlobal;
    AscendC::GlobalTensor<B_T> wGlobal;
    AscendC::GlobalTensor<C_T> cGlobal;   // GEMM result in workspace
    AscendC::GlobalTensor<BiasT> biasGlobal;
    AscendC::GlobalTensor<C_T> yGlobal;

    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ A_T*>(xGM), B * K);
    wGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ B_T*>(wGM), N * K);
    // workspace for C: [B, N]
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(workspaceGM), B * N);
    biasGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ BiasT*>(bGM), N);
    // output: [B, 1]
    yGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ C_T*>(yGM), B * 1);

    // Create Matmul instance: NORMAL layout
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, A_T, false, LayoutMode::NORMAL> aType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, B_T, true,  LayoutMode::NORMAL> bType; // weight [N,K], transpose -> [K,N]
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, C_T, false, LayoutMode::NORMAL> cType;
    typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, BiasT> biasType;
    constexpr static MatmulConfig MM_CFG = GetNormalConfig(false, false, false, BatchMode::BATCH_LESS_THAN_L1);

    TPipe pipe;
    matmul::Matmul<aType, bType, cType, biasType, MM_CFG> mm;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm);
    TCubeTiling mmTiling = tiling.mmTiling();
    mm.Init(&mmTiling);
    mm.SetTensorA(aGlobal, false);
    mm.SetTensorB(wGlobal, true);
    mm.SetWorkspace(workspaceGM, 0);
    if (tiling.hasBias()) {
        mm.SetBias(biasGlobal);
    }
    // Compute GEMM to workspace cGlobal
    mm.Run(cGlobal);

    // After GEMM: x = max(x, dim=1, keepdim=True)
    // Then subtract mean over dim=1 (keepdim=True), then GELU
    // We produce final yGlobal [B, 1]

    // Queues for row reduce and scalar ops
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQ;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQ;
    pipe.InitBuffer(inQ, 1, N * sizeof(C_T));
    pipe.InitBuffer(outQ, 1, 1 * sizeof(C_T));

    for (uint32_t b = 0; b < B; ++b) {
        // Copy one row [N]
        AscendC::LocalTensor<C_T> rowLocal = inQ.AllocTensor<C_T>();
        AscendC::DataCopy(rowLocal, cGlobal[b * N], N);
        inQ.EnQue(rowLocal);

        // DeQue and reduce max to [1]
        rowLocal = inQ.DeQue<C_T>();
        AscendC::LocalTensor<C_T> maxLocal = outQ.AllocTensor<C_T>();
        // ReduceMax: compute maximum value across last axis
        AscendC::ReduceMax(maxLocal, rowLocal, N);

        // mean over dim=1 (for [1] vector, mean == itself)
        AscendC::LocalTensor<C_T> meanLocal = outQ.AllocTensor<C_T>();
        AscendC::Mean(meanLocal, maxLocal, 1);

        // subtract: max - mean (will be 0 for [1] input)
        AscendC::LocalTensor<C_T> subLocal = outQ.AllocTensor<C_T>();
        AscendC::Sub(subLocal, maxLocal, meanLocal, 1);

        // GELU
        AscendC::LocalTensor<C_T> geluLocal = outQ.AllocTensor<C_T>();
        AscendC::Gelu(geluLocal, subLocal, 1);

        // write out [1]
        AscendC::DataCopy(yGlobal[b * 1], geluLocal, 1);

        // free locals
        outQ.FreeTensor(geluLocal);
        outQ.FreeTensor(subLocal);
        outQ.FreeTensor(meanLocal);
        outQ.FreeTensor(maxLocal);
        inQ.FreeTensor(rowLocal);
    }
}
