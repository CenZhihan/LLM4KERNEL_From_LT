kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_transposed_2d_asymmetric_input_square_kernel_custom_tiling_data.h"

// 仅支持: stride=1, padding=0, output_padding=0, groups=1, 无bias
// 布局: x[N, Cin, Hin, Win], w[Cin, Cout, K, K], y[N, Cout, Hout, Wout]
// 计算: Hout=Hin+K-1, Wout=Win+K-1
// 算法: 直接定义式逐元素累加（示范用），可按需用 Matmul/IM2COL/CUBE 优化

using DTYPE_X = float;
using DTYPE_W = float;
using DTYPE_Y = float;

constexpr int32_t BUFFER_NUM = 2;

class KernelDeconv2D {
public:
    __aicore__ inline KernelDeconv2D() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y, const ConvTransposed2dAsymmetricInputSquareKernelCustomTilingData &t)
    {
        this->N    = t.N();
        this->Cin  = t.Cin();
        this->Cout = t.Cout();
        this->Hin  = t.Hin();
        this->Win  = t.Win();
        this->K    = t.K();
        this->Hout = t.Hout();
        this->Wout = t.Wout();
        this->tileH = t.tileH();
        this->tileW = t.tileW();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x, (uint32_t)(N * Cin * Hin * Win));
        wGm.SetGlobalBuffer((__gm__ DTYPE_W*)w, (uint32_t)(Cin * Cout * K * K));
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y, (uint32_t)(N * Cout * Hout * Wout));

        // 同时保留GM原始指针用于标量读
        wPtr = reinterpret_cast<__gm__ DTYPE_W*>(w);

        const uint32_t tileSize = tileH * tileW;
        pipe.InitBuffer(xBufQ, BUFFER_NUM, tileSize * sizeof(DTYPE_X)); // 用于输入补丁
        pipe.InitBuffer(accBufQ, BUFFER_NUM, tileSize * sizeof(DTYPE_Y)); // 用于累加
    }

    __aicore__ inline void Process()
    {
        // 多核按Cout维度轮询分摊
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        for (uint32_t n = 0; n < N; ++n) {
            for (uint32_t oc = blockIdx; oc < Cout; oc += blockNum) {
                for (uint32_t ho = 0; ho < Hout; ho += tileH) {
                    for (uint32_t wo = 0; wo < Wout; wo += tileW) {
                        ComputeTile(n, oc, ho, wo);
                    }
                }
            }
        }
    }

private:
    __aicore__ inline void ZeroAcc(AscendC::LocalTensor<DTYPE_Y>& acc, uint32_t validSize)
    {
        // 将累加缓冲清零
        AscendC::Duplicate(acc, (DTYPE_Y)0, validSize);
    }

    __aicore__ inline void ComputeTile(uint32_t n, uint32_t oc, uint32_t ho0, uint32_t wo0)
    {
        const uint32_t validH = AscendC::min((uint32_t)tileH, (uint32_t)(Hout - ho0));
        const uint32_t validW = AscendC::min((uint32_t)tileW, (uint32_t)(Wout - wo0));
        const uint32_t tileSize = validH * validW;

        AscendC::LocalTensor<DTYPE_Y> accLocal = accBufQ.AllocTensor<DTYPE_Y>();
        ZeroAcc(accLocal, tileSize);

        // 遍历所有 Cin 和 KxK
        for (uint32_t ic = 0; ic < Cin; ++ic) {
            for (uint32_t kh = 0; kh < K; ++kh) {
                for (uint32_t kw = 0; kw < K; ++kw) {
                    // 对应的输入起始坐标（注意反卷积几何关系）
                    // ho = hi + (K-1 - kh), wo = wi + (K-1 - kw)  ==> hi = ho - (K-1 - kh)
                    // 因此 tile 入射输入块左上角:
                    int32_t hi0 = (int32_t)ho0 - (int32_t)(K - 1 - kh);
                    int32_t wi0 = (int32_t)wo0 - (int32_t)(K - 1 - kw);

                    // 读取当前 (ic, oc, kh, kw) 的权重标量
                    // 权重布局: [Cin, Cout, K, K]
                    uint32_t wOff = (((ic * Cout + oc) * K + kh) * K + kw);
                    DTYPE_W wVal = wPtr[wOff]; // 从GM读标量

                    // 逐行处理 tile
                    for (uint32_t rh = 0; rh < validH; ++rh) {
                        int32_t hi = hi0 + (int32_t)rh;
                        if (hi < 0 || hi >= (int32_t)Hin) continue;

                        // 本行在输出tile的线性偏移
                        uint32_t accRowOffset = rh * validW;

                        int32_t wi_start = wi0;
                        // 计算有效区间 [wL, wR)
                        int32_t wL = 0;
                        if (wi_start < 0) {
                            wL = -wi_start;
                        }
                        if (wL >= (int32_t)validW) continue;

                        int32_t maxCanRead = (int32_t)Win - wi_start;
                        int32_t validWThis = (int32_t)validW - wL;
                        int32_t takeW = AscendC::min(AscendC::max(0, maxCanRead), validWThis);
                        if (takeW <= 0) continue;

                        // 从输入中复制 [hi, wi_start + wL : wi_start + wL + takeW)
                        AscendC::LocalTensor<DTYPE_X> xLocal = xBufQ.AllocTensor<DTYPE_X>();
                        uint32_t xOff = (((n * Cin + ic) * Hin + (uint32_t)hi) * Win + (uint32_t)(wi_start + wL));
                        AscendC::DataCopy(xLocal, xGm[xOff], (uint32_t)takeW);

                        // 将 xLocal 按标量 wVal 进行缩放（Muls: 向量按标量乘）
                        AscendC::Muls(xLocal, xLocal, wVal, (uint32_t)takeW);

                        // 将缩放结果累加到 accLocal 对应切片
                        // 先把 acc 对应位置取出成一个行切片，再做逐元素相加
                        // 由于缺少直接切片接口，我们按两段拷入/拷出实现:
                        // 1) 取出 acc 目标位置到临时缓冲
                        AscendC::LocalTensor<DTYPE_Y> accRowSeg = accBufQ.AllocTensor<DTYPE_Y>();
                        // 将 acc 对应行片段拷贝到 accRowSeg
                        AscendC::DataCopy(accRowSeg, accLocal[accRowOffset + (uint32_t)wL], (uint32_t)takeW);
                        // 2) accRowSeg += xLocal
                        AscendC::Add(accRowSeg, accRowSeg, xLocal, (uint32_t)takeW);
                        // 3) 写回 acc 这一段
                        AscendC::DataCopy(accLocal[accRowOffset + (uint32_t)wL], accRowSeg, (uint32_t)takeW);

                        accBufQ.FreeTensor(accRowSeg);
                        xBufQ.FreeTensor(xLocal);
                    } // rh
                } // kw
            } // kh
        } // ic

        // 写回 y 的 tile
        for (uint32_t rh = 0; rh < validH; ++rh) {
            uint32_t accRowOffset = rh * validW;
            uint32_t yOff = (((n * Cout + oc) * Hout + (ho0 + rh)) * Wout + wo0);
            AscendC::DataCopy(yGm[yOff], accLocal[accRowOffset], validW);
        }

        accBufQ.FreeTensor(accLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM>  xBufQ;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> accBufQ;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_W> wGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    __gm__ DTYPE_W* wPtr;

    uint32_t N, Cin, Cout, Hin, Win, K, Hout, Wout;
    uint32_t tileH, tileW;
};

extern "C" __global__ __aicore__ void conv_transposed_2d_asymmetric_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(t, tiling);
    KernelDeconv2D op;
    op.Init(x, weight, y, t);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    用自定义 AscendC 算子替换 nn.ConvTranspose2d（限定 stride=1, padding=0, output_padding=0, groups=1, bias=False）
    权重布局与 PyTorch ConvTranspose2d 一致: [Cin, Cout, K, K]
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        # 与 nn.ConvTranspose2d 权重一致: [Cin, Cout, K, K]
        w = torch.empty(in_channels, out_channels, kernel_size, kernel_size)
        torch.nn.init.kaiming_uniform_(w, a=5**0.5)
        self.weight = torch.nn.Parameter(w)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return custom_ops_lib.conv_transposed_2d_asymmetric_input_square_kernel_custom(x, self.weight)
