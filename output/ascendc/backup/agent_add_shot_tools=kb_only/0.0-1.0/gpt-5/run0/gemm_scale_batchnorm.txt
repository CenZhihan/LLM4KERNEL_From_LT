kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "gemm_scale_batchnorm_custom_tiling_data.h"
#include "lib/matmul_intf.h"

// This kernel fuses: GEMM (x @ W^T + b) -> scale (*) -> BatchNorm
// References:
// - Matmul tiling and usage: CANN C API Reference 5.2.2.1.1 and 5.2.1.24 IterateNBatch
// - BatchNorm high-level API: CANN C API Reference 5.4.5 BatchNorm
// Constraints to note:
// - BatchNorm input expects ND layout; S*H needs 32-byte alignment. If not, pad to zero.

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 1;

template <typename T>
inline __aicore__ uint32_t AlignToBlock(const uint32_t inputValue)
{
    constexpr uint32_t ONE_BLK_SIZE = 32;
    uint32_t alignUnit = ONE_BLK_SIZE / sizeof(T);
    return (inputValue + alignUnit - 1) / alignUnit * alignUnit;
}

class KernelGemmScaleBatchnorm {
public:
    __aicore__ inline void Init(GM_ADDR xGmAddr, GM_ADDR wGmAddr, GM_ADDR bGmAddr,
                                GM_ADDR scaleGmAddr, GM_ADDR bnWgtGmAddr, GM_ADDR bnBiasGmAddr,
                                GM_ADDR yGmAddr, const optiling::GemmScaleBatchnormCustomTilingData &tiling)
    {
        B = tiling.batch();
        K = tiling.in_feat();
        N = tiling.out_feat();
        epsilon = tiling.epsilon();
        totalLen = tiling.totalLength();

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(xGmAddr), B * K);
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(wGmAddr), N * K);
        biasGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(bGmAddr), N);
        scaleGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(scaleGmAddr), N);
        bnWgtGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(bnWgtGmAddr), N);
        bnBiasGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(bnBiasGmAddr), N);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(yGmAddr), B * N);

        // Internal temporary buffer for GEMM output before BN
        gemmOutGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(yGmAddr), B * N); // reuse y gm as workspace before BN

        // Queues for vector IO
        pipe.InitBuffer(inQueueA, BUFFER_NUM, B * K * sizeof(DTYPE)); // placeholders; will use GM tensors directly for GEMM
        pipe.InitBuffer(outQueueC, BUFFER_NUM, B * N * sizeof(DTYPE));

        // For BN formatting: we'll treat tensor as [Bch = N, S = B, H = 1]
        bLength = N;
        sLength = B;
        hLength = 1;
        shLength = sLength * hLength;
        bshLength = bLength * shLength;

        // Queues for BN
        pipe.InitBuffer(inQueueBNInput, 1, sizeof(DTYPE) * bshLength);
        pipe.InitBuffer(inQueueGamma, 1, sizeof(DTYPE) * bLength);
        pipe.InitBuffer(inQueueBeta, 1, sizeof(DTYPE) * bLength);
        pipe.InitBuffer(outQueueBNOut, 1, sizeof(DTYPE) * bshLength);
        pipe.InitBuffer(outQueueMean, 1, sizeof(DTYPE) * shLength);
        pipe.InitBuffer(outQueueVar, 1, sizeof(DTYPE) * shLength);
    }

    __aicore__ inline void Process()
    {
        // 1) GEMM: C = X * W^T + b
        GemmCompute();

        // 2) Scale: C = C * scale (per feature N)
        ScaleCompute();

        // 3) BatchNorm along batch dimension per feature
        BatchNormCompute();
    }

private:
    // Matmul using library interface (NORMAL layout, single call)
    __aicore__ inline void GemmCompute()
    {
        // Define MatmulType: A[B,K], B[N,K] (we set transposeB=true to match W^T), C[B,N]
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, DTYPE, false, LayoutMode::NORMAL> aType;
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, DTYPE, true,  LayoutMode::NORMAL> bType;
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, DTYPE, false, LayoutMode::NORMAL> cType;
        typedef matmul::MatmulType <AscendC::TPosition::GM, CubeFormat::ND, DTYPE> biasType;

        // Build GM views
        AscendC::GlobalTensor<DTYPE> aGlobal, bGlobal, cGlobal, biasGlobal;
        aGlobal.SetGlobalBuffer(const_cast<__gm__ DTYPE*>(xGm.GetPhyAddr()), B * K);
        bGlobal.SetGlobalBuffer(const_cast<__gm__ DTYPE*>(wGm.GetPhyAddr()), N * K);
        cGlobal.SetGlobalBuffer(const_cast<__gm__ DTYPE*>(gemmOutGm.GetPhyAddr()), B * N);
        biasGlobal.SetGlobalBuffer(const_cast<__gm__ DTYPE*>(biasGm.GetPhyAddr()), N);

        // Create Matmul obj
        matmul::Matmul<aType, bType, cType, biasType> mm;
        AscendC::TPipe mmPipe;
        g_cubeTPipePtr = &mmPipe;
        REGIST_MATMUL_OBJ(&mmPipe, GetSysWorkSpacePtr(), mm);

        // Build tiling quickly on device side is not supported; use default init from host tiling context is typical.
        // Here we assume default internal config for single GEMM (for illustration).
        optiling::TCubeTiling dummyTiling;
        // Minimal init path
        mm.Init(&dummyTiling);
        mm.SetTensorA(aGlobal[0], false);
        mm.SetTensorB(bGlobal[0], true);
        mm.SetWorkspace(nullptr, 0);
        mm.SetBias(biasGlobal[0]);

        // Single GEMM
        mm.IterateNBatch(1, 1, 1, false);
    }

    __aicore__ inline void ScaleCompute()
    {
        // z = z * scale, broadcast on last dim N
        // Tile by rows
        for (uint32_t row = 0; row < B; ++row) {
            AscendC::LocalTensor<DTYPE> rowLocal = outQueueC.AllocTensor<DTYPE>();
            AscendC::LocalTensor<DTYPE> scaleLocal = inQueueA.AllocTensor<DTYPE>();
            AscendC::DataCopy(rowLocal, gemmOutGm[row * N], N);
            AscendC::DataCopy(scaleLocal, scaleGm, N);
            // Elementwise multiply
            AscendC::Mul(rowLocal, rowLocal, scaleLocal, N);
            // Write back
            AscendC::DataCopy(gemmOutGm[row * N], rowLocal, N);
            outQueueC.FreeTensor(rowLocal);
            inQueueA.FreeTensor(scaleLocal);
        }
    }

    __aicore__ inline void BatchNormCompute()
    {
        // Re-interpret C[B,N] as [Bch=N, S=B, H=1] by transposing to match expected layout.
        // We will transpose: dstBN[bch, s, h=1] = src[s, bch].
        AscendC::LocalTensor<DTYPE> bnInputLocal = inQueueBNInput.AllocTensor<DTYPE>();
        // Prepare transposed buffer
        for (uint32_t s = 0; s < sLength; ++s) {
            // copy one row from gemmOut
            AscendC::LocalTensor<DTYPE> row = inQueueA.AllocTensor<DTYPE>();
            AscendC::DataCopy(row, gemmOutGm[s * N], N);
            // scatter to bnInputLocal at [bch, s, 0]
            for (uint32_t bch = 0; bch < bLength; ++bch) {
                bnInputLocal[bch * shLength + s] = row[bch];
            }
            inQueueA.FreeTensor(row);
        }

        // gamma/beta for BN are bnWgtGm/bnBiasGm, length = N (bLength)
        AscendC::LocalTensor<DTYPE> gammaLocal = inQueueGamma.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> betaLocal  = inQueueBeta.AllocTensor<DTYPE>();
        AscendC::DataCopy(gammaLocal, bnWgtGm, bLength);
        AscendC::DataCopy(betaLocal,  bnBiasGm, bLength);

        AscendC::LocalTensor<DTYPE> bnOutLocal = outQueueBNOut.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> outMeanLocal = outQueueMean.AllocTensor<DTYPE>();
        AscendC::LocalTensor<DTYPE> outVarLocal  = outQueueVar.AllocTensor<DTYPE>();

        BatchNormTiling bnTiling;
        // Basic dimensions setup for BN
        // In practical usage, host should provide bn tiling. Here we rely on framework defaults.

        AscendC::BatchNorm<DTYPE, false, false>(
            bnOutLocal,
            outMeanLocal,
            outVarLocal,
            bnInputLocal,
            gammaLocal,
            betaLocal,
            (DTYPE)epsilon,
            bnTiling);

        // Transpose back to [B, N] -> write to yGm
        for (uint32_t s = 0; s < sLength; ++s) {
            AscendC::LocalTensor<DTYPE> row = outQueueC.AllocTensor<DTYPE>();
            for (uint32_t bch = 0; bch < bLength; ++bch) {
                row[bch] = bnOutLocal[bch * shLength + s];
            }
            AscendC::DataCopy(yGm[s * N], row, N);
            outQueueC.FreeTensor(row);
        }

        // Free locals
        outQueueBNOut.FreeTensor(bnOutLocal);
        outQueueMean.FreeTensor(outMeanLocal);
        outQueueVar.FreeTensor(outVarLocal);
        inQueueGamma.FreeTensor(gammaLocal);
        inQueueBeta.FreeTensor(betaLocal);
        inQueueBNInput.FreeTensor(bnInputLocal);
    }

private:
    // GM tensors
    AscendC::GlobalTensor<DTYPE> xGm;       // [B,K]
    AscendC::GlobalTensor<DTYPE> wGm;       // [N,K]
    AscendC::GlobalTensor<DTYPE> biasGm;    // [N]
    AscendC::GlobalTensor<DTYPE> scaleGm;   // [N]
    AscendC::GlobalTensor<DTYPE> bnWgtGm;   // [N]
    AscendC::GlobalTensor<DTYPE> bnBiasGm;  // [N]
    AscendC::GlobalTensor<DTYPE> gemmOutGm; // [B,N]
    AscendC::GlobalTensor<DTYPE> yGm;       // [B,N]

    // Queues
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueueA;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueC;

    // BN queues
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueBNInput;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueGamma;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueBeta;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueBNOut;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueMean;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueVar;

    // Dims
    uint32_t B, K, N;
    float epsilon;
    uint32_t totalLen;

    // BN dims (map [B,N] -> [Bch=N, S=B, H=1])
    uint32_t bLength;   // N
    uint32_t sLength;   // B
    uint32_t hLength;   // 1
    uint32_t shLength;  // B
    uint32_t bshLength; // N*B
};

extern "C" __global__ __aicore__ void gemm_scale_batchnorm_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR scale, GM_ADDR bn_weight, GM_ADDR bn_bias,
    GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelGemmScaleBatchnorm op;
    op.Init(x, weight, bias, scale, bn_weight, bn_bias, y, tilingData);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, scale_shape, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.weight = torch.nn.Parameter(torch.empty(out_features, in_features))
        self.bias = torch.nn.Parameter(torch.zeros(out_features))
        torch.nn.init.kaiming_uniform_(self.weight, a=math.sqrt(5))
        if in_features > 0:
            fan_in, _ = torch.nn.init._calculate_fan_in_and_fan_out(self.weight)
            bound = 1 / math.sqrt(fan_in)
            torch.nn.init.uniform_(self.bias, -bound, bound)
        self.scale = torch.nn.Parameter(torch.rand(scale_shape))
        self.bn_weight = torch.nn.Parameter(torch.ones(out_features))
        self.bn_bias = torch.nn.Parameter(torch.zeros(out_features))
        self.eps = eps
        self.momentum = momentum

    def forward(self, x):
        return custom_ops_lib.gemm_scale_batchnorm_custom(x, self.weight, self.bias, self.scale, self.bn_weight, self.bn_bias)

import math

batch_size = 128
in_features = 1024
out_features = 512
scale_shape = (out_features,)

def get_inputs():
    return [torch.rand(batch_size, in_features)]

def get_init_inputs():
    return [in_features, out_features, scale_shape]
