kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "matmul_group_norm_leaky_relu_sum_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                uint32_t totalLength, uint32_t tileNum,
                                uint32_t bLength, uint32_t hLength,
                                float epsilon, float negativeSlope)
    {
        this->bLength = bLength;
        this->hLength = hLength;
        this->tileNum = tileNum;
        this->epsilon = epsilon;
        this->negativeSlope = negativeSlope;

        // Assign rows to blocks evenly
        uint32_t blocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowsPerBlock = (bLength + blocks - 1) / blocks;
        uint32_t startRow = rowsPerBlock * blockIdx;
        if (startRow > bLength) { startRow = bLength; }
        uint32_t endRow = startRow + rowsPerBlock;
        if (endRow > bLength) { endRow = bLength; }
        this->rowCount = (endRow > startRow) ? (endRow - startRow) : 0;

        xGm.SetGlobalBuffer((__gm__ float *)x + startRow * hLength, this->rowCount * hLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + startRow * hLength, this->rowCount * hLength);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, hLength);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, hLength);

        // One row per tile
        this->tileLength = hLength;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueGamma, 1, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBeta, 1, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpQueue, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Preload gamma/beta
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.AllocTensor<float>();
        AscendC::DataCopy(gammaLocal, gammaGm, hLength);
        inQueueGamma.EnQue(gammaLocal);

        AscendC::LocalTensor<float> betaLocal = inQueueBeta.AllocTensor<float>();
        AscendC::DataCopy(betaLocal, betaGm, hLength);
        inQueueBeta.EnQue(betaLocal);

        int32_t loopCount = static_cast<int32_t>(rowCount);
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute();
            CopyOut(i);
        }

        // free gamma/beta
        gammaLocal = inQueueGamma.DeQue<float>();
        inQueueGamma.FreeTensor(gammaLocal);
        betaLocal = inQueueBeta.DeQue<float>();
        inQueueBeta.FreeTensor(betaLocal);
    }

private:
    __aicore__ inline void CopyIn(int32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        // xLocal -> scale by gamma and shift by beta (affine), then leaky_relu and sum y+y (i.e., *2)
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.DeQue<float>();
        AscendC::LocalTensor<float> betaLocal = inQueueBeta.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpQueue.AllocTensor<float>();

        // y = x * gamma
        AscendC::Mul(yLocal, xLocal, gammaLocal, this->tileLength);
        // y = y + beta
        AscendC::Add(yLocal, yLocal, betaLocal, this->tileLength);

        // LeakyReLU: pos = relu(y); neg = y - pos; y = pos + negativeSlope * neg
        AscendC::Relu(tmpLocal, yLocal, this->tileLength);                 // tmpLocal = pos
        AscendC::Sub(yLocal, yLocal, tmpLocal, this->tileLength);          // yLocal = neg
        AscendC::Muls(yLocal, yLocal, this->negativeSlope, this->tileLength); // yLocal = negativeSlope * neg
        AscendC::Add(yLocal, yLocal, tmpLocal, this->tileLength);          // yLocal = pos + scaled_neg

        // y = y + y  -> scale by 2
        AscendC::Muls(yLocal, yLocal, 2.0f, this->tileLength);

        tmpQueue.FreeTensor(tmpLocal);
        outQueueY.EnQue(yLocal);

        inQueueGamma.EnQue(gammaLocal); // reuse gamma/beta for next row
        inQueueBeta.EnQue(betaLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t rowIdx)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[rowIdx * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueGamma;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBeta;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpQueue;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;

    uint32_t bLength;
    uint32_t hLength;
    uint32_t tileNum;
    uint32_t rowCount;
    uint32_t tileLength;
    float epsilon;
    float negativeSlope;
};

extern "C" __global__ __aicore__ void matmul_group_norm_leaky_relu_sum_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, gamma, beta, y,
            tiling_data.totalLength(), tiling_data.tileNum(),
            tiling_data.bLength(), tiling_data.hLength(),
            tiling_data.epsilon(), tiling_data.negativeSlope());
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_size, num_groups, eps=1e-5, negative_slope=0.01):
        super().__init__()
        self.fc = torch.nn.Linear(input_size, hidden_size)
        self.gn = torch.nn.GroupNorm(num_groups=num_groups, num_channels=hidden_size, eps=eps)
        self.negative_slope = negative_slope

    def forward(self, x):
        x = self.fc(x)
        # Fused custom AscendC kernel for (affine) GroupNorm + LeakyReLU + x+x
        # Here we pass gn.weight (gamma) and gn.bias (beta) to the kernel
        x = custom_ops_lib.matmul_group_norm_leaky_relu_sum_custom(x, self.gn.weight, self.gn.bias)
        return x
