namespace optiling {

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // Fetch input shape
    const gert::Shape* inShape = context->GetInputShape(0);
    auto originShape = inShape->GetOriginShape();
    uint32_t dimNum = static_cast<uint32_t>(originShape.GetDimNum());
    // Reduce over last dimension inside kernel. The Python wrapper will permute input if needed.
    uint64_t totalLength = originShape.GetShapeSize();
    uint32_t reduceLen = static_cast<uint32_t>(originShape.GetDim(dimNum - 1));
    uint32_t outerRows = static_cast<uint32_t>(totalLength / reduceLen);

    // Decide block dim
    uint32_t blockDim = outerRows >= 8 ? 8 : (outerRows == 0 ? 1 : outerRows);
    context->SetBlockDim(blockDim);

    // rows per core (ceil-div)
    uint32_t rowsPerBlock = (outerRows + blockDim - 1) / blockDim;

    // workLocal temporary buffer in UB (bytes). For ReduceMax, workLocal is required on Atlas.
    // Per CANN API reference, workLocal can share the same size as source operand and is <= source.
    // We provision reduceLen * sizeof(half) here.
    const uint32_t typeSize = 2; // float16
    uint32_t tmpBufferSize = reduceLen * typeSize;

    MaxReductionOverADimensionCustomTilingData tiling;
    tiling.set_outerRows(outerRows);
    tiling.set_reduceLen(reduceLen);
    tiling.set_rowsPerBlock(rowsPerBlock);
    tiling.set_tmpBufferSize(tmpBufferSize);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t* workspaces = context->GetWorkspaceSizes(1);
    workspaces[0] = 0;
    return ge::GRAPH_SUCCESS;
}

namespace ge {
// Output shape = input shape with last dimension reduced.
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);

    // Build output dims by removing the last axis
    auto origin = x_shape->GetOriginShape();
    int64_t ndim = origin.GetDimNum();
    std::vector<int64_t> outDims;
    outDims.reserve(ndim > 0 ? ndim - 1 : 0);
    for (int64_t i = 0; i < ndim - 1; ++i) {

    for (size_t i = 0; i < outDims.size(); ++i) {
        y_shape->SetDim(i, outDims[i]);

    return GRAPH_SUCCESS;

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "max_reduction_over_a_dimension_custom_tiling_data.h"

// This kernel reduces along the last dimension. Python wrapper permutes input if needed.
using srcType = half;
using dstType = half;

class KernelMaxReduceLastDim {
public:
    __aicore__ inline KernelMaxReduceLastDim() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t outerRows, uint32_t reduceLen,
                                uint32_t rowsPerBlock, uint32_t tmpBufferSize)
    {
        this->outerRows = outerRows;
        this->reduceLen = reduceLen;
        this->rowsPerBlock = rowsPerBlock;
        this->tmpBytes = tmpBufferSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t startRow = rowsPerBlock * blockIdx;
        // clamp count for last block
        uint32_t maxRowsThisBlock = (startRow >= outerRows) ? 0 : AscendC::Min(rowsPerBlock, outerRows - startRow);
        this->rowsThisBlock = maxRowsThisBlock;

        // Offset GM pointers to this core's start row
        uint64_t offsetElems = static_cast<uint64_t>(startRow) * static_cast<uint64_t>(reduceLen);
        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ srcType*>(x) + offsetElems, static_cast<uint64_t>(rowsThisBlock) * reduceLen);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ dstType*>(y) + startRow, rowsThisBlock);

        // Allocate UB queues
        // One full row for input, one tmp buffer for workLocal, and one element for output
        pipe.InitBuffer(inQueue, 1, static_cast<uint64_t>(reduceLen) * sizeof(srcType));
        pipe.InitBuffer(tmpQueue, 1, tmpBufferSize); // workLocal
        pipe.InitBuffer(outQueue, 1, sizeof(dstType));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t r = 0; r < rowsThisBlock; ++r) {
            CopyIn(r);
            Compute(r);
            CopyOut(r);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t r)
    {
        AscendC::LocalTensor<srcType> srcLocal = inQueue.AllocTensor<srcType>();
        AscendC::DataCopy(srcLocal, xGm[static_cast<uint64_t>(r) * reduceLen], reduceLen);
        inQueue.EnQue(srcLocal);
    }

    __aicore__ inline void Compute(uint32_t /*r*/)
    {
        AscendC::LocalTensor<srcType> srcLocal = inQueue.DeQue<srcType>();
        AscendC::LocalTensor<srcType> workLocal = tmpQueue.AllocTensor<srcType>(); // workLocal required by ReduceMax
        AscendC::LocalTensor<dstType> dstLocal = outQueue.AllocTensor<dstType>();  // store one max value

        // Use ReduceMax over the first 'reduceLen' elements of the row (contiguous)
        // Reference: CANN Ascend C API Reference 4.1.8.1 ReduceMax (first n elements)
        AscendC::ReduceMax<dstType>(dstLocal, srcLocal, workLocal, static_cast<int32_t>(reduceLen), false);

        outQueue.EnQue<dstType>(dstLocal);
        inQueue.FreeTensor(srcLocal);
        tmpQueue.FreeTensor(workLocal);
    }

    __aicore__ inline void CopyOut(uint32_t r)
    {
        AscendC::LocalTensor<dstType> dstLocal = outQueue.DeQue<dstType>();
        AscendC::DataCopy(yGm[r], dstLocal, 1);
        outQueue.FreeTensor(dstLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueue;

    AscendC::GlobalTensor<srcType> xGm;
    AscendC::GlobalTensor<dstType> yGm;

    uint32_t outerRows;
    uint32_t reduceLen;
    uint32_t rowsPerBlock;
    uint32_t rowsThisBlock;
    uint32_t tmpBytes;
};

extern "C" __global__ __aicore__ void max_reduction_over_a_dimension_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelMaxReduceLastDim op;
    op.Init(x, y, tilingData.outerRows, tilingData.reduceLen, tilingData.rowsPerBlock, tilingData.tmpBufferSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Max reduction over an arbitrary dimension using a custom AscendC kernel
    that reduces over the last axis. We permute input if needed.
    """
    def __init__(self, dim: int):
        super().__init__()
        self.dim = dim

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Kernel supports float16 and reduces over the last dimension.
        # Move target dimension to last by permutation when necessary.
        dim = self.dim
        ndim = x.dim()
        if dim < 0:
            dim += ndim
        assert 0 <= dim < ndim, "Invalid reduction dimension"

        x = x.to(torch.float16)
        if dim != ndim - 1:
            order = list(range(ndim))
            # move dim to the end while keeping relative order of other axes
            order = [i for i in range(ndim) if i != dim] + [dim]
            x = x.permute(order).contiguous()

        # Call custom op: reduce over last dimension
        y = custom_ops_lib.max_reduction_over_a_dimension_custom(x)
        return y
