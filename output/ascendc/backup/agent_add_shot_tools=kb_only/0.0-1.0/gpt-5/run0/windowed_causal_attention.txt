#include "windowed_causal_attention_custom_tiling.h"

namespace optiling {
static const uint32_t DEFAULT_BLOCK_DIM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 读取输入形状：q 的形状 [B, N, H, Dh]
    const gert::Shape* qShape = context->GetInputShape(0);
    if (qShape->GetDimNum() != 4) {
        return ge::GRAPH_FAILED;
    }
    uint32_t B  = static_cast<uint32_t>(qShape->GetDim(0));
    uint32_t N  = static_cast<uint32_t>(qShape->GetDim(1));
    uint32_t H  = static_cast<uint32_t>(qShape->GetDim(2));
    uint32_t Dh = static_cast<uint32_t>(qShape->GetDim(3));

    // 读取可选属性 window_size，若不存在则给默认 256
    int64_t windowSizeAttr = 256;
    auto* attr = context->GetAttr("window_size");
    if (attr != nullptr) {
        windowSizeAttr = attr->GetInt();
        if (windowSizeAttr <= 0) windowSizeAttr = 256;
    }
    uint32_t windowSize = static_cast<uint32_t>(windowSizeAttr);

    // block 配置
    uint32_t blockNum = DEFAULT_BLOCK_DIM;
    if (N < blockNum) blockNum = N; // 简单约束：块数不超过序列长度
    context->SetBlockDim(blockNum);

    // 每个block负责的 token 数
    uint32_t tokensPerBlock = (N + blockNum - 1) / blockNum;

    // softmax缩放因子
    float scale = 1.0f / std::sqrt(static_cast<float>(Dh));

    WindowedCausalAttentionCustomTilingData tiling;
    tiling.set_B(B);
    tiling.set_N(N);
    tiling.set_H(H);
    tiling.set_Dh(Dh);
    tiling.set_windowSize(windowSize);
    tiling.set_tokensPerBlock(tokensPerBlock);
    tiling.set_blockNum(blockNum);
    tiling.set_scale(scale);

    // 写入 tiling buffer
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    // workspace 暂不使用
    size_t* workspaceSizes = context->GetWorkspaceSizes(1);
    workspaceSizes[0] = 0;
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "windowed_causal_attention_custom_tiling_data.h"

// 简化实现：直接从 GM 访问并三遍扫描（max、sumExp、输出聚合）。
// 形状均为 float32，布局 [B, N, H, Dh]。
extern "C" __global__ __aicore__ void windowed_causal_attention_custom(
    GM_ADDR qGM, GM_ADDR kGM, GM_ADDR vGM, GM_ADDR oGM, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    const uint32_t B  = tiling_data.B;
    const uint32_t N  = tiling_data.N;
    const uint32_t H  = tiling_data.H;
    const uint32_t Dh = tiling_data.Dh;
    const uint32_t windowSize = tiling_data.windowSize;
    const uint32_t tokensPerBlock = tiling_data.tokensPerBlock;
    const uint32_t blockNum = tiling_data.blockNum;
    const float scale = tiling_data.scale;

    // 映射 GM
    __gm__ float* q = reinterpret_cast<__gm__ float*>(qGM);
    __gm__ float* k = reinterpret_cast<__gm__ float*>(kGM);
    __gm__ float* v = reinterpret_cast<__gm__ float*>(vGM);
    __gm__ float* o = reinterpret_cast<__gm__ float*>(oGM);

    // 当前 block 负责的 token 范围
    uint32_t blockIdx = AscendC::GetBlockIdx();
    uint32_t startTok = blockIdx * tokensPerBlock;
    uint32_t endTok = startTok + tokensPerBlock;
    if (startTok >= N) return;
    if (endTok > N) endTok = N;

    // 索引计算辅助：返回线性下标
    auto qIndex = [=](__aicore__ uint32_t b, uint32_t n, uint32_t h, uint32_t dh) -> uint64_t {
        return (((uint64_t)b * N + n) * H + h) * Dh + dh;
    };
    auto kvIndex = qIndex; // 同布局

    // 遍历 batch、head、token
    for (uint32_t b = 0; b < B; ++b) {
        for (uint32_t h = 0; h < H; ++h) {
            for (uint32_t i = startTok; i < endTok; ++i) {
                // 计算窗口起点（只看历史，含当前位置 i）
                uint32_t start = (i > windowSize) ? (i - windowSize) : 0;
                uint32_t L = i - start + 1; // 有效 key 数

                // Pass 1: 计算最大分数 maxScore
                float maxScore = -3.4e38f;
                for (uint32_t j = start; j <= i; ++j) {
                    // dot(Q[i], K[j]) * scale
                    float dot = 0.f;
                    uint64_t qiBase = qIndex(b, i, h, 0);
                    uint64_t kjBase = kvIndex(b, j, h, 0);
                    for (uint32_t m = 0; m < Dh; ++m) {
                        dot += q[qiBase + m] * k[kjBase + m];
                    }
                    float score = dot * scale;
                    if (score > maxScore) maxScore = score;
                }

                // Pass 2: 计算 sumExp
                float sumExp = 0.f;
                for (uint32_t j = start; j <= i; ++j) {
                    float dot = 0.f;
                    uint64_t qiBase = qIndex(b, i, h, 0);
                    uint64_t kjBase = kvIndex(b, j, h, 0);
                    for (uint32_t m = 0; m < Dh; ++m) {
                        dot += q[qiBase + m] * k[kjBase + m];
                    }
                    float score = dot * scale;
                    float e = AscendC::Exp(score - maxScore);
                    sumExp += e;
                }
                float invDenom = 1.0f / sumExp;

                // Pass 3: 逐通道聚合输出
                uint64_t outBase = qIndex(b, i, h, 0);
                for (uint32_t m = 0; m < Dh; ++m) {
                    float acc = 0.f;
                    for (uint32_t j = start; j <= i; ++j) {
                        float dot = 0.f;
                        uint64_t qiBase = qIndex(b, i, h, 0);
                        uint64_t kjBase = kvIndex(b, j, h, 0);
                        for (uint32_t t = 0; t < Dh; ++t) {
                            dot += q[qiBase + t] * k[kjBase + t];
                        }
                        float score = dot * scale;
                        float weight = AscendC::Exp(score - maxScore) * invDenom;
                        uint64_t vjBase = kvIndex(b, j, h, 0);
                        acc += weight * v[vjBase + m];
                    }
                    o[outBase + m] = acc;
                }
            }
        }
    }
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import math
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    """
    使用自定义 AscendC 算子加速的滑动窗口因果注意力
    算子名: WindowedCausalAttentionCustom
    内核函数: windowed_causal_attention_custom
    """
    def __init__(self, d_model=1024, num_heads=16, window_size=256):
        super().__init__()
        self.q_proj = nn.Linear(d_model, d_model)
        self.k_proj = nn.Linear(d_model, d_model)
        self.v_proj = nn.Linear(d_model, d_model)
        self.out_proj = nn.Linear(d_model, d_model)
        self.num_heads = num_heads
        self.window_size = window_size

    def forward(self, x):
        B, N, D = x.shape
        H = self.num_heads
        d_h = D // H

        # 线性映射后 reshape 为 [B, N, H, d_h]
        Q = self.q_proj(x).view(B, N, H, d_h).contiguous()
        K = self.k_proj(x).view(B, N, H, d_h).contiguous()
        V = self.v_proj(x).view(B, N, H, d_h).contiguous()

        # 调用自定义算子：完成带窗口的因果注意力
        O = custom_ops_lib.windowed_causal_attention_custom(Q, K, V, int(self.window_size))

        return self.out_proj(O.reshape(B, N, D))
'''
