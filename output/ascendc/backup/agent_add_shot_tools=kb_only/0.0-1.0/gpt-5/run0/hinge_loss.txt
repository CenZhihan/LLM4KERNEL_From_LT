kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "hinge_loss_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelHingeLoss {
public:
    __aicore__ inline KernelHingeLoss() {}
    __aicore__ inline void Init(GM_ADDR pred, GM_ADDR target, GM_ADDR z,
                                uint32_t totalLength, uint32_t tileNum, float invN)
    {
        ASSERT(AscendC::GetBlockNum() != 0 && "block dim can not be zero!");
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        ASSERT(tileNum != 0 && "tile num can not be zero!");
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->invN = invN;

        predGm.SetGlobalBuffer((__gm__ DTYPE_X *)pred + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        tgtGm.SetGlobalBuffer((__gm__ DTYPE_Y *)target + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        // 输出为单元素，全核原子累加到同一地址
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, 1);

        pipe.InitBuffer(inQueuePred, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueTgt,  BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(calcQueue,   BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(outQueueZ,   BUFFER_NUM, sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        // 仅 block 0 清零输出，随后全核同步，避免竞争
        if (AscendC::GetBlockIdx() == 0) {
            AscendC::LocalTensor<DTYPE_Z> zeroLocal = outQueueZ.AllocTensor<DTYPE_Z>();
            AscendC::Duplicate(zeroLocal, (DTYPE_Z)0, 1);
            AscendC::DataCopy(zGm, zeroLocal, 1);
            outQueueZ.FreeTensor(zeroLocal);
        }
        AscendC::SyncAll();

        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> predLocal = inQueuePred.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> tgtLocal  = inQueueTgt.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(predLocal, predGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(tgtLocal,  tgtGm[progress * this->tileLength],  this->tileLength);
        inQueuePred.EnQue(predLocal);
        inQueueTgt.EnQue(tgtLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> predLocal = inQueuePred.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> tgtLocal  = inQueueTgt.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> tmpLocal  = calcQueue.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_Z> sumLocal  = outQueueZ.AllocTensor<DTYPE_Z>();

        // tmp = 1 - predictions * targets
        AscendC::Mul(tmpLocal, predLocal, tgtLocal, this->tileLength);
        AscendC::Muls(tmpLocal, tmpLocal, (DTYPE_Z)(-1.0f), this->tileLength);
        AscendC::Adds(tmpLocal, tmpLocal, (DTYPE_Z)1.0f, this->tileLength);

        // clamp(min=0): tmp = max(tmp, 0)
        AscendC::Maxs(tmpLocal, tmpLocal, (DTYPE_Z)0.0f, this->tileLength);

        // reduce sum -> 标量
        AscendC::ReduceSum(sumLocal, tmpLocal, this->tileLength);

        // 乘以 invN 得到均值贡献
        AscendC::Muls(sumLocal, sumLocal, (DTYPE_Z)this->invN, 1);

        outQueueZ.EnQue(sumLocal);
        calcQueue.FreeTensor(tmpLocal);
        inQueuePred.FreeTensor(predLocal);
        inQueueTgt.FreeTensor(tgtLocal);
    }

    __aicore__ inline void CopyOut(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_Z> sumLocal = outQueueZ.DeQue<DTYPE_Z>();
        // 原子加到全局输出 z[0]
        AscendC::SetAtomicAdd<DTYPE_Z>();
        AscendC::DataCopy(zGm, sumLocal, 1);
        AscendC::SetAtomicNone();
        outQueueZ.FreeTensor(sumLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> inQueuePred, inQueueTgt;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> calcQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::GlobalTensor<DTYPE_X> predGm;
    AscendC::GlobalTensor<DTYPE_Y> tgtGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    float invN;
};

extern "C" __global__ __aicore__ void hinge_loss_custom(GM_ADDR predictions, GM_ADDR targets, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelHingeLoss op;
    op.Init(predictions, targets, z, tiling_data.totalLength, tiling_data.tileNum, tiling_data.invN);
    if (TILING_KEY_IS(1)) {
        op.Process();
    }
}
