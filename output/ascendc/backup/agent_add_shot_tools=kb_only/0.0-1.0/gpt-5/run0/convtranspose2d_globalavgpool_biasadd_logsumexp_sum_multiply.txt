kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "convtranspose2d_globalavgpool_biasadd_logsumexp_sum_multiply_custom_tiling_data.h"

using T = DTYPE_X;
constexpr int32_t IN_Q_NUM = 1;
constexpr int32_t OUT_Q_NUM = 1;

class KernelReduceLastSum {
public:
    __aicore__ inline KernelReduceLastSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t tmpSize, uint32_t outter, uint32_t inner, uint32_t n)
    {
        this->outter = outter;
        this->inner = inner;
        this->n = n;
        this->tmpSize = tmpSize;

        xGm.SetGlobalBuffer((__gm__ T*)x);
        yGm.SetGlobalBuffer((__gm__ T*)y);

        // UB buffers
        pipe.InitBuffer(inQueue, IN_Q_NUM, inner * sizeof(T));
        // dst buffer must be 32-byte aligned even if only 1 element is valid
        uint32_t outAlignedBytes = ((sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE) * AscendC::ONE_BLK_SIZE;
        pipe.InitBuffer(outQueue, OUT_Q_NUM, outAlignedBytes);
        pipe.InitBuffer(tmpQueue, 1, tmpSize);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < outter; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        srcLocal = inQueue.AllocTensor<T>();
        AscendC::DataCopy(srcLocal, xGm[row * n], n); // copy only valid n elements
        inQueue.EnQue<T>(srcLocal);
    }

    __aicore__ inline void Compute(uint32_t /*row*/)
    {
        srcLocal = inQueue.DeQue<T>();
        tmpLocal = tmpQueue.AllocTensor<uint8_t>();

        // allocate aligned dst buffer
        dstLocal = outQueue.AllocTensor<T>();
        // zero-init aligned area
        uint32_t outAlignedElems = ((sizeof(T) + AscendC::ONE_BLK_SIZE - 1) / AscendC::ONE_BLK_SIZE) * AscendC::ONE_BLK_SIZE / sizeof(T);
        T zero(0);
        AscendC::Duplicate<T>(dstLocal, zero, outAlignedElems);

        AscendC::SumParams params;
        params.outter = 1;     // reduce one row each time
        params.inner = inner;  // padded length
        params.n = n;          // valid elements
        AscendC::Sum<T, -1, false, false>(dstLocal, srcLocal, tmpLocal, params);

        outQueue.EnQue<T>(dstLocal);
        tmpQueue.FreeTensor(tmpLocal);
        inQueue.FreeTensor(srcLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        dstLocal = outQueue.DeQue<T>();
        // write only the first valid scalar
        AscendC::DataCopy(yGm[row], dstLocal, 1);
        outQueue.FreeTensor(dstLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, IN_Q_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, OUT_Q_NUM> outQueue;
    AscendC::TQue<AscendC::QuePosition::VECCALC, 1> tmpQueue;

    AscendC::GlobalTensor<T> xGm;
    AscendC::GlobalTensor<T> yGm;

    AscendC::LocalTensor<T> srcLocal;
    AscendC::LocalTensor<T> dstLocal;
    AscendC::LocalTensor<uint8_t> tmpLocal;

    uint32_t outter;
    uint32_t inner;
    uint32_t n;
    uint32_t tmpSize;
};

extern "C" __global__ __aicore__ void convtranspose2d_globalavgpool_biasadd_logsumexp_sum_multiply_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelReduceLastSum op;
    op.Init(x, y, tiling_data.tmpSize, tiling_data.outter, tiling_data.inner, tiling_data.n);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Fused custom op replaces channel-wise reduce-sum in the pipeline using AscendC Sum API.
    """
    def __init__(self, in_channels, out_channels, kernel_size, bias_shape):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv_transpose(x)                           # [N, C, H, W]
        x = torch.mean(x, dim=(2, 3), keepdim=True)         # Global average: [N, C, 1, 1]
        x = x + self.bias                                   # Bias add: [N, C, 1, 1]
        x = torch.exp(x)                                    # exp before logsumexp
        x = x.permute(0, 2, 3, 1).contiguous()              # [N, 1, 1, C] (channels to last dim)
        s = custom_ops_lib.convtranspose2d_globalavgpool_biasadd_logsumexp_sum_multiply_custom(x)  # [N, 1, 1, 1], sum over C
        s = torch.log(s)                                    # log(sum(exp(.))) -> [N, 1, 1, 1]
        s = torch.sum(s, dim=(2, 3))                        # remove spatial dims -> [N, 1]
        s = s * 10.0                                        # multiply
        return s
