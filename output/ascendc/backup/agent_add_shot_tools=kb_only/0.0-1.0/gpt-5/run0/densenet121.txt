host_operator_src="""
#include "densenet121_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // 获取输入形状 NCHW
    const gert::Shape* x_shape = context->GetInputShape(0);
    const gert::Shape* y_shape = context->GetInputShape(1);

    uint32_t n = static_cast<uint32_t>(x_shape->GetOriginShape().GetDim(0));
    uint32_t c1 = static_cast<uint32_t>(x_shape->GetOriginShape().GetDim(1));
    uint32_t h = static_cast<uint32_t>(x_shape->GetOriginShape().GetDim(2));
    uint32_t w = static_cast<uint32_t>(x_shape->GetOriginShape().GetDim(3));
    uint32_t c2 = static_cast<uint32_t>(y_shape->GetOriginShape().GetDim(1));

    // 设置block数，按batch并行，兼容大batch
    uint32_t maxBlock = 32; // 910B常见上限
    uint32_t blockDim = (n < maxBlock) ? n : maxBlock;
    context->SetBlockDim(blockDim);

    // 设定tiling参数
    optiling::Densenet121CustomTilingData tiling;
    tiling.set_n(n);
    tiling.set_c1(c1);
    tiling.set_c2(c2);
    tiling.set_h(h);
    tiling.set_w(w);
    // 设置单次处理的元素数（可调）
    tiling.set_tileEle(4096);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t* currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    const gert::Shape* y_shape = context->GetInputShape(1);
    gert::Shape* z_shape = context->GetOutputShape(0);

    int64_t n = x_shape->GetOriginShape().GetDim(0);
    int64_t c1 = x_shape->GetOriginShape().GetDim(1);
    int64_t h = x_shape->GetOriginShape().GetDim(2);
    int64_t w = x_shape->GetOriginShape().GetDim(3);
    int64_t c2 = y_shape->GetOriginShape().GetDim(1);

    gert::Shape out;
    out.SetDimNum(4);
    out.SetDim(0, n);
    out.SetDim(1, c1 + c2);
    out.SetDim(2, h);
    out.SetDim(3, w);
    *z_shape = out;
    return GRAPH_SUCCESS;
}

{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "densenet121_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelCatC {
public:
    __aicore__ inline KernelCatC() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t n, uint32_t c1, uint32_t c2,
                                uint32_t h, uint32_t w, uint32_t tileEle)
    {
        this->n = n;
        this->c1 = c1;
        this->c2 = c2;
        this->h = h;
        this->w = w;
        this->tileEle = tileEle;

        this->elePerBatchX = c1 * h * w;
        this->elePerBatchY = c2 * h * w;
        this->elePerBatchZ = (c1 + c2) * h * w;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, (uint32_t)(n * elePerBatchX));
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, (uint32_t)(n * elePerBatchY));
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, (uint32_t)(n * elePerBatchZ));

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->tileEle * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileEle * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        for (uint32_t b = blockIdx; b < n; b += blockNum) {
            // 复制X到Z: [b, 0:c1, :, :]
            CopySegment(elePerBatchX,
                        b * elePerBatchX,
                        b * elePerBatchZ + 0);

            // 复制Y到Z: [b, c1:c1+c2, :, :]
            CopySegment(elePerBatchY,
                        b * elePerBatchY,
                        b * elePerBatchZ + elePerBatchX,
                        /*isY*/ true);
        }
    }

private:
    __aicore__ inline void CopySegment(uint32_t totalEle, uint32_t inBase, uint32_t outBase, bool isY = false)
    {
        uint32_t remain = totalEle;
        uint32_t offset = 0;
        int32_t progress = 0;
        // 简单双缓冲流水线
        while (remain > 0) {
            uint32_t cur = remain > tileEle ? tileEle : remain;

            // CopyIn
            AscendC::LocalTensor<DTYPE_X> inLocal = inQueue.AllocTensor<DTYPE_X>();
            if (!isY) {
                AscendC::DataCopy(inLocal, xGm[inBase + offset], cur);
            } else {
                AscendC::DataCopy(inLocal, yGm[inBase + offset], cur);
            }
            inQueue.EnQue(inLocal);

            // Compute(直通)
            AscendC::LocalTensor<DTYPE_X> tmp = inQueue.DeQue<DTYPE_X>();
            AscendC::LocalTensor<DTYPE_Z> outLocal = outQueue.AllocTensor<DTYPE_Z>();
            AscendC::DataCopy(outLocal, tmp, cur);
            outQueue.EnQue<DTYPE_Z>(outLocal);
            inQueue.FreeTensor(tmp);

            // CopyOut
            AscendC::LocalTensor<DTYPE_Z> zLocal = outQueue.DeQue<DTYPE_Z>();
            AscendC::DataCopy(zGm[outBase + offset], zLocal, cur);
            outQueue.FreeTensor(zLocal);

            offset += cur;
            remain -= cur;
            progress++;
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;

    uint32_t n, c1, c2, h, w;
    uint32_t tileEle;
    uint32_t elePerBatchX, elePerBatchY, elePerBatchZ;
};

extern "C" __global__ __aicore__ void densenet121_custom(GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelCatC op;
    op.Init(x, y, z,
            tilingData.n, tilingData.c1, tilingData.c2,
            tilingData.h, tilingData.w, tilingData.tileEle);
    op.Process();
}
"""

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class DenseBlock(nn.Module):
    def __init__(self, num_layers: int, num_input_features: int, growth_rate: int):
        super(DenseBlock, self).__init__()
        layers = []
        in_feats = num_input_features
        for i in range(num_layers):
            layers.append(self._make_layer(in_feats, growth_rate))
            in_feats += growth_rate
        self.layers = nn.ModuleList(layers)

    def _make_layer(self, in_features: int, growth_rate: int):
        return nn.Sequential(
            nn.BatchNorm2d(in_features),
            nn.ReLU(inplace=True),
            nn.Conv2d(in_features, growth_rate, kernel_size=3, padding=1, bias=False),
            nn.Dropout(0.0)
        )

    def forward(self, x):
        for layer in self.layers:
            new_feature = layer(x)
            # 使用自定义AscendC内核替换channel维度拼接
            x = custom_ops_lib.densenet121_custom(x, new_feature)
        return x

class TransitionLayer(nn.Module):
    def __init__(self, num_input_features: int, num_output_features: int):
        super(TransitionLayer, self).__init__()
        self.transition = nn.Sequential(
            nn.BatchNorm2d(num_input_features),
            nn.ReLU(inplace=True),
            nn.Conv2d(num_input_features, num_output_features, kernel_size=1, bias=False),
            nn.AvgPool2d(kernel_size=2, stride=2)
        )

    def forward(self, x):
        return self.transition(x)

class ModelNew(nn.Module):
    def __init__(self, growth_rate: int = 32, num_classes: int = 1000):
        super(ModelNew, self).__init__()

        self.features = nn.Sequential(
            nn.Conv2d(3, 64, kernel_size=7, stride=2, padding=3, bias=False),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2, padding=1)
        )

        num_features = 64
        block_layers = [6, 12, 24, 16]

        self.dense_blocks = nn.ModuleList()
        self.transition_layers = nn.ModuleList()

        for i, num_layers in enumerate(block_layers):
            block = DenseBlock(num_layers=num_layers, num_input_features=num_features, growth_rate=growth_rate)
            self.dense_blocks.append(block)
            num_features = num_features + num_layers * growth_rate

            if i != len(block_layers) - 1:
                transition = TransitionLayer(num_input_features=num_features, num_output_features=num_features // 2)
                self.transition_layers.append(transition)
                num_features = num_features // 2

        self.final_bn = nn.BatchNorm2d(num_features)
        self.classifier = nn.Linear(num_features, num_classes)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.features(x)

        for i, block in enumerate(self.dense_blocks):
            x = block(x)
            if i != len(self.dense_blocks) - 1:
                x = self.transition_layers[i](x)

        x = self.final_bn(x)
        x = F.relu(x, inplace=True)
        x = F.adaptive_avg_pool2d(x, (1, 1)).view(x.size(0), -1)
        x = self.classifier(x)
        return x
'''
