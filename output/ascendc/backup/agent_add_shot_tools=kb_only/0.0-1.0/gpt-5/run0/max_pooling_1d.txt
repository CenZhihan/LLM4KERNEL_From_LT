const uint32_t BLOCK_DIM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    MaxPooling1dCustomTilingData tiling;

    const gert::StorageShape *in_shape = context->GetInputShape(0);
    uint32_t n = 1, c = 1, inLen = 1;
    if (in_shape->GetStorageShape().GetDimNum() == 3) {
        n = in_shape->GetStorageShape().GetDim(0);
        c = in_shape->GetStorageShape().GetDim(1);
        inLen = in_shape->GetStorageShape().GetDim(2);
    } else {
        // 兼容性处理：按(N,C,L)假定
        return ge::GRAPH_PARAM_INVALID;
    }

    const gert::RuntimeAttrs *attrs = context->GetAttrs();
    uint32_t kernel_size = *(attrs->GetAttrPointer<uint32_t>(0));
    uint32_t stride      = *(attrs->GetAttrPointer<uint32_t>(1));
    uint32_t padding     = *(attrs->GetAttrPointer<uint32_t>(2));
    // return_indices 无需参与计算

    if (stride == 0) {
        stride = kernel_size;
    }

    // 输出长度：floor((L + 2*pad - K)/stride) + 1，确保非负
    int64_t numer = static_cast<int64_t>(inLen) + 2 * static_cast<int64_t>(padding) - static_cast<int64_t>(kernel_size);
    int64_t outLen64 = numer >= 0 ? (numer / static_cast<int64_t>(stride) + 1) : 0;
    if (outLen64 < 0) outLen64 = 0;
    uint32_t outLen = static_cast<uint32_t>(outLen64);

    uint64_t totalInLength = static_cast<uint64_t>(n) * c * inLen;
    uint64_t totalOutLength = static_cast<uint64_t>(n) * c * outLen;

    context->SetBlockDim(BLOCK_DIM);

    tiling.set_n(n);
    tiling.set_c(c);
    tiling.set_inLen(inLen);
    tiling.set_outLen(outLen);
    tiling.set_kernelSize(kernel_size);
    tiling.set_stride(stride);
    tiling.set_padding(padding);
    tiling.set_totalInLength(static_cast<uint32_t>(totalInLength));
    tiling.set_totalOutLength(static_cast<uint32_t>(totalOutLength));
    tiling.set_tileNum(TILE_NUM);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "max_pooling_1d_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // 双缓冲

class KernelMaxPool1D {
public:
    __aicore__ inline KernelMaxPool1D() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalInLength, uint32_t totalOutLength,
                                uint32_t n, uint32_t c,
                                uint32_t inLen, uint32_t outLen,
                                uint32_t kernelSize, uint32_t stride, uint32_t padding,
                                uint32_t tileNum)
    {
        this->n = n;
        this->c = c;
        this->inLen = inLen;
        this->outLen = outLen;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->tileNum = tileNum;

        // 将输入设置为完整区间，输出按block切分
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, totalInLength);
        uint32_t blockOut = totalOutLength / AscendC::GetBlockNum();
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + blockOut * AscendC::GetBlockIdx(), blockOut);
        this->blockOutLen = blockOut;

        // 每tile输出个数
        this->tileOutLen = (blockOut / tileNum) / BUFFER_NUM;
        if (this->tileOutLen == 0) {
            this->tileOutLen = 1;
        }

        // 对应单个tile最大全窗口跨度：(tileOutLen-1)*stride + kernelSize
        this->maxSpanPerTile = (this->tileOutLen > 0)
                                 ? ((this->tileOutLen - 1) * this->stride + this->kernelSize)
                                 : this->kernelSize;

        // 初始化队列缓冲
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->maxSpanPerTile * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileOutLen * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; ++i) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CalcTileRange(int32_t progress,
                                         uint32_t &globalOutStart,
                                         uint32_t &bcIdx,
                                         uint32_t &p0,
                                         uint32_t &outCount,
                                         uint32_t &inStart,
                                         uint32_t &span,
                                         uint32_t &clampStart)
    {
        // 本block起始输出下标
        uint32_t blockStart = this->blockOutLen * AscendC::GetBlockIdx();
        globalOutStart = blockStart + progress * this->tileOutLen;

        // 映射到展平后的 (bc, p)
        bcIdx = globalOutStart / this->outLen;
        p0 = globalOutStart % this->outLen;

        // 限制本tile不跨越 bc 边界
        uint32_t pLast = p0 + this->tileOutLen - 1;
        if (pLast >= this->outLen) {
            pLast = this->outLen - 1;
        }
        outCount = pLast >= p0 ? (pLast - p0 + 1) : 0;

        // 输入起点(全局)与跨度(按clamp到合法范围)
        int64_t startRaw = static_cast<int64_t>(p0) * this->stride - static_cast<int64_t>(this->padding);
        int64_t clampS = startRaw < 0 ? 0 : startRaw;
        clampStart = static_cast<uint32_t>(clampS);

        int64_t lastStartRaw = static_cast<int64_t>(pLast) * this->stride - static_cast<int64_t>(this->padding);
        int64_t lastEndRaw = lastStartRaw + static_cast<int64_t>(this->kernelSize); // 半开区间右端
        int64_t clampE = lastEndRaw > static_cast<int64_t>(this->inLen) ? static_cast<int64_t>(this->inLen) : lastEndRaw;

        uint32_t baseIn = bcIdx * this->inLen;
        inStart = baseIn + static_cast<uint32_t>(clampS);
        span = static_cast<uint32_t>(clampE - clampS);
        if (span == 0) span = 1; // 避免零拷贝
    }

    __aicore__ inline void CopyIn(int32_t progress)
    {
        uint32_t globalOutStart, bcIdx, p0, outCount, inStart, span, clampStart;
        CalcTileRange(progress, globalOutStart, bcIdx, p0, outCount, inStart, span, clampStart);

        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[inStart], span);
        inQueueX.EnQue(xLocal);

        // 记录当前tile的元信息
        cur_p0 = p0;
        cur_bcIdx = bcIdx;
        cur_outCount = outCount;
        cur_inSpan = span;
        cur_clampStart = clampStart;
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        // 对每个输出位置j，计算窗口最大值（窗口越界按0填充）
        for (uint32_t j = 0; j < cur_outCount; ++j) {
            uint32_t p = cur_p0 + j;
            int64_t startRaw = static_cast<int64_t>(p) * this->stride - static_cast<int64_t>(this->padding);
            int64_t rel = startRaw - static_cast<int64_t>(cur_clampStart);
            if (rel < 0) rel = 0;

            DTYPE_Y maxv = static_cast<DTYPE_Y>(0.0f); // 隐式零填充
            bool inited = false;

            for (uint32_t k = 0; k < this->kernelSize; ++k) {
                int64_t inPosRaw = startRaw + static_cast<int64_t>(k);
                if (inPosRaw >= 0 && inPosRaw < static_cast<int64_t>(this->inLen)) {
                    int64_t idx = rel + static_cast<int64_t>(k);
                    if (idx >= 0 && idx < static_cast<int64_t>(cur_inSpan)) {
                        DTYPE_Y v = static_cast<DTYPE_Y>(xLocal[static_cast<uint32_t>(idx)]);
                        if (!inited) {
                            maxv = v;
                            inited = true;
                        } else {
                            maxv = v > maxv ? v : maxv;
                        }
                    }
                } else {
                    // 越界按0参与最大
                    if (!inited) {
                        maxv = static_cast<DTYPE_Y>(0.0f);
                        inited = true;
                    } else {
                        DTYPE_Y z = static_cast<DTYPE_Y>(0.0f);
                        maxv = z > maxv ? z : maxv;
                    }
                }
            }
            yLocal[j] = maxv;
        }

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        uint32_t globalOutStart, bcIdx, p0, outCount, inStart, span, clampStart;
        CalcTileRange(progress, globalOutStart, bcIdx, p0, outCount, inStart, span, clampStart);

        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        // 将结果写回到当前block的输出区间
        uint32_t blockLocalStart = progress * this->tileOutLen;
        AscendC::DataCopy(yGm[blockLocalStart], yLocal, outCount);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN,  BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t n, c;
    uint32_t inLen, outLen;
    uint32_t kernelSize, stride, padding;

    uint32_t tileNum;
    uint32_t blockOutLen;
    uint32_t tileOutLen;
    uint32_t maxSpanPerTile;

    // 当前tile的元信息
    uint32_t cur_p0 = 0;
    uint32_t cur_bcIdx = 0;
    uint32_t cur_outCount = 0;
    uint32_t cur_inSpan = 0;
    uint32_t cur_clampStart = 0;
};

extern "C" __global__ __aicore__ void max_pooling_1d_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMaxPool1D op;
    op.Init(x, y,
            tiling_data.totalInLength, tiling_data.totalOutLength,
            tiling_data.n, tiling_data.c,
            tiling_data.inLen, tiling_data.outLen,
            tiling_data.kernelSize, tiling_data.stride, tiling_data.padding,
            tiling_data.tileNum);
    op.Process();
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

static inline int64_t calc_out_len_1d(int64_t in_len, int64_t kernel_size, int64_t stride, int64_t padding) {
    if (stride <= 0) stride = kernel_size;
    const int64_t numer = in_len + 2 * padding - kernel_size;
    if (numer < 0) return 0;
    return numer / stride + 1;
}

at::Tensor max_pooling_1d_custom_impl_npu(const at::Tensor& x, int64_t kernel_size, int64_t stride, int64_t padding, bool return_indices) {
    TORCH_CHECK(x.dim() == 3, "Input must be 3D (N, C, L)");
    auto sizes = x.sizes();
    const int64_t N = sizes[0];
    const int64_t C = sizes[1];
    const int64_t L = sizes[2];
    if (stride <= 0) stride = kernel_size;

    const int64_t Lout = calc_out_len_1d(L, kernel_size, stride, padding);
    at::Tensor y = at::empty({N, C, Lout}, x.options());

    // 通过EXEC_NPU_CMD传递attrs与输出
    EXEC_NPU_CMD(aclnnMaxPooling1dCustom, x, (int)kernel_size, (int)stride, (int)padding, (bool)return_indices, y);
    return y;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("max_pooling_1d_custom", &max_pooling_1d_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("max_pooling_1d_custom", &max_pooling_1d_custom_impl_npu, "MaxPool1d custom (x, k, s, p, return_indices=false)");
}
