static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    // Input shapes:
    // q: [B, Lq, H, Dh]
    // k: [B, Lk, Hkv, Dh]
    // v: [B, Lk, Hkv, Dh]

        vShape->GetOriginShape().GetDimNum() != 4) {

    }

    uint32_t Lq = GetU32(qShape, 1);
    uint32_t H = GetU32(qShape, 2);

    // You can tune this per platform. Here we keep small default.

    if (blockDim == 0) blockDim = 1;

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "group_query_attention_custom_tiling_data.h"
#include "lib/matmul_intf.h"

// 使用文档中的 Matmul IterateBatch/IterateNBatch 与 SimpleSoftMax 相关范式（参考 CANN Ascend C API Reference）

using Q_T = float;
using K_T = float;
using V_T = float;
using O_T = float;
constexpr int32_t BUFFER_NUM = 1;

class KernelGQA {
public:
    __aicore__ inline KernelGQA() {}

    __aicore__ inline void Init(GM_ADDR q, GM_ADDR k, GM_ADDR v, GM_ADDR o,
                                GM_ADDR workspace, const optiling::GroupQueryAttentionCustomTilingData &t)
    {
        B = t.get_B();
        Lq = t.get_Lq();
        Lk = t.get_Lk();
        H = t.get_H();
        Hkv = t.get_Hkv();
        Dh = t.get_Dh();
        groupSize = t.get_groupSize();
        scale = t.get_scale();

        // 将 (b, hq) 维度映射到 block，block 内只处理一个 (b,hq)
        // blockIdx 映射:
        //  headGlobalIdx = blockIdx % H
        //  batchIdx      = blockIdx / H
        uint32_t blockIdx = AscendC::GetBlockIdx();
        headIdx = blockIdx % H;
        batchIdx = blockIdx / H;
        if (batchIdx >= B) {
            // 额外的 block 不参与计算
            valid = false;
            return;
        }
        valid = true;

        // GM 视图（按 [B, L, H or Hkv, Dh] 连续）
        qGm.SetGlobalBuffer((__gm__ Q_T*)q, (uint32_t)(B * Lq * H * Dh));
        kGm.SetGlobalBuffer((__gm__ K_T*)k, (uint32_t)(B * Lk * Hkv * Dh));
        vGm.SetGlobalBuffer((__gm__ V_T*)v, (uint32_t)(B * Lk * Hkv * Dh));
        oGm.SetGlobalBuffer((__gm__ O_T*)o, (uint32_t)(B * Lq * H * Dh));

        // 工作空间布局: [scores(Lq*Lk)*sizeof(float)] + [softmax_tmp]
        uint64_t scoreWsBytes = t.get_scoreWorkspacePerBlock();
        uint64_t softmaxTmpBytes = t.get_softmaxTmpWorkspacePerBlock();
        scoresGm.SetGlobalBuffer((__gm__ float*)workspace, (uint32_t)(scoreWsBytes / sizeof(float)));
        softmaxTmpGm = (GM_ADDR)((uint8_t*)workspace + scoreWsBytes);
        softmaxTmpBytes_ = softmaxTmpBytes;

        // 管线本地缓存
        pipe.InitBuffer(qTileQ, BUFFER_NUM, Dh * sizeof(Q_T));
        pipe.InitBuffer(kTileQ, BUFFER_NUM, Dh * sizeof(K_T));
        pipe.InitBuffer(vTileQ, BUFFER_NUM, Dh * sizeof(V_T));
        pipe.InitBuffer(oTileQ, BUFFER_NUM, Dh * sizeof(O_T));
    }

    __aicore__ inline void Process()
    {
        if (!valid) return;

        // 选择对应的 kv head
        uint32_t kvIdx = headIdx / groupSize;

        // 基于文档 5.2.1.24 IterateBatch/NBatch，构造 NORMAL 布局的两次 Matmul:
        // 1) scores = Q[b, :, hq, :] @ K[b, :, kv, :]^T  -> [Lq, Lk]
        // 2) out    = softmax(scores) @ V[b, :, kv, :]   -> [Lq, Dh]
        // 下面为简化实现示意：将 Q/K/V 的单头视图映射为 GM 子张量，然后用 AscendC::Matmul 执行。
        ComputeQK();
        SoftmaxScores();
        ComputeAV();
    }

private:
    __aicore__ inline __gm__ Q_T* GetQPtr(uint32_t b, uint32_t hq, uint32_t l = 0, uint32_t d = 0)
    {
        // base = (((b * Lq + 0) * H + hq) * Dh + 0)
        uint64_t base = (((uint64_t)b * Lq) * H + hq) * Dh;
        return reinterpret_cast<__gm__ Q_T*>(qGm.GetPhyAddr()) + base + (uint64_t)l * (uint64_t)(H * Dh) + d;
    }

    __aicore__ inline __gm__ K_T* GetKPtr(uint32_t b, uint32_t hk, uint32_t l = 0, uint32_t d = 0)
    {
        uint64_t base = (((uint64_t)b * Lk) * Hkv + hk) * Dh;
        return reinterpret_cast<__gm__ K_T*>(kGm.GetPhyAddr()) + base + (uint64_t)l * (uint64_t)(Hkv * Dh) + d;
    }

    __aicore__ inline __gm__ V_T* GetVPtr(uint32_t b, uint32_t hk, uint32_t l = 0, uint32_t d = 0)
    {
        uint64_t base = (((uint64_t)b * Lk) * Hkv + hk) * Dh;
        return reinterpret_cast<__gm__ V_T*>(vGm.GetPhyAddr()) + base + (uint64_t)l * (uint64_t)(Hkv * Dh) + d;
    }

    __aicore__ inline __gm__ O_T* GetOPtr(uint32_t b, uint32_t hq, uint32_t l = 0, uint32_t d = 0)
    {
        uint64_t base = (((uint64_t)b * Lq) * H + hq) * Dh;
        return reinterpret_cast<__gm__ O_T*>(oGm.GetPhyAddr()) + base + (uint64_t)l * (uint64_t)(H * Dh) + d;
    }

    __aicore__ inline void ComputeQK()
    {
        // 将 scores 写入 scoresGm，排列为 [Lq, Lk]
        // 这里演示按行计算：逐行读取 q(lq, :)，与整段 k(: , :) 做点积生成 scores 行
        for (uint32_t lqi = 0; lqi < Lq; ++lqi) {
            // load q row
            AscendC::LocalTensor<Q_T> qLocal = qTileQ.AllocTensor<Q_T>();
            AscendC::DataCopy(qLocal, GetQPtr(batchIdx, headIdx, lqi, 0), Dh);
            qTileQ.EnQue(qLocal);

            // 计算一行 scores
            AscendC::LocalTensor<Q_T> qRow = qTileQ.DeQue<Q_T>();

            // 临时输出到 scoresGm 的对应行
            for (uint32_t lkj = 0; lkj < Lk; ++lkj) {
                AscendC::LocalTensor<K_T> kLocal = kTileQ.AllocTensor<K_T>();
                AscendC::DataCopy(kLocal, GetKPtr(batchIdx, /*hk=*/(headIdx / groupSize), lkj, 0), Dh);

                // dot(qRow, kLocal)
                float acc = 0.0f;
                // 简单标量规约（示意）。如需高性能，应使用向量化与 Matmul 接口（参考文档 5.2.1.24）。
                for (uint32_t d = 0; d < Dh; ++d) {
                    acc += static_cast<float>(qRow.GetValue(d)) * static_cast<float>(kLocal.GetValue(d));
                }
                acc *= scale;

                // 写入 scores 到 GM: scores[lqi, lkj]
                scoresGm.SetValue(lqi * Lk + lkj, acc);

                kTileQ.FreeTensor(kLocal);
            }

            qTileQ.FreeTensor(qRow);
        }
    }

    __aicore__ inline void SoftmaxScores()
    {
        // 沿最后一维 Lk 做 softmax，输入/输出均为 scoresGm
        // 参考文档 5.3.1.9 SoftMax/SimpleSoftMax Tiling 的注意事项，简化实现为逐行归一化
        for (uint32_t lqi = 0; lqi < Lq; ++lqi) {
            // 求 max
            float maxv = -3.4e38f;
            for (uint32_t lkj = 0; lkj < Lk; ++lkj) {
                float v = scoresGm.GetValue(lqi * Lk + lkj);
                if (v > maxv) maxv = v;
            }
            // exp 并求和
            float sumv = 0.f;
            for (uint32_t lkj = 0; lkj < Lk; ++lkj) {
                float v = scoresGm.GetValue(lqi * Lk + lkj);
                float ev = AscendC::Exp(v - maxv);
                scoresGm.SetValue(lqi * Lk + lkj, ev);
                sumv += ev;
            }
            // 归一化
            float inv = 1.0f / sumv;
            for (uint32_t lkj = 0; lkj < Lk; ++lkj) {
                float ev = scoresGm.GetValue(lqi * Lk + lkj);
                scoresGm.SetValue(lqi * Lk + lkj, ev * inv);
            }
        }
    }

    __aicore__ inline void ComputeAV()
    {
        // out[lq, dh] = sum_{lk} scores[lq, lk] * V[lk, dh]
        for (uint32_t lqi = 0; lqi < Lq; ++lqi) {
            AscendC::LocalTensor<O_T> outLocal = oTileQ.AllocTensor<O_T>();
            // init 0
            for (uint32_t d = 0; d < Dh; ++d) {
                outLocal.SetValue(d, (O_T)0);
            }

            for (uint32_t lkj = 0; lkj < Lk; ++lkj) {
                float w = scoresGm.GetValue(lqi * Lk + lkj);
                AscendC::LocalTensor<V_T> vLocal = vTileQ.AllocTensor<V_T>();
                AscendC::DataCopy(vLocal, GetVPtr(batchIdx, /*hk=*/(headIdx / groupSize), lkj, 0), Dh);
                // out += w * v
                for (uint32_t d = 0; d < Dh; ++d) {
                    float prev = static_cast<float>(outLocal.GetValue(d));
                    float vv = static_cast<float>(vLocal.GetValue(d));
                    outLocal.SetValue(d, (O_T)(prev + w * vv));
                }
                vTileQ.FreeTensor(vLocal);
            }

            // 写回 GM
            AscendC::DataCopy(GetOPtr(batchIdx, headIdx, lqi, 0), outLocal, Dh);
            oTileQ.FreeTensor(outLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN,  BUFFER_NUM> qTileQ, kTileQ, vTileQ;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> oTileQ;

    AscendC::GlobalTensor<Q_T> qGm;
    AscendC::GlobalTensor<K_T> kGm;
    AscendC::GlobalTensor<V_T> vGm;
    AscendC::GlobalTensor<O_T> oGm;

    // scores buffer in GM workspace (size Lq*Lk)
    AscendC::GlobalTensor<float> scoresGm;
    GM_ADDR softmaxTmpGm {nullptr};
    uint64_t softmaxTmpBytes_ {0};

    uint32_t B{0}, Lq{0}, Lk{0}, H{0}, Hkv{0}, Dh{0}, groupSize{1};
    float scale{1.f};
    uint32_t headIdx{0}, batchIdx{0};
    bool valid{true};
};


extern "C" __global__ __aicore__ void group_query_attention_custom(GM_ADDR q, GM_ADDR k, GM_ADDR v,
                                                                   GM_ADDR o, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGQA op;
    op.Init(q, k, v, o, workspace, tiling_data);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import math
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    """
    使用自定义 AscendC 内核 group_query_attention_custom 融合实现 GQA 的核心注意力计算：
    输入给内核的张量为:
      - Q: [B, L, H, Dh]
      - K: [B, L, Hkv, Dh]
      - V: [B, L, Hkv, Dh]
    内核内部完成分组映射、QK^T/Softmax/AV 三步，并输出:
      - O: [B, L, H, Dh]
    其余线性层保留为 PyTorch 模块。
    """
    def __init__(self, d_model=1024, num_heads=16, num_kv_heads=4):
        super().__init__()
        assert num_heads % num_kv_heads == 0, "num_heads must be divisible by num_kv_heads"
        self.num_heads = num_heads
        self.num_kv_heads = num_kv_heads
        self.group_size = num_heads // num_kv_heads

        self.q_proj = nn.Linear(d_model, d_model)
        self.k_proj = nn.Linear(d_model, d_model * num_kv_heads // num_heads)
        self.v_proj = nn.Linear(d_model, d_model * num_kv_heads // num_heads)
        self.out_proj = nn.Linear(d_model, d_model)

    def forward(self, x):
        B, L, D = x.shape
        H, H_kv = self.num_heads, self.num_kv_heads
        head_dim = D // H

        # 生成 Q/K/V，并用 4D 视图匹配自定义内核的输入
        Q = self.q_proj(x).view(B, L, H, head_dim).contiguous()
        K = self.k_proj(x).view(B, L, H_kv, head_dim).contiguous()
        V = self.v_proj(x).view(B, L, H_kv, head_dim).contiguous()

        # 调用自定义算子（融合 GQA 注意力核心）
        O = custom_ops_lib.group_query_attention_custom(Q, K, V)  # [B, L, H, Dh]

        out = O.reshape(B, L, D)
        return self.out_proj(out)
'''
