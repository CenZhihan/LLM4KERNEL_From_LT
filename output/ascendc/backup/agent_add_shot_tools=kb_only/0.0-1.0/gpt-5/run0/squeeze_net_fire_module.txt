namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    // 输出形状为 [N, C1+C3, H, W]
    const gert::Shape* x1_shape = context->GetInputShape(0);
    const gert::Shape* x3_shape = context->GetInputShape(1);

    std::vector<int64_t> s0 = x1_shape->GetOriginShape().GetDims();
    std::vector<int64_t> s1 = x3_shape->GetOriginShape().GetDims();

        return GRAPH_FAILED;
    }
    std::vector<int64_t> out = {s0[0], s0[1] + s1[1], s0[2], s0[3]};
    y_shape->SetDimNum(4);
    for (size_t i = 0; i < 4; ++i) {
        y_shape->SetDim(i, out[i]);
    }
    return GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "squeeze_net_fire_module_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // 双缓冲

using DTYPE = float;

class KernelFireCatRelu {
public:
    __aicore__ inline KernelFireCatRelu() {}

    __aicore__ inline void Init(GM_ADDR x1, GM_ADDR x3, GM_ADDR y,
                                uint32_t n, uint32_t c1, uint32_t c3,
                                uint32_t h, uint32_t w, uint32_t tileNum)
    {
        this->n = n;
        this->c1 = c1;
        this->c3 = c3;
        this->h = h;
        this->w = w;
        this->tileNum = tileNum;

        this->hw = h * w;
        this->lenPerSample1 = c1 * this->hw;
        this->lenPerSample3 = c3 * this->hw;
        this->lenPerSampleY = (c1 + c3) * this->hw;

        x1Gm.SetGlobalBuffer((__gm__ DTYPE *)x1, static_cast<uint32_t>(n) * lenPerSample1);
        x3Gm.SetGlobalBuffer((__gm__ DTYPE *)x3, static_cast<uint32_t>(n) * lenPerSample3);
        yGm.SetGlobalBuffer((__gm__ DTYPE *)y,  static_cast<uint32_t>(n) * lenPerSampleY);

        // 简单按样本划分多核
        this->blockNum = AscendC::GetBlockNum();
        this->blockIdx = AscendC::GetBlockIdx();
        this->samplesPerBlock = (n + blockNum - 1) / blockNum;
        this->startSample = blockIdx * samplesPerBlock;
        this->endSample = (startSample + samplesPerBlock) > n ? n : (startSample + samplesPerBlock);

        // 计算每 tile 长度，保证至少为 1
        uint32_t baseTile = lenPerSampleY / (tileNum * BUFFER_NUM);
        this->tileLength = baseTile > 0 ? baseTile : 1;

        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(DTYPE));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t nIdx = startSample; nIdx < endSample; ++nIdx) {
            // 先处理 x1 分支：ReLU 后写到 y 的前半部分
            ProcessBranch(x1Gm, nIdx, lenPerSample1, 0);

            // 再处理 x3 分支：ReLU 后写到 y 的后半部分
            ProcessBranch(x3Gm, nIdx, lenPerSample3, lenPerSample1);
        }
    }

private:
    __aicore__ inline void ProcessBranch(AscendC::GlobalTensor<DTYPE> &inGm,
                                         uint32_t nIdx,
                                         uint32_t branchLen,
                                         uint32_t outOffsetBase)
    {
        uint32_t processed = 0;
        while (processed < branchLen) {
            uint32_t currTile = (processed + tileLength <= branchLen) ? tileLength : (branchLen - processed);

            // CopyIn
            AscendC::LocalTensor<DTYPE> inLocal = inQueueA.AllocTensor<DTYPE>();
            AscendC::DataCopy(inLocal, inGm[nIdx * branchLen + processed], currTile);
            inQueueA.EnQue(inLocal);

            // Compute (ReLU)
            AscendC::LocalTensor<DTYPE> inL = inQueueA.DeQue<DTYPE>();
            AscendC::LocalTensor<DTYPE> outL = outQueue.AllocTensor<DTYPE>();
            AscendC::Relu(outL, inL, currTile);
            outQueue.EnQue<DTYPE>(outL);
            inQueueA.FreeTensor(inL);

            // CopyOut 到输出的相应位置
            AscendC::LocalTensor<DTYPE> outLL = outQueue.DeQue<DTYPE>();
            AscendC::DataCopy(yGm[nIdx * lenPerSampleY + outOffsetBase + processed], outLL, currTile);
            outQueue.FreeTensor(outLL);

            processed += currTile;
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;

    AscendC::GlobalTensor<DTYPE> x1Gm;
    AscendC::GlobalTensor<DTYPE> x3Gm;
    AscendC::GlobalTensor<DTYPE> yGm;

    uint32_t n, c1, c3, h, w;
    uint32_t hw;
    uint32_t lenPerSample1, lenPerSample3, lenPerSampleY;
    uint32_t tileNum, tileLength;

    uint32_t blockNum, blockIdx;
    uint32_t samplesPerBlock, startSample, endSample;
};

extern "C" __global__ __aicore__ void squeeze_net_fire_module_custom(GM_ADDR x1, GM_ADDR x3, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFireCatRelu op;
    op.Init(x1, x3, y,
            tiling_data.n, tiling_data.c1, tiling_data.c3,
            tiling_data.h, tiling_data.w, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, squeeze_channels, expand1x1_channels, expand3x3_channels):
        super().__init__()
        self.squeeze = torch.nn.Conv2d(in_channels, squeeze_channels, kernel_size=1)
        self.squeeze_activation = torch.nn.ReLU(inplace=True)
        self.expand1x1 = torch.nn.Conv2d(squeeze_channels, expand1x1_channels, kernel_size=1)
        self.expand3x3 = torch.nn.Conv2d(squeeze_channels, expand3x3_channels, kernel_size=3, padding=1)

    def forward(self, x):
        x = self.squeeze_activation(self.squeeze(x))
        y1 = self.expand1x1(x)   # pre-activation
        y3 = self.expand3x3(x)   # pre-activation
        # 使用自定义 AscendC 算子完成两个分支的 ReLU 与通道拼接
        return custom_ops_lib.squeeze_net_fire_module_custom(y1, y3)
