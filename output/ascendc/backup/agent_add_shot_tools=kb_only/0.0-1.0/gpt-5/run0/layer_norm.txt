host_operator_src="""
#include "layer_norm_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    LayerNormCustomTilingData tiling;

    // x shape
    uint32_t totalLength = static_cast<uint32_t>(context->GetInputShape(0)->GetOriginShape().GetShapeSize());
    // gamma shape equals normalized_shape (flattened size = H)
    uint32_t hLength = static_cast<uint32_t>(context->GetInputShape(1)->GetOriginShape().GetShapeSize());
    if (hLength == 0) {
        return ge::GRAPH_FAILED;
    }
    uint32_t bLength = totalLength / hLength;

    context->SetBlockDim(BLOCK_DIM);
    tiling.set_totalLength(totalLength);
    tiling.set_tileNum(TILE_NUM);
    tiling.set_bLength(bLength);
    tiling.set_hLength(hLength);
    tiling.set_epsilon(1e-5f);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "layer_norm_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 1;

template <typename dataType>
class KernelLayerNorm {
public:
    __aicore__ inline KernelLayerNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                uint32_t bLength, uint32_t hLength, float epsilon)
    {
        this->bLength = bLength;
        this->hLength = hLength;
        this->bshLength = bLength * hLength;
        this->epsilon = static_cast<dataType>(epsilon);

        xGm.SetGlobalBuffer((__gm__ dataType *)x, this->bshLength);
        gammaGm.SetGlobalBuffer((__gm__ dataType *)gamma, this->hLength);
        betaGm.SetGlobalBuffer((__gm__ dataType *)beta, this->hLength);
        yGm.SetGlobalBuffer((__gm__ dataType *)y, this->bshLength);

        // Allocate queues: one-shot for simplicity
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->bshLength * sizeof(dataType));
        pipe.InitBuffer(inQueueGamma, BUFFER_NUM, this->hLength * sizeof(dataType));
        pipe.InitBuffer(inQueueBeta, BUFFER_NUM, this->hLength * sizeof(dataType));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->bshLength * sizeof(dataType));
        // Temporary buffer if needed by high-level API
        // Heuristic tmp size similar to RmsNorm reference
        uint32_t tmpBufferSize = this->bshLength * 2 + this->bLength;
        pipe.InitBuffer(tmpQueue, BUFFER_NUM, tmpBufferSize);
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<dataType> xLocal = inQueueX.AllocTensor<dataType>();
        AscendC::DataCopy(xLocal, xGm, this->bshLength);
        inQueueX.EnQue(xLocal);

        AscendC::LocalTensor<dataType> gammaLocal = inQueueGamma.AllocTensor<dataType>();
        AscendC::DataCopy(gammaLocal, gammaGm, this->hLength);
        inQueueGamma.EnQue(gammaLocal);

        AscendC::LocalTensor<dataType> betaLocal = inQueueBeta.AllocTensor<dataType>();
        AscendC::DataCopy(betaLocal, betaGm, this->hLength);
        inQueueBeta.EnQue(betaLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<dataType> xLocal = inQueueX.DeQue<dataType>();
        AscendC::LocalTensor<dataType> gammaLocal = inQueueGamma.DeQue<dataType>();
        AscendC::LocalTensor<dataType> betaLocal = inQueueBeta.DeQue<dataType>();
        AscendC::LocalTensor<dataType> yLocal = outQueueY.AllocTensor<dataType>();
        AscendC::LocalTensor<uint8_t> stackBuffer = tmpQueue.AllocTensor<uint8_t>();

        // Use high-level LayerNorm API as per 5.4.1 LayerNorm reference style
        // Normalize input of shape [B, H] where H equals product of normalized_shape
        // The API is expected to internally handle mean/variance and use shared buffer.
        AscendC::LayerNorm<dataType>(yLocal, xLocal, gammaLocal, betaLocal, this->epsilon, stackBuffer,
                                     /*bLength*/ this->bLength, /*hLength*/ this->hLength);

        inQueueX.FreeTensor(xLocal);
        inQueueGamma.FreeTensor(gammaLocal);
        inQueueBeta.FreeTensor(betaLocal);
        tmpQueue.FreeTensor(stackBuffer);
        outQueueY.EnQue(yLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<dataType> yLocal = outQueueY.DeQue<dataType>();
        AscendC::DataCopy(yGm, yLocal, this->bshLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueGamma;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueBeta;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::QuePosition::VECCALC, BUFFER_NUM> tmpQueue;

    AscendC::GlobalTensor<dataType> xGm;
    AscendC::GlobalTensor<dataType> gammaGm;
    AscendC::GlobalTensor<dataType> betaGm;
    AscendC::GlobalTensor<dataType> yGm;

    uint32_t bLength;
    uint32_t hLength;
    uint32_t bshLength;
    dataType epsilon;
};

extern "C" __global__ __aicore__ void layer_norm_custom(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelLayerNorm<DTYPE_X> op;
    op.Init(x, gamma, beta, y,
            tiling_data.bLength,
            tiling_data.hLength,
            tiling_data.epsilon);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model using custom AscendC LayerNorm kernel.
    """
    def __init__(self, normalized_shape: tuple, eps: float = 1e-5) -> None:
        super().__init__()
        self.normalized_shape = normalized_shape
        self.eps = eps
        # Mimic nn.LayerNorm parameterization
        weight = torch.ones(normalized_shape, dtype=torch.float32)
        bias = torch.zeros(normalized_shape, dtype=torch.float32)
        self.weight = torch.nn.Parameter(weight)
        self.bias = torch.nn.Parameter(bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Custom kernel expects gamma/beta shaped as normalized_shape
        return custom_ops_lib.layer_norm_custom(x, self.weight, self.bias)
'''
