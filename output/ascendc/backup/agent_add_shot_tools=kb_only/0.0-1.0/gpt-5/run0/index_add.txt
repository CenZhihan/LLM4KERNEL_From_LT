kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "index_add_custom_tiling_data.h"

// 参考文档：
// - AscendC::InitGlobalMemory: 在 Init 中清零 GM 输出缓冲区，且必须在 InitBuffer 之前调用。
// - AscendC::SetAtomicAdd<float>: 使用 VECOUT/本地到 GM 的原子累加保障并发写正确性。

constexpr int32_t BUFFER_NUM = 2;

class KernelIndexAdd {
public:
    __aicore__ inline KernelIndexAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR indices, GM_ADDR values, GM_ADDR z,
                                uint32_t nRows, uint32_t nCols, uint32_t nUpdates,
                                uint32_t tileNumX, uint32_t tileNumK)
    {
        this->nRows = nRows;
        this->nCols = nCols;
        this->nUpdates = nUpdates;
        this->tileNumX = tileNumX;
        this->tileNumK = tileNumK;

        uint32_t totalX = nRows * nCols;
        uint32_t totalV = nUpdates * nCols;

        // 全局缓冲设置：z 用于写，x/values/indices 用于读
        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(x), totalX);
        zGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(z), totalX);
        valuesGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(values), totalV);
        indicesGm.SetGlobalBuffer(reinterpret_cast<__gm__ int32_t*>(indices), nUpdates);

        // 先将输出 z 清零，确保原子累加的起点正确
        AscendC::InitGlobalMemory(zGm, totalX, 0.0f);

        // 为分块拷贝/计算初始化本地队列缓冲
        // Phase A: 将 x 累加到 z（并行对 totalX 分块）
        this->blockXLength = totalX / AscendC::GetBlockNum();
        this->blockXStart = this->blockXLength * AscendC::GetBlockIdx();
        this->tileXLength = this->blockXLength / (this->tileNumX * BUFFER_NUM);
        if (this->tileXLength == 0) { this->tileXLength = 1; }

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileXLength * sizeof(float));

        // Phase B: 处理 updates（并行对 nUpdates 分块）
        this->blockKLength = nUpdates / AscendC::GetBlockNum();
        this->blockKStart = this->blockKLength * AscendC::GetBlockIdx();
        if (this->blockKLength == 0 && AscendC::GetBlockIdx() == 0) {
            // 将余数留给第0个block处理（简单处理）
            this->blockKLength = nUpdates;
            this->blockKStart = 0;
        }

        // 每次处理一行 values 与一个 index，避免 UB 压力
        pipe.InitBuffer(inQueueIdx, 1, sizeof(int32_t));
        pipe.InitBuffer(inQueueValues, 1, this->nCols * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AddBaseXToZ();
        ScatterAddValues();
    }

private:
    __aicore__ inline void AddBaseXToZ()
    {
        int32_t loopCount = (this->tileNumX * BUFFER_NUM);
        for (int32_t i = 0; i < loopCount; ++i) {
            uint32_t offset = this->blockXStart + i * this->tileXLength;
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[offset], this->tileXLength);

            // 使用原子累加将 x 局部块加到 z 上，保证多核/多迭代安全
            AscendC::SetAtomicAdd<float>();
            AscendC::DataCopy(zGm[offset], xLocal, this->tileXLength);
            AscendC::SetAtomicNone();

            inQueueX.FreeTensor(xLocal);
        }
    }

    __aicore__ inline void ScatterAddValues()
    {
        for (uint32_t k = 0; k < this->blockKLength; ++k) {
            uint32_t gk = this->blockKStart + k;

            // 读取单个 index
            AscendC::LocalTensor<int32_t> idxLocal = inQueueIdx.AllocTensor<int32_t>();
            AscendC::DataCopy(idxLocal, indicesGm[gk], 1);
            inQueueIdx.EnQue(idxLocal);
            AscendC::LocalTensor<int32_t> idxLocalUse = inQueueIdx.DeQue<int32_t>();
            // 假定可从 LocalTensor 读取标量（具体接口以实际 AscendC 版本为准）
            int32_t rowIdx = idxLocalUse.GetValue(0);
            inQueueIdx.FreeTensor(idxLocalUse);

            // 读取对应的一整行 values
            AscendC::LocalTensor<float> valLocal = inQueueValues.AllocTensor<float>();
            AscendC::DataCopy(valLocal, valuesGm[gk * this->nCols], this->nCols);

            // 原子累加到 z 的指定行
            uint32_t rowOffset = static_cast<uint32_t>(rowIdx) * this->nCols;
            AscendC::SetAtomicAdd<float>();
            AscendC::DataCopy(zGm[rowOffset], valLocal, this->nCols);
            AscendC::SetAtomicNone();

            inQueueValues.FreeTensor(valLocal);
        }
    }

private:
    AscendC::TPipe pipe;

    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueIdx;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueValues;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    AscendC::GlobalTensor<float> valuesGm;
    AscendC::GlobalTensor<int32_t> indicesGm;

    uint32_t nRows;
    uint32_t nCols;
    uint32_t nUpdates;

    uint32_t tileNumX;
    uint32_t tileNumK;

    uint32_t blockXStart;
    uint32_t blockXLength;
    uint32_t tileXLength;

    uint32_t blockKStart;
    uint32_t blockKLength;
};

extern "C" __global__ __aicore__ void index_add_custom(GM_ADDR x, GM_ADDR indices, GM_ADDR values, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelIndexAdd op;
    op.Init(x, indices, values, z,
            tiling_data.nRows, tiling_data.nCols, tiling_data.nUpdates,
            tiling_data.tileNumX, tiling_data.tileNumK);
    op.Process();
}
