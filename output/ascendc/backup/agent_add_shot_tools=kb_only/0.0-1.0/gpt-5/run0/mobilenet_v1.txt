kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "mobilenet_v1_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // queue depth

class KernelMobilenetV1 {
public:
    __aicore__ inline KernelMobilenetV1() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        // Element-wise ReLU using basic vector API
        AscendC::Relu(yLocal, xLocal, this->tileLength);
        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void mobilenet_v1_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMobilenetV1 op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}
"""

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, num_classes=1000, input_channels=3, alpha=1.0):
        super().__init__()
        self.num_classes = num_classes
        self.input_channels = input_channels
        self.alpha = alpha

        def conv_bn(inp, oup, stride):
            # remove ReLU module and call custom op in forward
            return torch.nn.Sequential(
                torch.nn.Conv2d(inp, oup, 3, stride, 1, bias=False),
                torch.nn.BatchNorm2d(oup),
            )

        def conv_dw(inp, oup, stride):
            # remove ReLU modules and call custom op in forward after each BN
            return torch.nn.Sequential(
                torch.nn.Conv2d(inp, inp, 3, stride, 1, groups=inp, bias=False),
                torch.nn.BatchNorm2d(inp),

                torch.nn.Conv2d(inp, oup, 1, 1, 0, bias=False),
                torch.nn.BatchNorm2d(oup),
            )

        a = self.alpha
        self.layer0 = conv_bn(input_channels, int(32 * a), 2)
        self.layer1 = conv_dw(int(32 * a), int(64 * a), 1)
        self.layer2 = conv_dw(int(64 * a), int(128 * a), 2)
        self.layer3 = conv_dw(int(128 * a), int(128 * a), 1)
        self.layer4 = conv_dw(int(128 * a), int(256 * a), 2)
        self.layer5 = conv_dw(int(256 * a), int(256 * a), 1)
        self.layer6 = conv_dw(int(256 * a), int(512 * a), 2)
        self.layer7 = conv_dw(int(512 * a), int(512 * a), 1)
        self.layer8 = conv_dw(int(512 * a), int(512 * a), 1)
        self.layer9 = conv_dw(int(512 * a), int(512 * a), 1)
        self.layer10 = conv_dw(int(512 * a), int(512 * a), 1)
        self.layer11 = conv_dw(int(512 * a), int(512 * a), 1)
        self.layer12 = conv_dw(int(512 * a), int(1024 * a), 2)
        self.layer13 = conv_dw(int(1024 * a), int(1024 * a), 1)
        self.avgpool = torch.nn.AvgPool2d(7)
        self.fc = torch.nn.Linear(int(1024 * a), self.num_classes)

    def act(self, x):
        # custom AscendC ReLU
        return custom_ops_lib.mobilenet_v1_custom(x)

    def forward_block_no_relu(self, block, x):
        # helper for blocks defined as Sequential without ReLU
        # For conv_bn: [conv, bn]
        if len(block) == 2:
            x = block[0](x)
            x = block[1](x)
            x = self.act(x)
            return x
        # For conv_dw: [dw_conv, dw_bn, pw_conv, pw_bn]
        elif len(block) == 4:
            x = block[0](x)
            x = block[1](x)
            x = self.act(x)
            x = block[2](x)
            x = block[3](x)
            x = self.act(x)
            return x
        else:
            # fallback
            for m in block:
                x = m(x)
            return x

    def forward(self, x):
        x = self.forward_block_no_relu(self.layer0, x)
        x = self.forward_block_no_relu(self.layer1, x)
        x = self.forward_block_no_relu(self.layer2, x)
        x = self.forward_block_no_relu(self.layer3, x)
        x = self.forward_block_no_relu(self.layer4, x)
        x = self.forward_block_no_relu(self.layer5, x)
        x = self.forward_block_no_relu(self.layer6, x)
        x = self.forward_block_no_relu(self.layer7, x)
        x = self.forward_block_no_relu(self.layer8, x)
        x = self.forward_block_no_relu(self.layer9, x)
        x = self.forward_block_no_relu(self.layer10, x)
        x = self.forward_block_no_relu(self.layer11, x)
        x = self.forward_block_no_relu(self.layer12, x)
        x = self.forward_block_no_relu(self.layer13, x)
        x = self.avgpool(x)
        x = x.view(x.size(0), -1)
        x = self.fc(x)
        return x
'''
