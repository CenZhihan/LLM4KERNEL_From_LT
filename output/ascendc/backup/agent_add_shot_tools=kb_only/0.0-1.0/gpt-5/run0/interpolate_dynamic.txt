#include "interpolate_dynamic_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    InterpolateDynamicCustomTilingData tiling;

    // 获取输入形状 NCHW
    const gert::Shape* xShape = context->GetInputShape(0);
    auto dims = xShape->GetOriginShape().GetDims();
    // 兼容 ND：不足4维时按[N, C, H, W]回填
    uint64_t n = dims.size() > 0 ? dims[0] : 1;
    uint64_t c = dims.size() > 1 ? dims[1] : 1;
    uint64_t hIn = dims.size() > 2 ? dims[2] : 1;
    uint64_t wIn = dims.size() > 3 ? dims[3] : 1;

    // 读取属性 out_h / out_w / align_corners
    int64_t outH = 0;
    int64_t outW = 0;
    bool alignCorners = false;
    (void)context->GetAttr("out_h", outH);
    (void)context->GetAttr("out_w", outW);
    (void)context->GetAttr("align_corners", alignCorners);

    // 设置BlockDim
    context->SetBlockDim(BLOCK_DIM);

    tiling.set_n(static_cast<uint32_t>(n));
    tiling.set_c(static_cast<uint32_t>(c));
    tiling.set_hIn(static_cast<uint32_t>(hIn));
    tiling.set_wIn(static_cast<uint32_t>(wIn));
    tiling.set_hOut(static_cast<uint32_t>(outH));
    tiling.set_wOut(static_cast<uint32_t>(outW));
    tiling.set_totalLengthOut(static_cast<uint32_t>(n * c * outH * outW));
    tiling.set_tileNum(TILE_NUM);
    tiling.set_alignCorners(alignCorners ? 1U : 0U);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t* workspaces = context->GetWorkspaceSizes(1);
    workspaces[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    // 输出形状由输入 N、C 与属性 out_h/out_w 决定
    const gert::Shape* xShape = context->GetInputShape(0);
    int64_t outH = 0, outW = 0;
    bool alignCorners = false;
    (void)context->GetAttr("out_h", outH);
    (void)context->GetAttr("out_w", outW);
    (void)context->GetAttr("align_corners", alignCorners);

    auto inDims = xShape->GetOriginShape().GetDims();
    int64_t n = inDims.size() > 0 ? inDims[0] : 1;
    int64_t c = inDims.size() > 1 ? inDims[1] : 1;

    gert::Shape* yShape = context->GetOutputShape(0);
    gert::Shape out;
    out.SetDimNum(4);
    out.SetDim(0, n);
    out.SetDim(1, c);
    out.SetDim(2, outH);
    out.SetDim(3, outW);
    *yShape = out;
    return GRAPH_SUCCESS;
}

static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

namespace ops {
class InterpolateDynamicCustom : public OpDef {
public:
    explicit InterpolateDynamicCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)

            .Format({ge::FORMAT_NCHW, ge::FORMAT_ND})
            .UnknownShapeFormat({ge::FORMAT_NCHW, ge::FORMAT_ND});

        this->Attr("out_h").Int(1);
        this->Attr("out_w").Int(1);
        this->Attr("align_corners").Bool(false);

        this->AICore().SetTiling(optiling::TilingFunc);
        this->AICore().AddConfig("ascend910b");
    }
};

OP_ADD(InterpolateDynamicCustom);
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "interpolate_dynamic_custom_tiling_data.h"

// 说明：本示例基于AscendC ISASI中的BilinearInterpolation能力（参考Ascend C API Reference 1: ISASI APIs）。
// 为了演示结构，采用一次性拷入/拷出的简单流程。实际工程中请结合TPipe/TQue进行分块(tiling)处理，降低UB占用。

class KernelInterpolateDynamic {
public:
    __aicore__ inline KernelInterpolateDynamic() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t n, uint32_t c,
                                uint32_t hIn, uint32_t wIn,
                                uint32_t hOut, uint32_t wOut,
                                uint32_t tileNum, bool alignCorners)
    {
        this->n = n;
        this->c = c;
        this->hIn = hIn;
        this->wIn = wIn;
        this->hOut = hOut;
        this->wOut = wOut;
        this->tileNum = tileNum;
        this->alignCorners = alignCorners;

        // 以Block划分输出元素范围
        this->outElem = static_cast<uint64_t>(n) * c * hOut * wOut;
        this->inElem  = static_cast<uint64_t>(n) * c * hIn  * wIn;
        this->blockOutElem = outElem / AscendC::GetBlockNum();
        this->blockInElem  = inElem  / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(x) + this->blockInElem * AscendC::GetBlockIdx(), this->blockInElem);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(y) + this->blockOutElem * AscendC::GetBlockIdx(), this->blockOutElem);

        pipe.InitBuffer(inQueueX, 1, this->blockInElem * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, this->blockOutElem * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm, this->blockInElem);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // 这里调用ISASI BilinearInterpolation以完成双线性插值（垂直+水平迭代）
        // 实际接口参数以CANN发行版为准，本处示意用法：
        // AscendC::BilinearInterpolation(dst, src, n, c, hIn, wIn, hOut, wOut, alignCorners);
        AscendC::BilinearInterpolation(yLocal, xLocal,
                                       this->n, this->c,
                                       this->hIn, this->wIn,
                                       this->hOut, this->wOut,
                                       this->alignCorners);

        outQueueY.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm, yLocal, this->blockOutElem);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t n, c, hIn, wIn, hOut, wOut;
    uint32_t tileNum;
    bool alignCorners;
    uint64_t inElem, outElem;
    uint64_t blockInElem, blockOutElem;
};

extern "C" __global__ __aicore__ void interpolate_dynamic_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelInterpolateDynamic op;
    bool alignCorners = (tilingData.alignCorners != 0);
    op.Init(x, y,
            tilingData.n, tilingData.c,
            tilingData.hIn, tilingData.wIn,
            tilingData.hOut, tilingData.wOut,
            tilingData.tileNum, alignCorners);
    op.Process();
}
