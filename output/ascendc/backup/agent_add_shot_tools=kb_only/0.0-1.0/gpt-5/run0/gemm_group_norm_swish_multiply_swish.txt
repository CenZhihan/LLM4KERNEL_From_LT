kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "gemm_group_norm_swish_multiply_swish_custom_tiling_data.h"

constexpr int32_t BUFFER_NUM = 2; // double buffer

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR z, uint32_t totalLength, uint32_t tileNum, uint32_t featureLength)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum(); // per-block length (B_local * F)
        this->featureLength = featureLength;                      // F
        this->tileNum = tileNum;                                  // tiles along feature axis
        this->tileLength = this->featureLength / this->tileNum;   // feature split per tile
        this->rows = this->blockLength / this->featureLength;     // B_local

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(x) + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        wGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(w), this->featureLength);
        zGm.SetGlobalBuffer(reinterpret_cast<__gm__ float *>(z) + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // iterate over rows (batch) and feature tiles
        for (uint32_t r = 0; r < this->rows; ++r) {
            for (uint32_t t = 0; t < this->tileNum; ++t) {
                uint32_t progress = r * this->tileNum + t;
                CopyIn(progress);
                Compute(progress);
                CopyOut(progress);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        uint32_t rowIdx = progress / this->tileNum;
        uint32_t tileIdx = progress % this->tileNum;

        uint32_t xOffset = rowIdx * this->featureLength + tileIdx * this->tileLength;
        uint32_t wOffset = tileIdx * this->tileLength;

        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();

        AscendC::DataCopy(xLocal, xGm[xOffset], this->tileLength);
        AscendC::DataCopy(wLocal, wGm[wOffset], this->tileLength);

        inQueueX.EnQue(xLocal);
        inQueueW.EnQue(wLocal);
    }

    __aicore__ inline void Compute(uint32_t /*progress*/)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> wLocal = inQueueW.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        // z = Swish(x) = x * sigmoid(x)
        AscendC::Swish(zLocal, xLocal, this->tileLength);
        // z = z * w (channel-wise weight, broadcast along batch)
        AscendC::Mul(zLocal, zLocal, wLocal, this->tileLength);
        // z = Swish(z)
        AscendC::Swish(zLocal, zLocal, this->tileLength);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueW.FreeTensor(wLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        uint32_t rowIdx = progress / this->tileNum;
        uint32_t tileIdx = progress % this->tileNum;

        uint32_t zOffset = rowIdx * this->featureLength + tileIdx * this->tileLength;

        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[zOffset], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueueW;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueueZ;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t blockLength;
    uint32_t featureLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t rows;
};

extern "C" __global__ __aicore__ void gemm_group_norm_swish_multiply_swish_custom(GM_ADDR x, GM_ADDR w, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelFused op;
    op.Init(x, w, z, tilingData.totalLength, tilingData.tileNum, tilingData.featureLength);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    GEMM + GroupNorm by PyTorch, then fused Swish * Multiply * Swish by custom AscendC kernel.
    """
    def __init__(self, in_features, out_features, num_groups, multiply_weight_shape):
        super(ModelNew, self).__init__()
        self.gemm = torch.nn.Linear(in_features, out_features)
        self.group_norm = torch.nn.GroupNorm(num_groups, out_features)
        self.multiply_weight = torch.nn.Parameter(torch.randn(multiply_weight_shape))

    def forward(self, x):
        x = self.gemm(x)                 # (B, F)
        x = self.group_norm(x)           # (B, F)
        x = custom_ops_lib.gemm_group_norm_swish_multiply_swish_custom(x, self.multiply_weight)  # fused elementwise
        return x
