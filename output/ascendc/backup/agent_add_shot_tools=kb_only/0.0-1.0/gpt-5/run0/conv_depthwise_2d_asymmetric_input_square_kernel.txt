kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv_depthwise_2d_asymmetric_input_square_kernel_custom_tiling_data.h"

// 使用 AscendC 基本 API 与数据搬移 API，参见知识库：DataCopy、Add、Mul、Reduce、TPipe/TQue 等。
// 本 kernel 实现按通道（groups=in_channels）的 depthwise 2D 卷积（NCHW, float）。

using DTYPE = float;
#define DTYPE_X DTYPE
#define DTYPE_W DTYPE
#define DTYPE_B DTYPE
#define DTYPE_Y DTYPE

constexpr int32_t BUFFER_NUM = 1;

class KernelDwConv2D {
public:
    __aicore__ inline KernelDwConv2D() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
                                uint32_t n, uint32_t c,
                                uint32_t hIn, uint32_t wIn,
                                uint32_t hOut, uint32_t wOut,
                                uint32_t k, uint32_t stride, uint32_t padding,
                                uint32_t totalLengthOut, uint32_t tileNum)
    {
        this->n = n; this->c = c;
        this->hIn = hIn; this->wIn = wIn;
        this->hOut = hOut; this->wOut = wOut;
        this->k = k; this->stride = stride; this->padding = padding;
        this->totalLengthOut = totalLengthOut;
        this->tileNum = tileNum;
        this->blockLengthOut = totalLengthOut / AscendC::GetBlockNum();
        this->tileLength = (this->blockLengthOut + tileNum * BUFFER_NUM - 1) / (tileNum * BUFFER_NUM); // 向上取整
        // 对齐到 32B，避免尾部越界时使用 mask
        // 简化实现：逐元素处理时不强制对齐

        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x, n * c * hIn * wIn);
        wGm.SetGlobalBuffer((__gm__ DTYPE_W*)w, c * k * k);          // w: [C,1,K,K]
        if (b != 0) {
            bGm.SetGlobalBuffer((__gm__ DTYPE_B*)b, c);             // b: [C]
            hasBias = true;
        } else {
            hasBias = false;
        }
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y, n * c * hOut * wOut);

        // 分配队列缓冲
        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(DTYPE_X));  // 仅占位，不直接批量搬移使用
        pipe.InitBuffer(inQueueW, 1, k * k * sizeof(DTYPE_W));
        pipe.InitBuffer(inQueueB, 1, sizeof(DTYPE_B));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        const uint32_t loops = (this->blockLengthOut + this->tileLength - 1) / this->tileLength;
        for (uint32_t i = 0; i < loops; ++i) {
            ComputeTile(i);
        }
    }

private:
    // 线性下标 -> (n, c, ho, wo)
    __aicore__ inline void LinearToNCHW(uint32_t linear, uint32_t &on, uint32_t &oc, uint32_t &oh, uint32_t &ow) {
        uint32_t perC = hOut * wOut;
        uint32_t perN = c * perC;
        on = linear / perN;
        uint32_t rem1 = linear % perN;
        oc = rem1 / perC;
        uint32_t rem2 = rem1 % perC;
        oh = rem2 / wOut;
        ow = rem2 % wOut;
    }

    __aicore__ inline void ComputeTile(uint32_t tileIdx)
    {
        uint32_t start = this->blockBaseStart() + tileIdx * this->tileLength;
        uint32_t end = AscendC::min(start + this->tileLength, this->blockBaseStart() + this->blockLengthOut);
        if (start >= end) { return; }

        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        // 为了便于逐元素写回，先清零
        AscendC::Duplicate(yLocal, (DTYPE_Y)0, (end - start));

        // 临时缓冲：权重、输入补丁、乘积
        AscendC::LocalTensor<DTYPE_W> wLocal = inQueueW.AllocTensor<DTYPE_W>();
        AscendC::LocalTensor<DTYPE_X> patchLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> prodLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();

        for (uint32_t idx = start; idx < end; ++idx) {
            uint32_t on=0, oc=0, oh=0, ow=0;
            LinearToNCHW(idx, on, oc, oh, ow);

            // 加载单通道的权重 [K*K]
            {
                uint32_t wOffset = oc * k * k;
                AscendC::DataCopy(wLocal, wGm[wOffset], k * k);
            }
            // 可选 bias
            DTYPE_B biasVal = (DTYPE_B)0;
            if (hasBias) {
                AscendC::DataCopy(bLocal, bGm[oc], 1);
                // bLocal[0] 取值：此处使用 Duplicate 写回
                // 为兼容 API，我们将 biasVal 赋值为 0，然后用 Add 加到输出
                // 在最后输出写回时再用 Adds 完成
            }

            // 构造输入补丁 patchLocal[K*K]
            // 补丁左上角坐标
            int32_t in_h0 = static_cast<int32_t>(oh * stride) - static_cast<int32_t>(padding);
            int32_t in_w0 = static_cast<int32_t>(ow * stride) - static_cast<int32_t>(padding);

            // 清零补丁，越界部分视为 0
            AscendC::Duplicate(patchLocal, (DTYPE_X)0, k * k);

            for (uint32_t kh = 0; kh < k; ++kh) {
                int32_t ih = in_h0 + static_cast<int32_t>(kh);
                if (ih < 0 || ih >= static_cast<int32_t>(hIn)) {
                    continue;
                }
                // 连续拷贝一行的有效段
                uint32_t rowBase = kh * k;

                // 计算本行有效起止列
                int32_t iwStart = in_w0;
                int32_t iwEnd = in_w0 + static_cast<int32_t>(k); // [iwStart, iwEnd)
                int32_t validStart = AscendC::max(0, iwStart);
                int32_t validEnd = AscendC::min(static_cast<int32_t>(wIn), iwEnd);

                if (validStart < validEnd) {
                    uint32_t copyLen = static_cast<uint32_t>(validEnd - validStart);
                    // 计算输入全局偏移：((on*C + oc)*H + ih)*W + validStart
                    uint32_t xOffset = (((on * c + oc) * hIn) + static_cast<uint32_t>(ih)) * wIn + static_cast<uint32_t>(validStart);
                    // 拷贝到补丁的 [rowBase + (validStart - iwStart), ...]
                    uint32_t patchColStart = static_cast<uint32_t>(validStart - iwStart);
                    AscendC::DataCopy(patchLocal[ rowBase + patchColStart ], xGm[xOffset], copyLen);
                }
            }

            // 逐元素乘积
            AscendC::Mul(prodLocal, patchLocal, wLocal, k * k);
            // 对乘积求和，得到单个标量结果
            // 使用 ReduceSum 将 prodLocal 所有元素求和到 prodLocal[0]（约定 API 支持 calCount）
            AscendC::ReduceSum(prodLocal, prodLocal, k * k);

            // 将单元素结果写到 yLocal 对应位置
            uint32_t localPos = idx - start;
            AscendC::DataCopy(yLocal[localPos], prodLocal, 1);
            // 加 bias
            if (hasBias) {
                AscendC::Adds(yLocal[localPos], yLocal[localPos], bLocal, 1);
            }
        }

        inQueueW.FreeTensor(wLocal);
        inQueueX.FreeTensor(patchLocal);
        inQueueX.FreeTensor(prodLocal);
        inQueueB.FreeTensor(bLocal);

        outQueueY.EnQue<DTYPE_Y>(yLocal);

        // 写回
        AscendC::LocalTensor<DTYPE_Y> yOutLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[start], yOutLocal, (end - start));
        outQueueY.FreeTensor(yOutLocal);
    }

    __aicore__ inline uint32_t blockBaseStart() const {
        return this->blockLengthOut * AscendC::GetBlockIdx();
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_W> wGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t n, c, hIn, wIn, hOut, wOut, k, stride, padding;
    uint32_t totalLengthOut, tileNum, blockLengthOut, tileLength;
    bool hasBias{false};
};

extern "C" __global__ __aicore__ void conv_depthwise_2d_asymmetric_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelDwConv2D op;
    op.Init(
        x, w, b, y,
        tilingData.n(), tilingData.c(),
        tilingData.hIn(), tilingData.wIn(),
        tilingData.hOut(), tilingData.wOut(),
        tilingData.k(), tilingData.stride(), tilingData.padding(),
        tilingData.totalLengthOut(), tilingData.tileNum()
    );
    op.Process();
}
