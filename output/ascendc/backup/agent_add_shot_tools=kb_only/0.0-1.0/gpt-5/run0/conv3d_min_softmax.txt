host_operator_src="""
#include "conv3d_min_softmax_custom_tiling.h"
#include "register/op_def_registry.h"
#include "tiling/softmax_tiling.h"

namespace optiling {
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    Conv3dMinSoftmaxCustomTilingData tiling;

    // Input tensor after min reduction is expected to be [B, C, H, W].
    const gert::Shape* inShape = context->GetInputShape(0);
    auto dims = inShape->GetStorageShape().GetDims();
    if (dims.size() < 2) {
        return ge::GRAPH_FAILED;
    }
    // Treat last axis as reduction axis for SoftMax (channel-last).
    // ModelNew will permute to NHWC so last axis is C.
    uint32_t m = 1;
    for (size_t i = 0; i + 1 < dims.size(); ++i) {
        m *= static_cast<uint32_t>(dims[i]);
    }
    uint32_t n = static_cast<uint32_t>(dims.back());
    uint32_t srcSize = m * n;

    // SoftMax tiling and tmp space
    std::vector<int64_t> shapeVec = {static_cast<int64_t>(m), static_cast<int64_t>(n)};
    ge::Shape srcShape(shapeVec);

    const uint32_t dtypeSize = sizeof(float);
    // Get minimum temporary space size required by SoftMax
    uint32_t localWorkSpaceSize = AscendC::GetSoftMaxMinTmpSize(srcShape, dtypeSize, false);

    // Get tiling parameters for SoftMax
    AscendC::SoftMaxTilingFunc(srcShape, dtypeSize, localWorkSpaceSize, tiling.softmaxTilingData);

    // Save tiling data
    tiling.set_m(m);
    tiling.set_n(n);
    tiling.set_srcSize(srcSize);
    tiling.set_tmpBufSize(localWorkSpaceSize);

    // Block dim can be set according to model; use 8 as a common choice
    context->SetBlockDim(8);

    // Workspace (not used in kernel; using sharedTmpBuffer from UB)
    size_t *workspaceSizes = context->GetWorkspaceSizes(1);
    workspaceSizes[0] = 0;

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());
    context->SetTilingKey(1);
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape; // softmax does not change shape
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv3d_min_softmax_custom_tiling_data.h"

using DTYPE = float;
constexpr int32_t BUFFER_NUM = 1;

class KernelConv3dMinSoftmax {
public:
    __aicore__ inline KernelConv3dMinSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t srcSize, uint32_t tmpBufSize,
                                const SoftMaxTiling &softmaxTiling)
    {
        this->dataSize = srcSize;
        this->tmpBufSize = tmpBufSize;
        this->softmaxTiling = softmaxTiling;

        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(x), dataSize);
        yGm.SetGlobalBuffer(reinterpret_cast<__gm__ DTYPE *>(y), dataSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, dataSize * sizeof(DTYPE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, dataSize * sizeof(DTYPE));
        if (tmpBufSize > 0) {
            pipe.InitBuffer(tmpUb, tmpBufSize); // tmp size is in bytes
        }
    }
    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.AllocTensor<DTYPE>();
        AscendC::DataCopy(xLocal, xGm, dataSize);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE> xLocal = inQueueX.DeQue<DTYPE>();
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.AllocTensor<DTYPE>();
        AscendC::LocalTensor<uint8_t> tmp;
        if (tmpBufSize > 0) {
            tmp = tmpUb.Get<uint8_t>();
            AscendC::SoftMax<DTYPE>(yLocal, xLocal, tmp, softmaxTiling);
        } else {
            AscendC::SoftMax<DTYPE>(yLocal, xLocal, softmaxTiling);
        }
        outQueueY.EnQue<DTYPE>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<DTYPE> yLocal = outQueueY.DeQue<DTYPE>();
        AscendC::DataCopy(yGm, yLocal, dataSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpUb;
    AscendC::GlobalTensor<DTYPE> xGm;
    AscendC::GlobalTensor<DTYPE> yGm;
    uint32_t dataSize = 0;
    uint32_t tmpBufSize = 0;
    SoftMaxTiling softmaxTiling;
};

extern "C" __global__ __aicore__ void conv3d_min_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMinSoftmax op;
    op.Init(x, y, tiling_data.srcSize, tiling_data.tmpBufSize, tiling_data.softmaxTilingData);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Applies Conv3d in NCDHW, min-reduces along depth (dim=2),
    then performs softmax along channel dimension via custom AscendC kernel.
    """
    def __init__(self, in_channels, out_channels, kernel_size, dim):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.dim = dim
        assert self.dim == 2, "Conv3dMinSoftmaxCustom kernel expects min reduction along depth (dim=2)."

    def forward(self, x):
        # x: [N, C_in, D, H, W]
        x = self.conv(x)                      # [N, C_out, D', H', W']
        x = torch.min(x, dim=self.dim)[0]     # [N, C_out, H', W']
        # Move channel to last axis to meet kernel expectation of reducing on last axis
        x = x.permute(0, 2, 3, 1).contiguous()  # [N, H', W', C_out], last axis is softmax axis
        x = custom_ops_lib.conv3d_min_softmax_custom(x)
        x = x.permute(0, 3, 1, 2).contiguous()  # [N, C_out, H', W']
        return x
'''
