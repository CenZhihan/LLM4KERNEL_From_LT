kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "conv2d_subtract_tanh_subtract_avg_pool_custom_tiling_data.h"

// 说明：该自定义核完成逐元素的 (tanh(x - c1) - c2) 计算。
// 其中 c1, c2 由 tiling 传入，来自算子属性。
// Conv2d 与 AvgPool 仍由上层网络结构负责，保持模块化组合。

constexpr int32_t BUFFER_NUM = 2; // 每个队列的 tensor 数

class KernelSubTanhSub {
public:
    __aicore__ inline KernelSubTanhSub() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t tileNum, float c1, float c2)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        // 将属性转为计算精度的数据类型
        this->sub1 = static_cast<DTYPE_X>(c1);
        this->sub2 = static_cast<DTYPE_X>(c2);

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_X *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        // 计算用临时缓冲：1 个 tile 的大小
        pipe.InitBuffer(tmpCalcQue, 1, this->tileLength * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t /*progress*/)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> tmpLocal = tmpCalcQue.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> yLocal = outQueueY.AllocTensor<DTYPE_X>();

        // 1) x - sub1  => 使用 Add 和负常量实现减法
        DTYPE_X negSub1 = static_cast<DTYPE_X>(-sub1);
        AscendC::Duplicate<DTYPE_X>(tmpLocal, negSub1, this->tileLength); // 先把 tmpLocal 填充为常量
        AscendC::Add(tmpLocal, xLocal, tmpLocal, this->tileLength);       // tmpLocal = x + (-sub1)

        // 2) tanh(tmpLocal) 到 yLocal
        AscendC::Tanh(yLocal, tmpLocal, this->tileLength);

        // 3) yLocal - sub2  => 再次使用 Add 和负常量实现
        DTYPE_X negSub2 = static_cast<DTYPE_X>(-sub2);
        AscendC::Duplicate<DTYPE_X>(tmpLocal, negSub2, this->tileLength); // 复用 tmpLocal 作为常量缓冲
        AscendC::Add(yLocal, yLocal, tmpLocal, this->tileLength);

        outQueueY.EnQue<DTYPE_X>(yLocal);
        tmpCalcQue.FreeTensor(tmpLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> yLocal = outQueueY.DeQue<DTYPE_X>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECCALC, 1> tmpCalcQue;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_X> yGm;

    uint32_t blockLength = 0;
    uint32_t tileNum = 0;
    uint32_t tileLength = 0;
    DTYPE_X sub1 = static_cast<DTYPE_X>(0.0f);
    DTYPE_X sub2 = static_cast<DTYPE_X>(0.0f);
};

extern "C" __global__ __aicore__ void conv2d_subtract_tanh_subtract_avg_pool_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSubTanhSub op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.tileNum, tiling_data.subtract1, tiling_data.subtract2);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    优化版：保留原始 Conv2d 与 AvgPool2d 层，
    用自定义 AscendC 核实现逐元素的 x - c1 -> tanh -> - c2 融合计算。
    """
    def __init__(self, in_channels, out_channels, kernel_size, subtract1_value, subtract2_value, kernel_size_pool):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.subtract1_value = float(subtract1_value)
        self.subtract2_value = float(subtract2_value)
        self.avgpool = torch.nn.AvgPool2d(kernel_size_pool)

    def forward(self, x):
        x = self.conv(x)
        x = custom_ops_lib.conv2d_subtract_tanh_subtract_avg_pool_custom(x, self.subtract1_value, self.subtract2_value)
        x = self.avgpool(x)
        return x
