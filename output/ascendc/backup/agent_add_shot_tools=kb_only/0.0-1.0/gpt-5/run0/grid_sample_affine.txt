kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "grid_sample_affine_custom_tiling_data.h"

__aicore__ inline float clampf(float v, float lo, float hi) {
    return v < lo ? lo : (v > hi ? hi : v);
}

class KernelGridSampleAffine {
public:
    __aicore__ inline KernelGridSampleAffine() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR theta, GM_ADDR y,
                                uint32_t n, uint32_t c, uint32_t h, uint32_t w)
    {
        this->n = n;
        this->c = c;
        this->h = h;
        this->w = w;

        xPtr = reinterpret_cast<__gm__ float*>(x);
        thetaPtr = reinterpret_cast<__gm__ float*>(theta);
        yPtr = reinterpret_cast<__gm__ float*>(y);

        blockNum = AscendC::GetBlockNum();
        blockIdx = AscendC::GetBlockIdx();
        batchPerBlock = (n + blockNum - 1) / blockNum;
        bStart = blockIdx * batchPerBlock;
        bEnd = bStart + batchPerBlock;
        if (bEnd > n) bEnd = n;
    }

    __aicore__ inline void Process()
    {
        Compute();
    }

private:
    __aicore__ inline void Compute()
    {
        // align_corners = false, mode = bilinear, padding_mode = zeros
        for (uint32_t b = bStart; b < bEnd; ++b) {
            // theta: [N,2,3] row-major
            float a00 = thetaPtr[b * 6 + 0];
            float a01 = thetaPtr[b * 6 + 1];
            float a02 = thetaPtr[b * 6 + 2];
            float a10 = thetaPtr[b * 6 + 3];
            float a11 = thetaPtr[b * 6 + 4];
            float a12 = thetaPtr[b * 6 + 5];

            for (uint32_t oy = 0; oy < h; ++oy) {
                float yn = ((static_cast<float>(oy) + 0.5f) / static_cast<float>(h)) * 2.f - 1.f;
                for (uint32_t ox = 0; ox < w; ++ox) {
                    float xn = ((static_cast<float>(ox) + 0.5f) / static_cast<float>(w)) * 2.f - 1.f;

                    float xs_n = a00 * xn + a01 * yn + a02;
                    float ys_n = a10 * xn + a11 * yn + a12;

                    float xs = (xs_n + 1.f) * 0.5f * static_cast<float>(w) - 0.5f;
                    float ys = (ys_n + 1.f) * 0.5f * static_cast<float>(h) - 0.5f;

                    int x0 = static_cast<int>(floorf(xs));
                    int y0 = static_cast<int>(floorf(ys));
                    int x1 = x0 + 1;
                    int y1 = y0 + 1;

                    float wx1 = xs - static_cast<float>(x0);
                    float wy1 = ys - static_cast<float>(y0);
                    float wx0 = 1.f - wx1;
                    float wy0 = 1.f - wy1;

                    bool x0_in = (x0 >= 0) && (x0 < static_cast<int>(w));
                    bool x1_in = (x1 >= 0) && (x1 < static_cast<int>(w));
                    bool y0_in = (y0 >= 0) && (y0 < static_cast<int>(h));
                    bool y1_in = (y1 >= 0) && (y1 < static_cast<int>(h));

                    uint64_t out_base = (static_cast<uint64_t>(b) * c * h + oy) * w + ox;
                    for (uint32_t ch = 0; ch < c; ++ch) {
                        float v00 = 0.f, v01 = 0.f, v10 = 0.f, v11 = 0.f;

                        if (x0_in && y0_in) {
                            uint64_t idx = ((static_cast<uint64_t>(b) * c + ch) * h + static_cast<uint32_t>(y0)) * w + static_cast<uint32_t>(x0);
                            v00 = xPtr[idx];
                        }
                        if (x1_in && y0_in) {
                            uint64_t idx = ((static_cast<uint64_t>(b) * c + ch) * h + static_cast<uint32_t>(y0)) * w + static_cast<uint32_t>(x1);
                            v01 = xPtr[idx];
                        }
                        if (x0_in && y1_in) {
                            uint64_t idx = ((static_cast<uint64_t>(b) * c + ch) * h + static_cast<uint32_t>(y1)) * w + static_cast<uint32_t>(x0);
                            v10 = xPtr[idx];
                        }
                        if (x1_in && y1_in) {
                            uint64_t idx = ((static_cast<uint64_t>(b) * c + ch) * h + static_cast<uint32_t>(y1)) * w + static_cast<uint32_t>(x1);
                            v11 = xPtr[idx];
                        }

                        float top = wx0 * v00 + wx1 * v01;
                        float bot = wx0 * v10 + wx1 * v11;
                        float val = wy0 * top + wy1 * bot;

                        uint64_t out_idx = out_base + static_cast<uint64_t>(ch) * static_cast<uint64_t>(h) * static_cast<uint64_t>(w);
                        yPtr[out_idx] = val;
                    }
                }
            }
        }
    }

private:
    __gm__ float* xPtr;
    __gm__ float* thetaPtr;
    __gm__ float* yPtr;

    uint32_t n, c, h, w;
    uint32_t blockNum, blockIdx;
    uint32_t batchPerBlock, bStart, bEnd;
};

extern "C" __global__ __aicore__ void grid_sample_affine_custom(GM_ADDR x, GM_ADDR theta, GM_ADDR y,
                                                                GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGridSampleAffine op;
    op.Init(x, theta, y, tiling_data.n, tiling_data.c, tiling_data.h, tiling_data.w);
    op.Process();
}
