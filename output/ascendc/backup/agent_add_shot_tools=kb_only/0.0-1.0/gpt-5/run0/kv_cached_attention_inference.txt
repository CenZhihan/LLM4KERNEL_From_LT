    tiling.set_kv_len(kv_len);

kernel_src="""
#include "kernel_operator.h"
#define __NPU_TILING__
#include "kv_cached_attention_inference_custom_tiling_data.h"

// 本核实现对输入logits进行SoftMax（沿最后一维kv_len），用于KV缓存推理中的注意力权重计算。
// 注意：为方便工程集成，本核的GM参数名沿用(q, k_cache, v_cache)，但仅使用q作为logits输入。

class KernelKvCachedAttentionInfer {
public:
    __aicore__ inline KernelKvCachedAttentionInfer() {}

    __aicore__ inline void Init(GM_ADDR logits, GM_ADDR dummy_k, GM_ADDR out_probs,
                                uint32_t batch, uint32_t q_len, uint32_t kv_len, uint32_t d_model,
                                const SoftMaxTiling &softmaxTilingData)
    {
        (void)dummy_k; // 未使用
        this->batch = batch;
        this->q_len = q_len;
        this->kv_len = kv_len;
        this->d_model = d_model;
        this->height = batch * q_len;
        this->width = kv_len;
        this->tiling = softmaxTilingData;

        srcGlobal.SetGlobalBuffer((__gm__ float*)logits);
        dstGlobal.SetGlobalBuffer((__gm__ float*)out_probs);

        // 缓冲区：一次性处理 [height, width]
        pipe.InitBuffer(inQueueSrc, 1, height * width * sizeof(float));
        pipe.InitBuffer(outQueueDst, 1, height * width * sizeof(float));
        // SoftMax临时空间（参考最小空间，按kv_len尺度分配）
        pipe.InitBuffer(tmpBuf, width * sizeof(float) * 4); // 参考文档：最小临时空间可保证功能正确
    }

    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<float> srcLocal = inQueueSrc.AllocTensor<float>();
        AscendC::DataCopy(srcLocal, srcGlobal, height * width);
        inQueueSrc.EnQue(srcLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> srcLocal = inQueueSrc.DeQue<float>();
        AscendC::LocalTensor<float> dstLocal = outQueueDst.AllocTensor<float>();
        AscendC::LocalTensor<uint8_t> tmpLocal = tmpBuf.Get<uint8_t>();

        AscendC::SoftMaxShapeInfo shapeInfo = { height, width, height, width };
        // 使用SoftMax高阶API（参考5.3.1.9 SoftMax/SimpleSoftMax Tiling）
        AscendC::SoftMax<float>(dstLocal, srcLocal, tmpLocal, tiling, shapeInfo);

        outQueueDst.EnQue<float>(dstLocal);
        inQueueSrc.FreeTensor(srcLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<float> dstLocal = outQueueDst.DeQue<float>();
        AscendC::DataCopy(dstGlobal, dstLocal, height * width);
        outQueueDst.FreeTensor(dstLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> inQueueSrc;
    AscendC::TQue<AscendC::QuePosition::VECOUT, 1> outQueueDst;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;

    AscendC::GlobalTensor<float> srcGlobal;
    AscendC::GlobalTensor<float> dstGlobal;

    uint32_t batch;
    uint32_t q_len;
    uint32_t kv_len;
    uint32_t d_model;
    uint32_t height;
    uint32_t width;
    SoftMaxTiling tiling;
};

extern "C" __global__ __aicore__ void kv_cached_attention_inference_custom(GM_ADDR q, GM_ADDR k_cache, GM_ADDR v_cache, GM_ADDR workspace, GM_ADDR tiling)
{
    (void)workspace;
    GET_TILING_DATA(tilingData, tiling);
    KernelKvCachedAttentionInfer op;
    op.Init(q, k_cache, v_cache,
            tilingData.batch, tilingData.q_len, tilingData.kv_len, tilingData.d_model,
            tilingData.softmaxTilingData);
    op.Process();
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>
#include <cmath>

at::Tensor kv_cached_attention_inference_custom_impl_npu(const at::Tensor& Q,
                                                         const at::Tensor& K_cache,
                                                         const at::Tensor& V_cache) {
    // 计算注意力logits: [B, q_len, kv_len]
    auto d_model = Q.size(-1);
    at::Tensor Kt = K_cache.transpose(-2, -1); // [B, D, kv_len]
    at::Tensor logits = at::matmul(Q, Kt) / std::sqrt(static_cast<double>(d_model));

    // 调用自定义核进行SoftMax（沿最后一维kv_len）
    at::Tensor probs = at::empty_like(logits);
    // 说明：为匹配算子签名(3输入1输出)，这里将V_cache作为第三个输入传入，但内核仅使用第一个输入(logits)
    EXEC_NPU_CMD(aclnnKvCachedAttentionInferenceCustom, logits, K_cache, probs);

    // 输出：attn @ V_cache -> [B, q_len, D]
    at::Tensor out = at::matmul(probs, V_cache);
    return out;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("kv_cached_attention_inference_custom", &kv_cached_attention_inference_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("kv_cached_attention_inference_custom", &kv_cached_attention_inference_custom_impl_npu, "KV Cached Attention Inference (SoftMax fused)");
}
