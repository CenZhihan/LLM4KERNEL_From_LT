host_operator_src="""
#include "log_softmax_custom_tiling.h"
#include "register/op_def_registry.h"

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    LogSoftmaxCustomTilingData tiling;
    auto in_shape = context->GetInputShape(0)->GetOriginShape();
    uint32_t dims = in_shape.GetDimNum();
    uint32_t rows = 1;
    uint32_t cols = 1;
    if (dims == 0) {
        return ge::GRAPH_FAILED;
    } else if (dims == 1) {
        rows = 1;
        cols = static_cast<uint32_t>(in_shape.GetDim(0));
    } else {
        rows = static_cast<uint32_t>(in_shape.GetDim(0));
        cols = static_cast<uint32_t>(in_shape.GetDim(dims - 1));
        for (uint32_t i = 1; i + 1 < dims; ++i) {
            rows *= static_cast<uint32_t>(in_shape.GetDim(i));
        }
    }

    context->SetBlockDim(BLOCK_DIM);
    tiling.set_rows(rows);
    tiling.set_cols(cols);
    tiling.set_tileNum(TILE_NUM);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    // workspace stores two arrays (per-row): row_max and row_sumexp
    currentWorkspace[0] = static_cast<size_t>(rows) * 2 * sizeof(float);
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"
#include <cfloat>

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelLogSoftmax {
public:
    __aicore__ inline KernelLogSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, uint32_t rows, uint32_t cols, uint32_t tileNum)
    {
        this->rows = rows;
        this->cols = cols;
        this->tileNum = tileNum;
        // Each block handles a contiguous range of rows
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (rows + blockNum - 1) / blockNum;
        uint32_t startRow = blockIdx * this->rowsPerBlock;
        if (startRow + this->rowsPerBlock > rows) {
            this->rowsPerBlock = (rows > startRow) ? (rows - startRow) : 0;
        }
        this->startRow = startRow;

        // tile length along columns
        uint32_t totalTiles = tileNum * BUFFER_NUM;
        this->tileLength = (cols + totalTiles - 1) / totalTiles;
        if (this->tileLength == 0) this->tileLength = cols > 0 ? cols : 1;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startRow * cols, this->rowsPerBlock * cols);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + startRow * cols, this->rowsPerBlock * cols);

        // workspace layout: [rows floats of max][rows floats of sumExp]
        maxWs = (__gm__ float*)((char*)workspace) + 0;
        sumExpWs = (__gm__ float*)((char*)workspace) + rows;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        if (rowsPerBlock == 0 || cols == 0) {
            return;
        }
        // Pass 1: compute per-row max
        for (uint32_t r = 0; r < rowsPerBlock; ++r) {
            float rmax = -FLT_MAX;
            uint32_t processed = 0;
            int32_t loopCount = (cols + tileLength - 1) / tileLength;
            for (int32_t i = 0; i < loopCount; ++i) {
                uint32_t curLen = AscendC::Min(tileLength, cols - processed);
                CopyIn(r, processed, curLen);
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
                float tmax = AscendC::ReduceMax(xLocal, curLen);
                if (tmax > rmax) rmax = tmax;
                inQueueX.FreeTensor(xLocal);
                processed += curLen;
            }
            maxWs[startRow + r] = rmax;
        }

        // Pass 2: compute per-row sum(exp(x - max))
        for (uint32_t r = 0; r < rowsPerBlock; ++r) {
            float rmax = maxWs[startRow + r];
            float rsum = 0.f;
            uint32_t processed = 0;
            int32_t loopCount = (cols + tileLength - 1) / tileLength;
            for (int32_t i = 0; i < loopCount; ++i) {
                uint32_t curLen = AscendC::Min(tileLength, cols - processed);
                CopyIn(r, processed, curLen);
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();

                // xLocal = xLocal - rmax
                AscendC::Sub(xLocal, xLocal, rmax, curLen);
                // xLocal = exp(xLocal)
                AscendC::Exp(xLocal, xLocal, curLen);
                float tsum = AscendC::ReduceSum(xLocal, curLen);
                rsum += tsum;

                inQueueX.FreeTensor(xLocal);
                processed += curLen;
            }
            sumExpWs[startRow + r] = rsum;
        }

        // Pass 3: y = x - (rmax + log(sumExp))
        for (uint32_t r = 0; r < rowsPerBlock; ++r) {
            float rmax = maxWs[startRow + r];
            float rsum = sumExpWs[startRow + r];
            float logsum = AscendC::Log(rsum);
            float shift = rmax + logsum;

            uint32_t processed = 0;
            int32_t loopCount = (cols + tileLength - 1) / tileLength;
            for (int32_t i = 0; i < loopCount; ++i) {
                uint32_t curLen = AscendC::Min(tileLength, cols - processed);
                // read
                CopyIn(r, processed, curLen);
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();

                // yLocal = xLocal - shift
                AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
                AscendC::Sub(yLocal, xLocal, shift, curLen);

                outQueueY.EnQue<DTYPE_Y>(yLocal);
                inQueueX.FreeTensor(xLocal);

                // write
                CopyOut(r, processed, curLen);
                processed += curLen;
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t r, uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[r * cols + offset], len);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t r, uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[r * cols + offset], yLocal, len);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    __gm__ float* maxWs;
    __gm__ float* sumExpWs;

    uint32_t rows;
    uint32_t cols;
    uint32_t tileNum;
    uint32_t tileLength;

    uint32_t startRow;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void log_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLogSoftmax op;
    op.Init(x, y, workspace, tiling_data.rows, tiling_data.cols, tiling_data.tileNum);
    op.Process();
}
"""
