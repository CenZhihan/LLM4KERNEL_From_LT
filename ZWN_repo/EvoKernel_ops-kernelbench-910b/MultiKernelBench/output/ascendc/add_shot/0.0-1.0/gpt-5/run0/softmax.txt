host_operator_src="""
#include "softmax_custom_tiling.h"
#include "register/op_def_registry.h"

namespace {
inline uint32_t u32_max(uint32_t a, uint32_t b) { return a > b ? a : b; }
}

namespace optiling {
const uint32_t BLOCK_DIM = 8;
const uint32_t TILE_NUM = 8;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    SoftmaxCustomTilingData tiling;

    auto in_shape = context->GetInputShape(0)->GetOriginShape();
    uint32_t rank = static_cast<uint32_t>(in_shape.GetDimNum());

    uint64_t rows64 = 1;
    uint64_t cols64 = 1;
    if (rank == 0) {
        rows64 = 1;
        cols64 = 1;
    } else if (rank == 1) {
        rows64 = 1;
        cols64 = static_cast<uint64_t>(in_shape.GetDim(0));
    } else {
        rows64 = static_cast<uint64_t>(in_shape.GetDim(0));
        cols64 = 1;
        for (uint32_t i = 1; i < rank; ++i) {
            cols64 *= static_cast<uint64_t>(in_shape.GetDim(i));
        }
    }

    uint32_t rows = static_cast<uint32_t>(rows64);
    uint32_t cols = static_cast<uint32_t>(cols64);

    context->SetBlockDim(BLOCK_DIM);

    uint32_t rowsPerBlock = (rows + BLOCK_DIM - 1) / BLOCK_DIM;
    uint32_t tileNum = TILE_NUM;
    uint32_t tileCols = u32_max(1u, cols / tileNum);

    tiling.set_rows(rows);
    tiling.set_cols(cols);
    tiling.set_rowsPerBlock(rowsPerBlock);
    tiling.set_tileCols(tileCols);
    tiling.set_tileNum(tileNum);

    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *workspaceSizes = context->GetWorkspaceSizes(1);
    workspaceSizes[0] = 0;

    return ge::GRAPH_SUCCESS;
}
}


namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* in_shape = context->GetInputShape(0);
    gert::Shape* out_shape = context->GetOutputShape(0);
    *out_shape = *in_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}


namespace ops {
class SoftmaxCustom : public OpDef {
public:
    explicit SoftmaxCustom(const char* name) : OpDef(name)
    {
        this->Input("x")
            .ParamType(REQUIRED)
            .DataType({ge::DT_FLOAT})
            .Format({ge::FORMAT_ND})

            .DataType({ge::DT_FLOAT})

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // buffers for queues

class KernelSoftmax {
public:
    __aicore__ inline KernelSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t rows, uint32_t cols, uint32_t rowsPerBlock, uint32_t tileCols)
    {
        this->rows = rows;
        this->cols = cols;
        this->rowsPerBlock = rowsPerBlock;
        this->tileCols = tileCols;

        uint32_t startRow = AscendC::GetBlockIdx() * this->rowsPerBlock;
        uint32_t rowsThisBlock = GetRowsThisBlock(startRow);
        this->rowsThisBlock = rowsThisBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + static_cast<uint64_t>(startRow) * cols, static_cast<uint64_t>(rowsThisBlock) * cols);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + static_cast<uint64_t>(startRow) * cols, static_cast<uint64_t>(rowsThisBlock) * cols);

        // init queue buffers
        pipe.InitBuffer(inQueue, BUFFER_NUM, this->tileCols * sizeof(DTYPE_X));
        pipe.InitBuffer(tmpQueue, BUFFER_NUM, this->tileCols * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileCols * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        // Process each row assigned to this core
        for (uint32_t r = 0; r < rowsThisBlock; ++r) {
            RowSoftmax(r);
        }
    }

private:
    __aicore__ inline uint32_t GetRowsThisBlock(uint32_t startRow) const
    {
        uint32_t remain = (startRow < rows) ? (rows - startRow) : 0;
        return remain < rowsPerBlock ? remain : rowsPerBlock;
    }

    __aicore__ inline uint32_t NumTiles() const
    {
        return (cols + tileCols - 1) / tileCols;
    }

    __aicore__ inline void LoadTile(AscendC::LocalTensor<DTYPE_X>& xLocal, uint32_t rowOffset, uint32_t t)
    {
        uint32_t offset = t * tileCols;
        uint32_t len = (offset + tileCols <= cols) ? tileCols : (cols - offset);
        AscendC::DataCopy(xLocal, xGm[rowOffset + offset], len);
        currentTileLen = len;
    }

    __aicore__ inline void StoreTile(const AscendC::LocalTensor<DTYPE_Y>& yLocal, uint32_t rowOffset, uint32_t t)
    {
        uint32_t offset = t * tileCols;
        AscendC::DataCopy(yGm[rowOffset + offset], yLocal, currentTileLen);
    }

    __aicore__ inline void RowSoftmax(uint32_t r)
    {
        uint32_t rowOffset = r * cols;
        uint32_t numTiles = NumTiles();

        // Pass 1: compute row max
        DTYPE_X rowMax;
        bool maxInit = false;
        for (uint32_t t = 0; t < numTiles; ++t) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.AllocTensor<DTYPE_X>();
            LoadTile(xLocal, rowOffset, t);
            DTYPE_X tileMax = AscendC::ReduceMax(xLocal, currentTileLen);
            inQueue.FreeTensor(xLocal);
            if (!maxInit) {
                rowMax = tileMax;
                maxInit = true;
            } else {
                rowMax = tileMax > rowMax ? tileMax : rowMax;
            }
        }

        // Pass 2: compute exp(x - rowMax), accumulate sum, and store numerators in output
        DTYPE_Y sumExp = static_cast<DTYPE_Y>(0);
        for (uint32_t t = 0; t < numTiles; ++t) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.AllocTensor<DTYPE_X>();
            AscendC::LocalTensor<DTYPE_Y> yLocal = tmpQueue.AllocTensor<DTYPE_Y>();

            LoadTile(xLocal, rowOffset, t);
            // yLocal = exp(xLocal - rowMax)
            AscendC::Sub(yLocal, xLocal, rowMax, currentTileLen);
            AscendC::Exp(yLocal, yLocal, currentTileLen);

            DTYPE_Y tileSum = AscendC::ReduceSum(yLocal, currentTileLen);
            sumExp += tileSum;

            // Store temporary numerator to yGm for now
            StoreTile(yLocal, rowOffset, t);

            tmpQueue.FreeTensor(yLocal);
            inQueue.FreeTensor(xLocal);
        }

        // Pass 3: normalize by sumExp
        for (uint32_t t = 0; t < numTiles; ++t) {
            AscendC::LocalTensor<DTYPE_Y> yLocal = outQueue.AllocTensor<DTYPE_Y>();
            uint32_t offset = t * tileCols;
            currentTileLen = (offset + tileCols <= cols) ? tileCols : (cols - offset);
            AscendC::DataCopy(yLocal, yGm[rowOffset + offset], currentTileLen);
            AscendC::Div(yLocal, yLocal, sumExp, currentTileLen);
            AscendC::DataCopy(yGm[rowOffset + offset], yLocal, currentTileLen);
            outQueue.FreeTensor(yLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> tmpQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t rows;
    uint32_t cols;
    uint32_t rowsPerBlock;
    uint32_t tileCols;
    uint32_t rowsThisBlock;
    uint32_t currentTileLen;
};

extern "C" __global__ __aicore__ void softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);

    KernelSoftmax op;
    op.Init(x, y,
            tiling_data.rows,
            tiling_data.cols,
            tiling_data.rowsPerBlock,
            tiling_data.tileCols);
    op.Process();
}
